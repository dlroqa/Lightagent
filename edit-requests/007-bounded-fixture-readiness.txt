Bound the MCP test fixture's readiness probe

The python3 probe in installed_mcp_tool_is_listed_callable_and_removed
had no limit, and took 163 s under injected disk contention (#82). It
now runs through run_helper, which gives it 120 s, discards its output
so nothing blocks on a pipe, and kills and reaps it at the deadline or
on cancellation (kill_on_drop covers a dropped future). A probe that
does not finish in time fails the test as a precondition failure; a
missing python3, or one that cannot run the fixture, still skips it.

Deterministic tests run this test binary as the helper (exit 0, exit 3,
never exit) to cover readiness, a nonzero exit, a spawn failure, the
deadline and cancellation, each with reaping. The production MCP
timeout and the test's assertions are unchanged.

    /// How long the MCP fixture's interpreter may take to start before
    /// `installed_mcp_tool_is_listed_callable_and_removed` fails. A quiet
    /// cold start took under a second on every Windows runner measured, and
    /// the slowest whole lib suite seen without injected load took 77 s
    /// (#82); this exceeds both with room, and leaves most of the 60-minute
    /// check job to report a start that is stuck rather than slow.
    const FIXTURE_READY_WITHIN: Duration = Duration::from_secs(120);
    /// How long a killed helper may take to exit and be reaped.
    const HELPER_REAP_WITHIN: Duration = Duration::from_secs(10);

    /// What running a test helper to completion found. It holds no paths,
    /// arguments or error text: only what a test acts on.
    #[derive(Debug)]
    enum Readiness {
        /// It exited successfully.
        Ready,
        /// It could not be started, as when it is not installed.
        NotStarted(std::io::ErrorKind),
        /// It exited unsuccessfully, with this code where there is one.
        Exited(Option<i32>),
        /// It was killed before it exited; `reaped` says whether it was then
        /// waited for within `HELPER_REAP_WITHIN`.
        Stopped { why: Stop, reaped: bool },
    }

    #[derive(Debug, PartialEq, Eq)]
    enum Stop {
        /// It had not exited by the deadline.
        Deadline,
        /// The caller cancelled first.
        Cancelled,
        /// Its exit could not be observed.
        WaitFailed,
    }

    /// Runs `command` to completion within `within`, with no input and its
    /// output discarded, so neither side can block on a pipe. A helper still
    /// running at the deadline or on `cancel` is killed and waited for, so it
    /// is reaped rather than left running; if this future is dropped instead,
    /// `kill_on_drop` kills it. Starting the process is a blocking system call
    /// that the deadline counts but cannot interrupt.
    async fn run_helper(
        mut command: tokio::process::Command,
        within: Duration,
        cancel: &CancellationToken,
    ) -> Readiness {
        let started = Instant::now();
        let mut child = match command
            .stdin(std::process::Stdio::null())
            .stdout(std::process::Stdio::null())
            .stderr(std::process::Stdio::null())
            .kill_on_drop(true)
            .spawn()
        {
            Ok(child) => child,
            Err(error) => return Readiness::NotStarted(error.kind()),
        };
        let remaining = within.saturating_sub(started.elapsed());
        let why = tokio::select! {
            waited = tokio::time::timeout(remaining, child.wait()) => match waited {
                Ok(Ok(status)) if status.success() => return Readiness::Ready,
                Ok(Ok(status)) => return Readiness::Exited(status.code()),
                Ok(Err(_)) => Stop::WaitFailed,
                Err(_) => Stop::Deadline,
            },
            () = cancel.cancelled() => Stop::Cancelled,
        };
        // Killing a helper that exited meanwhile fails harmlessly, and the
        // wait then reaps it either way.
        child.start_kill().ok();
        let reaped = matches!(
            tokio::time::timeout(HELPER_REAP_WITHIN, child.wait()).await,
            Ok(Ok(_))
        );
        Readiness::Stopped { why, reaped }
    }

    /// The helper process of the `run_helper` tests: this test binary, run
    /// again with `LIGHTAGENT_TEST_HELPER` set, exits with code 3 for `exit`
    /// and never exits for `hold`. Run any other way it does nothing.
    #[test]
    #[ignore = "run only as the helper process of the run_helper tests"]
    fn run_helper_process() {
        match std::env::var("LIGHTAGENT_TEST_HELPER").as_deref() {
            Ok("exit") => std::process::exit(3),
            Ok("hold") => loop {
                std::thread::park();
            },
            _ => {}
        }
    }

    fn helper(behaviour: &str) -> tokio::process::Command {
        let mut command = tokio::process::Command::new(std::env::current_exe().unwrap());
        command
            .args(["--exact", "chat::model_tests::run_helper_process", "--ignored"])
            .env("LIGHTAGENT_TEST_HELPER", behaviour);
        command
    }

    async fn a_helper_still_running_at_the_deadline_is_killed_and_reaped() {

        let started = Instant::now();
        let Readiness::Stopped { why, reaped } =
            run_helper(helper("hold"), within, &CancellationToken::new()).await
        else {
            panic!("a helper that never exits is stopped");
        };
        assert_eq!(why, Stop::Deadline);
        assert!(reaped, "the killed helper was waited for");
        assert!(started.elapsed() < within + HELPER_REAP_WITHIN);
    }
