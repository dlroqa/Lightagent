=== message
D5a-2: a turn its runtime did not live to record is Interrupted, never rerun

A session turn the API admits saves the prompt at once and records the run
when it ends. A runtime that died in between (a crash, a kill) left the
prompt with no record and no word of what happened.

- The store keeps a run journal, `sessions/.runs/<id>.run`, written when
  an API turn starts and removed once it is recorded. One that outlives the
  process holding the session's run lease marks an interrupted turn.
- `recover_interrupted` records it: an append turn gets a run record with
  stop reason `Interrupted` and no answer; an interrupted regeneration
  leaves the saved answer and its record exactly as they were. Then the
  journal goes. Nothing is run again, so no tool call is repeated.
- When: as `lightagent serve` (the desktop runtime included) starts, for
  every session no live process holds; and when a session's next turn is
  admitted, under its lease, before that turn.
- The Web UI names the stop reason: interrupted, not run again.

Evidence:
- store tests: recorded once, a live holder's journal left alone, an
  interrupted regeneration keeps the answer, a recorded turn leaves none;
- crates/lightagent/tests/desktop_interrupted.rs (separate processes, all
  platforms): a runtime killed mid-turn; the next runtime records the turn
  as Interrupted, the provider gets no second request, the session takes
  the next turn; an interrupted regeneration keeps the saved answer;
- desktop/e2e/runs.py `crash` (the real host): an approved write runs, the
  runtime is killed while the model's next turn is in flight; the host
  reports the lost stream and restarts its runtime once; the session says
  Interrupted with no answer; the turn and the write are never repeated;
  the next run is admitted.

    /// Record session `id`'s interrupted turn, if a journal outlived the
    /// process that wrote it. An appended turn gets a run record with stop
    /// reason [`INTERRUPTED`] and no answer; an interrupted regeneration
    /// leaves the saved answer and its record as they were. Then the journal
    /// goes. Nothing is run again. The caller holds `id`'s run lease, so no
    /// live turn owns the journal. Returns whether there was one.
    pub fn recover_interrupted(
        &self,
        id: &SessionId,
        _lease: &RunLease,
    ) -> Result<bool, StoreError> {
        let path = self.journal_path(id);
        let bytes = match std::fs::read(&path) {
            Ok(bytes) => bytes,
            Err(err) if err.kind() == std::io::ErrorKind::NotFound => return Ok(false),
            Err(err) => {
                return Err(StoreError::Unreadable {
                    id: id.as_str().to_owned(),
                    reason: err.to_string(),
                });
            }
        };
        // An unreadable journal still marks an interruption; only its
        // details are lost.
        let journal: Option<RunJournal> = serde_json::from_slice(&bytes).ok();
        if !journal.as_ref().is_some_and(|journal| journal.replacing) {
            let record = RunRecord {
                run_id: journal
                    .as_ref()
                    .map(|journal| journal.run_id.clone())
                    .unwrap_or_default(),
                started_at: journal
                    .as_ref()
                    .map_or_else(SystemTime::now, |journal| journal.started_at),
                ended_at: Some(SystemTime::now()),
                stop_reason: Some(INTERRUPTED.to_owned()),
                tools: Vec::new(),
            };
            let recorded = self.update(id, |session| {
                session.runs.push(record);
                session.touch();
                Ok::<(), Infallible>(())
            });
            match recorded {
                Ok(_) | Err(SessionUpdateError::Store(StoreError::NotFound(_))) => {}
                Err(SessionUpdateError::Store(error)) => return Err(error),
                Err(SessionUpdateError::Change(never)) => match never {},
            }
        }
        let _ = std::fs::remove_file(&path);
        Ok(true)
    }

    /// Record every interrupted turn in this store
    /// ([`Self::recover_interrupted`]). A session whose lease a live process
    /// holds is left alone. Returns the sessions recorded.
    pub fn recover_all_interrupted(&self) -> Vec<SessionId> {
        let Ok(entries) = std::fs::read_dir(self.directory.join(RUNS_DIR)) else {
            return Vec::new();
        };
        let mut recovered = Vec::new();
        for entry in entries.flatten() {
            let name = entry.file_name();
            let Some(stem) = name.to_str().and_then(|name| name.strip_suffix(".run")) else {
                continue;
            };
            let Ok(id) = SessionId::parse(stem) else {
                continue;
            };
            let Ok(lease) = self.lease_run(&id) else {
                continue;
            };
            if self.recover_interrupted(&id, &lease).unwrap_or(false) {
                recovered.push(id);
            }
        }
        recovered
    }

//! runtime starts again on the same home, the session records that turn as
//! `Interrupted` and the turn is never run again (the provider gets no

const KEY: &str = "92a3b4c5d6e7f8091a2b3c4d5e6f708192a3b4c5d6e7f8091a2b3c4d5e6f7081";
const MODELS: &str = r#"{"object":"list","data":[{"id":"m1"}]}"#;
const ANSWER: &str = "HTTP/1.1 200 OK\r\ncontent-type: text/event-stream\r\nconnection: close\r\n\r\n\

struct Runtime {
    child: Child,
    _lifeline: ChildStdin,
    addr: String,
}

impl Drop for Runtime {
    fn drop(&mut self) {
        let _ = self.child.kill();
        let _ = self.child.wait();
    }
}

fn scratch_home(label: &str) -> PathBuf {
    let nanos = SystemTime::now()
        .duration_since(UNIX_EPOCH)
        .map(|elapsed| elapsed.as_nanos())
        .unwrap_or(0);
    let home = std::env::temp_dir().join(format!(
        "lightagent-desktop-interrupted-{label}-{}-{nanos}",
        std::process::id()
    ));
    std::fs::create_dir_all(&home).expect("home");
    home
}

fn start(home: &Path) -> Runtime {
    let mut child = Command::new(env!("CARGO_BIN_EXE_lightagent"))
        .args(["serve", "--desktop", "--host", "127.0.0.1", "--port", "0"])
        .env("LIGHTAGENT_HOME", home)
        .stdin(Stdio::piped())
        .stdout(Stdio::piped())
        .stderr(Stdio::null())
        .spawn()
        .expect("start lightagent serve --desktop");
    let mut lifeline = child.stdin.take().expect("stdin");
    writeln!(lifeline, "{KEY}").expect("key");
    lifeline.flush().expect("flush");
    let stdout = child.stdout.take().expect("stdout");
    let (tx, rx) = mpsc::channel();
    std::thread::spawn(move || {
        let mut line = String::new();
        let _ = BufReader::new(stdout).read_line(&mut line);
        let _ = tx.send(line);
    });
    let line = rx.recv_timeout(Duration::from_secs(60)).expect("handshake");
    let handshake: Value = serde_json::from_str(&line).expect("handshake JSON");
    let addr = handshake["addr"].as_str().expect("addr").to_owned();
    Runtime {
        child,
        _lifeline: lifeline,
        addr,
    }
}

/// One authenticated request: the status and the raw body.
fn call(runtime: &Runtime, method: &str, path: &str, body: Option<&str>) -> (u16, String) {
    let mut stream = TcpStream::connect(&runtime.addr).expect("connect");
    stream
        .set_read_timeout(Some(Duration::from_secs(60)))
        .expect("timeout");
    let body = body.unwrap_or("");
    let head = format!(
        "{method} {path} HTTP/1.1\r\nHost: {}\r\nAuthorization: Bearer {KEY}\r\nContent-Type: application/json\r\nContent-Length: {}\r\nConnection: close\r\n\r\n{body}",
        runtime.addr,
        body.len()
    );
    stream.write_all(head.as_bytes()).expect("send");
    let mut response = String::new();
    let _ = stream.read_to_string(&mut response);
    let status = response
        .split_whitespace()
        .nth(1)
        .and_then(|code| code.parse().ok())
        .unwrap_or(0);
    let raw = response
        .split_once("\r\n\r\n")
        .map(|(_, body)| body.to_owned())
        .unwrap_or_default();
    (status, raw)
}

/// Read one request, its body included; return its head.
fn read_request(stream: &mut TcpStream) -> Option<String> {
    let _ = stream.set_read_timeout(Some(Duration::from_secs(10)));
    let mut bytes = Vec::new();
    let mut buffer = [0u8; 4096];
    loop {
        if let Some(end) = bytes.windows(4).position(|window| window == b"\r\n\r\n") {
            let head = String::from_utf8_lossy(&bytes[..end]).into_owned();
            let length = head
                .lines()
                .find_map(|line| {
                    line.to_ascii_lowercase()
                        .strip_prefix("content-length:")
                        .map(|value| value.trim().parse::<usize>().unwrap_or(0))
                })
                .unwrap_or(0);
            if bytes.len() >= end + 4 + length {
                return Some(head);
            }
        }
        match stream.read(&mut buffer) {
            Ok(0) | Err(_) => return None,
            Ok(read) => bytes.extend_from_slice(&buffer[..read]),
        }
    }
}

/// A provider that answers the first `answered` chat requests with "First."
/// and holds every later one open, counting them all.
fn provider(answered: usize) -> (String, Arc<AtomicUsize>) {
    let listener = TcpListener::bind("127.0.0.1:0").expect("listener");
    let url = format!("http://{}", listener.local_addr().expect("address"));
    let chats = Arc::new(AtomicUsize::new(0));
    let counter = Arc::clone(&chats);
    std::thread::spawn(move || {
        for stream in listener.incoming() {
            let Ok(mut stream) = stream else { continue };
            let counter = Arc::clone(&counter);
            std::thread::spawn(move || {
                let Some(head) = read_request(&mut stream) else {
                    return;
                };
                if head.starts_with("GET") {
                    let response = format!(
                        "HTTP/1.1 200 OK\r\nContent-Type: application/json\r\nContent-Length: {}\r\nConnection: close\r\n\r\n{MODELS}",
                        MODELS.len()
                    );
                    let _ = stream.write_all(response.as_bytes());
                    return;
                }
                if counter.fetch_add(1, Ordering::SeqCst) < answered {
                    let _ = stream.write_all(ANSWER.as_bytes());
                } else {
                    std::thread::sleep(Duration::from_secs(300));
                }
            });
        }
    });
    (url, chats)
}

fn wait_until(what: &str, mut probe: impl FnMut() -> bool) {
    let deadline = Instant::now() + Duration::from_secs(60);
    while !probe() {
        assert!(Instant::now() < deadline, "timed out waiting for {what}");
        std::thread::sleep(Duration::from_millis(50));
    }
}

fn run_on(runtime: &Runtime, id: &str, message: &str) -> (u16, String) {

#[test]
fn a_turn_interrupted_by_a_crash_is_recorded_and_never_run_again() {
    let home = scratch_home("append");
    let (url, chats) = provider(0);
    cli_ok(&home, &["config", "set", "inference.base_url", &url]);
    let runtime = start(&home);
    let id = new_session(&runtime);
    let (status, raw) = run_on(&runtime, &id, "hello");
    assert_eq!(status, 202, "{raw}");
    wait_until("the turn to reach the provider", || {
        chats.load(Ordering::SeqCst) == 1
    });
    // Killed mid-turn: nothing records the turn.
    drop(runtime);

    let runtime = start(&home);
    let saved = session(&runtime, &id);
    let runs = saved["runs"].as_array().expect("runs");
    assert_eq!(runs.len(), 1, "{saved}");
    assert_eq!(runs[0]["stop_reason"], "Interrupted", "{saved}");
    assert_eq!(
        saved["messages"].as_array().map(Vec::len),
        Some(1),
        "the prompt only, no answer: {saved}"
    );
    assert_eq!(
        chats.load(Ordering::SeqCst),
        1,
        "the interrupted turn was sent again"
    );

    // The session takes the next turn, and the interruption stays recorded once.
    let (status, raw) = run_on(&runtime, &id, "again");
    assert_eq!(status, 202, "{raw}");
    wait_until("the next turn to reach the provider", || {
        chats.load(Ordering::SeqCst) == 2
    });
    let saved = session(&runtime, &id);
    assert_eq!(saved["runs"].as_array().map(Vec::len), Some(1), "{saved}");
}

    except psutil.Error:
