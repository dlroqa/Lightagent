D7-3: knowledge (RAG) through one shared operation set; the index is written under a lock

- lightagent-rag: RagStore wrote its in-memory copy back with
  std::fs::write, so two writers (the CLI and, from now on, the API)
  lost each other's documents, and a reader could see a half-written
  index. add embeds first, then, under the index's lock file, reads the
  index again, replaces that source's chunks and writes it atomically and
  owner-only (paths::write_private, as the module doc promises); clear
  does the same. Single-writer behavior is unchanged. A test: two stores
  opened on one index each add a source; both remain (the old code kept
  only the second).
- crate lightagent rag: one implementation of add (a file, or every file
  in a folder; the remote index is still synced and a remote failure
  never drops the local document), search (remote first when configured,
  else hybrid local), list and clear. The CLI prints as before; its
  --json gains `skipped` and `notes` (what it printed to stderr), and
  rag::perform answers the API with the same JSON.
- ops::RagRequest and RunFactory::rag (default: unavailable).
- API (operator): GET|POST|DELETE /knowledge, GET /knowledge/search
  (`q`, `top_k`). POST takes an absolute file or folder path. Contract
  knowledge.*; the host refuses add and clear during a profile switch.
  Four backend inventory rows (not counted).
- Tests (separate processes): CLI and API list, search and add alike; a
  relative path refused; a missing path reported as skipped by both;
  CLI and API adds of different documents at the same time keep both;
  clear over the API empties the CLI's list.

/// Resolve the active profile's index path and the loaded config.
fn active_index() -> Result<(PathBuf, Config, String), String> {
    let paths = LightagentPaths::resolve().map_err(|error| error.to_string())?;
    let config = ConfigStore::at(&paths)
        .load()
        .map_err(|error| error.to_string())?;
    let store = ProfileStore::new(paths.root());
    let active = store
        .active()
        .map_err(|error| error.to_string())?
        .ok_or_else(|| "no active profile — run `lightagent init` first".to_owned())?;
    let dir = store.handle(&active).dir().to_path_buf();
    Ok((index_path(&dir), config, active.to_string()))
}

/// `rag add <path>` — index a file, or every file in a directory.
pub async fn add(path: PathBuf, source: Option<String>, json: bool) -> Result<(), String> {
    let (index, config, profile) = active_index()?;
    let mut store = RagStore::open(&index).map_err(|error| error.to_string())?;
    let embedder = HashingEmbedder;
    let semantic = semantic_embedder(&config);

    let mut targets = Vec::new();
    if path.is_dir() {
        let entries = std::fs::read_dir(&path).map_err(|error| error.to_string())?;
        for entry in entries.flatten() {
            if entry.path().is_file() {
                targets.push(entry.path());
            }
        }
        targets.sort();
    } else {
        targets.push(path.clone());
    }

    let mut total = 0;
    let mut indexed = Vec::new();
    for target in targets {
        let text = match std::fs::read_to_string(&target) {
            Ok(text) => text,
            Err(error) => {
                eprintln!("· skipping {}: {error}", target.display());
                continue;
            }
        };
        let name = source.clone().unwrap_or_else(|| {
            target
                .file_name()
                .map(|n| n.to_string_lossy().into_owned())
                .unwrap_or_else(|| target.display().to_string())
        });
        let added = store
            .add(
                &name,
                &text,
                &embedder,
                semantic.as_deref(),
                config.rag.max_chunk_chars,
                config.rag.chunk_overlap_chars,
            )
            .await
            .map_err(|error| error.to_string())?;
        // The local write succeeds independently. A remote outage must never
        // make an indexed document disappear from this profile's RAG store.
        if let Some(remote) = RemoteRag::from_config(&config, &profile)
            && let Err(error) = remote
                .upsert(
                    &name,
                    &text,
                    config.rag.max_chunk_chars,
                    config.rag.chunk_overlap_chars,
                )
                .await
        {
            eprintln!("· remote RAG sync skipped for {name}: {error}");
        }
        total += added;
        indexed.push((name, added));
    }

    if json {
        let value = serde_json::json!({
            "indexed": indexed.iter().map(|(n, c)| serde_json::json!({ "source": n, "chunks": c })).collect::<Vec<_>>(),
            "total_chunks": total,
        });
        println!("{value:#}");
    } else {
        for (name, added) in &indexed {
            println!("indexed {name} ({added} chunks)");
        }
        println!("{total} chunk(s) added.");
    }
    Ok(())
}

/// `rag search <query>` — the best passages for a query.
pub async fn search(query: String, top_k: Option<usize>, json: bool) -> Result<(), String> {
    let (index, config, profile) = active_index()?;
    let store = RagStore::open(&index).map_err(|error| error.to_string())?;
    let k = top_k.unwrap_or(config.rag.top_k).max(1);
    let semantic = semantic_embedder(&config);
    let hits = match RemoteRag::from_config(&config, profile) {
        Some(remote) => match remote.search(&query, k).await {
            Ok(hits) if !hits.is_empty() => hits,
            _ => {
                store
                    .search(&query, &HashingEmbedder, semantic.as_deref(), k)
                    .await
            }
        },
        None => {
            store
                .search(&query, &HashingEmbedder, semantic.as_deref(), k)
                .await
        }
    };

    if json {
        let value = serde_json::json!({
            "query": query,
            "hits": hits.iter().map(|hit| serde_json::json!({
                "source": hit.source, "score": hit.score, "text": hit.text,
            })).collect::<Vec<_>>(),
        });
        println!("{value:#}");
        return Ok(());
    }
    if hits.is_empty() {
        println!("No relevant passages found.");
        return Ok(());
    }
    for (rank, hit) in hits.iter().enumerate() {
        println!("[{}] {} (score {:.2})", rank + 1, hit.source, hit.score);
        println!("{}\n", hit.text);
    }
    Ok(())
}

/// `rag list` — the indexed sources and their chunk counts.

    let store = RagStore::open(&index).map_err(|error| error.to_string())?;
    let sources = store.sources();
    if json {
        let value = serde_json::json!({

            "total_chunks": store.len(),

        println!("{value:#}");
        return Ok(());

    }
    Ok(())
}

fn active_index_at(paths: &LightagentPaths) -> Result<ActiveIndex, OpError> {
    let config = ConfigStore::at(paths).load().map_err(op_internal)?;
    let store = ProfileStore::new(paths.root());

/// What an add did: each indexed source and its chunks, the files that
/// could not be read, and remote-index notes.
struct Added {
    indexed: Vec<(String, usize)>,
    total: usize,
    skipped: Vec<(String, String)>,
    notes: Vec<String>,
}

async fn op_add(active: &ActiveIndex, path: &Path, source: Option<&str>) -> Result<Added, OpError> {
    let mut store = RagStore::open(&active.index).map_err(op_internal)?;
    let embedder = HashingEmbedder;
    let semantic = semantic_embedder(&active.config);
    let config = &active.config;

    let mut targets = Vec::new();
    if path.is_dir() {
        let entries = std::fs::read_dir(path).map_err(op_internal)?;
        for entry in entries.flatten() {
            if entry.path().is_file() {
                targets.push(entry.path());
            }
        }
        targets.sort();
    } else {
        targets.push(path.to_path_buf());
    }

    let mut added = Added {
        indexed: Vec::new(),
        total: 0,
        skipped: Vec::new(),
        notes: Vec::new(),
    };
    for target in targets {
        let text = match std::fs::read_to_string(&target) {
            Ok(text) => text,
            Err(error) => {
                added
                    .skipped
                    .push((target.display().to_string(), error.to_string()));
                continue;
            }
        };
        let name = source.map(str::to_owned).unwrap_or_else(|| {
            target
                .file_name()
                .map(|n| n.to_string_lossy().into_owned())
                .unwrap_or_else(|| target.display().to_string())
        });
        let count = store
            .add(
                &name,
                &text,
                &embedder,
                semantic.as_deref(),
                config.rag.max_chunk_chars,
                config.rag.chunk_overlap_chars,
            )
            .await
            .map_err(op_internal)?;
        // The local write succeeds independently. A remote outage must never
        // make an indexed document disappear from this profile's RAG store.
        if let Some(remote) = RemoteRag::from_config(config, active.profile.as_str())
            && let Err(error) = remote
                .upsert(
                    &name,
                    &text,
                    config.rag.max_chunk_chars,
                    config.rag.chunk_overlap_chars,
                )
                .await
        {
            added
                .notes
                .push(format!("remote RAG sync skipped for {name}: {error}"));
        }
        added.total += count;
        added.indexed.push((name, count));
    }
    Ok(added)
}

async fn op_search(active: &ActiveIndex, query: &str, top_k: Option<usize>) -> Result<Vec<Hit>, OpError> {
    let store = RagStore::open(&active.index).map_err(op_internal)?;
    let k = top_k.unwrap_or(active.config.rag.top_k).max(1);
    let semantic = semantic_embedder(&active.config);
    Ok(match RemoteRag::from_config(&active.config, active.profile.as_str()) {
        Some(remote) => match remote.search(query, k).await {
            Ok(hits) if !hits.is_empty() => hits,
            _ => {
                store
                    .search(query, &HashingEmbedder, semantic.as_deref(), k)
                    .await
            }
        },
        None => {
            store
                .search(query, &HashingEmbedder, semantic.as_deref(), k)
                .await
        }
    })
}

/// One knowledge operation for the HTTP API (`RunFactory::rag`), on the
/// active profile's index, answered with the JSON `lightagent rag … --json`
/// prints.
pub(crate) async fn perform(paths: &LightagentPaths, request: RagRequest) -> Result<Value, OpError> {
    let active = active_index_at(paths)?;
    Ok(match request {
        RagRequest::List => {
            let (sources, total) = op_list(&active)?;
            list_value(&sources, total)
        }
        RagRequest::Add { path, source } => {
            added_value(&op_add(&active, Path::new(&path), source.as_deref()).await?)
        }
        RagRequest::Search { query, top_k } => {
            search_value(&query, &op_search(&active, &query, top_k).await?)
        }
        RagRequest::Clear => {
            op_clear(&active)?;
            json!({ "cleared": true })
        }
    })
}

//! runtime's API and `lightagent rag` act on the active profile's index

use std::io::{BufRead as _, BufReader, Read as _, Write as _};
use std::net::TcpStream;
use std::path::{Path, PathBuf};
use std::process::{Child, ChildStdin, Command, Output, Stdio};
use std::sync::mpsc;
use std::time::{Duration, SystemTime, UNIX_EPOCH};

use serde_json::{Value, json};

const KEY: &str = "92a3b4c5d6e7f8091a2b3c4d5e6f708192a3b4c5d6e7f8091a2b3c4d5e6f7081";
const KNOWLEDGE: &str = "/api/lightagent/v1/knowledge";

struct Runtime {
    child: Child,
    _lifeline: ChildStdin,
    addr: String,
}

impl Drop for Runtime {
    fn drop(&mut self) {
        let _ = self.child.kill();
        let _ = self.child.wait();
    }
}

fn scratch_home(label: &str) -> PathBuf {
    let nanos = SystemTime::now()
        .duration_since(UNIX_EPOCH)
        .map(|elapsed| elapsed.as_nanos())
        .unwrap_or(0);
    let home = std::env::temp_dir().join(format!(
        "lightagent-desktop-knowledge-{label}-{}-{nanos}",
        std::process::id()
    ));
    std::fs::create_dir_all(&home).expect("home");
    home
}

fn start(home: &Path) -> Runtime {
    let mut child = Command::new(env!("CARGO_BIN_EXE_lightagent"))
        .args(["serve", "--desktop", "--host", "127.0.0.1", "--port", "0"])
        .env("LIGHTAGENT_HOME", home)
        .stdin(Stdio::piped())
        .stdout(Stdio::piped())
        .stderr(Stdio::null())
        .spawn()
        .expect("start lightagent serve --desktop");
    let mut lifeline = child.stdin.take().expect("stdin");
    writeln!(lifeline, "{KEY}").expect("key");
    lifeline.flush().expect("flush");
    let stdout = child.stdout.take().expect("stdout");
    let (tx, rx) = mpsc::channel();
    std::thread::spawn(move || {
        let mut line = String::new();
        let _ = BufReader::new(stdout).read_line(&mut line);
        let _ = tx.send(line);
    });
    let line = rx.recv_timeout(Duration::from_secs(60)).expect("handshake");
    let handshake: Value = serde_json::from_str(&line).expect("handshake JSON");
    let addr = handshake["addr"].as_str().expect("addr").to_owned();
    Runtime {
        child,
        _lifeline: lifeline,
        addr,
    }
}

/// One authenticated request: the status and the raw body.
fn call(runtime: &Runtime, method: &str, path: &str, body: Option<&str>) -> (u16, String) {
    let mut stream = TcpStream::connect(&runtime.addr).expect("connect");
    stream
        .set_read_timeout(Some(Duration::from_secs(60)))
        .expect("timeout");
    let body = body.unwrap_or("");
    let head = format!(
        "{method} {path} HTTP/1.1\r\nHost: {}\r\nAuthorization: Bearer {KEY}\r\nContent-Type: application/json\r\nContent-Length: {}\r\nConnection: close\r\n\r\n{body}",
        runtime.addr,
        body.len()
    );
    stream.write_all(head.as_bytes()).expect("send");
    let mut response = String::new();
    let _ = stream.read_to_string(&mut response);
    let status = response
        .split_whitespace()
        .nth(1)
        .and_then(|code| code.parse().ok())
        .unwrap_or(0);
    let raw = response
        .split_once("\r\n\r\n")
        .map(|(_, body)| body.to_owned())
        .unwrap_or_default();
    (status, raw)
}

fn parsed(raw: &str) -> Value {

fn cli_json(home: &Path, args: &[&str]) -> Value {

fn sources(listing: &Value) -> Vec<String> {
    let mut names: Vec<String> = listing["sources"]
        .as_array()
        .expect("sources")
        .iter()
        .filter_map(|source| source["source"].as_str().map(str::to_owned))
        .collect();
    names.sort();
    names
}

fn the_cli_and_the_api_keep_one_index() {
    let home = scratch_home("one");
    cli_ok(&home, &["config", "set", "extensions.enabled", "true"]);
    cli_ok(&home, &["profile", "create", "work"]);
    cli_ok(&home, &["profile", "use", "work"]);
    let docs = home.join("docs");
    std::fs::create_dir_all(docs.join("folder")).expect("docs");
    std::fs::write(
        docs.join("rivers.md"),
        "Rivers carry water from mountains to the sea.\n",
    )
    .expect("doc");
    std::fs::write(
        docs.join("folder").join("a.md"),
        "Glaciers carve valleys over centuries.\n",
    )
    .expect("doc");
    std::fs::write(
        docs.join("folder").join("b.md"),
        "Deltas form where rivers slow down.\n",
    )
    .expect("doc");
    let runtime = start(&home);

    // A file indexed over the API, listed by the CLI; the listings are equal.
    let body = json!({ "path": docs.join("rivers.md") }).to_string();
    let (status, raw) = call(&runtime, "POST", KNOWLEDGE, Some(&body));
    assert_eq!(status, 200, "{raw}");
    let added = parsed(&raw);
    assert_eq!(added["indexed"][0]["source"], "rivers.md", "{added}");
    assert_eq!(added["skipped"], json!([]), "{added}");
    let (_, raw) = call(&runtime, "GET", KNOWLEDGE, None);
    assert_eq!(parsed(&raw), cli_json(&home, &["rag", "list"]));

    // A folder indexed by the CLI, listed over the API.
    cli_json(&home, &["rag", "add", arg(&docs.join("folder"))]);
    let (_, raw) = call(&runtime, "GET", KNOWLEDGE, None);
    assert_eq!(
        sources(&parsed(&raw)),
        vec!["a.md".to_owned(), "b.md".to_owned(), "rivers.md".to_owned()]
    );

    // Search: the API answers what the CLI prints.
    let (status, raw) = call(
        &runtime,
        "GET",
        &format!("{KNOWLEDGE}/search?q=rivers%20sea&top_k=2"),
        None,
    );
    assert_eq!(status, 200, "{raw}");
    assert_eq!(
        parsed(&raw),
        cli_json(&home, &["rag", "search", "rivers sea", "--top-k", "2"])
    );

    // A relative path is refused; a missing one is reported as skipped by both.
    let body = json!({ "path": "docs/rivers.md" }).to_string();
    assert_eq!(call(&runtime, "POST", KNOWLEDGE, Some(&body)).0, 400);
    let missing = docs.join("missing.md");
    let body = json!({ "path": missing }).to_string();
    let (status, raw) = call(&runtime, "POST", KNOWLEDGE, Some(&body));
    assert_eq!(status, 200, "{raw}");
    let by_api = parsed(&raw);
    let by_cli = cli_json(&home, &["rag", "add", arg(&missing)]);
    assert_eq!(by_api["skipped"].as_array().map(Vec::len), Some(1), "{by_api}");
    assert_eq!(by_api, by_cli);

    // Adds from both at once keep both documents (the index is written
    // under its lock).
    for round in 0..4 {
        let api_doc = docs.join(format!("api-{round}.md"));
        let cli_doc = docs.join(format!("cli-{round}.md"));
        std::fs::write(&api_doc, format!("API document {round} about lakes.\n")).expect("doc");
        std::fs::write(&cli_doc, format!("CLI document {round} about hills.\n")).expect("doc");
        let body = json!({ "path": api_doc }).to_string();
        std::thread::scope(|scope| {
            let api = scope.spawn(|| call(&runtime, "POST", KNOWLEDGE, Some(&body)).0);
            let cli = scope.spawn(|| cli_json(&home, &["rag", "add", arg(&cli_doc)]));
            assert_eq!(api.join().expect("api add"), 200);
            cli.join().expect("cli add");
        });
        let names = sources(&cli_json(&home, &["rag", "list"]));
        assert!(
            names.contains(&format!("api-{round}.md")) && names.contains(&format!("cli-{round}.md")),
            "round {round}: {names:?}"
        );
    }

    // Cleared over the API: the CLI lists nothing.
    let (status, raw) = call(&runtime, "DELETE", KNOWLEDGE, None);
    assert_eq!(status, 200, "{raw}");
    assert_eq!(cli_json(&home, &["rag", "list"])["sources"], json!([]));
}
