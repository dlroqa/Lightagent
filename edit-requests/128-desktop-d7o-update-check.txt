                .unwrap_or(CHECK_TIMEOUT);
            Ok((url, timeout))
        }
        _ => Ok((LATEST_RELEASE_API.to_owned(), CHECK_TIMEOUT)),
    }

/// timed-out, malformed or tagless answer is an error that says which.
pub(crate) async fn latest_from(
    client: &reqwest::Client,
    api: &str,
    timeout: Duration,
) -> Result<Release, String> {
    let timed_out = || {
        format!(
            "checking for updates timed out after {} ms",
            timeout.as_millis()
        )
    };
    let response = client
        .get(api)
        .header("Accept", "application/vnd.github+json")
        .timeout(timeout)
        .send()
        .await
        .map_err(|error| {
            if error.is_timeout() {
                timed_out()
            } else {
                format!("could not check for updates: {error}")
            }
        })?;
    let status = response.status();
    if !status.is_success() {
        let header = |name: &str| {
            response
                .headers()
                .get(name)
                .and_then(|value| value.to_str().ok())
                .map(str::to_owned)
        };
        if matches!(status.as_u16(), 403 | 429)
            && header("x-ratelimit-remaining").as_deref() == Some("0")
        {
            return Err(match header("x-ratelimit-reset") {
                Some(reset) => format!(
                    "GitHub's API rate limit is used up (HTTP {}); it resets at Unix time {reset}",
                    status.as_u16()
                ),
                None => format!(
                    "GitHub's API rate limit is used up (HTTP {})",
                    status.as_u16()
                ),
            });
        }
        return Err(format!(
            "GitHub answered HTTP {} while checking for updates",
            status.as_u16()
        ));
    }
    let release = response.json::<Release>().await.map_err(|error| {
        if error.is_timeout() {
            timed_out()
        } else {
            format!("could not read the latest release: {error}")
        }
    })?;
    if release.version().is_empty() {
        return Err("could not read the latest release: it names no version".to_owned());
    }
    Ok(release)
}

struct Runtime {

impl Drop for Runtime {
    fn drop(&mut self) {

/// What the fixture answers the next request with.
#[derive(Clone)]
enum Answer {
    Respond(String),
    /// Accept and say nothing, for longer than any check waits.
    Hold,
}

struct Fixture {
    url: String,
    requests: Arc<Mutex<Vec<String>>>,
    answer: Arc<Mutex<Answer>>,
}

fn fixture() -> Fixture {
    let listener = TcpListener::bind("127.0.0.1:0").expect("bind the fixture");
    let url = format!(
        "http://{}{RELEASE_PATH}",
        listener.local_addr().expect("fixture address")
    );
    let requests = Arc::new(Mutex::new(Vec::new()));
    let answer = Arc::new(Mutex::new(Answer::Hold));
    let (seen, current) = (Arc::clone(&requests), Arc::clone(&answer));
    std::thread::spawn(move || {
        for stream in listener.incoming().flatten() {
            let (seen, current) = (Arc::clone(&seen), Arc::clone(&current));
            std::thread::spawn(move || serve(stream, &seen, &current));
        }
    });
    Fixture {
        url,
        requests,
        answer,
    }
}

fn serve(mut stream: TcpStream, seen: &Mutex<Vec<String>>, current: &Mutex<Answer>) {
    let mut head = Vec::new();
    let mut byte = [0u8; 1];
    while !head.ends_with(b"\r\n\r\n") {
        match stream.read(&mut byte) {
            Ok(1) => head.push(byte[0]),
            _ => return,
        }
    }
    let line = String::from_utf8_lossy(&head)
        .lines()
        .next()
        .unwrap_or_default()
        .to_owned();
    seen.lock().expect("requests").push(line);
    let answer = current.lock().expect("answer").clone();
    match answer {
        Answer::Respond(raw) => {
            let _ = stream.write_all(raw.as_bytes());
        }
        Answer::Hold => std::thread::sleep(Duration::from_secs(5)),
    }
}

/// An HTTP response; each of `headers` ends with CRLF.

}

fn start(home: &Path, env: &[(&str, &str)]) -> Runtime {
    let mut command = Command::new(env!("CARGO_BIN_EXE_lightagent"));
    command
        .args(["serve", "--desktop", "--host", "127.0.0.1", "--port", "0"])
        .stdin(Stdio::piped())
        .stdout(Stdio::piped())
        .stderr(Stdio::null());
    quiet(&mut command, home, env);
    let mut child = command.spawn().expect("start lightagent serve --desktop");
    let mut lifeline = child.stdin.take().expect("stdin");
    writeln!(lifeline, "{KEY}").expect("key");
    lifeline.flush().expect("flush");
    let stdout = child.stdout.take().expect("stdout");
    let (tx, rx) = mpsc::channel();
    std::thread::spawn(move || {
        let mut reader = BufReader::new(stdout);
        let mut line = String::new();
        let _ = reader.read_line(&mut line);
        let _ = tx.send(line);
        let mut rest = String::new();
        while reader.read_line(&mut rest).is_ok_and(|read| read > 0) {
            rest.clear();
        }
    });
    let line = rx.recv_timeout(Duration::from_secs(60)).expect("handshake");
    let handshake: Value = serde_json::from_str(&line).expect("handshake JSON");
    let addr = handshake["addr"].as_str().expect("addr").to_owned();
    Runtime {
        child,
        _lifeline: lifeline,
        addr,
    }
}

fn call(runtime: &Runtime, method: &str, path: &str) -> (u16, String) {
    let mut stream = TcpStream::connect(&runtime.addr).expect("connect");
    stream
        .set_read_timeout(Some(Duration::from_secs(60)))
        .expect("timeout");
    let head = format!(
        "{method} {path} HTTP/1.1\r\nHost: {}\r\nAuthorization: Bearer {KEY}\r\nContent-Length: 0\r\nConnection: close\r\n\r\n",
        runtime.addr
    );
    stream.write_all(head.as_bytes()).expect("send");
    let mut response = String::new();
    let _ = stream.read_to_string(&mut response);
    let status = response
        .split_whitespace()
        .nth(1)
        .and_then(|code| code.parse().ok())
        .unwrap_or(0);
    let raw = response
        .split_once("\r\n\r\n")
        .map(|(_, body)| body.to_owned())
        .unwrap_or_default();
    (status, raw)
}

// The Diagnostics screen's Updates card against the real harness (D7-12):
// render-panel.sh serves a controlled release API on loopback naming
// v99.0.0, so Check for updates reports that release as available. It is
// read-only: the API reports what `update --check --json` prints.
// crates/lightagent/tests/desktop_update_check.rs is the fixture evidence
// for every answer; .github/workflows/update-check-live.yml is the live one.

export async function checkUpdate(browser, base, outDir) {
  const failures = [];
  const page = await browser.newPage({ viewport: { width: 1280, height: 1400 } });
  const errors = [];
  page.on("pageerror", (error) => errors.push(error.message));
  try {
    await page.goto(`${base}/#/diagnostics`);
    const screen = page.getByTestId("diagnostics-screen");
    await screen.waitFor({ timeout: 15000 });
    await screen.getByText("Checking installs nothing", { exact: false }).waitFor({ timeout: 10000 });
    await screen.getByRole("button", { name: "Check for updates" }).click();
    await page.getByTestId("diagnostics-update").getByText("Lightagent 99.0.0 is available", { exact: false })
      .waitFor({ timeout: 15000 });
    const report = await (await fetch(`${base}/api/lightagent/v1/update/check`)).json();
    if (report.latest !== "99.0.0" || report.update_available !== true) {
      failures.push(`update: the API reports ${JSON.stringify(report)}`);
    }
    await page.screenshot({ path: `${outDir}/diagnostics-update.png`, fullPage: true });
    if (errors.length) failures.push(`update: page errors: ${errors.join("; ")}`);
  } catch (err) {
    failures.push(`update: ${err instanceof Error ? err.message.split("\n")[0] : String(err)}`);
    await page.screenshot({ path: `${outDir}/update-failed.png`, fullPage: true }).catch(() => {});
  } finally {
    await page.close();
  }
  return failures;
}
