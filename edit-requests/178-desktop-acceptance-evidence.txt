"""A stdio MCP server for the lifecycle checks that does not leave on its own.

  python stubborn_mcp.py <marker>

It answers `initialize`, `tools/list` and `ping`, and keeps running after its
standard input ends: only being stopped ends it. A check that still finds it
after its runtime has gone has found a process the runtime left behind, not
one that was merely slow to notice. The marker argument is not used except to
make the process findable by its command line.
"""

import json
import sys
import time


def send(message):
    sys.stdout.buffer.write(json.dumps(message).encode("utf-8") + b"\n")
    sys.stdout.buffer.flush()


def answer(message):
    method = message.get("method")
    if method == "initialize":
        version = (message.get("params") or {}).get("protocolVersion") or "2025-06-18"
        return {"protocolVersion": version, "capabilities": {"tools": {}},
                "serverInfo": {"name": "stubborn", "version": "1.0.0"}}
    if method == "tools/list":
        return {"tools": [{"name": "wait", "description": "Does nothing.",
                           "inputSchema": {"type": "object", "properties": {}}}]}
    if method == "ping":
        return {}
    return None


def main():
    for line in sys.stdin.buffer:
        try:
            message = json.loads(line)
        except ValueError:
            continue
        if not isinstance(message, dict) or "id" not in message or "method" not in message:
            continue
        result = answer(message)
        if result is None:
            send({"jsonrpc": "2.0", "id": message["id"], "error": {"code": -32601, "message": "not supported"}})
        else:
            send({"jsonrpc": "2.0", "id": message["id"], "result": result})
    # The end of input: stay, on purpose.
    while True:
        time.sleep(60)


if __name__ == "__main__":
    main()

"""The user data a Desktop install, upgrade or uninstall must leave alone (D9 acceptance).

  python user_data.py seed <cli> <baseline.json>
  python user_data.py verify <cli> <baseline.json> <key> <description> <result.json>

`seed` creates one of each kind of user data with the real CLI and runtime
(`<cli>` is the `lightagent` binary the job built), in the Lightagent home
the app uses (LIGHTAGENT_HOME, else ~/.lightagent): a second profile, a
setting, a saved conversation with an attachment, a memory, an indexed
knowledge source, a provider API key in the secrets store and a sealed
remote credential in the vault (its key file outside the home, as a person
keeps it). It records every file of the home with its SHA-256.

`verify` checks that every recorded file is still there, unchanged, and
reads each kind of data back through the CLI and a keyed `lightagent serve`,
which also shows that both protected credentials still open. It writes
`<result.json>` (no secret, no absolute path) and exits 1 on any difference.

No Lightagent Desktop release has been published, so there is no earlier
version to upgrade from. The installers' checks therefore simulate one: the
home is seeded before the data checks, and the same installer is then run
again over its own installation. That is reported as a simulated upgrade,
never as an upgrade from a release.
"""

def lightagent_home():
    return os.environ.get("LIGHTAGENT_HOME") or os.path.join(os.path.expanduser("~"), ".lightagent")


def manifest(root):
    """Every file of the home with its SHA-256. Lock files are left out: they
    are created and released as the app works, and hold no data."""
    files = {}
    for base, _dirs, names in os.walk(root):
        for name in names:
            if name.endswith(".lock"):
                continue
            path = os.path.join(base, name)
            rel = os.path.relpath(path, root).replace(os.sep, "/")
            try:
                with open(path, "rb") as handle:
                    files[rel] = hashlib.sha256(handle.read()).hexdigest()
            except OSError as error:
                files[rel] = f"unreadable ({type(error).__name__})"
    return files


def free_port():
    with socket.socket() as probe:
        probe.bind(("127.0.0.1", 0))
        return probe.getsockname()[1]


class Server:
    """A keyed `lightagent serve` on loopback, for what the CLI does not show."""

    def __init__(self, cli):
        self.key = "la-data-" + secrets.token_urlsafe(24)
        self.port = free_port()
        env = dict(os.environ, LIGHTAGENT_API_KEY=self.key)
        self.process = subprocess.Popen(
            [cli, "serve", "--host", "127.0.0.1", "--port", str(self.port), "--key-env", "LIGHTAGENT_API_KEY"],
            env=env, stdin=subprocess.DEVNULL, stdout=subprocess.DEVNULL, stderr=subprocess.DEVNULL)
        if not harness.wait_until(self.listening, timeout=60) or self.process.poll() is not None:
            self.stop()
            raise RuntimeError("lightagent serve did not start")

    def listening(self):
        if self.process.poll() is not None:
            return True
        try:
            with socket.create_connection(("127.0.0.1", self.port), timeout=1):
                return True
        except OSError:
            return False

    def request(self, method, path, body=None, headers=None):
        data = body if isinstance(body, bytes) or body is None else json.dumps(body).encode()
        request = urllib.request.Request(f"http://127.0.0.1:{self.port}{API}{path}", data=data, method=method)
        request.add_header("Authorization", f"Bearer {self.key}")
        if body is not None and not isinstance(body, bytes):
            request.add_header("Content-Type", "application/json")
        for name, value in (headers or {}).items():
            request.add_header(name, value)
        try:
            with urllib.request.urlopen(request, timeout=60) as response:
                raw = response.read()
                return response.status, json.loads(raw) if raw else None
        except urllib.error.HTTPError as error:
            return error.code, error.read().decode("utf-8", "replace")[:300]

    def stop(self):
        if self.process.poll() is None:
            self.process.terminate()
            try:
                self.process.wait(timeout=15)
            except subprocess.TimeoutExpired:
                self.process.kill()
                self.process.wait()


def ids_in(value):
    if isinstance(value, dict):
        for key in ("sessions", "items", "data"):
            if isinstance(value.get(key), list):
                return ids_in(value[key])
        return []
    if isinstance(value, list):
        return [item["id"] for item in value if isinstance(item, dict) and "id" in item]
    return []


def answered(body):
    messages = (body or {}).get("messages", []) if isinstance(body, dict) else []
    return any(ANSWER in harness.content_of(m) for m in messages if isinstance(m, dict) and m.get("role") == "assistant")


def find_session(server):
    status, listing = server.request("GET", "/sessions")
    for session in ids_in(listing):
        found, body = server.request("GET", f"/sessions/{session}")
        if found == 200 and answered(body):
            return session
    raise RuntimeError(f"the seeded conversation is not listed (sessions.list {status})")


def active_profile(cli):
    listing = json.loads(harness.run_cli(cli, "--json", "profile", "list"))
    if isinstance(listing, dict):
        if isinstance(listing.get("active"), str):
            return listing["active"]
        listing = listing.get("profiles", [])
    for row in listing if isinstance(listing, list) else []:
        if isinstance(row, dict) and row.get("active"):
            return row.get("id") or row.get("name")
    return "default"

def verify(cli, baseline_path, key, description, result_path):
    with open(baseline_path, encoding="utf-8") as handle:
        baseline = json.load(handle)
    root = baseline["home"]
    os.environ["LIGHTAGENT_CREDENTIAL_KEY_FILE"] = baseline["credential_key_file"]
    now, before = manifest(root), baseline["files"]
    missing = sorted(set(before) - set(now))
    changed = sorted(name for name in before if name in now and now[name] != before[name])
    added = sorted(set(now) - set(before))
    checks = {}

    def check(name, probe):
        try:
            ok, seen = probe()
        except Exception as error:  # a failed read is a failed check, with why
            ok, seen = False, repr(error)[:300]
        checks[name] = {"ok": bool(ok), "seen": seen}

    def listed(args, wanted):
        return wanted in harness.run_cli(cli, "--json", *args), None

    def attachment():
        with open(os.path.join(root, baseline["attachment"]), "rb") as handle:
            return hashlib.sha256(handle.read()).hexdigest() == baseline["attachment_sha256"], None

    check("the second profile is listed", lambda: listed(["profile", "list"], PROFILE))
    check("the setting is kept", lambda: (harness.run_cli(cli, "config", "get", "web.enabled").strip() == "true", None))
    check("the memory is listed", lambda: listed(["memory", "list"], MEMORY))
    check("the knowledge source is listed", lambda: listed(["rag", "list"], SOURCE))
    check("the vault credential is listed", lambda: listed(["remote-credential", "list"], SERVICE))
    check("the attachment is byte for byte the same", attachment)
    try:
        server = Server(cli)
    except RuntimeError as error:
        server = None
        checks["the runtime starts on the home"] = {"ok": False, "seen": str(error)}
    if server:
        try:
            def conversation():
                status, body = server.request("GET", f"/sessions/{baseline['session']}")
                return status == 200 and answered(body), status

            def provider_key():
                status, body = server.request("GET", "/provider/credential")
                state = body.get("state") if isinstance(body, dict) else None
                return status == 200 and state == "ready", state

            def vault_opens():
                status, body = server.request("GET", "/remote-credentials/vault")
                unopenable = body.get("unopenable_with_configured_key") if isinstance(body, dict) else None
                return status == 200 and unopenable == 0, {"status": status, "unopenable": unopenable}

            check("the conversation and its answer are kept", conversation)
            check("the provider key is still usable", provider_key)
            check("the vault still opens with its key", vault_opens)
        finally:
            server.stop()
    ok = not missing and not changed and all(c["ok"] for c in checks.values())
    result = {"key": key, "description": description, "ok": ok, "files": len(before),
              "missing": missing[:20], "changed": changed[:20], "added": added[:20], "checks": checks}
    with open(result_path, "w", encoding="utf-8") as handle:
        json.dump(result, handle, indent=1)
    print(json.dumps({"key": key, "ok": ok, "missing": len(missing), "changed": len(changed),
                      "failed": [name for name, c in checks.items() if not c["ok"]]}))
    return 0 if ok else 1


def main():
    if sys.argv[1] == "seed":
        return seed(sys.argv[2], sys.argv[3])
    return verify(sys.argv[2], sys.argv[3], sys.argv[4], sys.argv[5], sys.argv[6])


if __name__ == "__main__":
    sys.exit(main())

"""The shipped AppImage's lifecycle, driven as a person drives it (D9 acceptance).

  python appimage_lifecycle.py <out-dir> <app.AppImage> <cli>

Run as the smoke account inside its own X display and session bus, with its
own Lightagent home, against the packaged AppImage (no test hooks). The
AppImage's runtime starts the desktop host as its child, so every check
follows the actual processes: the host, the runtime the host started, and
the MCP server a run starts; the AppImage's wrapper is checked last.

  normal_quit     the window is asked to close (WM_DELETE_WINDOW, what a
                  window manager's close button sends) with nothing working:
                  the host, its runtime and the wrapper exit
  quit_with_work  a run is working (its MCP server running, its model call
                  held): the close request leaves everything running and the
                  app asks; Tab then Enter choose "Stop runs and quit", and the
                  host, its runtime and the MCP server end
  host_crash      the host is killed (SIGKILL) while a run works: its runtime
                  follows it, and the run's MCP server does not outlive them

The message is typed into the composer with xdotool, as OS key presses. The
MCP server (stubborn_mcp.py) ignores the end of its input on purpose, so it
ends only if the runtime ends it. Writes appimage-lifecycle.json and exits 1
when a check fails.
"""

import ctypes
import ctypes.util
import json
import os
import signal
import subprocess
import sys
import time

HERE = os.path.dirname(os.path.abspath(__file__))
sys.path.insert(0, HERE)
import harness  # noqa: E402
import psutil  # noqa: E402

READY = ["RUNTIME_READY", "PAGE_LOADED", "UI_READY", "API_OK sessions.list"]
FIXTURE = os.path.join(HERE, "stubborn_mcp.py")


def sh(args, timeout=60):
    try:
        out = subprocess.run(args, capture_output=True, text=True, timeout=timeout)
        return out.returncode, (out.stdout or "") + (out.stderr or "")
    except (OSError, subprocess.SubprocessError) as error:
        return -1, repr(error)


def running(pid):
    try:
        return pid is not None and psutil.Process(pid).status() != psutil.STATUS_ZOMBIE
    except psutil.Error:
        return False


def binary_of(process):
    try:
        return os.path.basename(process.exe())
    except psutil.Error:
        return ""


def below(pid, binary):
    """The first process under `pid` running `binary` (by its executable, not
    its name: the AppImage starts the host through a link)."""
    try:
        for process in psutil.Process(pid).children(recursive=True):
            if binary_of(process) == binary:
                return process.pid
    except psutil.Error:
        pass
    return None


def marked(marker):
    found = []
    for process in psutil.process_iter(["pid", "cmdline"]):
        try:
            if process.pid != os.getpid() and marker in " ".join(process.info["cmdline"] or []) and running(process.pid):
                found.append(process.pid)
        except psutil.Error:
            pass
    return found


def ended(pids, seconds):
    return bool(harness.wait_until(lambda: not [pid for pid in pids if running(pid)], timeout=seconds, every=0.25))


def app_windows():
    _, text = sh(["xdotool", "search", "--name", "^Lightagent$"])
    return [int(word) for word in text.split() if word.isdigit()]


class ClientMessage(ctypes.Structure):
    _fields_ = [("type", ctypes.c_int), ("serial", ctypes.c_ulong), ("send_event", ctypes.c_int),
                ("display", ctypes.c_void_p), ("window", ctypes.c_ulong), ("message_type", ctypes.c_ulong),
                ("format", ctypes.c_int), ("data", ctypes.c_long * 5)]


class XEvent(ctypes.Union):
    _fields_ = [("client", ClientMessage), ("pad", ctypes.c_long * 24)]


def close_request(window):
    """WM_DELETE_WINDOW to `window`, as a window manager's close button sends it."""
    x11 = ctypes.CDLL(ctypes.util.find_library("X11") or "libX11.so.6")
    x11.XOpenDisplay.restype = ctypes.c_void_p
    x11.XOpenDisplay.argtypes = [ctypes.c_char_p]
    x11.XInternAtom.restype = ctypes.c_ulong
    x11.XInternAtom.argtypes = [ctypes.c_void_p, ctypes.c_char_p, ctypes.c_int]
    x11.XSendEvent.argtypes = [ctypes.c_void_p, ctypes.c_ulong, ctypes.c_int, ctypes.c_long, ctypes.c_void_p]
    x11.XFlush.argtypes = [ctypes.c_void_p]
    x11.XCloseDisplay.argtypes = [ctypes.c_void_p]
    display = x11.XOpenDisplay(None)
    if not display:
        return False
    event = XEvent()
    event.client.type = 33  # ClientMessage
    event.client.send_event = 1
    event.client.display = display
    event.client.window = window
    event.client.message_type = x11.XInternAtom(display, b"WM_PROTOCOLS", 0)
    event.client.format = 32
    event.client.data[0] = x11.XInternAtom(display, b"WM_DELETE_WINDOW", 0)
    event.client.data[1] = 0  # CurrentTime
    sent = x11.XSendEvent(display, window, 0, 0, ctypes.byref(event))
    x11.XFlush(display)
    x11.XCloseDisplay(display)
    return bool(sent)


def close_app():
    return [close_request(window) for window in app_windows()]


def type_message(text):
    for window in app_windows():
        sh(["xdotool", "windowfocus", str(window)])
    time.sleep(0.5)
    sh(["xdotool", "type", "--delay", "30", text], timeout=120)
    time.sleep(0.3)
    sh(["xdotool", "key", "Return"])


def screenshot(out, name):
    sh(["import", "-window", "root", os.path.join(out, f"appimage-{name}.png")], timeout=30)


class App:
    """The AppImage started as a person starts it, and its actual processes."""

    def __init__(self, out, image, case):
        self.report = os.path.join(out, f"report-{case}.txt")
        if os.path.exists(self.report):
            os.remove(self.report)
        self.log = open(os.path.join(out, f"app-{case}.log"), "w", encoding="utf-8")
        self.wrapper = subprocess.Popen([image], env=dict(os.environ, LIGHTAGENT_DESKTOP_E2E_REPORT=self.report),
                                        stdin=subprocess.DEVNULL, stdout=self.log, stderr=subprocess.STDOUT)

        def ready():
            lines = (harness.read(self.report) or "").splitlines()
            return self.wrapper.poll() is not None or all(any(line.startswith(tag) for line in lines) for tag in READY)

        self.ready = bool(harness.wait_until(ready, timeout=150)) and self.wrapper.poll() is None
        try:
            own = binary_of(psutil.Process(self.wrapper.pid))
        except psutil.Error:
            own = ""
        self.host = self.wrapper.pid if own == "lightagent-desktop" else below(self.wrapper.pid, "lightagent-desktop")
        self.runtime = below(self.host, "lightagent-sidecar") if self.host else None

    def facts(self):
        return {"ready": self.ready, "host_found": bool(self.host), "host_is_the_wrapper": self.host == self.wrapper.pid,
                "runtime_found": bool(self.runtime)}

    def end(self):
        for pid in (self.runtime, self.host):
            if running(pid):
                try:
                    os.kill(pid, signal.SIGKILL)
                except OSError:
                    pass
        if self.wrapper.poll() is None:
            self.wrapper.kill()
        try:
            self.wrapper.wait(timeout=30)
        except subprocess.TimeoutExpired:
            pass
        self.log.close()


def wrapper_exits(app):
    return bool(harness.wait_until(lambda: app.wrapper.poll() is not None, timeout=30))


def normal_quit(app, provider, marker, check, out):
    sent = close_app()
    check.expect("the window takes a close request (WM_DELETE_WINDOW)", any(sent), sent)
    check.expect("with nothing working the host exits", ended([app.host], 30))
    check.expect("its runtime exits with it", ended([app.runtime], 15))
    check.expect("the AppImage's wrapper exits after its host", wrapper_exits(app), app.wrapper.poll())


def working(app, provider, scenario, marker, check, out):
    type_message(f"[e2e:{scenario}] Hold on, please.")
    reached = provider.wait_for("gate", scenario, timeout=90, gate=scenario)
    mcp = harness.wait_until(lambda: marked(marker), timeout=30, every=0.5) or []
    screenshot(out, f"{scenario}-working")
    check.expect("a run works: its model call is held and its MCP server runs", reached and bool(mcp),
                 {"model_call_held": reached, "mcp_servers": len(mcp)})
    return mcp


def quit_with_work(app, provider, marker, check, out):
    mcp = working(app, provider, "appimage-quit", marker, check, out)
    sent = close_app()
    time.sleep(5)
    kept = running(app.host) and running(app.runtime) and bool(mcp) and all(running(pid) for pid in mcp)
    screenshot(out, "quit-asked")
    check.expect("a close request with a run working keeps the app, its runtime and the run (it asks)",

"""Install, launch and uninstall the desktop installers on a GitHub-hosted runner (D9).

Usage: python install_smoke.py <out-dir> <label> <bundle-dir>

Each installer the runner built is used as a person would use it:

  linux    the .deb installed through apt, launched by a new user account
           (la-smoke, with its own home), and removed; the AppImage run by
           that user and deleted
  macos    the .dmg mounted, its app copied to /Applications, launched with
           a fresh home, and removed (a GUI session for a second account is
           not available on the runner)
  windows  the NSIS setup installed silently for the current user and the
           MSI for the machine, each launched and uninstalled silently (the
           app's data folder is the runner account's; Windows resolves it
           through the known-folder API, not the environment)

A launch is run.py's `ready` check against the installed binary: the
bundled runtime starts, the UI loads and makes its first API call, no launch
key shows in any process's arguments or environment, and the runtime leaves
with the app. Also recorded: each installer's SHA-256, the signature state
(unsigned is expected), and the minimum OS the binaries declare, which is a
declaration only: no older system is run.

User data (user_data.py): the account's own Lightagent home is seeded with
one of each kind of data (a profile, a setting, a conversation with an
attachment, a memory, a knowledge source, a provider key and a sealed vault
credential). The installed app is launched on it, the same installer is run
again over its installation and the app launched again, and the app is
uninstalled; after each step every seeded file must be unchanged and each
kind of data must read back, the credentials included. No Desktop release
exists, so "after the reinstall" is a simulated upgrade, reported as such.

Notices: THIRD_PARTY_NOTICES.txt and LICENSE.txt must be in each installed
app, identical to the ones the job generated and bundled.

On Linux the shipped AppImage's lifecycle is driven as well
(appimage_lifecycle.py): its actual host process is closed and killed, not
only the AppImage's wrapper.

Writes installers-<label>.json and SHA256SUMS-<label>.txt, prints a Markdown
summary, and exits non-zero when a step fails.
"""

def sh(args, timeout=600, **kw):
    try:
        out = subprocess.run(args, capture_output=True, text=True, timeout=timeout, **kw)
        return out.returncode, (out.stdout or "") + (out.stderr or "")
    except (OSError, subprocess.SubprocessError) as error:
        return -1, repr(error)


def sha256(path):
    digest = hashlib.sha256()
    with open(path, "rb") as handle:
        for block in iter(lambda: handle.read(1 << 20), b""):
            digest.update(block)
    return digest.hexdigest()


def ready_report(directory):
    try:
        with open(os.path.join(directory, "e2e-ready.json"), encoding="utf-8") as handle:
            data = json.load(handle)
    except (OSError, ValueError):
        return None
    return {key: data.get(key) for key in ("milestones_s", "missing", "key_leaks", "runtime_survived_host_termination")}


def judged(code, report, tail, **extra):
    ok = code == 0 and report is not None and not report["missing"] and not report["key_leaks"]
    return {"ok": ok, "exit": code, "report": report, "tail": None if ok else tail[-900:], **extra}


def launch(out, name, app, env=None):
    """run.py ready against an installed app, as this account, with a fresh
    Lightagent home (and, when given, other environment)."""
    target = os.path.join(out, f"launch-{name}")
    os.makedirs(target, exist_ok=True)
    home = tempfile.mkdtemp(prefix="la-install-home-")
    run_env = dict(os.environ, LIGHTAGENT_HOME=os.path.join(home, "lightagent"), E2E_REQUIRE_NO_ORPHAN="1")
    run_env.update(env or {})
    code, tail = sh([sys.executable, RUN, target, "ready", "--", app], timeout=300, env=run_env)
    return judged(code, ready_report(target), tail, account="runner")


def own_home_env():

def user_data(action, key, description, baseline, prefix=(), script=DATA, cli=CLI, env=None):
    """user_data.py seed or verify; the step's result, with `ok`."""
    result_path = f"{baseline}.{key}.json"
    args = [*prefix, sys.executable, script, action, cli, baseline]
    if action == "verify":
        args += [key, description, result_path]
    code, tail = sh(args, timeout=900, env=env)
    if action == "seed":
        result = {"ok": code == 0, "description": description}
    else:
        try:
            with open(result_path, encoding="utf-8") as handle:
                result = json.load(handle)
        except (OSError, ValueError):
            result = {"description": description}
        result["ok"] = bool(result.get("ok")) and code == 0
    if not result["ok"]:
        result["tail"] = tail[-900:]
    return result


def local_data(action, key, description, baseline):
    return user_data(action, key, description, baseline, env=own_home_env())


def with_steps(result, **steps):
    result["steps"] = steps
    result["ok"] = bool(result.get("ok")) and all(steps.values())
    return result


def data_cycle(out, entry, name, app, reinstall, how, baseline=None):
    """Seed the account's own home (unless `baseline` names an earlier seed),
    launch the installed app on it, run the installer again over its
    installation and launch again, checking the data after each. Returns the
    baseline once seeded."""
    if baseline is None:
        if os.path.exists(os.path.join(os.path.expanduser("~"), ".lightagent")):
            entry["data_seed"] = {"ok": False, "seen": "this account already has a Lightagent home; it is left untouched"}
            return None
        baseline = os.path.join(tempfile.mkdtemp(prefix="la-user-data-"), "baseline.json")
        entry["data_seed"] = local_data("seed", "seed", "seeded in the account's own home (simulated earlier data)", baseline)
        if not entry["data_seed"]["ok"]:
            return None
    launched = local_launch(out, f"{name}-existing-data", app)
    entry["data_after_launch"] = with_steps(local_data("verify", f"{name}-after-launch", "after the installed app ran on it", baseline),
                                            launch=launched["ok"])
    code, tail = reinstall()
    relaunched = local_launch(out, f"{name}-reinstalled", app) if code == 0 else {"ok": False}
    entry["data_after_reinstall"] = with_steps(local_data("verify", f"{name}-after-reinstall", how, baseline),
                                               reinstall=code == 0, launch=relaunched["ok"])
    if code != 0:
        entry["data_after_reinstall"]["reinstall_tail"] = tail[-600:]
    return baseline


def notices_check(paths):
    """The packaged notices: in the installed app, and identical to the generated ones."""
    beside = {os.path.dirname(p) for p in paths if os.path.basename(p) == NOTICE_FILES[0]}
    seen = {}
    for name in NOTICE_FILES:
        candidates = [p for p in paths if os.path.basename(p) == name and os.path.isfile(p)]
        found = next((p for p in candidates if os.path.dirname(p) in beside), candidates[0] if candidates else None)
        generated = os.path.join(NOTICES, name)
        seen[name] = {"packaged": found is not None,
                      "same_as_generated": bool(found) and os.path.isfile(generated) and sha256(found) == sha256(generated)}
    return {"ok": all(v["packaged"] and v["same_as_generated"] for v in seen.values()), **seen}

def glibc_needed(paths):
    found = set()
    for path in paths:
        _, text = sh(["objdump", "-T", path])
        found.update(re.findall(r"GLIBC_([0-9]+(?:\.[0-9]+)+)", text))
    return max(found, key=lambda v: tuple(int(x) for x in v.split(".")), default=None)

def plist_value(app, key):
    code, text = sh(["/usr/libexec/PlistBuddy", "-c", f"Print :{key}", os.path.join(app, "Contents", "Info.plist")])
    return text.strip() if code == 0 else None


def macos(out, bundle, entries):
    baseline = None
    for dmg in sorted(glob.glob(os.path.join(bundle, "dmg", "*.dmg"))):
        entry = {"installer": os.path.basename(dmg), "sha256": sha256(dmg)}
        mount = tempfile.mkdtemp(prefix="la-dmg-")
        for attempt in (1, 2, 3):
            code, tail = sh(["hdiutil", "attach", "-nobrowse", "-readonly", "-mountpoint", mount, dmg], timeout=300)
            if code == 0 or "Resource busy" not in tail:
                break
            entry.setdefault("attach_retries", []).append(attempt)
            time.sleep(15)
        apps = glob.glob(os.path.join(mount, "*.app"))
        target = os.path.join("/Applications", os.path.basename(apps[0])) if apps else None
        if target and os.path.exists(target):
            entry["install"] = {"ok": False, "tail": f"{target} already exists; it is not overwritten"}
            sh(["hdiutil", "detach", mount])
            entries.append(entry)
            continue
        copied = sh(["ditto", apps[0], target]) if target else (-1, "no app in the image")
        binary = os.path.join(target or "", "Contents", "MacOS", "lightagent-desktop")
        entry["install"] = {"ok": code == 0 and copied[0] == 0 and os.path.exists(binary),
                            "how": "mounted, the app copied to /Applications", "tail": None if copied[0] == 0 else copied[1][-600:]}
        if entry["install"]["ok"]:
            entry["signature"] = sh(["codesign", "-dv", "--verbose=2", target])[1].strip()[-500:]
            entry["gatekeeper"] = sh(["spctl", "--assess", "--type", "execute", "-v", target])[1].strip()[-300:]
            entry["min_os"] = {"LSMinimumSystemVersion": plist_value(target, "LSMinimumSystemVersion"),
                               "declared_by": "Info.plist and each binary's minos (declared, not run on older systems)"}
            fresh = tempfile.mkdtemp(prefix="la-macos-home-")
            entry["launch"] = launch(out, "dmg", binary, env={"HOME": fresh})
            entry["notices"] = notices_check([os.path.join(target, "Contents", "Resources", name) for name in NOTICE_FILES])

            def replace_app():
                shutil.rmtree(target, ignore_errors=True)
                return sh(["ditto", apps[0], target])

            baseline = data_cycle(out, entry, "dmg", binary, replace_app,
                                  "simulated upgrade: the app replaced from the image (as Finder's Replace does), then a launch", baseline)
            shutil.rmtree(target, ignore_errors=True)
        sh(["hdiutil", "detach", mount])
        entry["uninstall"] = {"ok": bool(target) and not os.path.exists(target), "how": "the app removed from /Applications"}
        if baseline:
            entry["data_after_uninstall"] = local_data("verify", "dmg-after-uninstall", "after the app was removed from /Applications", baseline)
        entries.append(entry)

def pe_versions(path):
    with open(path, "rb") as handle:
        data = handle.read(4096)
    header = int.from_bytes(data[0x3C:0x40], "little")
    optional = header + 24

    def word(offset):
        return int.from_bytes(data[optional + offset:optional + offset + 2], "little")

    return {"os_version": f"{word(40)}.{word(42)}", "subsystem_version": f"{word(48)}.{word(50)}"}

def windows_input(action):
    """Run one WINDOWS_INPUT action; its key=value answer, or why it failed."""
    try:
        out = subprocess.run(["powershell", "-NoProfile", "-Command", WINDOWS_INPUT.replace("ACTION", action)],
                             capture_output=True, text=True, timeout=120)
    except (OSError, subprocess.SubprocessError) as error:
        return {"error": repr(error)[:200]}
    lines = [line for line in (out.stdout or "").splitlines() if "=" in line]
    if out.returncode != 0 or not lines:
        return {"error": ((out.stderr or "") + (out.stdout or "")).strip()[-300:]}
    return dict(part.split("=", 1) for part in lines[-1].split("|") if "=" in part)

def marked(marker):
    """Live processes whose command line carries `marker`."""
    import psutil

    found = []
    for process in psutil.process_iter(["pid", "cmdline"]):
        try:
            carries = marker in " ".join(process.info["cmdline"] or [])
            if process.pid != os.getpid() and carries and process.status() != psutil.STATUS_ZOMBIE:
                found.append(process.pid)
        except psutil.Error:
            pass
    return found


def live(pids):
    import psutil

    left = []
    for pid in pids:
        try:
            if psutil.Process(pid).status() != psutil.STATUS_ZOMBIE:
                left.append(pid)
        except psutil.Error:
            pass
    return left


def end_all(pids):
    import psutil

    for pid in pids:
        try:
            psutil.Process(pid).kill()
        except psutil.Error:
            pass

def sha256_file(path):
    with open(path, "rb") as handle:
        return hashlib.sha256(handle.read()).hexdigest()


def is_proc_macro(package):
    return any("proc-macro" in target.get("kind", []) for target in package.get("targets", []))


def runtime_linked(metadata, root_name):
    """The ids of the packages in the binary itself: `linked`, without
    procedural macros and what only they depend on (they run at build time)."""
    packages = {package["id"]: package for package in metadata["packages"]}
    nodes = {node["id"]: node for node in metadata["resolve"]["nodes"]}
    stack = [p["id"] for p in metadata["packages"] if p["name"] == root_name and p.get("source") is None]
    seen = set()
    while stack:
        current = stack.pop()
        if current in seen or (current in packages and is_proc_macro(packages[current])):
            continue
        seen.add(current)
        for dep in nodes.get(current, {}).get("deps", []):
            if any(kind.get("kind") is None for kind in dep.get("dep_kinds", [])):
                stack.append(dep["pkg"])
    return seen


def lock_checksums(root):
    """(name, version) -> checksum, from both workspaces' Cargo.lock."""
    sums = {}
    for name in ("Cargo.lock", os.path.join("desktop", "Cargo.lock")):
        path = os.path.join(root, name)
        if not os.path.exists(path):
            continue
        with open(path, encoding="utf-8") as handle:
            text = handle.read()
        for block in text.split("[[package]]"):
            fields = dict(re.findall(r'^(name|version|checksum) = "([^"]*)"$', block, re.M))
            if fields.get("checksum"):
                sums[(fields.get("name"), fields.get("version"))] = fields["checksum"]
    return sums


def patched(root):
    """Manifests or Cargo configuration that patch or replace crates."""
    found = []
    for name in ("Cargo.toml", os.path.join("desktop", "Cargo.toml"), os.path.join(".cargo", "config.toml"),
                 os.path.join("desktop", ".cargo", "config.toml")):
        path = os.path.join(root, name)
        if os.path.exists(path):
            with open(path, encoding="utf-8") as handle:
                if re.search(r"^\s*\[(patch|replace)\b", handle.read(), re.M):
                    found.append(name)
    return found


def license_texts(directory, license_file=None):
    names = []
    if directory and os.path.isdir(directory):
        names = sorted(n for n in os.listdir(directory) if LICENSE_NAME.match(n) and os.path.isfile(os.path.join(directory, n)))
        if license_file:
            extra = os.path.relpath(license_file if os.path.isabs(license_file) else os.path.join(directory, license_file), directory)
            if extra not in names and os.path.isfile(os.path.join(directory, extra)):
                names.append(extra)
    texts = []
    for name in names:
        try:
            with open(os.path.join(directory, name), encoding="utf-8", errors="replace") as handle:
                texts.append((name, handle.read().strip("\n")))
        except OSError:
            pass
    return texts

def mpl_sources(metadata_dir, out, root):
    """The exact published archives of every MPL-2.0 crate any target links, checked against Cargo.lock."""
    os.makedirs(out, exist_ok=True)
    sums = lock_checksums(root)
    wanted = {}
    for target in TARGETS:
        for workspace, binary in (("desktop", "lightagent-desktop"), ("root", "lightagent")):
            for package in linked(load(os.path.join(metadata_dir, f"{workspace}-{target}.json")), binary):
                if package.get("source") and MPL.search(package.get("license") or ""):
                    wanted[(package["name"], package["version"])] = package["source"]
    cache = os.path.join(os.environ.get("CARGO_HOME") or os.path.join(os.path.expanduser("~"), ".cargo"), "registry", "cache")
    lines, problems, listing = ["### MPL-2.0 source archives (the exact published crates this build uses)"], [], []
    for (name, version), source in sorted(wanted.items()):
        expected = sums.get((name, version))
        found = sorted(glob.glob(os.path.join(cache, "*", f"{name}-{version}.crate")))
        if source != CRATES_IO or not expected or not found:
            why = "not from crates.io" if source != CRATES_IO else "no Cargo.lock checksum" if not expected else "its archive is not in the cargo cache"
            problems.append(f"{name}@{version}: {why}")
            continue
        actual = sha256_file(found[0])
        if actual != expected:
            problems.append(f"{name}@{version}: the archive's SHA-256 {actual} is not Cargo.lock's {expected}")
            continue
        shutil.copy(found[0], os.path.join(out, f"{name}-{version}.crate"))
        listing.append(f"{actual}  {name}-{version}.crate")
        lines.append(f"- {name} {version}: SHA-256 `{actual}`, equal to Cargo.lock's checksum (the published source, unmodified)")
    with open(os.path.join(out, "SHA256SUMS"), "w", encoding="utf-8", newline="\n") as handle:
        handle.write("".join(line + "\n" for line in listing))
    with open(os.path.join(out, "README.txt"), "w", encoding="utf-8", newline="\n") as handle:
        handle.write("The Source Code Form of the MPL-2.0 components in Lightagent Desktop: each archive is the\n"
                     "crate exactly as published on crates.io (its SHA-256 equals the checksum Cargo.lock pins),\n"
                     "used unmodified. Each archive holds the component's license text.\n")
    problems += [f"{name} patches or replaces crates" for name in patched(root)]
    lines += [f"- FAILED {problem}" for problem in problems]
    print("\n".join(lines))
    return 1 if problems else 0


def shortest_path(metadata, root_name, wanted):
    packages = {package["id"]: package for package in metadata["packages"]}
    nodes = {node["id"]: node for node in metadata["resolve"]["nodes"]}
    start = [p["id"] for p in metadata["packages"] if p["name"] == root_name and p.get("source") is None]
    parent = {ident: None for ident in start}
    queue = list(start)
    while queue:
        current = queue.pop(0)
        if packages[current]["name"] == wanted:
            path = []
            while current is not None:
                path.append(f"{packages[current]['name']}@{packages[current]['version']}")
                current = parent[current]
            return list(reversed(path))
        for dep in nodes.get(current, {}).get("deps", []):
            if dep["pkg"] not in parent and any(kind.get("kind") is None for kind in dep.get("dep_kinds", [])):
                parent[dep["pkg"]] = current
                queue.append(dep["pkg"])
    return None


def call_sites(metadata, root_name, crate, pattern):
    found = []
    for package in linked(metadata, root_name):
        if package["name"] == crate:
            continue
        directory = os.path.dirname(package["manifest_path"])
        for base, dirs, names in os.walk(directory):
            dirs[:] = [d for d in dirs if d not in ("target", "node_modules", ".git")]
            for name in names:
                if not name.endswith(".rs"):
                    continue
                try:
                    with open(os.path.join(base, name), encoding="utf-8", errors="replace") as handle:
                        source = handle.read().splitlines()
                except OSError:
                    continue
                for number, line in enumerate(source, 1):
                    if pattern.search(line):
                        found.append(f"{package['name']}@{package['version']}:{os.path.relpath(os.path.join(base, name), directory)}:{number}")
    return found


def advisory_paths(metadata_dir):
    """Where each reviewed advisory's crate is reached, whether it is in the
    binary or only used at build time, and the call sites of its affected
    code in every crate the binary is built from (exact locked versions)."""
    lines = ["### Advisory crates: dependency paths and call sites (exact versions)"]
    for advisory, crate, pattern in ADVISORY_CODE:
        for target in TARGETS:
            for workspace, binary in (("desktop", "lightagent-desktop"), ("root", "lightagent")):
                metadata = load(os.path.join(metadata_dir, f"{workspace}-{target}.json"))
                path = shortest_path(metadata, binary, crate)
                if not path:
                    continue
                packages = {p["id"]: p for p in metadata["packages"]}
                in_binary = runtime_linked(metadata, binary)
                versions = sorted({p["version"] for p in linked(metadata, binary) if p["name"] == crate})
                built_in = any(packages[i]["name"] == crate for i in in_binary if i in packages)
                sites = call_sites(metadata, binary, crate, pattern) if pattern else []
                lines.append(f"- {advisory} {crate} {', '.join(versions)} in `{binary}` for `{target}`: "
                             f"{'in the binary' if built_in else 'build time only'}; path {' > '.join(path)}"
                             + (f"; call sites of the affected code: {len(sites)}" if pattern else ""))
                lines += [f"  - {site}" for site in sites[:20]]
    print("\n".join(lines))
    return 0
