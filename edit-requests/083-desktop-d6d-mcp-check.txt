/// The most tool names an MCP check reports.
const CHECK_TOOLS_SHOWN: usize = 200;
/// The most characters of an MCP check's error.
const CHECK_ERROR_CHARS: usize = 600;

/// The values an MCP server is configured with that must never be shown:
/// its environment and header values and its auth secret.
fn mcp_secret_values(entry: &lightagent_core::McpServerEntry) -> Vec<String> {
    let mut values: Vec<String> = match entry {
        lightagent_core::McpServerEntry::Stdio { env, .. } => env.values().cloned().collect(),
        lightagent_core::McpServerEntry::Http { headers, auth, .. } => {
            let mut values: Vec<String> = headers.values().cloned().collect();
            if let Some(lightagent_core::SecretLookup::Value(value)) =
                auth.as_ref().map(lightagent_core::SecretRef::lookup)
            {
                values.push(value);
            }
            values
        }
    };
    values.retain(|value| value.len() >= 4);
    values
}

fn mcp_redacted(text: &str, secrets: &[String]) -> String {
    let mut text = text.to_owned();
    for secret in secrets {
        text = text.replace(secret.as_str(), "[redacted]");
    }
    text.chars().take(CHECK_ERROR_CHARS).collect()
}

/// An explicit check of the MCP server `name` (D6-4), connected exactly as
/// an operator run connects it: the same resolver, timeout and HTTP client,
/// and an extension's folder as a stdio server's working folder. It lists
/// the server's tools and disconnects; a stdio server's process is killed
/// when its connection drops. A stdio server's command is started, so only
/// an explicit request calls this: viewing or listing servers never does.
/// It changes no setting. `None` when neither the configuration nor an
/// active extension declares the name.
pub(crate) async fn check_mcp_server(
    config: &Config,
    extensions: &ExtensionStore,
    name: &str,
) -> Option<serde_json::Value> {
    let configured = config
        .mcp
        .servers
        .iter()
        .find(|entry| entry.name() == name)
        .map(|entry| (entry, "configuration".to_owned()));
    let (entry, declared_by) = configured.or_else(|| {
        extensions.active(&config.extensions).find_map(|ext| {
            ext.mcp_servers
                .iter()
                .find(|entry| entry.name() == name)
                .map(|entry| (entry, format!("extension {}", ext.name)))
        })
    })?;
    let transport = match entry {
        lightagent_core::McpServerEntry::Stdio { .. } => "stdio",
        lightagent_core::McpServerEntry::Http { .. } => "http",
    };
    let secrets = mcp_secret_values(entry);
    let report = |ok: bool, tools: Vec<String>, more: usize, error: Option<String>, ms: u64| {
        serde_json::json!({
            "name": name, "transport": transport, "declared_by": declared_by,
            "ok": ok, "tools": tools, "more_tools": more, "error": error,
            "elapsed_ms": ms, "mcp_enabled": config.mcp.enabled,
        })
    };
    let started = Instant::now();
    let elapsed = || u64::try_from(started.elapsed().as_millis()).unwrap_or(u64::MAX);
    let (specs, refused) = plan_mcp_servers(config, extensions, RemoteAccess::Operator);
    if let Some(refusal) = refused.iter().find(|refusal| refusal.server == name) {
        return Some(report(false, Vec::new(), 0, Some(refusal.to_string()), elapsed()));
    }
    let Some(spec) = specs.into_iter().find(|spec| spec.name == name) else {
        return Some(report(
            false,
            Vec::new(),
            0,
            Some("the server could not be prepared as a run would connect it".to_owned()),
            elapsed(),
        ));
    };
    let timeout = Duration::from_secs(config.mcp.timeout_secs.clamp(1, 600));
    lightagent_provider_lightweight::ensure_provider();
    let client = match reqwest::Client::builder()
        .timeout(timeout)
        .redirect(reqwest::redirect::Policy::none())
        .build()
    {
        Ok(client) => client,
        Err(error) => {
            return Some(report(
                false,
                Vec::new(),
                0,
                Some(format!("could not build the MCP HTTP client: {error}")),
                elapsed(),
            ));
        }
    };
    // Bounded twice: each request by the timeout, and the whole check here.
    // Giving up drops the connection, which kills a stdio server.
    let limit = timeout.saturating_mul(3);
    let connected =
        tokio::time::timeout(limit, McpHub::connect(vec![spec], timeout, client)).await;
    let checked = match connected {
        Err(_) => report(
            false,
            Vec::new(),
            0,
            Some(format!("no answer within {} seconds", limit.as_secs())),
            elapsed(),
        ),
        Ok(hub) => match hub.errors.first() {
            Some((_, error)) => report(
                false,
                Vec::new(),
                0,
                Some(mcp_redacted(error, &secrets)),
                elapsed(),
            ),
            None => {
                let mut tools: Vec<String> = hub
                    .tools
                    .iter()
                    .map(|tool| tool.definition().name.clone())
                    .collect();
                tools.sort();
                let more = tools.len().saturating_sub(CHECK_TOOLS_SHOWN);
                tools.truncate(CHECK_TOOLS_SHOWN);
                // The tools hold the connection; dropping the hub ends it.
                drop(hub);
                report(true, tools, more, None, elapsed())
            }
        },
    };
    Some(checked)
}

/// A minimal streamable-HTTP MCP server with one tool, `ping`.
fn spawn_http_mcp() -> u16 {
    fn reply(body: &str) -> Option<String> {
        let message: Value = serde_json::from_str(body).ok()?;
        let id = message.get("id").cloned().unwrap_or(Value::Null);
        match message.get("method").and_then(Value::as_str).unwrap_or("") {
            "initialize" => Some(json!({"jsonrpc":"2.0","id":id,"result":{"protocolVersion":"2025-06-18","capabilities":{"tools":{}}}}).to_string()),
            "notifications/initialized" => None,
            "tools/list" => Some(json!({"jsonrpc":"2.0","id":id,"result":{"tools":[{"name":"ping","description":"Ping","inputSchema":{"type":"object"}}]}}).to_string()),
            _ => Some(json!({"jsonrpc":"2.0","id":id,"error":{"code":-32601,"message":"no"}}).to_string()),
        }
    }
    let listener = std::net::TcpListener::bind("127.0.0.1:0").expect("bind");
    let port = listener.local_addr().expect("addr").port();
    std::thread::spawn(move || {
        for mut stream in listener.incoming().flatten() {
            let mut buffer = Vec::new();
            let mut chunk = [0u8; 2048];
            let mut body = String::new();
            loop {
                if let Some(end) = buffer.windows(4).position(|w| w == b"\r\n\r\n") {
                    let head = String::from_utf8_lossy(&buffer[..end]).to_ascii_lowercase();
                    let length: usize = head
                        .lines()
                        .find_map(|line| line.strip_prefix("content-length:"))
                        .and_then(|value| value.trim().parse().ok())
                        .unwrap_or(0);
                    if buffer.len() >= end + 4 + length {
                        body = String::from_utf8_lossy(&buffer[end + 4..end + 4 + length]).into_owned();
                        break;
                    }
                }
                match stream.read(&mut chunk) {
                    Ok(0) | Err(_) => break,
                    Ok(read) => buffer.extend_from_slice(&chunk[..read]),
                }
            }
            let response = match reply(&body) {
                Some(json) => format!(
                    "HTTP/1.1 200 OK\r\nContent-Type: application/json\r\nContent-Length: {}\r\nConnection: close\r\n\r\n{json}",
                    json.len()
                ),
                None => "HTTP/1.1 202 Accepted\r\nContent-Length: 0\r\nConnection: close\r\n\r\n".to_owned(),
            };
            let _ = stream.write_all(response.as_bytes());
        }
    });
    port
}

/** `NAME=value` or `Name: value` lines into a map; blank lines are skipped. */
function pairs(text: string, separator: "=" | ":"): Record<string, string> | string {
  const out: Record<string, string> = {};
  for (const line of text.split("\n").map((value) => value.trim()).filter(Boolean)) {
    const at = line.indexOf(separator);
    if (at <= 0) return `"${line}" needs the form name${separator}value`;
    out[line.slice(0, at).trim()] = line.slice(at + 1).trim();
  }
  return out;
}

        Connected in {check.elapsed_ms} ms and disconnected. {check.tools.length + check.more_tools} tool
        {check.tools.length + check.more_tools === 1 ? "" : "s"}: {names}
        {check.more_tools > 0 ? ` and ${check.more_tools} more` : ""}.{runs}
      </div>
    );
  }
  return (
    <div className="notice notice--warn" role="alert">
      Could not connect: {check.error ?? "no answer"}.{runs}
    </div>
  );
}

/** A minimal streamable-HTTP MCP server with one tool, `ping`. */
async function httpMcp() {
  const server = createServer(async (request, response) => {
    const chunks = [];
    for await (const chunk of request) chunks.push(chunk);
    let message = {};
    try {
      message = JSON.parse(Buffer.concat(chunks).toString("utf8") || "{}");
    } catch {
      message = {};
    }
    const reply = (result) => {
      response.writeHead(200, { "content-type": "application/json" });
      response.end(JSON.stringify({ jsonrpc: "2.0", id: message.id ?? null, ...result }));
    };
    if (message.method === "initialize") return reply({ result: { protocolVersion: "2025-06-18", capabilities: { tools: {} } } });
    if (message.method === "tools/list") {
      return reply({ result: { tools: [{ name: "ping", description: "Ping", inputSchema: { type: "object" } }] } });
    }
    if (typeof message.method === "string" && message.method.startsWith("notifications/")) {
      response.writeHead(202);
      return response.end();
    }
    return reply({ error: { code: -32601, message: "no" } });
  });
  await new Promise((resolve) => server.listen(0, "127.0.0.1", resolve));
  return { server, url: `http://127.0.0.1:${server.address().port}/mcp` };
}

async function until(probe, timeout = 15000) {
  const deadline = Date.now() + timeout;
  while (Date.now() < deadline) {
    if (await probe()) return true;
    await new Promise((resolve) => setTimeout(resolve, 150));
  }
  return Boolean(await probe());
}

def scenario_mcp(host):
    """D6-4 through the host transport: a stdio server is added and listed
    without starting; a check starts it once, reports why it failed with no
    environment value, and its process is gone afterwards; MCP stays off."""
    import shutil
    import tempfile

    check = Checks("mcp")
    work = tempfile.mkdtemp(prefix="lightagent-e2e-mcp-")
    pidfile = os.path.join(work, "pid")
    secret = "e2e-mcp-env-value-5c1e"
    script = ("import os, sys, time; open(sys.argv[1], 'w').write(str(os.getpid())); "
              "sys.stderr.write('token ' + os.environ.get('TOKEN', '') + '\\n'); sys.stderr.flush(); time.sleep(300)")

    def gone(pid):
        deadline = time.time() + 15
        while time.time() < deadline:
            try:
                with open(f"/proc/{pid}/stat", encoding="utf-8") as handle:
                    state = handle.read().rsplit(")", 1)[1].split()[0]
            except OSError:
                return True
            if state == "Z":
                return True
            time.sleep(0.1)
        return False

    try:
        status, _ = host.fetch("PUT", f"{API}/mcp", {"timeout_secs": 2})
        check.expect("timeout set through the host", status == 200, status)
        server = {"transport": "stdio", "name": "hostprobe", "command": sys.executable,
                  "args": ["-c", script, pidfile], "env": {"TOKEN": secret}}
        status, added = host.fetch("POST", f"{API}/mcp/servers", {"server": server})
        check.expect("added through the host, value withheld", status == 201 and secret not in json.dumps(added),
                     (status, added))
        status, listed = host.fetch("GET", f"{API}/mcp")
        check.expect("listing starts nothing", status == 200 and not os.path.exists(pidfile),
                     (status, os.path.exists(pidfile)))
        status, result = host.fetch("POST", f"{API}/mcp/servers/hostprobe/check")
        started = os.path.exists(pidfile)
        check.expect("a check starts it and reports the failure without the value",
                     status == 200 and started and (result or {}).get("ok") is False and secret not in json.dumps(result),
                     (status, result, started))
        pid = None
        if started:
            with open(pidfile, encoding="utf-8") as handle:
                pid = handle.read().strip()
        check.expect("its process is gone after the check", pid is not None and gone(pid), pid)
        status, view = host.fetch("GET", f"{API}/mcp")
        check.expect("a check does not turn MCP on", ((view or {}).get("mcp") or {}).get("enabled") is False, view)
        status, _ = host.fetch("DELETE", f"{API}/mcp/servers/hostprobe")
        check.expect("removed through the host", status == 200, status)
    finally:
        shutil.rmtree(work, ignore_errors=True)
    return check
