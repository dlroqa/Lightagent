/// Where the configured key comes from, in the order the CLI and the
/// server resolve it. Never the key or its path.
fn key_source() -> Option<&'static str> {
    if std::env::var_os("LIGHTAGENT_CREDENTIAL_KEY_FILE").is_some_and(|value| !value.is_empty()) {
        Some("file")
    } else if std::env::var("LIGHTAGENT_CREDENTIAL_KEY").is_ok_and(|value| !value.is_empty()) {
        Some("env")
    } else {
        None
    }
}

/// `lightagent remote-credential …` for the API (which admits only the
/// local operator), with the key the runtime was started with. A secret is
/// read only from a file named by an absolute path: it is never in a
/// request or an answer, and no error names the file. A configured key
/// that cannot be used refuses every command.
pub(crate) fn perform(home: &Path, request: RemoteCredentialRequest) -> Result<Value, OpError> {
    let invalid = |message: String| OpError::new(OpCategory::Invalid, message);
    let absolute = |raw: String, what: &str| {
        let path = PathBuf::from(raw);
        if path.is_absolute() {
            Ok(path)
        } else {
            Err(invalid(format!("{what} must be an absolute path")))
        }
    };
    let from_file = |raw: Option<String>| {
        let raw = raw.ok_or_else(|| {
            invalid(
                "over the API a secret is read only from a file: name it (an absolute path)"
                    .to_owned(),
            )
        })?;
        Ok::<_, OpError>(SecretInput {
            secret_file: Some(absolute(raw, "the secret file")?),
            secret_stdin: false,
        })
    };
    let action = match request {
        RemoteCredentialRequest::Status => RemoteCredentialAction::Status,
        RemoteCredentialRequest::Keygen { path } => RemoteCredentialAction::Keygen {
            path: absolute(path, "the key file")?,
        },
        RemoteCredentialRequest::Add {
            profile,
            service,
            destinations,
            operations,
            expires_in,
            max_requests_per_run,
            secret_file,
        } => {
            if destinations.is_empty() {
                return Err(invalid("name at least one destination".to_owned()));
            }
            RemoteCredentialAction::Add {
                profile,
                service,
                destinations,
                operations,
                expires_in,
                max_requests_per_run,
                secret: from_file(secret_file)?,
            }
        }
        RemoteCredentialRequest::Rotate { id, secret_file } => RemoteCredentialAction::Rotate {
            id,
            secret: from_file(secret_file)?,
        },
        RemoteCredentialRequest::Revoke(id) => RemoteCredentialAction::Revoke { id },
        RemoteCredentialRequest::List { profile } => RemoteCredentialAction::List { profile },
        RemoteCredentialRequest::Audit { limit } => RemoteCredentialAction::Audit {
            limit: limit.unwrap_or(50),
        },
        RemoteCredentialRequest::Rekey { previous_key_file } => RemoteCredentialAction::Rekey {
            previous_key_file: absolute(previous_key_file, "the previous key file")?,
        },
    };
    let key = MasterKey::from_environment(home)
        .map_err(|error| invalid(format!("the configured credential key is unusable: {error}")))?;
    let secret = match &action {
        RemoteCredentialAction::Add { secret, .. }
        | RemoteCredentialAction::Rotate { secret, .. } => Some(secret.read().map_err(invalid)?),
        _ => None,
    };
    execute(home, &operator(), now_unix(), key.as_ref(), &action, secret).map_err(invalid)
}

/// A `lightagent remote-credential …` command through the embedder (see
/// [`ops::RemoteCredentialRequest`]), for the local operator only. No
/// request or answer carries a secret.
async fn remote_credential_operation(
    state: &AppState,
    headers: &HeaderMap,
    admitted: FromDesktop,
    request: ops::RemoteCredentialRequest,
) -> Response {
    if let Some(rejection) = local_operator(state, headers, admitted.is_some()) {
        return rejection;
    }
    match state.manager.remote_credentials(request).await {
        Ok(value) => Json(value).into_response(),
        Err(error) => error.into_response(),
    }
}

    remote_credential_operation(&state, &headers, admitted, request).await
}

//! Remote credentials through one shared operation set (D7-9): `lightagent
//! remote-credential` and the desktop runtime's API manage one vault as the
//! local operator, with the key each process is started with. What one
//! writes the other lists, and the API answers with the JSON the CLI
//! prints. A secret is read only from a file and is never in an answer,
//! the audit, or a file under the home in plaintext; refusals name no
//! file; a configured key that cannot be used refuses every command; a
//! keyed server refuses every route. Every process here is separate.

use std::io::{BufRead as _, BufReader, Read as _, Write as _};
use std::net::TcpStream;
use std::path::{Path, PathBuf};
use std::process::{Child, ChildStdin, Command, Output, Stdio};
use std::sync::mpsc;
use std::time::{Duration, SystemTime, UNIX_EPOCH};

use serde_json::{Value, json};

const KEY: &str = "b4c5d6e7f8091a2b3c4d5e6f708192a3b4c5d6e7f8091a2b3c4d5e6f708192a3";
const VAULT_HEX_KEY: &str = "0f1e2d3c4b5a69788796a5b4c3d2e1f00f1e2d3c4b5a69788796a5b4c3d2e1f0";
const CREDENTIALS: &str = "/api/lightagent/v1/remote-credentials";

struct Runtime {
    child: Child,
    _lifeline: Option<ChildStdin>,
    addr: String,
}

impl Drop for Runtime {
    fn drop(&mut self) {
        let _ = self.child.kill();
        let _ = self.child.wait();
    }
}

fn scratch(label: &str) -> PathBuf {
    let nanos = SystemTime::now()
        .duration_since(UNIX_EPOCH)
        .map(|elapsed| elapsed.as_nanos())
        .unwrap_or(0);
    let dir = std::env::temp_dir().join(format!(
        "lightagent-desktop-credentials-{label}-{}-{nanos}",
        std::process::id()
    ));
    std::fs::create_dir_all(&dir).expect("scratch dir");
    dir
}

/// The key environment a process runs with: none, a key file, or a key.
#[derive(Clone, Copy)]
enum KeyEnv<'a> {
    None,
    File(&'a Path),
    Hex(&'a str),
}

fn with_key(command: &mut Command, key: KeyEnv<'_>) {
    command
        .env_remove("LIGHTAGENT_CREDENTIAL_KEY_FILE")
        .env_remove("LIGHTAGENT_CREDENTIAL_KEY");
    match key {
        KeyEnv::None => {}
        KeyEnv::File(path) => {
            command.env("LIGHTAGENT_CREDENTIAL_KEY_FILE", path);
        }
        KeyEnv::Hex(hex) => {
            command.env("LIGHTAGENT_CREDENTIAL_KEY", hex);
        }
    }
}

/// The first line `child` prints, within a minute; the rest is drained.
fn first_line(child: &mut Child) -> String {
    let stdout = child.stdout.take().expect("stdout");
    let (tx, rx) = mpsc::channel();
    std::thread::spawn(move || {
        let mut reader = BufReader::new(stdout);
        let mut line = String::new();
        let _ = reader.read_line(&mut line);
        let _ = tx.send(line);
        let mut rest = String::new();
        while reader.read_line(&mut rest).is_ok_and(|read| read > 0) {
            rest.clear();
        }
    });
    rx.recv_timeout(Duration::from_secs(60))
        .expect("first line")
}

fn start_desktop(home: &Path, key: KeyEnv<'_>) -> Runtime {
    let mut command = Command::new(env!("CARGO_BIN_EXE_lightagent"));
    command
        .args(["serve", "--desktop", "--host", "127.0.0.1", "--port", "0"])
        .env("LIGHTAGENT_HOME", home)
        .stdin(Stdio::piped())
        .stdout(Stdio::piped())
        .stderr(Stdio::null());
    with_key(&mut command, key);
    let mut child = command.spawn().expect("start lightagent serve --desktop");
    let mut lifeline = child.stdin.take().expect("stdin");
    writeln!(lifeline, "{KEY}").expect("key");
    lifeline.flush().expect("flush");
    let line = first_line(&mut child);
    let handshake: Value = serde_json::from_str(&line).expect("handshake JSON");
    let addr = handshake["addr"].as_str().expect("addr").to_owned();
    Runtime {
        child,
        _lifeline: Some(lifeline),
        addr,
    }
}

/// `lightagent serve` on loopback with an API key: reachable from elsewhere.
fn start_keyed(home: &Path) -> Runtime {
    let mut child = Command::new(env!("CARGO_BIN_EXE_lightagent"))
        .args(["serve", "--host", "127.0.0.1", "--port", "0"])
        .args(["--key-env", "LIGHTAGENT_TEST_CREDENTIALS_KEY"])
        .env("LIGHTAGENT_TEST_CREDENTIALS_KEY", KEY)
        .env("LIGHTAGENT_HOME", home)
        .stdin(Stdio::null())
        .stdout(Stdio::piped())
        .stderr(Stdio::null())
        .spawn()
        .expect("start lightagent serve");
    let line = first_line(&mut child);
    let addr = line
        .split("http://")
        .nth(1)
        .and_then(|rest| rest.split('/').next())
        .unwrap_or_else(|| panic!("no address in {line:?}"))
        .to_owned();
    Runtime {
        child,
        _lifeline: None,
        addr,
    }
}

fn call(runtime: &Runtime, method: &str, path: &str, body: Option<&str>) -> (u16, String) {
    let mut stream = TcpStream::connect(&runtime.addr).expect("connect");
    stream
        .set_read_timeout(Some(Duration::from_secs(60)))
        .expect("timeout");
    let body = body.unwrap_or("");
    let head = format!(
        "{method} {path} HTTP/1.1\r\nHost: {}\r\nAuthorization: Bearer {KEY}\r\nContent-Type: application/json\r\nContent-Length: {}\r\nConnection: close\r\n\r\n{body}",
        runtime.addr,
        body.len()
    );
    stream.write_all(head.as_bytes()).expect("send");
    let mut response = String::new();
    let _ = stream.read_to_string(&mut response);
    let status = response
        .split_whitespace()
        .nth(1)
        .and_then(|code| code.parse().ok())
        .unwrap_or(0);
    let raw = response
        .split_once("\r\n\r\n")
        .map(|(_, body)| body.to_owned())
        .unwrap_or_default();
    (status, raw)
}

fn parsed(raw: &str) -> Value {
    serde_json::from_str(raw).unwrap_or_else(|error| panic!("not JSON ({error}): {raw}"))
}

fn ok(runtime: &Runtime, method: &str, path: &str, body: Option<Value>) -> (Value, String) {
    let body = body.map(|body| body.to_string());
    let (status, raw) = call(runtime, method, path, body.as_deref());
    assert_eq!(status, 200, "{method} {path}: {raw}");
    (parsed(&raw), raw)
}

fn cli(home: &Path, key: KeyEnv<'_>, args: &[&str]) -> Output {
    let mut command = Command::new(env!("CARGO_BIN_EXE_lightagent"));
    command
        .args(args)
        .env("LIGHTAGENT_HOME", home)
        .stdin(Stdio::null());
    with_key(&mut command, key);
    command.output().expect("run lightagent")
}

fn cli_ok(home: &Path, key: KeyEnv<'_>, args: &[&str]) {
    let output = cli(home, key, args);
    assert!(
        output.status.success(),
        "lightagent {args:?}: {}",
        String::from_utf8_lossy(&output.stderr)
    );
}

fn cli_json(home: &Path, key: KeyEnv<'_>, args: &[&str]) -> Value {
    let mut all = vec!["--json"];
    all.extend_from_slice(args);
    let output = cli(home, key, &all);
    assert!(
        output.status.success(),
        "lightagent {args:?}: {}",
        String::from_utf8_lossy(&output.stderr)
    );
    serde_json::from_slice(&output.stdout).expect("JSON output")
}

fn arg(path: &Path) -> &str {
    path.to_str().expect("UTF-8 path")
}

/// Fail if any file under `dir` holds `secret` in plaintext.
fn assert_nowhere(dir: &Path, secret: &str) {
    let Ok(entries) = std::fs::read_dir(dir) else {
        return;
    };
    for entry in entries.flatten() {
        let path = entry.path();
        if path.is_dir() {
            assert_nowhere(&path, secret);
        } else if let Ok(bytes) = std::fs::read(&path) {
            assert!(
                !String::from_utf8_lossy(&bytes).contains(secret),
                "{} holds a secret in plaintext",
                path.display()
            );
        }
    }
}

fn grant<'a>(listing: &'a Value, id: &str) -> &'a Value {
    listing["credentials"]
        .as_array()
        .and_then(|rows| rows.iter().find(|row| row["id"] == id))
        .unwrap_or(&Value::Null)
}

// source and the grant that does not open; a key file is created
// (owner-only; an existing file is refused); a grant is added from a
// secret file (the secret is in neither the page nor browser storage; a
// missing file is refused without naming it), rotated from a new file and
// revoked (asks first); the old grant is re-sealed from the previous key
// (asks first); the audit shows the records asked for.
// crates/lightagent/tests/desktop_remote_credentials.rs proves the API's
// vault is the CLI's; desktop/e2e/runs.py `remote_credentials` covers the
// host.

import { mkdtemp, rm, stat, writeFile } from "node:fs/promises";
import { tmpdir } from "node:os";
import { join } from "node:path";

const API = "/api/lightagent/v1/remote-credentials";

async function get(base, path) {
  return (await fetch(`${base}${path}`)).json();
}

async function until(probe, timeout = 15000) {
  const deadline = Date.now() + timeout;
  while (Date.now() < deadline) {
    if (await probe()) return true;
    await new Promise((resolve) => setTimeout(resolve, 150));
  }
  return Boolean(await probe());
}

export async function checkRemoteCredentials(browser, base, outDir) {
  const failures = [];
  const dir = await mkdtemp(join(tmpdir(), "lightagent-render-credentials-"));
  const page = await browser.newPage({ viewport: { width: 1280, height: 2000 } });
  const errors = [];
  page.on("pageerror", (error) => errors.push(error.message));
  const screen = page.getByTestId("credentials-screen");
  const announced = (text) => page.getByTestId("credentials-message").filter({ hasText: text }).waitFor({ timeout: 10000 });
  const step = async (name, body) => {
    try {
      await body();
    } catch (err) {
      failures.push(`credentials: ${name}: ${err instanceof Error ? err.message.split("\n")[0] : String(err)}`);
      await page.screenshot({ path: `${outDir}/credentials-failed-${name.replace(/\W+/g, "-")}.png`, fullPage: true }).catch(() => {});
    }
  };
  const secrets = ["render-previous-secret"];
  const noSecretShown = async (where) => {
    const seen = await page.evaluate(() => {
      const stored = (storage) => Object.keys(storage).map((key) => `${key}=${storage.getItem(key)}`).join("\n");
      return `${document.documentElement.outerHTML}\n${stored(localStorage)}\n${stored(sessionStorage)}`;
    });
    for (const secret of secrets) {
      if (seen.includes(secret)) throw new Error(`a secret is in the page or browser storage (${where})`);
    }
  };
  let id = "";
  try {
    const active = (await get(base, "/api/lightagent/v1/profiles")).active_profile;
    await page.goto(`${base}/#/admin/credentials`);
    await screen.waitFor({ timeout: 15000 });
    const status = page.getByTestId("credentials-status");
    await status.waitFor({ timeout: 15000 });

    await step("the key's source and the grant that does not open", async () => {
      await status.getByText("from a key file", { exact: false }).waitFor({ timeout: 10000 });
      await page.getByTestId("credentials-unopenable").waitFor({ timeout: 10000 });
    });

    await step("create a key file", async () => {
      const file = join(dir, "new-credential.key");
      await page.locator("#rc-keygen-path").fill(file);
      await screen.getByRole("button", { name: "Create key file" }).click();
      await announced("Created a key file");
      const mode = (await stat(file)).mode & 0o777;
      if (process.platform !== "win32" && mode !== 0o600) throw new Error(`the key file mode is ${mode.toString(8)}`);
      await page.locator("#rc-keygen-path").fill(file);
      await screen.getByRole("button", { name: "Create key file" }).click();
      await page.getByTestId("credentials-error").filter({ hasText: "could not create the key file" }).waitFor({ timeout: 10000 });
    });

    await step("add a grant from a secret file", async () => {
      const secret = `render-probe-secret-${Date.now()}`;
      secrets.push(secret);
      const file = join(dir, "probe.secret");
      await writeFile(file, `${secret}\n`);
      await page.locator("#rc-add-profile").selectOption(active);
      await page.locator("#rc-add-service").fill("mcp:render-probe");
      await page.locator("#rc-add-destinations").fill("https://probe.render.invalid");
      await page.locator("#rc-add-budget").fill("3");
      await page.locator("#rc-add-secret-file").fill(join(dir, "missing.secret"));
      await screen.getByRole("button", { name: "Add credential" }).click();
      const refused = page.getByTestId("credentials-error").filter({ hasText: "could not read the secret file" });
      await refused.waitFor({ timeout: 10000 });
      if ((await refused.innerText()).includes(dir)) throw new Error("the refusal names the file");
      await page.locator("#rc-add-secret-file").fill(file);
      await screen.getByRole("button", { name: "Add credential" }).click();
      await announced("Added ");
      const text = await page.getByTestId("credentials-message").innerText();
      id = /Added ([^:]+):/.exec(text)?.[1] ?? "";
      const grant = (await get(base, API)).credentials.find((item) => item.id === id);
      if (!grant || grant.service !== "mcp:render-probe" || grant.max_requests_per_run !== 3) {
        throw new Error(`the grant is ${JSON.stringify(grant)}`);
      }
      await page.getByTestId(`rc-${id}`).waitFor({ timeout: 10000 });
      await noSecretShown("after add");
      await page.screenshot({ path: `${outDir}/credentials-added.png`, fullPage: true });
    });

    await step("rotate from a new file, then revoke", async () => {
      const row = page.getByTestId(`rc-${id}`);
      const secret = `render-probe-rotated-${Date.now()}`;
      secrets.push(secret);
      const file = join(dir, "probe-rotated.secret");
      await writeFile(file, `${secret}\n`);
      await row.getByRole("button", { name: "Rotate…" }).click();
      await page.locator("#rc-rotate-file").fill(file);
      await row.getByRole("button", { name: `Rotate ${id}` }).click();
      await announced(`Rotated ${id} (version 2)`);
      const grant = async () => (await get(base, API)).credentials.find((item) => item.id === id);
      if ((await grant())?.version !== 2) throw new Error("the grant was not rotated");
      await row.getByRole("button", { name: "Revoke…" }).click();
      await screen.getByRole("button", { name: "Keep it" }).click();
      if ((await grant())?.state !== "active") throw new Error("Keep it revoked the grant");
      await row.getByRole("button", { name: "Revoke…" }).click();
      await row.getByRole("button", { name: `Confirm: revoke ${id}` }).click();
      await announced(`Revoked ${id}`);
      if (!(await until(async () => (await grant())?.state === "revoked"))) throw new Error("Confirm did not revoke");
      await noSecretShown("after rotate and revoke");
    });

    await step("re-seal the old grant from the previous key", async () => {
      const previous = process.env.RENDER_PREVIOUS_CREDENTIAL_KEY;
      if (!previous) throw new Error("the harness named no previous key (RENDER_PREVIOUS_CREDENTIAL_KEY)");
      await page.locator("#rc-rekey-previous").fill(previous);
      await screen.getByRole("button", { name: "Re-seal…" }).click();
      await screen.getByRole("button", { name: "Confirm: re-seal every active grant" }).click();
      await announced("Re-sealed 1 grant");
      const vault = await get(base, `${API}/vault`);
      if (vault.unopenable_with_configured_key !== 0) throw new Error(`still unopenable: ${JSON.stringify(vault)}`);
    });

    await step("the audit shows the records asked for", async () => {
      await page.locator("#rc-audit-limit").fill("3");
      await screen.getByRole("button", { name: "Show audit" }).click();
      const rows = page.getByTestId("credentials-audit").locator("li");
      if (!(await until(async () => (await rows.count()) === 3))) throw new Error(`${await rows.count()} audit rows`);
      await noSecretShown("with the audit");
    });

    if (errors.length) failures.push(`credentials: page errors: ${errors.join("; ")}`);
  } catch (err) {
    failures.push(`credentials: ${err instanceof Error ? err.message.split("\n")[0] : String(err)}`);
  } finally {
    await page.close();
    await rm(dir, { recursive: true, force: true });
  }
  return failures;
}
