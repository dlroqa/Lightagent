The remaining D4 gate items (DESKTOP_ACCEPTANCE.md):

- `jev-routing-report` is one shared operation,
  `lightagent_core::config_keys::jev_routing_report`, moved unchanged from
  the CLI. It is served read-only as GET /jev/routing-report (admin scope,
  like /config), listed as contract operation `jev.routing_report`, and
  shown in Settings as a dry-run table under the routing permission.
- An unreadable key file fails closed. `SecretRef::for_request` gives the
  value to send, and refuses a file reference that cannot be read with
  `UnreadableSecret`, whose message names neither the path nor any
  content. Every other case resolves exactly as `resolve` did. The
  runtime's run, history-window probe and model listing use it: a run
  fails before any request is made, never sent without the credential it
  was configured with.
- A Settings save refused with 409 no longer replaces the screen's values
  and asks for the change again. The change is kept and listed against
  the values saved now (a section such as Jev is named, never shown).
  "Apply my change to the current settings" saves it on the current
  revision; "Discard my change" shows the saved values.

  two homes is the same, with the home's own path as the only

- an unreadable key file fails the run with a redacted error, and the

/// `lightagent config jev-routing-report`, shared by the CLI and the HTTP
/// API: the routing-permission dry run. For every profile on disk (and the
/// built-in `default`), whether it may route now and under `strict`. Profile
/// ids and fixed vocabulary only; no message, session or key is read, and
/// nothing is changed.
pub fn jev_routing_report(config: &Config, profiles: &[crate::ProfileId]) -> serde_json::Value {
    let jev = &config.platform.jev;
    let strict = crate::JevConfig {
        routing_permission: crate::JevRoutingPermission::Strict,
        ..jev.clone()
    };
    let mut ids: Vec<String> = profiles.iter().map(|id| id.as_str().to_owned()).collect();
    if !ids.iter().any(|id| id == "default") {
        ids.push("default".to_owned());
    }
    ids.sort();
    ids.dedup();
    let rows: Vec<serde_json::Value> = ids
        .iter()
        .map(|id| {
            serde_json::json!({
                "profile": id,
                "now": jev.routing_grant(id).as_str(),
                "under_strict": strict.routing_grant(id).as_str(),
            })
        })
        .collect();
    let denied: Vec<&String> = ids
        .iter()
        .filter(|id| !strict.routing_grant(id).permits())
        .collect();
    let unknown: Vec<&String> = jev
        .routing_allowed_profiles
        .iter()
        .filter(|allowed| !ids.contains(allowed))
        .collect();
    let routing_active = jev.endpoint.enabled
        && (jev
            .allowed_models
            .iter()
            .any(|model| !model.trim().is_empty())
            || !jev.allowed_profiles.is_empty());
    serde_json::json!({
        "report": "jev-routing-permission-dry-run",
        "changes_made": false,
        "routing_enabled": routing_active,
        "routing_input": jev.routing_input.as_str(),
        "permission_policy": jev.routing_permission.as_str(),
        "compatibility": jev.routing_permission == crate::JevRoutingPermission::LegacyAllProfiles,
        "profiles": rows,
        "denied_under_strict": denied,
        "allowed_profiles_not_found": unknown,
    })
}

    State(state): State<Arc<AppState>>,
    headers: HeaderMap,
) -> Response {
    if let Some(rejection) = deny(&state, &headers, Scope::Admin) {
        return rejection;
    }

        return internal("Lightagent settings are unavailable in this embedding");

        Ok(config) => config,
        Err(error) => return config_error(error),

        Ok(profiles) => Json(lightagent_core::config_keys::jev_routing_report(
            &config, &profiles,
        ))
        .into_response(),
        Err(error) => internal(&error.to_string()),

    await notice.waitFor({ state: "detached", timeout: 5000 });

    if (await approval.inputValue() !== target) failures.push("settings: after discarding, the screen does not show the saved value");

    }));

    if (JSON.stringify(await settingsNow(base)).includes(SECRET)) failures.push("settings: the settings response carries the secret");

    if (errors.length) failures.push(`settings: uncaught page error — ${errors[0]}`);
    console.log(`  [${failures.length ? "FAIL" : "ok"}] settings conflict, routing dry run and renderer state`);

    const after = await settingsNow(base).catch(() => null);

        ...after,
        approval_policy: original.approval_policy,
        max_turns: original.max_turns,
      }).catch(() => {});
    }
  }
  return failures;
}

/// Every file under `root`, by `/`-separated relative path, with its text.
/// The one normalization: the home's own path, which differs between two
/// homes by construction, reads as `<HOME>` wherever a file records it
/// (raw, or escaped inside JSON).
fn files(root: &Path) -> std::collections::BTreeMap<String, String> {
    let home = root.display().to_string();
    let escaped = serde_json::to_string(&home).expect("JSON");
    let escaped = escaped.trim_matches('"').to_owned();
    let mut found = std::collections::BTreeMap::new();
    let mut pending = vec![root.to_path_buf()];
    while let Some(dir) = pending.pop() {
        for entry in std::fs::read_dir(&dir).expect("read dir").flatten() {
            let path = entry.path();
            if path.is_dir() {
                pending.push(path);
                continue;
            }
            let relative = path
                .strip_prefix(root)
                .expect("under the home")
                .to_string_lossy()
                .replace('\\', "/");
            let text = String::from_utf8_lossy(&std::fs::read(&path).expect("read"))
                .replace(&escaped, "<HOME>")
                .replace(&home, "<HOME>");
            found.insert(relative, text);
        }
    }
    found
}

    let mut value: serde_json::Value =
        serde_json::from_slice(&std::fs::read(&config_file).expect("read")).expect("config JSON");
    value["inference"]["api_key"] = serde_json::json!({ "kind": "file", "path": missing });

        "POST",
        "/api/lightagent/v1/runs",

        assert!(std::time::Instant::now() < deadline, "the run never failed: {state}");
