//! DIAGNOSTIC ONLY (branch diag/windows-stall, never merged): a timeline of
//! stdio server lifecycle events for issue #82, appended to the file that
//! `LIGHTAGENT_DIAG_TIMELINE` names. Event names, process ids, request ids
//! and times only: never commands, arguments, environments or contents.

use std::io::Write as _;
use std::sync::{LazyLock, Mutex};
use std::time::{Instant, SystemTime, UNIX_EPOCH};

static START: LazyLock<Instant> = LazyLock::new(Instant::now);
static SINK: LazyLock<Option<Mutex<std::fs::File>>> = LazyLock::new(|| {
    let path = std::env::var_os("LIGHTAGENT_DIAG_TIMELINE")?;
    std::fs::OpenOptions::new()
        .create(true)
        .append(true)
        .open(path)
        .ok()
        .map(Mutex::new)
});

/// Append one event, when a timeline file is configured.
pub fn event(what: &str) {
    let Some(sink) = SINK.as_ref() else {
        return;
    };
    let mono = START.elapsed().as_micros();
    let wall = SystemTime::now()
        .duration_since(UNIX_EPOCH)
        .map(|d| d.as_millis())
        .unwrap_or(0);
    let thread = std::thread::current();
    let line = format!(
        "{wall} pid={} mono_us={mono} thread={} {what}\n",
        std::process::id(),
        thread.name().unwrap_or("?")
    );
    if let Ok(mut file) = sink.lock() {
        let _ = file.write_all(line.as_bytes());
    }
}

name: diag-windows-stall

# DIAGNOSTIC ONLY (branch diag/windows-stall, never merged), for issue #82.
# Six fresh Windows runners each run the check job's steps cold, as
# check.yml does (same job id and cache key, so the same cache is restored;
# nothing is saved), under a resource sampler that records process names and
# CPU deltas only, never command lines or environments. The instrumented
# build appends the MCP stdio timeline. The same runner then repeats the lib
# suite and the affected test warm. Bounded: six replicas, fixed rounds.

on:
  push:
    branches: [diag/windows-stall]

permissions:
  contents: read

env:
  CARGO_INCREMENTAL: '0'

jobs:
  check:
    name: cold check (windows replica ${{ matrix.replica }})
    runs-on: windows-latest
    timeout-minutes: 75
    strategy:
      fail-fast: false
      matrix:
        replica: [1, 2, 3, 4, 5, 6]
    defaults:
      run:
        shell: bash
    steps:
      - uses: actions/checkout@v7
      - run: rustup show
      - uses: Swatinem/rust-cache@v2
        with:
          key: windows-x64
          save-if: false
      - uses: actions/setup-node@v7
        with:
          node-version: '22'
          cache: npm
          cache-dependency-path: frontend/package-lock.json
      - run: npm ci --prefix frontend
      - name: Runner facts (no environment dump)
        shell: pwsh
        run: |
          "processors=$env:NUMBER_OF_PROCESSORS"
          $os = Get-CimInstance Win32_OperatingSystem
          "memory_mb=$([math]::Round($os.TotalVisibleMemorySize/1024))"
          try { Get-MpComputerStatus | Select-Object AMRunningMode, RealTimeProtectionEnabled, AntivirusEnabled | Format-List } catch { "defender status unavailable" }
      - name: check.sh cold, then warm rounds, under the sampler
        env:
          LIGHTAGENT_DIAG_TIMELINE: ${{ runner.temp }}/timeline.txt
        run: |
          out="$RUNNER_TEMP/diag"; mkdir -p "$out"
          cat > "$RUNNER_TEMP/sampler.ps1" <<'PS'
          param($csv, $stop)
          "wall_ms,cpu_pct,disk_pct,avail_mb,python_procs,top_cpu_s" | Out-File $csv -Encoding utf8
          $prev = @{}
          while (-not (Test-Path $stop)) {
            $c = Get-Counter '\Processor(_Total)\% Processor Time','\PhysicalDisk(_Total)\% Disk Time','\Memory\Available MBytes' -SampleInterval 1 -MaxSamples 1 -ErrorAction SilentlyContinue
            $v = @($c.CounterSamples | ForEach-Object { [math]::Round($_.CookedValue, 1) })
            $procs = @(Get-Process | Select-Object Id, ProcessName, CPU)
            $deltas = foreach ($p in $procs) {
              $cpu = [double]$p.CPU
              $was = $prev[$p.Id]
              if ($null -eq $was) { $was = $cpu }
              $prev[$p.Id] = $cpu
              [pscustomobject]@{ n = $p.ProcessName; d = $cpu - $was }
            }
            $top = ($deltas | Sort-Object d -Descending | Select-Object -First 5 | ForEach-Object { "$($_.n):$([math]::Round($_.d, 2))" }) -join ' '
            $py = @($procs | Where-Object { $_.ProcessName -like 'python*' }).Count
            "$([DateTimeOffset]::UtcNow.ToUnixTimeMilliseconds()),$($v -join ','),$py,$top" | Add-Content $csv
          }
          PS
          pwsh -NoProfile -File "$RUNNER_TEMP/sampler.ps1" "$out/sampler.csv" "$RUNNER_TEMP/sampler.stop" &
          sampler=$!
          mark() { echo "$(date +%s%3N) phase $1" | tee -a "$LIGHTAGENT_DIAG_TIMELINE" >> "$out/phases.txt"; }
          mark cold-check.begin
          bash ./scripts/check.sh 2>&1 | tee "$out/check.log"; status=${PIPESTATUS[0]}
          mark "cold-check.end exit=$status"
          for round in 1 2 3; do
            mark "warm-suite.$round.begin"
            cargo test -p lightagent --lib 2>&1 | tee "$out/warm-suite-$round.log" | grep -E '^test result|FAILED|panicked'
            mark "warm-suite.$round.end exit=${PIPESTATUS[0]}"
          done
          for round in 1 2 3 4 5; do
            mark "focused.$round.begin"
            cargo test -p lightagent --lib installed_mcp_tool_is_listed_callable_and_removed 2>&1 | tee "$out/focused-$round.log" | grep -E '^test result|FAILED|panicked'
            mark "focused.$round.end exit=${PIPESTATUS[0]}"
          done
          touch "$RUNNER_TEMP/sampler.stop"; wait $sampler || true
          cp "$LIGHTAGENT_DIAG_TIMELINE" "$out/timeline.txt" 2>/dev/null || true
          echo "$status" > "$out/check-exit.txt"
      - name: Summarise on the runner
        if: always()
        run: |
          python - "$RUNNER_TEMP/diag" <<'PY' | tee -a "$GITHUB_STEP_SUMMARY"
          import re, sys, collections
          out = sys.argv[1]
          def read(name):
              try:
                  return open(f"{out}/{name}", encoding="utf-8", errors="replace").read().splitlines()
              except FileNotFoundError:
                  return []
          print(f"## replica ${{ matrix.replica }}")
          print("check.sh exit:", (read("check-exit.txt") or ["?"])[0])
          def lib_result(lines):
              for l in lines:
                  m = re.match(r"test result: (\w+)\. (\d+) passed; (\d+) failed.*finished in ([\d.]+)s", l)
                  if m and int(m.group(2)) + int(m.group(3)) >= 280:
                      return f"{m.group(1)} {m.group(2)} passed {m.group(3)} failed in {m.group(4)} s"
              return "n/a"
          print("cold lightagent lib suite:", lib_result(read("check.log")))
          for r in (1, 2, 3):
              print(f"warm lib suite {r}:", lib_result(read(f"warm-suite-{r}.log")))
          fails = sorted({l.split()[1] for l in read("check.log") if l.startswith("test ") and l.endswith("FAILED")})
          print("cold failing tests:", ", ".join(fails) or "none")
          phases = [(int(l.split()[0]), l.split(" phase ")[1]) for l in read("phases.txt")]
          def phase_at(ms):
              name = "before"
              for t, p in phases:
                  if t <= ms and p.endswith(".begin"):
                      name = p[:-6]
              return name
          timeline = read("timeline.txt")
          answers = collections.defaultdict(list)
          for l in timeline:
              m = re.search(r"stdio\.request\.(answered|timeout|closed) .*after_ms=(\d+)", l)
              if m:
                  answers[(phase_at(int(l.split()[0])), m.group(1))].append(int(m.group(2)))
          print("stdio requests by phase (count / max ms):")
          for (p, kind), v in sorted(answers.items()):
              print(f"  {p} {kind}: {len(v)} / {max(v)}")
          for l in timeline:
              if "test.installed_mcp" in l or "stdio.request.timeout" in l:
                  print("  ", re.sub(r" thread=\S+", "", l))
          rows = []
          for l in read("sampler.csv")[1:]:
              parts = l.split(",")
              if len(parts) >= 6:
                  try:
                      rows.append((int(parts[0]), float(parts[1]), float(parts[2]), float(parts[3]), int(parts[4]), parts[5]))
                  except ValueError:
                      pass
          cold = [r for r in rows if phase_at(r[0]) == "cold-check"]
          if cold:
              cpu = sorted(r[1] for r in cold)
              print(f"sampler cold-check: {len(cold)} samples, cpu median {cpu[len(cpu)//2]} max {cpu[-1]}, samples >=95% cpu: {sum(c >= 95 for c in cpu)}")
              starts = [int(l.split()[0]) for l in timeline if "test.installed_mcp.start" in l]
              if starts:
                  t0 = starts[0]
                  print("sampler around the affected test (-20 s .. +40 s):")
                  for r in rows:
                      if t0 - 20000 <= r[0] <= t0 + 40000:
                          print(f"  {(r[0]-t0)/1000:+.0f}s cpu={r[1]} disk={r[2]} avail_mb={r[3]:.0f} python={r[4]} top={r[5]}")
              heavy = collections.Counter()
              for r in cold:
                  if r[1] >= 90:
                      for item in r[5].split():
                          name, _, d = item.rpartition(":")
                          try:
                              if float(d) > 0.5:
                                  heavy[name] += 1
                          except ValueError:
                              pass
              print("processes busy while cpu >= 90% (samples):", ", ".join(f"{n}:{c}" for n, c in heavy.most_common(8)) or "none")
          PY
      - uses: actions/upload-artifact@v7
        if: always()
        with:
          name: diag-windows-stall-${{ matrix.replica }}
          path: ${{ runner.temp }}/diag
          retention-days: 30
      - name: Fail when the cold check failed
        if: always()
        run: test "$(cat "$RUNNER_TEMP/diag/check-exit.txt" 2>/dev/null)" = 0
