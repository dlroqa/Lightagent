// Saved conversations against the real harness (D7-5): a seeded
// conversation is listed and opens with its message; Delete asks first;
// Keep it keeps it; Confirm deletes it from the server and the list.
// crates/lightagent/tests/desktop_sessions.rs proves the API's sessions are
// the CLI's.

import { seededSession } from "./delivery.mjs";

const TITLE = "Sessions probe conversation";
export async function checkSessions(browser, base, outDir) {
  const failures = [];
  const page = await browser.newPage({ viewport: { width: 1280, height: 1000 } });
  const errors = [];
  page.on("pageerror", (error) => errors.push(error.message));
  let id = null;
  const exists = async () => (await fetch(`${base}/api/lightagent/v1/sessions/${id}`)).status === 200;
  const step = async (name, body) => {
    try {
      await body();
      return true;
    } catch (err) {
      const text = await page.evaluate(() => document.body.innerText).catch(() => "");
      const message = err instanceof Error ? err.message.split("\n")[0] : String(err);
      failures.push(`sessions: ${name}: ${message} [page ends: ${text.replace(/\s+/g, " ").slice(-300)}]`);
      await page.screenshot({ path: `${outDir}/sessions-failed-${name.replace(/\W+/g, "-")}.png` }).catch(() => {});
      return false;
    }
  };
  try {
    id = await seededSession(base, TITLE);
    const listed = (await (await fetch(`${base}/api/lightagent/v1/sessions`)).json()).sessions ?? [];
    const label = listed.find((session) => session.id === id)?.title || TITLE;
    await page.goto(`${base}/#/`);
    // Everything happens within this conversation's own sidebar row.
    const item = page.locator("li.session-row").filter({ hasText: label }).first();
    const openDelete = async () => {
      await item.getByRole("button", { name: `More actions for ${label}` }).click({ timeout: 10000 });
      await page.getByRole("menu", { name: `Actions for ${label}` }).getByRole("menuitem", { name: /Delete/ }).click({ timeout: 10000 });
    };

    const listedOk = await step("the conversation is listed and opens", async () => {
      await item.waitFor({ timeout: 15000 });
      await item.getByRole("button", { name: label, exact: true }).click({ timeout: 10000 });
      await page.getByText(TITLE, { exact: false }).first().waitFor({ timeout: 15000 });
    });
    if (listedOk) {
      await step("delete asks first and Keep it keeps it", async () => {
        await openDelete();
        await item.getByRole("button", { name: `Confirm: delete ${label}` }).waitFor({ timeout: 5000 });
        await page.screenshot({ path: `${outDir}/sessions-confirm-delete.png` });
        if (!(await exists())) throw new Error("Delete removed the conversation before it was confirmed");
        await item.getByRole("button", { name: "Keep it" }).click({ timeout: 10000 });
        if (!(await exists())) throw new Error("Keep it removed the conversation");
      });
      const more = item.getByRole("button", { name: `More actions for ${label}` });
      const opened = await step("the menu opens again", async () => {
        if (await more.isDisabled()) throw new Error("More actions is disabled");
        await more.click({ timeout: 10000 });
        await page.getByRole("menu", { name: `Actions for ${label}` }).waitFor({ timeout: 5000 });
      });
      const chosen = opened && await step("Delete is chosen again", async () => {
        await page.getByRole("menu", { name: `Actions for ${label}` }).getByRole("menuitem", { name: /Delete/ }).click({ timeout: 10000 });
        await item.getByRole("button", { name: `Confirm: delete ${label}` }).waitFor({ timeout: 5000 });
      });
      if (chosen) await step("confirm deletes it", async () => {
        await item.getByRole("button", { name: `Confirm: delete ${label}` }).click({ timeout: 10000 });
        const deadline = Date.now() + 15000;
        while (Date.now() < deadline && (await exists())) await new Promise((resolve) => setTimeout(resolve, 150));
        if (await exists()) throw new Error("Confirm did not delete the conversation");
        await page.locator("li.session-row").filter({ hasText: label }).waitFor({ state: "detached", timeout: 10000 });
      });
    }
    if (errors.length) failures.push(`sessions: page errors: ${errors.join("; ")}`);
  } catch (err) {
    failures.push(`sessions: ${err instanceof Error ? err.message.split("\n")[0] : String(err)}`);
  } finally {
    await page.close();
    if (id) await fetch(`${base}/api/lightagent/v1/sessions/${id}`, { method: "DELETE" }).catch(() => {});
  }
  return failures;
}
