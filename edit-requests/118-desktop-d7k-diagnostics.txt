//! helping them. It is built field by field from an explicit allowlist;
//! nothing is copied wholesale. It never holds a credential value, a
//! conversation or its title, memory or knowledge contents, identity
//! records, the desktop launch key, an environment variable, a profile id
//! or a path: paths are shown as `<home>`, and URLs as their origin only.
//! Nothing is uploaded: the bundle is returned to the caller, who decides
//! where to save it.

/// The bundle's top-level fields, in order: the only ones it has.
pub(crate) const SECTIONS: [&str; 10] = [
    "schema",
    "about",
    "included",
    "excluded",
    "lightagent",
    "doctor",
    "architecture",
    "settings",
    "counts",
    "stores",

const INCLUDED: [&str; 6] = [
    "the Lightagent version, operating system and architecture",
    "doctor: whether the home exists, the provider's kind and origin, whether a profile is active, and how many profiles and built-in tools there are",
    "architecture: which platform components are configured, by origin (scheme, host and port) only",
    "settings: on/off switches and limits from the configuration, never a value that can hold a secret",
    "counts: profiles, saved providers, MCP servers, and the active profile's sessions, memories and indexed documents",
    "stores: whether the identity store and the credential vault exist, whether a vault key is configured and where it comes from, and how many records they hold",
];

const EXCLUDED: [&str; 7] = [

    "conversations, their titles and transcripts",
    "memory and knowledge contents",

    "the desktop app's launch key and environment variables",
    "profile names and ids",
    "file paths, shown as <home>; URLs are reduced to scheme, host and port",
];

/// A URL as its origin (scheme, host and port): never a user, password,
/// path, query or fragment. Anything that is not a URL is left out.
fn origin(url: &str) -> Value {
    let Some((scheme, rest)) = url.split_once("://") else {
        return Value::Null;
    };
    let authority = rest.split(['/', '?', '#']).next().unwrap_or_default();
    let host = authority
        .rsplit_once('@')
        .map_or(authority, |(_, host)| host);
    if scheme.is_empty() || host.is_empty() {
        return Value::Null;
    }
    Value::String(format!(
        "{}://{}",
        scheme.to_ascii_lowercase(),
        host.to_ascii_lowercase()
    ))
}

fn endpoint(endpoint: &PlatformEndpointConfig) -> Value {
    if endpoint.enabled {
        json!({
            "configured": true,
            "origin": endpoint.base_url.as_deref().map_or(Value::Null, origin),
        })
    } else {
        json!({ "configured": false })
    }
}

/// How many entries `field` of an operation's answer holds, when it worked.
fn count(answer: Result<Value, OpError>, field: &str) -> Option<usize> {
    answer
        .ok()
        .and_then(|value| value[field].as_array().map(Vec::len))
}

/// The bundle for the home at `root`.
pub(crate) async fn bundle(root: &Path) -> Result<Value, OpError> {
    let internal = |error: String| OpError::new(OpCategory::Internal, error);
    let paths = LightagentPaths::rooted_at(root);
    let config = ConfigStore::at(&paths)
        .load()
        .map_err(|error| internal(error.to_string()))?;
    let profiles = ProfileStore::new(root);
    let active = profiles
        .active()
        .map_err(|error| internal(error.to_string()))?;
    let listed = profiles
        .list()
        .map_err(|error| internal(error.to_string()))?;
    let sessions = active.as_ref().and_then(|id| {
        SessionStore::at_profile(&profiles.handle(id))
            .list()
            .ok()
            .map(|rows| rows.len())
    });
    let memories = count(
        crate::memory::perform(&paths, MemoryRequest::List).await,
        "memories",
    );
    let documents = count(
        crate::rag::perform(&paths, RagRequest::List).await,
        "sources",
    );
    let identity = crate::identity_admin::perform(root, IdentityRequest::Status)
        .map(|status| {
            json!({
                "store": status["store"],
                "multi_user": status["multi_user"],
                "principals": status["principals"],
                "credentials": status["credentials"],
            })
        })
        .unwrap_or_else(|_| json!({ "error": "the identity store could not be read" }));
    let vault = crate::remote_credentials::perform(root, RemoteCredentialRequest::Status)
        .map(|status| {
            json!({
                "vault": status["vault"],
                "key_configured": status["key_configured"],
                "key_source": status["key_source"],
                "grants": status["grants"],
            })
        })
        .unwrap_or_else(|_| json!({ "error": "the configured credential key is unusable" }));
    let platform = &config.platform;
    let searxng = config
        .web
        .search
        .endpoint
        .as_deref()
        .filter(|_| config.web.enabled)
        .map_or(Value::Null, origin);
    Ok(json!({
        "schema": "lightagent-diagnostics/1",
        "about": "A diagnostics bundle to share with someone helping you with Lightagent. It holds only what `included` lists; everything `excluded` lists is left out. It is not sent anywhere: you choose where to save it.",
        "included": INCLUDED,
        "excluded": EXCLUDED,
        "lightagent": {
            "version": crate::VERSION,
            "os": std::env::consts::OS,
            "arch": std::env::consts::ARCH,
        },
        "doctor": {
            "home": "<home>",
            "home_exists": root.exists(),
            "config_file": "<home>/config.json",
            "provider": config.inference.provider,
            "provider_origin": origin(&config.inference.base_url),
            "active_profile_set": active.is_some(),
            "profiles": listed.len(),
            "tools": lightagent_tools::ToolRegistry::builtin().names().len(),
        },
        "architecture": {
            "jev": endpoint(&platform.jev.endpoint),
            "qdrant": endpoint(&platform.qdrant.endpoint),
            "infinity": endpoint(&platform.infinity.endpoint),
            "open_terminal": endpoint(&platform.open_terminal.endpoint),
            "searxng": searxng,
        },
        "settings": {
            "tools_enabled": config.tools.enabled,
            "terminal_allowed": config.tools.allow_terminal,
            "web_enabled": config.web.enabled,
            "mcp_enabled": config.mcp.enabled,
            "extensions_enabled": config.extensions.enabled,
            "agent_max_turns": config.agent.max_turns,
            "agent_max_tool_calls": config.agent.max_tool_calls,
            "context_tokens": config.runtime.n_ctx,
        },
        "counts": {
            "profiles": listed.len(),
            "saved_providers": config.inference.saved_providers.len(),
            "mcp_servers": config.mcp.servers.len(),
            "active_profile_sessions": sessions,
            "active_profile_memories": memories,
            "active_profile_documents": documents,
        },
        "stores": {
            "identity": identity,
            "remote_credentials": vault,
        },
    }))
}

        .unwrap_or(0);
    let dir = std::env::temp_dir().join(format!(
        "lightagent-desktop-diagnostics-{label}-{}-{nanos}",
        std::process::id()
    ));
    std::fs::create_dir_all(&dir).expect("scratch dir");
    dir
}

fn first_line(child: &mut Child) -> String {
    let stdout = child.stdout.take().expect("stdout");
    let (tx, rx) = mpsc::channel();
    std::thread::spawn(move || {
        let mut reader = BufReader::new(stdout);
        let mut line = String::new();
        let _ = reader.read_line(&mut line);
        let _ = tx.send(line);
        let mut rest = String::new();
        while reader.read_line(&mut rest).is_ok_and(|read| read > 0) {
            rest.clear();
        }
    });
    rx.recv_timeout(Duration::from_secs(60))
        .expect("first line")
}

fn start_desktop(home: &Path, env: &[(&str, &str)]) -> Runtime {
    let mut command = Command::new(env!("CARGO_BIN_EXE_lightagent"));
    command
        .args(["serve", "--desktop", "--host", "127.0.0.1", "--port", "0"])
        .env("LIGHTAGENT_HOME", home)
        .stdin(Stdio::piped())
        .stdout(Stdio::piped())
        .stderr(Stdio::null());
    for (name, value) in env {
        command.env(name, value);
    }
    let mut child = command.spawn().expect("start lightagent serve --desktop");
    let mut lifeline = child.stdin.take().expect("stdin");
    writeln!(lifeline, "{KEY}").expect("key");
    lifeline.flush().expect("flush");
    let line = first_line(&mut child);
    let handshake: Value = serde_json::from_str(&line).expect("handshake JSON");
    let addr = handshake["addr"].as_str().expect("addr").to_owned();
    Runtime {
        child,
        _lifeline: Some(lifeline),
        addr,
    }
}

fn start_keyed(home: &Path) -> Runtime {
    let mut child = Command::new(env!("CARGO_BIN_EXE_lightagent"))
        .args(["serve", "--host", "127.0.0.1", "--port", "0"])
        .args(["--key-env", "LIGHTAGENT_TEST_DIAGNOSTICS_KEY"])
        .env("LIGHTAGENT_TEST_DIAGNOSTICS_KEY", KEY)
        .env("LIGHTAGENT_HOME", home)
        .stdin(Stdio::null())
        .stdout(Stdio::piped())
        .stderr(Stdio::null())
        .spawn()
        .expect("start lightagent serve");
    let line = first_line(&mut child);
    let addr = line
        .split("http://")
        .nth(1)
        .and_then(|rest| rest.split('/').next())
        .unwrap_or_else(|| panic!("no address in {line:?}"))
        .to_owned();
    Runtime {
        child,
        _lifeline: None,
        addr,
    }
}

fn call(runtime: &Runtime, method: &str, path: &str, body: Option<&str>) -> (u16, String) {
    let mut stream = TcpStream::connect(&runtime.addr).expect("connect");
    stream
        .set_read_timeout(Some(Duration::from_secs(60)))
        .expect("timeout");
    let body = body.unwrap_or("");
    let head = format!(
        "{method} {path} HTTP/1.1\r\nHost: {}\r\nAuthorization: Bearer {KEY}\r\nContent-Type: application/json\r\nContent-Length: {}\r\nConnection: close\r\n\r\n{body}",
        runtime.addr,
        body.len()
    );
    stream.write_all(head.as_bytes()).expect("send");
    let mut response = String::new();
    let _ = stream.read_to_string(&mut response);
    let status = response
        .split_whitespace()
        .nth(1)
        .and_then(|code| code.parse().ok())
        .unwrap_or(0);
    let raw = response
        .split_once("\r\n\r\n")
        .map(|(_, body)| body.to_owned())
        .unwrap_or_default();
    (status, raw)
}

fn parsed(raw: &str) -> Value {
    serde_json::from_str(raw).unwrap_or_else(|error| panic!("not JSON ({error}): {raw}"))
}

fn ok(runtime: &Runtime, method: &str, path: &str, body: Option<Value>) -> (Value, String) {
    let body = body.map(|body| body.to_string());
    let (status, raw) = call(runtime, method, path, body.as_deref());
    assert_eq!(status, 200, "{method} {path}: {raw}");
    (parsed(&raw), raw)
}

  const errors = [];
  page.on("pageerror", (error) => errors.push(error.message));
  const marker = `render-diagnostics-private-${Date.now()}`;
  let memory = "";
  let session = "";
  try {
    const doctor = await call(base, "GET", "/diagnostics/doctor");
    const catalog = await call(base, "GET", "/profiles");
    memory = (await call(base, "POST", "/memories", { text: `${marker} memory` })).id ?? "";
    session = (await call(base, "POST", "/sessions", {})).id ?? "";
    if (session) await call(base, "PATCH", `/sessions/${session}`, { title: `${marker} title` });

    await page.goto(`${base}/#/diagnostics`);
    const screen = page.getByTestId("diagnostics-screen");
    await screen.waitFor({ timeout: 15000 });
    await page.getByTestId("diagnostics-doctor").getByText(String(doctor.home), { exact: true }).waitFor({ timeout: 15000 });
    await page.getByTestId("diagnostics-architecture").waitFor({ timeout: 10000 });

    await screen.getByRole("button", { name: "Prepare bundle" }).click();
    const shown = page.getByTestId("diagnostics-bundle");
    await shown.getByText("Left out", { exact: true }).waitFor({ timeout: 15000 });
    await page.screenshot({ path: `${outDir}/diagnostics-bundle.png`, fullPage: true });

    const [download] = await Promise.all([
      page.waitForEvent("download", { timeout: 15000 }),
      screen.getByRole("button", { name: "Save bundle…" }).click(),
    ]);
    const file = await download.path();
    const text = await readFile(file, "utf8");
    const saved = JSON.parse(text);
    const sections = Object.keys(saved).sort();
    if (JSON.stringify(sections) !== JSON.stringify([...SECTIONS].sort())) {
      failures.push(`diagnostics: the saved bundle's sections are ${JSON.stringify(sections)}`);
    }
    const forbidden = [marker, "render-previous-secret", String(doctor.home), ...catalog.profiles.map((p) => `"${p.id}"`)];
    for (const item of forbidden) {
      if (text.includes(item)) failures.push(`diagnostics: the saved bundle holds ${JSON.stringify(item)}`);
    }
    if ((saved.counts?.active_profile_memories ?? 0) < 1) {
      failures.push(`diagnostics: the bundle counts no memory: ${JSON.stringify(saved.counts)}`);
    }
    await page.getByRole("status").filter({ hasText: "It was not sent anywhere." }).waitFor({ timeout: 10000 });
    if (errors.length) failures.push(`diagnostics: page errors: ${errors.join("; ")}`);
  } catch (err) {
    failures.push(`diagnostics: ${err instanceof Error ? err.message.split("\n")[0] : String(err)}`);
    await page.screenshot({ path: `${outDir}/diagnostics-failed.png`, fullPage: true }).catch(() => {});
  } finally {
    await page.close();
    if (memory) await call(base, "DELETE", `/memories/items/${memory}`).catch(() => {});
    if (session) await call(base, "DELETE", `/sessions/${session}`).catch(() => {});
  }
  return failures;
}
