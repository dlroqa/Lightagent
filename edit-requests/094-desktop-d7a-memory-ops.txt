D7-1: memory through one shared operation set for the CLI, the API and the desktop

`lightagent memory` worked on the active profile's memory file directly,
and nothing else could: the API had no memory route.

- crate lightagent memory: one implementation of list, add, update,
  forget, clear, search (hybrid recall with the configured semantic
  embedder), reflect, candidates and promote, returning typed results.
  The CLI prints them as before (text and --json unchanged), and
  memory::perform answers the API with exactly the CLI's --json. Writes go
  through MemoryStore, which locks and re-reads before each change, so the
  CLI and the API never overwrite each other. New: an empty text is
  refused for add and update.
- ops::MemoryRequest and RunFactory::memory (default: unavailable), as
  profiles do; `lightagent serve` performs it.
- API (operator): GET|POST|DELETE /memories, GET /memories/search
  (`q`, `top_k`), GET /memories/reflect (`topic`), GET
  /memories/candidates (`session`), POST /memories/promote,
  PUT|DELETE /memories/items/{id}. Contract memory.* with each query
  listed; the host refuses the writes during a profile switch (memory
  belongs to the active profile). Nine backend inventory rows (not
  counted; the CLI rows are counted with the UI slice).
- Tests (separate processes): what the API writes the CLI lists and the
  reverse; list, search and reflect over the API equal the CLI's --json;
  update and forget across both; candidates and a refused promote on a
  saved session equal; empty text refused; clear over the API empties the
  CLI's list.

fn active_memory() -> Result<(PathBuf, Config), String> {
    let paths = LightagentPaths::resolve().map_err(|error| error.to_string())?;
    let config = ConfigStore::at(&paths)
        .load()
        .map_err(|error| error.to_string())?;
    let store = ProfileStore::new(paths.root());
    let active = store
        .active()
        .map_err(|error| error.to_string())?
        .ok_or_else(|| "no active profile — run `lightagent init` first".to_owned())?;
    let dir = store.handle(&active).dir().to_path_buf();
    Ok((memory_path(&dir), config))
}

/// Show user statements that look durable; nothing is saved until `promote`.
pub fn candidates(session_id: String, json: bool) -> Result<(), String> {
    let session = saved_session(&session_id)?;
    let candidates: Vec<_> = session
        .messages
        .iter()
        .enumerate()
        .filter(|(_, message)| {
            if message.role != "user" {
                return false;
            }
            let text = message.content.to_lowercase();
            [
                "prefer",
                "remember",
                "we decided",
                "always",
                "never",
                "my ",
                "our project",
                "uses ",
            ]
            .iter()
            .any(|marker| text.contains(marker))
        })
        .map(|(index, message)| (index + 1, message.content.as_str()))
        .collect();
    if json {
        println!(
            "{}",
            serde_json::json!({"session_id": session_id, "candidates": candidates.iter().map(|(index, text)| serde_json::json!({"message": index, "text": text})).collect::<Vec<_>>()})
        );
    } else if candidates.is_empty() {
        println!("No suggested facts. Any user message can still be promoted by its number.");
    } else {
        for (index, text) in candidates {
            println!("{index}: {}", text.replace('\n', " "));
        }
    }
    Ok(())
}

/// Save exactly one user-selected message, optionally edited, with provenance.
pub fn promote(
    session_id: String,
    message_number: usize,
    text: Option<String>,
    kind: Option<String>,
    tags: Vec<String>,
    json: bool,
) -> Result<(), String> {
    let session = saved_session(&session_id)?;
    let message = session
        .messages
        .get(message_number.saturating_sub(1))
        .filter(|message| message_number > 0 && message.role == "user")
        .ok_or_else(|| "choose a user message number from the saved session".to_owned())?;
    let fact = text.as_deref().unwrap_or(&message.content);
    let (path, _) = active_memory()?;
    let mut store = MemoryStore::open(&path).map_err(|error| error.to_string())?;
    let id = store
        .write_sourced(
            fact,
            kind.as_deref().unwrap_or("fact"),
            tags,
            Some(MemorySource {
                session_id,
                message_index: message_number,
            }),
            &HashingEmbedder,
            now_secs(),
        )
        .map_err(|error| error.to_string())?;
    if json {
        println!("{}", serde_json::json!({"id": id}));
    } else {
        println!("remembered ({id})");
    }
    Ok(())
}

/// Render the structured working knowledge derived from the bank.
pub fn reflect(topic: Option<String>, json: bool) -> Result<(), String> {
    let (path, _) = active_memory()?;
    let store = MemoryStore::open(&path).map_err(|e| e.to_string())?;
    let page = store.knowledge_page(topic.as_deref(), 32_000);
    if json {
        println!(
            "{}",
            serde_json::json!({ "topic": topic, "knowledge": page })
        );
    } else {
        println!("{page}");
    }
    Ok(())

/// `memory search <query>` — the most relevant memories.
pub async fn search(query: String, top_k: Option<usize>, json: bool) -> Result<(), String> {
    let (path, config) = active_memory()?;
    let store = MemoryStore::open(&path).map_err(|error| error.to_string())?;
    let k = top_k.unwrap_or(config.memory.top_k).max(1);
    let semantic = crate::rag::semantic_embedder(&config);
    let hits = store
        .search_hybrid(&query, &HashingEmbedder, semantic.as_deref(), k)
        .await;
    if json {
        let value = serde_json::json!({
            "query": query,
            "memories": hits.iter().map(|m| serde_json::json!({
                "id": m.id, "kind": m.kind, "source": m.source, "text": m.text,
            })).collect::<Vec<_>>(),
        });
        println!("{value:#}");
        return Ok(());
    }
    if hits.is_empty() {
        println!("No relevant memories.");
        return Ok(());
    }
    for memory in hits {
        let source = memory
            .source
            .as_ref()
            .map(|source| format!(" [from {}#{}]", source.session_id, source.message_index))
            .unwrap_or_default();
        println!("({}) {}{source}", memory.kind, memory.text);
    }
    Ok(())
}

use lightagent_api::ops::{MemoryRequest, OpCategory, OpError};

/// The active profile's memory file and saved sessions, and the config.
struct Active {
    path: PathBuf,
    sessions: SessionStore,
    config: Config,
}

fn internal(error: impl std::fmt::Display) -> OpError {
    OpError::new(OpCategory::Internal, error.to_string())
}

fn active_at(paths: &LightagentPaths) -> Result<Active, OpError> {
    let config = ConfigStore::at(paths).load().map_err(internal)?;
    let store = ProfileStore::new(paths.root());
    let active = store.active().map_err(internal)?.ok_or_else(|| {
        OpError::new(
            OpCategory::Invalid,
            "no active profile — run `lightagent init` first",
        )
    })?;
    let handle = store.handle(&active);
    Ok(Active {
        path: memory_path(handle.dir()),
        sessions: SessionStore::at_profile(&handle),
        config,
    })
}

fn cli_active() -> Result<Active, String> {
    let paths = LightagentPaths::resolve().map_err(|error| error.to_string())?;
    active_at(&paths).map_err(|error| error.to_string())
}

fn open(active: &Active) -> Result<MemoryStore, OpError> {
    MemoryStore::open(&active.path).map_err(internal)
}

fn saved_session(active: &Active, raw_id: &str) -> Result<Session, OpError> {
    let id = SessionId::parse(raw_id)
        .map_err(|error| OpError::new(OpCategory::Invalid, error.to_string()))?;
    active
        .sessions
        .load(&id)
        .map_err(|error| OpError::new(OpCategory::NotFound, error.to_string()))
}

fn require_text(text: &str) -> Result<(), OpError> {
    if text.trim().is_empty() {
        return Err(OpError::new(OpCategory::Invalid, "a memory needs text"));
    }
    Ok(())
}

                "prefer",
                "remember",

                "never",

                "uses ",
            ]

        .collect())
}

fn op_promote(
    active: &Active,
    session_id: &str,
    message_number: usize,
    text: Option<&str>,
    kind: Option<&str>,
    tags: Vec<String>,
) -> Result<String, OpError> {
    let session = saved_session(active, session_id)?;
    let message = session
        .messages
        .get(message_number.saturating_sub(1))
        .filter(|message| message_number > 0 && message.role == "user")

/// One memory operation for the HTTP API (`RunFactory::memory`), on the
/// active profile's memories, answered with the JSON
/// `lightagent memory … --json` prints.
pub(crate) async fn perform(
    paths: &LightagentPaths,
    request: MemoryRequest,
) -> Result<serde_json::Value, OpError> {
    let active = active_at(paths)?;
    Ok(match request {
        MemoryRequest::List => list_value(&op_list(&active)?),
        MemoryRequest::Add { text, kind, tags } => {
            serde_json::json!({ "id": op_add(&active, &text, kind.as_deref(), tags)? })
        }
        MemoryRequest::Update { id, text } => {
            serde_json::json!({ "updated": op_update(&active, &id, &text)? })
        }
        MemoryRequest::Forget(id) => serde_json::json!({ "removed": op_forget(&active, &id)? }),
        MemoryRequest::Clear => {
            op_clear(&active)?;
            serde_json::json!({ "cleared": true })
        }
        MemoryRequest::Search { query, top_k } => {
            search_value(&query, &op_search(&active, &query, top_k).await?)
        }
        MemoryRequest::Reflect(topic) => serde_json::json!({
            "topic": topic,
            "knowledge": op_reflect(&active, topic.as_deref())?,
        }),
        MemoryRequest::Candidates(session) => {
            candidates_value(&session, &op_candidates(&active, &session)?)
        }
        MemoryRequest::Promote {
            session,
            message,
            text,
            kind,
            tags,
        } => serde_json::json!({
            "id": op_promote(&active, &session, message, text.as_deref(), kind.as_deref(), tags)?
        }),
    })
}

/// `memory search <query>` — the most relevant memories.
pub async fn search(query: String, top_k: Option<usize>, json: bool) -> Result<(), String> {
    let active = cli_active()?;
    let hits = op_search(&active, &query, top_k)
        .await
        .map_err(|error| error.to_string())?;
    if json {
        println!("{:#}", search_value(&query, &hits));
        return Ok(());
    }
    if hits.is_empty() {
        println!("No relevant memories.");
        return Ok(());
    }
    for memory in &hits {
        let source = memory
            .source
            .as_ref()
            .map(|source| format!(" [from {}#{}]", source.session_id, source.message_index))
            .unwrap_or_default();
        println!("({}) {}{source}", memory.kind, memory.text);
    }
    Ok(())
}

//! Memory through one shared operation set (D7-1): the desktop runtime's
//! API and `lightagent memory` act on the active profile's memories alike.
//! What one writes the other lists, and the API answers each read with the
//! JSON the CLI prints. Every process here is separate.

use std::io::{BufRead as _, BufReader, Read as _, Write as _};
use std::net::TcpStream;
use std::path::{Path, PathBuf};
use std::process::{Child, ChildStdin, Command, Output, Stdio};
use std::sync::mpsc;
use std::time::{Duration, SystemTime, UNIX_EPOCH};

use serde_json::{Value, json};

const KEY: &str = "92a3b4c5d6e7f8091a2b3c4d5e6f708192a3b4c5d6e7f8091a2b3c4d5e6f7081";
const MEMORIES: &str = "/api/lightagent/v1/memories";

struct Runtime {
    child: Child,
    _lifeline: ChildStdin,
    addr: String,
}

impl Drop for Runtime {
    fn drop(&mut self) {
        let _ = self.child.kill();
        let _ = self.child.wait();
    }
}

fn scratch_home(label: &str) -> PathBuf {
    let nanos = SystemTime::now()
        .duration_since(UNIX_EPOCH)
        .map(|elapsed| elapsed.as_nanos())
        .unwrap_or(0);
    let home = std::env::temp_dir().join(format!(
        "lightagent-desktop-memory-{label}-{}-{nanos}",
        std::process::id()
    ));
    std::fs::create_dir_all(&home).expect("home");
    home
}

fn start(home: &Path) -> Runtime {
    let mut child = Command::new(env!("CARGO_BIN_EXE_lightagent"))
        .args(["serve", "--desktop", "--host", "127.0.0.1", "--port", "0"])
        .env("LIGHTAGENT_HOME", home)
        .stdin(Stdio::piped())
        .stdout(Stdio::piped())
        .stderr(Stdio::null())
        .spawn()
        .expect("start lightagent serve --desktop");
    let mut lifeline = child.stdin.take().expect("stdin");
    writeln!(lifeline, "{KEY}").expect("key");
    lifeline.flush().expect("flush");
    let stdout = child.stdout.take().expect("stdout");
    let (tx, rx) = mpsc::channel();
    std::thread::spawn(move || {
        let mut line = String::new();
        let _ = BufReader::new(stdout).read_line(&mut line);
        let _ = tx.send(line);
    });
    let line = rx.recv_timeout(Duration::from_secs(60)).expect("handshake");
    let handshake: Value = serde_json::from_str(&line).expect("handshake JSON");
    let addr = handshake["addr"].as_str().expect("addr").to_owned();
    Runtime {
        child,
        _lifeline: lifeline,
        addr,
    }
}

/// One authenticated request: the status and the raw body.
fn call(runtime: &Runtime, method: &str, path: &str, body: Option<&str>) -> (u16, String) {
    let mut stream = TcpStream::connect(&runtime.addr).expect("connect");
    stream
        .set_read_timeout(Some(Duration::from_secs(60)))
        .expect("timeout");
    let body = body.unwrap_or("");
    let head = format!(
        "{method} {path} HTTP/1.1\r\nHost: {}\r\nAuthorization: Bearer {KEY}\r\nContent-Type: application/json\r\nContent-Length: {}\r\nConnection: close\r\n\r\n{body}",
        runtime.addr,
        body.len()
    );
    stream.write_all(head.as_bytes()).expect("send");
    let mut response = String::new();
    let _ = stream.read_to_string(&mut response);
    let status = response
        .split_whitespace()
        .nth(1)
        .and_then(|code| code.parse().ok())
        .unwrap_or(0);
    let raw = response
        .split_once("\r\n\r\n")
        .map(|(_, body)| body.to_owned())
        .unwrap_or_default();
    (status, raw)
}

    assert_eq!(by_cli["memories"][0]["id"], id.as_str(), "{by_cli}");
    assert_eq!(by_cli["memories"][0]["kind"], "preference");

    // Written by the CLI: the API lists it.
