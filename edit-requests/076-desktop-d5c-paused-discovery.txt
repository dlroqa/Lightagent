D5 close-out: a paused run is found from its session

A run paused on its time budget keeps its session's run lease until it is
continued or stopped. Until now, once the page that started it was closed
or reloaded, nothing tied the session to that run: GET /runs is
operator-only and names no session, and a new turn on the session got a
bare 409. The run held the session until the runtime stopped.

- SessionStore::journaled_run reads the run id from the session's
  interruption journal (written at admission, removed once recorded).
- GET /sessions/{id} adds `active_run` {id, status} while this process
  runs one of the session's turns that the caller may see (the same
  authorized_run check as GET /runs/{id}); otherwise the response is
  unchanged. The in-process 409 on a busy session names it the same way.
- The Agent screen follows such a run when it loads the session, and after
  a busy 409, so the Continue / Answer now / Stop controls come back.
- Tests: the run is found from the session and from the 409; an undefined
  continue mode is refused and the run stays paused; two simultaneous
  continues resume the run exactly once; browser: a reload while paused
  brings the card back and Answer now sends mode answer.

/// The turn of session `id` this process is running, as `{id, status}`, if
/// the caller may see that run: a run paused on its time budget or waiting
/// on an approval stays reachable from its session after the page that
/// started it is gone. A journal another process wrote names no run here.
async fn session_active_run(
    state: &AppState,
    headers: &HeaderMap,
    sessions: &SessionStore,
    id: &SessionId,
) -> Option<serde_json::Value> {
    let run_id = sessions.journaled_run(id)?;
    let (run, _) = authorized_run(state, headers, &run_id, Scope::RunsRead)
        .await
        .ok()?;
    let status = run.status().await;
    if status.is_terminal() {
        return None;
    }
    Some(json!({ "id": run.id(), "status": status }))
}

/// A saved session as `GET /sessions/{id}` shows it: with `active_run` while
/// this process runs one of its turns that the caller may see.
async fn session_view(
    state: &AppState,
    headers: &HeaderMap,
    sessions: &SessionStore,
    id: &SessionId,
    session: Session,
) -> Response {
    let Some(active) = session_active_run(state, headers, sessions, id).await else {
        return Json(session).into_response();
    };
    match serde_json::to_value(&session) {
        Ok(mut view) => {
            if let Some(fields) = view.as_object_mut() {
                fields.insert("active_run".to_owned(), active);
            }
            Json(view).into_response()
        }
        Err(error) => internal(&error.to_string()),
    }
}

#[test]
fn a_paused_run_is_found_from_its_session_and_resumes_once() {
    let (runtime, bodies) = setup("found");
    let session = new_session(&runtime);
    let run = start_run(&runtime, &session, true);
    wait_until("the run to pause", || {
        run_info(&runtime, &run)["status"] == "awaiting_continue"
    });
    let session_path = format!("/api/lightagent/v1/sessions/{session}");

    // A client that lost the page finds the paused run from its session: on
    // the session itself, and on the refusal of a new turn.
    let (status, raw) = call(&runtime, "GET", &session_path, None);
    assert_eq!(status, 200, "{raw}");
    let saved: Value = serde_json::from_str(&raw).expect("session JSON");
    assert_eq!(saved["active_run"]["id"], run.as_str(), "{saved}");
    assert_eq!(saved["active_run"]["status"], "awaiting_continue", "{saved}");
    let body = json!({ "message": "Something else.", "session_id": session }).to_string();
    let (status, raw) = call(&runtime, "POST", "/api/lightagent/v1/runs", Some(&body));
    assert_eq!(status, 409, "{raw}");
    let refused: Value = serde_json::from_str(&raw).expect("refusal JSON");
    assert_eq!(refused["active_run"]["id"], run.as_str(), "{refused}");

    // A mode the API does not define is refused; the run stays paused.
    let (status, raw) = continue_run(&runtime, &run, "forever");
    assert!(
        (400..500).contains(&status) && status != 409,
        "{status}: {raw}"
    );
    assert_eq!(run_info(&runtime, &run)["status"], "awaiting_continue");

    // Two clients continue at once: exactly one resumes the run.
    let mut both: Vec<u16> = std::thread::scope(|scope| {
        let first = scope.spawn(|| continue_run(&runtime, &run, "extend").0);
        let second = scope.spawn(|| continue_run(&runtime, &run, "extend").0);
        vec![
            first.join().expect("first continue"),
            second.join().expect("second continue"),
        ]
    });
    both.sort_unstable();
    assert_eq!(both, vec![200, 409]);
    wait_until("the run to finish", || {
        run_info(&runtime, &run)["status"] == "completed"
    });
    assert_eq!(
        bodies.lock().expect("bodies").len(),
        2,
        "the paused run was resumed exactly once"
    );

    // Finished, it is no longer named on its session.
    let (status, raw) = call(&runtime, "GET", &session_path, None);
    assert_eq!(status, 200, "{raw}");
    let saved: Value = serde_json::from_str(&raw).expect("session JSON");
    assert!(saved.get("active_run").is_none(), "{saved}");
}
