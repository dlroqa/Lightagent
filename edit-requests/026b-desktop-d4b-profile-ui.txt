import { useCallback, useEffect, useRef, useState, type ReactNode, type RefObject } from "react";

import { invoke, isDesktop } from "../api/transport";
import "./desktop-gate.css";

/** The desktop host's view of its bundled runtime (`runtime_status`). */
interface RuntimeStatus {
  state: "starting" | "ready" | "failed" | "mismatch" | "duplicate";
  detail: string | null;
  desktop_version: string;
  runtime_version: string | null;
  profile: string | null;
  /** The window was asked to close while runs were working. */
  close_requested?: boolean;
}

/** How often the gate re-checks the runtime once the app is running. */
const WATCH_MS = 1000;

/**
 * In the desktop app: hold the UI until the bundled runtime is ready, keep
 * watching it afterwards, and explain plainly — with a way out — when it
 * cannot start, stops, does not match, or another window already owns it.
 * In the browser build this renders its children directly.
 */
export function DesktopGate({ children }: { children: ReactNode }) {
  const [status, setStatus] = useState<RuntimeStatus | null>(null);
  const [hostError, setHostError] = useState<string | null>(null);
  const [everReady, setEverReady] = useState(false);
  const [restarting, setRestarting] = useState(false);
  const [logs, setLogs] = useState<string[] | null>(null);
  const [generation, setGeneration] = useState(0);
  const heading = useRef<HTMLHeadingElement>(null);

  useEffect(() => {
    if (!isDesktop) return;
    let stopped = false;
    let timer: number | undefined;
    const poll = async () => {
      try {
        const next = await invoke<RuntimeStatus>("runtime_status");
        if (stopped) return;
        setStatus(next);
        setHostError(null);
        if (next.state === "ready") {
          setEverReady(true);
          setRestarting(false);
          void invoke("e2e_note", { note: "ui_ready" }).catch(() => undefined);
        } else if (next.state !== "starting") {
          setRestarting(false);
          void invoke("e2e_note", { note: "ui_error" }).catch(() => undefined);
        }
        timer = window.setTimeout(() => void poll(), next.state === "starting" ? 250 : WATCH_MS);
      } catch (cause) {
        if (!stopped) setHostError(cause instanceof Error ? cause.message : String(cause));
      }
    };
    void poll();
    return () => {
      stopped = true;
      if (timer !== undefined) window.clearTimeout(timer);
    };
  }, [generation]);

  const failed =
    hostError !== null ||
    status?.state === "failed" ||
    status?.state === "mismatch" ||
    status?.state === "duplicate";

  useEffect(() => {
    if (failed) heading.current?.focus();
  }, [failed]);

  const restart = useCallback(async () => {
    setRestarting(true);
    setLogs(null);
    try {
      await invoke("runtime_restart");
    } catch (cause) {
      setHostError(cause instanceof Error ? cause.message : String(cause));
      setRestarting(false);
    }
    setGeneration((value) => value + 1);
  }, []);

  const showLogs = useCallback(async () => {
    try {
      setLogs(await invoke<string[]>("runtime_logs", { limit: 200 }));
    } catch {
      setLogs([]);
    }
  }, []);

  if (!isDesktop) return <>{children}</>;
  const ready = status?.state === "ready" && hostError === null;
  if (ready && status?.close_requested) {
    return (
      <>
        <div aria-hidden="true" inert>
          {children}
        </div>
        <div className="desktop-gate desktop-gate--overlay" role="presentation">
          <CloseWithRuns onKeep={() => setGeneration((value) => value + 1)} />
        </div>
      </>
    );
  }
  if (ready) return <>{children}</>;

  const panel = failed && !restarting ? (
    <FailurePanel
      status={status}
      hostError={hostError}
      heading={heading}
      logs={logs}
      onRestart={() => void restart()}
      onShowLogs={() => void showLogs()}
    />
  ) : (
    <div className="desktop-gate__panel" role="status" aria-live="polite">
      <span className="desktop-gate__pulse" aria-hidden="true" />
      <h1 className="desktop-gate__title">
        {everReady || restarting ? "Restarting Lightagent's runtime…" : "Starting Lightagent…"}
      </h1>
      <p className="desktop-gate__text">Starting the bundled runtime on this computer.</p>
    </div>
  );

  // After the app has been running, keep it mounted under the notice so
  // nothing on screen is lost while the runtime recovers.
  if (everReady) {
    return (
      <>
        <div aria-hidden="true" inert>
          {children}
        </div>
        <div className="desktop-gate desktop-gate--overlay" role="presentation">
          {panel}
        </div>
      </>
    );
  }
  return <main className="desktop-gate">{panel}</main>;
}

function FailurePanel(props: {
  status: RuntimeStatus | null;
  hostError: string | null;
  heading: RefObject<HTMLHeadingElement | null>;
  logs: string[] | null;
  onRestart: () => void;
  onShowLogs: () => void;
}) {
  const { status, hostError, heading, logs } = props;
  const state = status?.state;
  const title =
    state === "mismatch"
      ? "The bundled runtime does not match this app"
      : state === "duplicate"
        ? "Lightagent is already open"
        : "Lightagent's runtime is not running";
  const remedy =
    state === "mismatch"
      ? "Reinstall Lightagent Desktop so the app and its runtime come from the same release. Nothing was run."
      : state === "duplicate"
        ? "Switch to the window that is already open. Close this one; it has not started anything."
        : "Restart the runtime. If it stops again, the runtime log shows why.";
  return (
    <div className="desktop-gate__panel desktop-gate__panel--error" role="alert">
      <h1 className="desktop-gate__title" tabIndex={-1} ref={heading}>
        {title}
      </h1>
      <p className="desktop-gate__text">{hostError ?? status?.detail ?? "The runtime stopped before it was ready."}</p>
      <p className="desktop-gate__text desktop-gate__text--muted">{remedy}</p>
      <dl className="desktop-gate__facts">
        <dt>App</dt>
        <dd>{status?.desktop_version ?? "unknown"}</dd>
        <dt>Runtime</dt>
        <dd>{status?.runtime_version ?? "not reported"}</dd>
      </dl>
      {state !== "duplicate" && (
        <div className="desktop-gate__actions">
          <button type="button" className="btn btn--primary" onClick={props.onRestart}>
            Restart runtime
          </button>
          <button type="button" className="btn" onClick={props.onShowLogs} aria-expanded={logs !== null}>
            Runtime log
          </button>
        </div>
      )}
      {logs !== null && (
        <pre className="desktop-gate__log" tabIndex={0} aria-label="Runtime log, most recent lines last">
          {logs.length ? logs.join("\n") : "The runtime wrote nothing before it stopped."}
        </pre>
      )}
    </div>
  );
}

/** Asked to close while runs are working: stop them and quit, or keep working. */
function CloseWithRuns({ onKeep }: { onKeep: () => void }) {
  const [busy, setBusy] = useState(false);
  const [error, setError] = useState<string | null>(null);
  const keep = useRef<HTMLButtonElement>(null);
  useEffect(() => keep.current?.focus(), []);
  return (
    <div className="desktop-gate__panel" role="alertdialog" aria-modal="true" aria-labelledby="close-with-runs-title">
      <h1 className="desktop-gate__title" id="close-with-runs-title">
        Runs are still working
      </h1>
      <p className="desktop-gate__text">
        Quitting stops them. A run waiting for your approval is recorded as rejected.
      </p>
      {error && (
        <p className="desktop-gate__text" role="alert">
          {error}
        </p>
      )}
      <div className="desktop-gate__actions">
        <button
          type="button"
          ref={keep}
          className="btn btn--primary"
          onClick={() => {
            void invoke("app_close_cancel").finally(onKeep);
          }}
        >
          Keep working
        </button>
        <button
          type="button"
          className="btn btn--danger"
          disabled={busy}
          onClick={() => {
            setBusy(true);
            setError(null);
            invoke("app_quit", { cancelRuns: true }).catch((cause: unknown) => {
              setError(cause instanceof Error ? cause.message : String(cause));
              setBusy(false);
            });
          }}
        >
          {busy ? "Stopping runs…" : "Stop runs and quit"}
        </button>
      </div>
    </div>
  );
}

import { useCallback, useEffect, useId, useState, type FormEvent } from "react";

import { agentApi, type AgentProfileDetail, type ProfileCatalog } from "../api/agent";
import { invoke, isDesktop } from "../api/transport";
import { Card } from "../components/Card";
import { TopBar } from "../components/Shell";
import "./profiles.css";

/** What the desktop host answers when asked to switch profile. */
interface SwitchOutcome {
  status: "switched" | "blocked";
  runs: { id: string; status: string }[];
  persisted: boolean;
  profile: string;
}

const PROFILE_ID = /^[a-z0-9][a-z0-9_-]{0,63}$/;

function message(cause: unknown): string {
  const text = cause instanceof Error ? cause.message : String(cause);
  return text.replace(/^\d{3}: /, "");
}

/**
 * Profiles: the same operations as `lightagent profile …`, on the same store.
 * In the desktop app, switching restarts the bundled runtime on the chosen
 * profile (start the new one, then commit; docs/desktop/DESKTOP_ARCHITECTURE.md
 * §7). In the browser, it sets the active profile the CLI and later servers
 * use; the server already running keeps serving its own until it restarts.
 */
export function ProfilesScreen() {
  const [catalog, setCatalog] = useState<ProfileCatalog | null>(null);
  const [loadError, setLoadError] = useState<string | null>(null);
  const [selected, setSelected] = useState<string | null>(null);
  const [notice, setNotice] = useState<string | null>(null);

  const load = useCallback(async () => {
    try {
      setCatalog(await agentApi.profiles());
      setLoadError(null);
    } catch (cause) {
      setLoadError(message(cause));
    }
  }, []);

  useEffect(() => {
    void load();
  }, [load]);

  const active = catalog?.active_profile ?? null;
  return (
    <>
      <TopBar
        title="Profiles"
        subtitle={`Persona, model and routing, shared with the lightagent CLI.${active ? ` Serving: ${active}.` : ""}`}
      />
      <div className="page profiles-page">
        {notice && (
          <div className="notice notice--info" role="status">
            {notice}
          </div>
        )}
        <Card title="Saved profiles">
          {loadError && (
            <p className="form-error" role="alert">
              Profiles could not be loaded: {loadError}
            </p>
          )}
          {!catalog && !loadError && <p className="muted">Loading profiles…</p>}
          {catalog && (
            <ul className="profile-list" aria-label="Saved profiles">
              {catalog.profiles.map((profile) => (
                <ProfileRow
                  key={profile.id}
                  id={profile.id}
                  name={profile.name}
                  model={profile.model}
                  active={profile.active}
                  onSelect={() => setSelected(profile.id)}
                  onChanged={(text) => {
                    setNotice(text);
                    void load();
                  }}
                />
              ))}
            </ul>
          )}
        </Card>
        {selected && <ProfileDetail id={selected} onChanged={() => void load()} />}
        <CreateProfile
          onCreated={(id) => {
            setNotice(`Created profile '${id}'.`);
            setSelected(id);
            void load();
          }}
        />
      </div>
    </>
  );
}

function ProfileRow(props: {
  id: string;
  name: string;
  model: string;
  active: boolean;
  onSelect: () => void;
  onChanged: (notice: string) => void;
}) {
  const { id, name, model, active } = props;
  const [busy, setBusy] = useState(false);
  const [error, setError] = useState<string | null>(null);
  const [blocked, setBlocked] = useState<SwitchOutcome["runs"] | null>(null);
  const [confirming, setConfirming] = useState(false);
  const [typed, setTyped] = useState("");
  const confirmId = useId();

  const use = async (cancelRuns: boolean) => {
    setBusy(true);
    setError(null);
    try {
      if (isDesktop) {
        const outcome = await invoke<SwitchOutcome>("profile_switch", { id, cancelRuns });
        if (outcome.status === "blocked") {
          setBlocked(outcome.runs);
          return;
        }
        setBlocked(null);
        // Everything on screen came from the previous runtime: start fresh.
        window.location.reload();
        return;
      }
      await agentApi.useProfile(id);
      props.onChanged(
        `'${id}' is now the active profile for the CLI and for servers started from now on. This server keeps serving its current profile until it restarts.`,
      );
    } catch (cause) {
      setError(message(cause));
    } finally {
      setBusy(false);
    }
  };

  const remove = async () => {
    setBusy(true);
    setError(null);
    try {
      await agentApi.deleteProfile(id);
      props.onChanged(`Deleted profile '${id}' and its local data.`);
    } catch (cause) {
      setError(message(cause));
    } finally {
      setBusy(false);
    }
  };

  return (
    <li className="profile-row">
      <div>
        <strong>{name}</strong>
        <span>
          {id} · {model}
        </span>
      </div>
      <div className="profile-row__actions">
        {active ? (
          <span className="status-chip">Active</span>
        ) : (
          <button type="button" className="btn" disabled={busy} onClick={() => void use(false)}>
            Use
          </button>
        )}
        <button type="button" className="btn btn--ghost" onClick={props.onSelect}>
          Details
        </button>
        {id !== "default" && !active && (
          <button
            type="button"
            className="btn btn--danger"
            aria-expanded={confirming}
            onClick={() => setConfirming((value) => !value)}
          >
            Delete…
          </button>
        )}
      </div>
      {blocked && (
        <div className="profile-row__confirm" role="alertdialog" aria-label={`Switch to ${id}`}>
          <p>
            {blocked.length === 1 ? "A run is" : `${blocked.length} runs are`} still working in this profile. Switching stops
            them; a pending approval is recorded as rejected.
          </p>
          <div className="profile-row__actions">
            <button type="button" className="btn btn--danger" disabled={busy} onClick={() => void use(true)}>
              Stop the runs and switch
            </button>
            <button type="button" className="btn" onClick={() => setBlocked(null)}>
              Keep working
            </button>
          </div>
        </div>
      )}
      {confirming && (
        <div className="profile-row__confirm">
          <label htmlFor={confirmId}>
            Deleting removes the profile and all of its sessions, memory and settings. Type <code>{id}</code> to confirm.
          </label>
          <input
            id={confirmId}
            className="input"
            value={typed}
            autoComplete="off"
            onChange={(event) => setTyped(event.target.value)}
          />
          <div className="profile-row__actions">
            <button type="button" className="btn btn--danger" disabled={busy || typed !== id} onClick={() => void remove()}>
              Delete {id}
            </button>
            <button type="button" className="btn" onClick={() => setConfirming(false)}>
              Cancel
            </button>
          </div>
        </div>
      )}
      {error && (
        <p className="form-error" role="alert">
          {error}
        </p>
      )}
    </li>
  );
}

function ProfileDetail({ id, onChanged }: { id: string; onChanged: () => void }) {
  const [detail, setDetail] = useState<AgentProfileDetail | null>(null);
  const [error, setError] = useState<string | null>(null);
  const [preset, setPreset] = useState("research@1");
  const [busy, setBusy] = useState(false);
  const [applied, setApplied] = useState<string | null>(null);
  const presetId = useId();

  useEffect(() => {
    setDetail(null);
    setError(null);
    setApplied(null);
    agentApi
      .profile(id)
      .then(setDetail)
      .catch((cause: unknown) => setError(message(cause)));
  }, [id]);

  const apply = async (event: FormEvent) => {
    event.preventDefault();
    setBusy(true);
    setError(null);
    try {
      await agentApi.presetProfile(id, preset.trim());
      setApplied(`Applied ${preset.trim()} to '${id}'.`);
      setDetail(await agentApi.profile(id));
      onChanged();
    } catch (cause) {
      setError(message(cause));
    } finally {
      setBusy(false);
    }
  };

  return (
    <Card title={`Profile: ${id}`}>
      {!detail && !error && <p className="muted">Loading…</p>}
      {detail && (
        <dl className="profile-detail">
          <dt>Name</dt>
          <dd>{detail.name}</dd>
          <dt>Model</dt>
          <dd>{detail.routing?.model ?? "default"}</dd>
          <dt>Persona</dt>
          <dd className="profile-detail__persona">{detail.persona}</dd>
        </dl>
      )}
      <form className="profile-form" onSubmit={(event) => void apply(event)}>
        <label htmlFor={presetId}>Capability preset (operator-owned)</label>
        <div className="profile-form__row">
          <input
            id={presetId}
            className="input"
            list={`${presetId}-known`}
            value={preset}
            onChange={(event) => setPreset(event.target.value)}

      {error && (
        <p className="form-error" role="alert">
          {error}
        </p>
      )}

function CreateProfile({ onCreated }: { onCreated: (id: string) => void }) {
  const [id, setId] = useState("");
  const [name, setName] = useState("");
  const [model, setModel] = useState("");
  const [persona, setPersona] = useState("");
  const [busy, setBusy] = useState(false);
  const [error, setError] = useState<string | null>(null);
  const ids = { id: useId(), name: useId(), model: useId(), persona: useId(), hint: useId() };
  const idValid = PROFILE_ID.test(id);

  const submit = async (event: FormEvent) => {
    event.preventDefault();
    if (!idValid) {
      setError("Use lowercase letters, digits, '-' and '_', starting with a letter or digit.");
      return;
    }
    setBusy(true);
    setError(null);
    try {
      await agentApi.createProfile({
        id,
        name: name.trim() || undefined,
        model: model.trim() || undefined,
        persona: persona.trim() || undefined,
      });
      onCreated(id);
      setId("");
      setName("");
      setModel("");
      setPersona("");
    } catch (cause) {
      setError(message(cause));
    } finally {
      setBusy(false);
    }
  };

  return (
    <Card title="Create a profile">
      <form className="profile-form" onSubmit={(event) => void submit(event)} noValidate>
        <label htmlFor={ids.id}>
          Profile id <span aria-hidden="true">*</span>
        </label>
        <input
          id={ids.id}
          className="input"
          required
          aria-required="true"
          aria-invalid={id !== "" && !idValid}
          aria-describedby={ids.hint}
          autoComplete="off"
          value={id}
          onChange={(event) => setId(event.target.value)}
        />
        <p id={ids.hint} className="card__note">
          Lowercase letters, digits, '-' and '_'. It names the profile's folder and cannot be changed later.
        </p>
        <label htmlFor={ids.name}>Display name</label>
        <input id={ids.name} className="input" value={name} onChange={(event) => setName(event.target.value)} />
        <label htmlFor={ids.model}>Model</label>
        <input
          id={ids.model}
          className="input"
          placeholder="default"
          value={model}
          onChange={(event) => setModel(event.target.value)}
        />
        <label htmlFor={ids.persona}>Persona</label>
        <textarea
          id={ids.persona}
          className="input"
          rows={3}
          placeholder="You are a helpful local agent."
          value={persona}
          onChange={(event) => setPersona(event.target.value)}
        />
        {error && (
          <p className="form-error" role="alert">
            {error}
          </p>
        )}
        <div className="profile-form__row">
          <button type="submit" className="btn btn--primary" disabled={busy}>
            {busy ? "Creating…" : "Create profile"}
          </button>
        </div>
      </form>
    </Card>
  );
}

//! Switching profile: start the new runtime, then commit
//! (docs/desktop/DESKTOP_ARCHITECTURE.md §7).
//!
//! 1. While runs are working the switch is refused, unless the person chose
//!    to stop them: pending approvals are then rejected explicitly (recorded,
//!    not lost), the runs are cancelled, and the switch goes ahead only once
//!    none is left.
//! 2. A second runtime starts on the chosen profile with its own key while
//!    the first keeps serving. If it does not become ready it is stopped, and
//!    nothing has changed.
//! 3. Commit: make the profile active through the new runtime (the CLI sees
//!    the same choice), route the app to it, end the old runtime's streams
//!    with `profile_switch`, and stop the old runtime.
//!
//! The same steps stop runs before the app quits.

use std::sync::Arc;
use std::sync::atomic::Ordering;
use std::time::{Duration, Instant};

use serde::Serialize;
use serde_json::{Value, json};

use crate::{Host, Runtime, lock, sidecar};

/// How long cancelled runs may take to reach a terminal state.
const DRAIN_TIMEOUT: Duration = Duration::from_secs(10);

#[derive(Debug, Serialize)]
pub struct Outcome {
    pub status: &'static str,
    pub runs: Vec<Value>,
    pub persisted: bool,
    pub profile: String,
}

/// What must happen before the runtime may be replaced.
#[derive(Debug, PartialEq, Eq)]
pub enum Plan {
    Proceed,
    Blocked,
    StopRunsThenProceed,
}

pub fn plan(active_runs: usize, stop_runs: bool) -> Plan {
    match (active_runs, stop_runs) {
        (0, _) => Plan::Proceed,
        (_, false) => Plan::Blocked,
        (_, true) => Plan::StopRunsThenProceed,
    }
}

/// One switch at a time.
struct Switching<'a>(&'a Host);

impl<'a> Switching<'a> {
    fn begin(host: &'a Host) -> Result<Self, String> {
        if host.switching.swap(true, Ordering::AcqRel) {
            return Err("a profile switch is already in progress".to_owned());
        }
        Ok(Self(host))
    }
}

impl Drop for Switching<'_> {
    fn drop(&mut self) {
        self.0.switching.store(false, Ordering::Release);
    }
}

async fn call(
    host: &Host,
    base: &str,
    authorization: &str,
    method: reqwest::Method,
    path: &str,
    body: Option<Value>,
) -> Result<Value, String> {
    let mut request = host
        .http
        .request(method, format!("{base}{path}"))
        .header(reqwest::header::AUTHORIZATION, authorization);
    if let Some(body) = body {
        request = request
            .header(reqwest::header::CONTENT_TYPE, "application/json")
            .body(body.to_string());
    }
    let response = request
        .send()
        .await
        .map_err(|error| format!("the runtime did not answer: {}", error.without_url()))?;
    let status = response.status();
    let text = response
        .text()
        .await
        .map_err(|error| format!("the runtime's answer was cut off: {}", error.without_url()))?;
    let value: Value = serde_json::from_str(&text).unwrap_or(Value::Null);
    if status.is_success() {
        return Ok(value);
    }
    let message = value["error"]["message"]
        .as_str()
        .or_else(|| value["error"].as_str())
        .unwrap_or("the request failed");
    Err(format!("{message} ({})", status.as_u16()))
}

fn runs_of(value: &Value) -> Vec<Value> {
    value["runs"].as_array().cloned().unwrap_or_default()
}

/// The runs the current runtime is still working on.
pub async fn active_runs(host: &Host) -> Result<Vec<Value>, String> {
    let (base, authorization) = host.connection()?;
    let runs = call(host, &base, &authorization, reqwest::Method::GET, "/api/lightagent/v1/runs", None).await?;
    Ok(runs_of(&runs))
}

/// Reject pending approvals, cancel every run, and wait until none is left.
pub async fn stop_runs(host: &Host) -> Result<(), String> {
    let (base, authorization) = host.connection()?;
    let approvals = call(host, &base, &authorization, reqwest::Method::GET, "/api/lightagent/v1/approvals", None).await?;
    for row in approvals["approvals"].as_array().into_iter().flatten() {
        if row["pending"].is_null() {
            continue;
        }
        if let Some(run) = row["run"].as_str() {
            call(
                host,
                &base,
                &authorization,
                reqwest::Method::POST,
                &format!("/api/lightagent/v1/approvals/{run}"),
                Some(json!({ "approve": false })),
            )
            .await?;
        }
    }
    let runs = call(host, &base, &authorization, reqwest::Method::GET, "/api/lightagent/v1/runs", None).await?;
    for run in runs_of(&runs) {
        if let Some(id) = run["id"].as_str() {
            // A run may finish on its own meanwhile; the drain below decides.
            let _ = call(
                host,
                &base,
                &authorization,
                reqwest::Method::POST,
                &format!("/api/lightagent/v1/runs/{id}/cancel"),
                Some(json!({})),
            )
            .await;
        }
    }
    let deadline = Instant::now() + DRAIN_TIMEOUT;
    loop {
        let runs = call(host, &base, &authorization, reqwest::Method::GET, "/api/lightagent/v1/runs", None).await?;
        if runs_of(&runs).is_empty() {
            return Ok(());
        }
        if Instant::now() >= deadline {
            return Err(format!(
                "the runs did not stop within {} s; nothing was changed",
                DRAIN_TIMEOUT.as_secs()
            ));
        }
        tokio::time::sleep(Duration::from_millis(200)).await;
    }
}

/// End the old runtime's relayed streams; their next poll reports the reason.
fn end_streams(host: &Host) {
    let subscriptions: Vec<_> = lock(&host.subscriptions).values().cloned().collect();
    for subscription in subscriptions {
        if let Some(task) = lock(&subscription.task).take() {
            task.abort();
        }
        {
            let mut state = lock(&subscription.state);
            state.closed = true;
            state.error = Some("profile_switch".to_owned());
        }
        subscription.notify.notify_waiters();
    }
}

pub async fn switch(host: &Arc<Host>, id: String, stop: bool) -> Result<Outcome, String> {
    // The id must be one the contract accepts as a path segment.
    host.contract
        .resolve("PUT", &format!("/api/lightagent/v1/profiles/{id}/active"))?;
    let _switching = Switching::begin(host)?;
    let runs = active_runs(host).await?;
    match plan(runs.len(), stop) {
        Plan::Blocked => {
            return Ok(Outcome { status: "blocked", runs, persisted: false, profile: id });
        }
        Plan::StopRunsThenProceed => stop_runs(host).await?,
        Plan::Proceed => {}
    }

    let binary = sidecar::bundled_binary()?;
    let profile = id.clone();
    let launched = tauri::async_runtime::spawn_blocking(move || {
        sidecar::launch(&binary, Some(&profile), crate::VERSION)
    })
    .await
    .map_err(|error| format!("could not start the runtime: {error}"))?;
    let fresh = launched.map_err(|error| match error {
        sidecar::LaunchError::Failed(reason) => {
            format!("the runtime for '{id}' did not start: {reason}. Nothing was changed.")
        }
        sidecar::LaunchError::Mismatch { expected, found } => {
            format!("the runtime reported {found}, not {expected}. Nothing was changed.")
        }
    })?;

    let persisted = call(
        host,
        &fresh.base_url(),
        &fresh.authorization(),
        reqwest::Method::PUT,
        &format!("/api/lightagent/v1/profiles/{id}/active"),
        None,
    )
    .await
    .is_ok();
    let previous = std::mem::replace(&mut *lock(&host.runtime), Runtime::Ready(fresh));
    end_streams(host);
    // The old runtime's lifeline closes here; let its grace period pass off the UI's path.
    std::thread::spawn(move || drop(previous));
    host.e2e.note_once("PROFILE_SWITCHED", &id);
    Ok(Outcome { status: "switched", runs: Vec::new(), persisted, profile: id })
}

            Runtime::Ready(sidecar) => Ok((sidecar.base_url(), sidecar.authorization())),

            Runtime::Failed(reason) => Err(format!("the runtime is unavailable: {reason}")),

/// Unattended-test evidence: when `LIGHTAGENT_DESKTOP_E2E_REPORT` names a
/// file, milestones are appended to it. Nothing secret is ever written.
struct E2e {
    path: Option<PathBuf>,
    started: Instant,
    seen: Mutex<HashSet<String>>,
}

impl E2e {
    fn from_env() -> Self {
        Self {
            path: std::env::var_os("LIGHTAGENT_DESKTOP_E2E_REPORT").map(PathBuf::from),
            started: Instant::now(),
            seen: Mutex::default(),
        }
    }

    fn note_once(&self, tag: &str, detail: &str) {
        let Some(path) = &self.path else { return };
        if !lock(&self.seen).insert(format!("{tag} {detail}")) {
            return;
        }
        let line = format!(
            "{tag} {detail} t_ms={}\n",
            self.started.elapsed().as_millis()
        );
        if let Ok(mut file) = std::fs::OpenOptions::new()
            .create(true)
            .append(true)
            .open(path)
        {
            let _ = file.write_all(line.as_bytes());
        }
    }
}

        Runtime::Duplicate => {
            status.state = "duplicate";
            status.detail = Some(
                "Lightagent Desktop is already open. Use that window; this one does not start a second runtime on the same data."
                    .to_owned(),
            );
        }
    }

/// Restart the runtime on request (a crashed or failed one, or a running one).
#[tauri::command]
fn runtime_restart(host: tauri::State<'_, Arc<Host>>) -> Result<(), String> {
    let previous = {
        let mut runtime = lock(&host.runtime);
        if matches!(*runtime, Runtime::Starting | Runtime::Duplicate) {
            return Err("the runtime cannot be restarted right now".to_owned());
        }
        std::mem::replace(&mut *runtime, Runtime::Starting)
    };
    let host = Arc::clone(host.inner());
    std::thread::spawn(move || {
        // Stopping the old runtime may take its grace period; not on the UI's thread.
        drop(previous);
        start_runtime(&host, true);
    });
    Ok(())
}

fn valid_filename(name: &str) -> bool {
    !name.is_empty()
        && name.len() <= 180
        && name != "."
        && name != ".."
        && name
            .chars()
            .all(|c| c.is_ascii_alphanumeric() || matches!(c, '.' | '_' | ' ' | '-'))
}

#[tauri::command]
async fn api_fetch(
    host: tauri::State<'_, Arc<Host>>,
    method: String,
    target: String,
    body: Option<String>,
    body_base64: Option<String>,
    filename: Option<String>,
) -> Result<FetchResult, String> {
    let operation = host.contract.resolve(&method, &target)?.clone();
    if operation.stream {
        return Err(format!(
            "{} is a stream; subscribe to it instead",
            operation.id
        ));
    }
    let (base, authorization) = host.connection()?;
    let verb =
        reqwest::Method::from_bytes(method.as_bytes()).map_err(|_| "invalid method".to_owned())?;
    let mut request = host
        .http
        .request(verb, format!("{base}{target}"))
        .header(reqwest::header::AUTHORIZATION, authorization);
    if operation.raw_body {
        let bytes = base64::engine::general_purpose::STANDARD
            .decode(body_base64.unwrap_or_default())
            .map_err(|_| "the attachment is not valid base64".to_owned())?;
        let filename = filename
            .filter(|name| valid_filename(name))
            .ok_or("an attachment needs a plain file name")?;
        request = request
            .header("X-Lightagent-Filename", filename)
            .body(bytes);
    } else if let Some(body) = body {
        request = request
            .header(reqwest::header::CONTENT_TYPE, "application/json")
            .body(body);
    }
    let response = request
        .send()
        .await
        .map_err(|error| format!("the runtime did not answer: {}", error.without_url()))?;
    let status = response.status().as_u16();
    let content_type = response
        .headers()
        .get(reqwest::header::CONTENT_TYPE)
        .and_then(|value| value.to_str().ok())
        .unwrap_or("")
        .to_owned();
    let body = response
        .text()
        .await
        .map_err(|error| format!("the runtime's answer was cut off: {}", error.without_url()))?;
    if (200..300).contains(&status) {
        host.e2e.note_once("API_OK", &operation.id);
    }
    Ok(FetchResult {
        status,
        content_type,
        body,
    })
}

#[tauri::command]
async fn api_subscribe(host: tauri::State<'_, Arc<Host>>, target: String) -> Result<u64, String> {
    let operation = host.contract.resolve("GET", &target)?;
    if !operation.stream {
        return Err(format!("{} is not a stream", operation.id));
    }
    let (base, authorization) = host.connection()?;
    let request = host
        .http
        .get(format!("{base}{target}"))
        .header(reqwest::header::AUTHORIZATION, authorization)
        .header(reqwest::header::ACCEPT, "text/event-stream");
    let id = host.next_subscription.fetch_add(1, Ordering::Relaxed);
    let subscription = Arc::new(Subscription::default());
    lock(&host.subscriptions).insert(id, Arc::clone(&subscription));
    let relay = Arc::clone(&subscription);
    let task = tauri::async_runtime::spawn(async move {
        let outcome = relay_stream(request, &relay).await;
        {
            let mut state = lock(&relay.state);
            state.closed = true;
            state.error = outcome.err();
        }
        relay.notify.notify_waiters();
    });
    *lock(&subscription.task) = Some(task);
    Ok(id)
}

async fn relay_stream(
    request: reqwest::RequestBuilder,
    subscription: &Subscription,
) -> Result<(), String> {
    let response = request
        .send()
        .await
        .map_err(|error| format!("the runtime did not answer: {}", error.without_url()))?;
    if !response.status().is_success() {
        return Err(format!(
            "the runtime refused the stream ({})",
            response.status().as_u16()
        ));
    }
    let mut parser = sse::SseParser::default();
    let mut stream = response.bytes_stream();
    while let Some(chunk) = stream.next().await {
        let chunk = chunk.map_err(|error| format!("the stream broke: {}", error.without_url()))?;
        let events = parser.push(&chunk);
        if !events.is_empty() {
            lock(&subscription.state).events.extend(events);
            subscription.notify.notify_waiters();
        }
    }
    Ok(())
}

#[derive(Serialize)]
struct PollResult {
    events: Vec<sse::SseEvent>,
    closed: bool,
    error: Option<String>,
}

fn drain(state: &mut SubscriptionState) -> PollResult {
    let events: Vec<_> = state.events.drain(..).collect();
    PollResult {
        events,
        closed: state.closed,
        error: state.error.clone(),
    }
}

#[tauri::command]
async fn api_events_poll(
    host: tauri::State<'_, Arc<Host>>,
    subscription: u64,
    wait_ms: u64,
) -> Result<PollResult, String> {
    let entry = lock(&host.subscriptions)
        .get(&subscription)
        .cloned()
        .ok_or("unknown subscription")?;
    // Register for a wake-up before looking, so an event that lands in
    // between is never missed.
    let mut notified = std::pin::pin!(entry.notify.notified());
    notified.as_mut().enable();
    let ready = {
        let state = lock(&entry.state);
        !state.events.is_empty() || state.closed
    };
    if !ready {
        let _ = tokio::time::timeout(Duration::from_millis(wait_ms).min(MAX_POLL), notified).await;
    }
    let result = drain(&mut lock(&entry.state));
    if result.closed {
        lock(&host.subscriptions).remove(&subscription);
    }
    Ok(result)
}

#[tauri::command]
fn api_unsubscribe(host: tauri::State<'_, Arc<Host>>, subscription: u64) {
    if let Some(entry) = lock(&host.subscriptions).remove(&subscription)
        && let Some(task) = lock(&entry.task).take()
    {
        task.abort();
    }
}

fn start_runtime(host: &Host, restart: bool) {
    let launched = sidecar::bundled_binary()
        .map_err(sidecar::LaunchError::Failed)
        .and_then(|binary| sidecar::launch(&binary, None, VERSION));
    let next = match launched {
        Ok(sidecar) => {
            host.e2e.note_once(
                if restart {
                    "RUNTIME_RESTARTED"
                } else {
                    "RUNTIME_READY"
                },
                &format!("version={}", sidecar.handshake.version),
            );
            Runtime::Ready(sidecar)
        }
        Err(sidecar::LaunchError::Failed(reason)) => {
            host.e2e.note_once("RUNTIME_FAILED", "");
            Runtime::Failed(reason)
        }
        Err(sidecar::LaunchError::Mismatch { expected, found }) => {
            host.e2e.note_once(
                "HANDSHAKE_MISMATCH",
                &format!("expected={expected} found={found}"),
            );
            Runtime::Mismatch { expected, found }
        }
    };
    *lock(&host.runtime) = next;
}

/// Crash detection: a runtime that exits while the host still wants it is
/// reported, its output kept, and it is restarted at most once per
/// [`AUTO_RESTART_WINDOW`].
fn watch_runtime(host: &Host) {
    loop {
        std::thread::sleep(WATCH_INTERVAL);
        let exited = match &mut *lock(&host.runtime) {
            Runtime::Ready(sidecar) => sidecar.exit_status().map(|status| {
                let logs: Vec<String> = lock(&sidecar.logs).iter().cloned().collect();
                (status, logs)
            }),
            _ => None,
        };
        let Some((status, logs)) = exited else {
            continue;
        };
        host.e2e.note_once("RUNTIME_EXITED", "");
        *lock(&host.last_logs) = logs;
        *lock(&host.runtime) =
            Runtime::Failed(format!("the runtime stopped unexpectedly ({status})"));
        let allowed = {
            let mut last = lock(&host.last_auto_restart);
            let allowed = last.is_none_or(|at| at.elapsed() >= AUTO_RESTART_WINDOW);
            if allowed {
                *last = Some(Instant::now());
            }
            allowed
        };
        if allowed {
            *lock(&host.runtime) = Runtime::Starting;
            start_runtime(host, true);
        }
    }
}

/// Claim this app's single-instance lock. `Ok(None)`: another instance holds it.
fn claim_instance(dir: Option<PathBuf>) -> Result<Option<std::fs::File>, String> {
    let dir = dir.ok_or("the app has no data directory")?;
    std::fs::create_dir_all(&dir).map_err(|error| error.to_string())?;
    let file = std::fs::OpenOptions::new()
        .create(true)
        .truncate(false)
        .write(true)
        .open(dir.join("instance.lock"))
        .map_err(|error| error.to_string())?;
    match file.try_lock() {
        Ok(()) => Ok(Some(file)),
        Err(std::fs::TryLockError::WouldBlock) => Ok(None),
        Err(std::fs::TryLockError::Error(error)) => Err(error.to_string()),
    }
}

fn main() {
    let contract = match contract::Contract::load() {
        Ok(contract) => contract,
        Err(error) => {
            eprintln!("{error}");
            std::process::exit(1);
        }
    };
    // Loopback only: never through a proxy from the environment, and never
    // following a redirect that would carry the credential elsewhere.
    let http = match reqwest::Client::builder()
        .no_proxy()
        .redirect(reqwest::redirect::Policy::none())
        .connect_timeout(Duration::from_secs(5))
        .build()
    {
        Ok(http) => http,
        Err(error) => {
            eprintln!("could not build the HTTP client: {error}");
            std::process::exit(1);
        }
    };
    let host = Arc::new(Host {
        contract,
        runtime: Mutex::new(Runtime::Starting),
        http,
        subscriptions: Mutex::default(),
        next_subscription: AtomicU64::new(1),
        e2e: E2e::from_env(),
        last_auto_restart: Mutex::new(None),
        last_logs: Mutex::default(),
        instance: Mutex::new(None),
        switching: AtomicBool::new(false),
        close_requested: AtomicBool::new(false),
    });
    let app = tauri::Builder::default()
        .manage(Arc::clone(&host))
        .invoke_handler(tauri::generate_handler![
            runtime_status,
            runtime_logs,
            runtime_restart,
            profile_switch,
            app_quit,
            app_close_cancel,
            api_fetch,
            api_subscribe,
            api_events_poll,
            api_unsubscribe,
            e2e_note
        ])
        .on_page_load(|webview, payload| {
            if matches!(payload.event(), tauri::webview::PageLoadEvent::Finished)
                && let Some(host) = webview.try_state::<Arc<Host>>()
            {
                host.e2e.note_once("PAGE_LOADED", "");
            }
        })
        .setup(|app| {
            let host = Arc::clone(app.state::<Arc<Host>>().inner());
            match claim_instance(app.path().app_local_data_dir().ok()) {
                Ok(Some(lock_file)) => *lock(&host.instance) = Some(lock_file),
                Ok(None) => {
                    host.e2e.note_once("DUPLICATE_INSTANCE", "");
                    *lock(&host.runtime) = Runtime::Duplicate;
                    return Ok(());
                }
                // Without a lock the app still runs; the runtime's own
                // atomic writes remain the guard against a second writer.
                Err(error) => eprintln!("single-instance lock unavailable: {error}"),
            }
            std::thread::spawn(move || {
                start_runtime(&host, false);
                watch_runtime(&host);
            });
            Ok(())
        })
        .build(tauri::generate_context!());
    let app = match app {
        Ok(app) => app,
        Err(error) => {
            eprintln!("could not start the window: {error}");
            std::process::exit(1);
        }
    };
    app.run(|app, event| match event {
        // Closing with runs at work: keep the window and let the UI ask
        // (stop them and quit, or keep working). With none, close as usual.
        tauri::RunEvent::WindowEvent {
            event: tauri::WindowEvent::CloseRequested { api, .. },
            ..
        } => {
            let Some(host) = app.try_state::<Arc<Host>>() else {
                return;
            };
            let host = Arc::clone(host.inner());
            if host.connection().is_err() {
                return;
            }
            api.prevent_close();
            let handle = app.clone();
            tauri::async_runtime::spawn(async move {
                match switch::active_runs(&host).await {
                    Ok(runs) if !runs.is_empty() => {
                        host.close_requested.store(true, Ordering::Release);
                        host.e2e.note_once("CLOSE_HELD", &format!("runs={}", runs.len()));
                    }
                    _ => handle.exit(0),
                }
            });
        }
        tauri::RunEvent::Exit => {
            if let Some(host) = app.try_state::<Arc<Host>>() {
                host.stop_runtime();
            }
        }
        _ => {}
    });
}

// The Profiles screen against the real harness API: create, inspect, preset,
// activate and delete a profile through the UI, and confirm the server's
// profile store (the store `lightagent profile` uses) changed each time.
// Activation is checked only as the UI reports it: the running server keeps
// serving its own profile, and crates/lightagent/tests/desktop_profiles.rs
// proves the API's activation is the CLI's.

export async function checkProfileStates(browser, base, outDir) {
  const failures = [];
  const page = await browser.newPage({ viewport: { width: 1280, height: 1000 } });
  const errors = [];
  page.on("pageerror", (error) => errors.push(error.message));
  const catalog = async () => (await fetch(`${base}/api/lightagent/v1/profiles`)).json();
  const has = (data, id) => data.profiles.some((profile) => profile.id === id);
  try {
    await page.goto(`${base}/#/profiles`);
    await page.getByRole("heading", { name: "Saved profiles" }).waitFor({ timeout: 15000 });

    // An invalid id is refused next to the form, before any request.
    await page.getByLabel("Profile id").fill("Bad Id");
    await page.getByRole("button", { name: "Create profile" }).click();
    await page.getByText("Use lowercase letters, digits").first().waitFor({ timeout: 5000 });
    if (has(await catalog(), "Bad Id")) failures.push("profiles: an invalid id reached the store");

    await page.getByLabel("Profile id").fill("e2e-work");
    await page.getByLabel("Display name").fill("E2E Work");
    await page.getByLabel("Model", { exact: true }).fill("e2e-model");
    await page.getByRole("button", { name: "Create profile" }).click();
    await page.getByText("Created profile 'e2e-work'.").waitFor({ timeout: 10000 });
    if (!has(await catalog(), "e2e-work")) failures.push("profiles: the created profile is not in the store");

    await page.getByRole("heading", { name: "Profile: e2e-work" }).waitFor({ timeout: 10000 });
    await page.getByRole("button", { name: "Apply preset" }).click();
    await page.getByText("Applied research@1 to 'e2e-work'.").waitFor({ timeout: 10000 });

    const row = page.getByRole("listitem").filter({ hasText: "e2e-work" });
    await row.getByRole("button", { name: "Use", exact: true }).click();
    await page.getByText("'e2e-work' is now the active profile").waitFor({ timeout: 10000 });

    await row.getByRole("button", { name: "Delete…" }).click();
    const confirm = row.getByRole("button", { name: "Delete e2e-work" });
    if (await confirm.isEnabled()) failures.push("profiles: delete was enabled before the id was typed");
    await row.getByLabel(/Type e2e-work to confirm/).fill("e2e-work");
    await confirm.click();
    await page.getByText("Deleted profile 'e2e-work'").waitFor({ timeout: 10000 });
    if (has(await catalog(), "e2e-work")) failures.push("profiles: the deleted profile is still in the store");

    // `default` is protected: the UI offers no delete for it.
    const defaultRow = page.getByRole("listitem").filter({ hasText: "default ·" });
    if (await defaultRow.getByRole("button", { name: "Delete…" }).count()) {
      failures.push("profiles: the protected default profile offers delete");
    }
    await page.screenshot({ path: `${outDir}/profiles.png`, fullPage: true });
  } catch (error) {
    failures.push(`profiles: ${error instanceof Error ? error.message : String(error)}`);
    await page.screenshot({ path: `${outDir}/profiles-failure.png`, fullPage: true }).catch(() => {});
  }
  if (errors.length) failures.push(`profiles: uncaught page error — ${errors[0]}`);
  await page.close();
  return failures;
}

### Audit findings that shape D1–D3

1. **The process model exists.** `lightagent serve` (`crates/lightagent/src/serve.rs:732`)
   serves `/api/lightagent/v1` (21 method/route pairs, `crates/lightagent-api/src/lib.rs:73-112`)
   and the React 19 + Vite WebUI from `--web-root` (`frontend/`; screens `/`, `/tools`, `/settings`).
2. **The API covers only part of the CLI.** It has no operation for config
   path/get/set and the Jev routing report, models, profile
   show/create/use/preset/delete, the extensions commands, worker
   assign/revoke, memory, RAG, identity, remote credentials, import, doctor,
   init/setup, onboard, reload, **continue** (resume after the time budget), or
   **steering** (input queued while a run is in progress).
3. **CLI handlers are print-bound.** They have the shape
   `fn(args…, json: bool) -> Result<(), String>` and print their results
   (for example `extensions.rs:27`, `rag.rs:697`, `import.rs:28`, `memory.rs`).
   Typed shared operations must be extracted (D3) so CLI, API, and desktop call
   the same code.
4. **Loopback is unauthenticated.** `serve.rs:746-758`: with no `--key-env`,
   a loopback bind uses `AuthConfig::open()`. `crates/lightagent-api` has no
   Host or Origin validation, so a DNS-rebinding page or any local process can
   drive tools. This is an input to the D1 security design.
5. **One profile per server process.** `AppState.session_profile` is fixed
   at start (`serve.rs:739-760`). `POST /runs {profile}` is honoured only in
   multi-user mode (`lib.rs:1148-1160`). The active-profile write
   `ProfileStore::set_active` is atomic (temp file + rename,
   `paths.rs:173-185`), but no lock coordinates a running server with a CLI
   that switches profile. This is an input to the D1 profile-switch decision.
6. **Secret authority.** Provider keys are stored in owner-only
   `~/.lightagent/secrets` and referenced as `SecretRef`. The
   remote-credential vault is sealed with
   `LIGHTAGENT_CREDENTIAL_KEY[_FILE]`. Identity secrets have strict delivery
   rules (`identity_admin.rs:10-14`).
7. **Targets and runners.** `check.yml` and `release.yml` already run
   natively on `macos-15-intel` (x86_64, `MACOSX_DEPLOYMENT_TARGET=10.14`),
   `macos-15` (arm64, 11.0), `windows-latest`, and `ubuntu-22.04`/`24.04`.
   The 10.14 floor is a declared deployment target; no hosted runner executes
   macOS older than 15. It stays an unverified target (D1).

## D3a — The real runtime in desktop mode

- `lightagent serve --desktop [--profile <id>]` (`crates/lightagent/src/serve.rs`, `run_desktop`):

  - writes one JSON handshake line to stdout and nothing else there;

## D3b — Host lifecycle

- **Crash detection:**
  - the host checks its runtime every 0.5 s;
  - an unexpected exit is reported (`failed`, with the exit status), and its output is kept for diagnostics;
  - the runtime is restarted automatically **at most once per 60 s**, so a runtime that keeps failing stays down and says so.
- **`runtime_restart`** (user-initiated) and `runtime_logs`, which also shows the last crashed runtime's output.
- **Single instance:**
  - an exclusive lock (`std::fs::File::try_lock`, no new crates) on `instance.lock` in the app's local data directory, held for the app's lifetime;
  - a second launch starts **no runtime** and explains itself;
  - the OS releases the lock if the app dies.
- **UI (`DesktopGate`):**
  - keeps watching after start-up; a stopped runtime shows a modal notice over the kept, inert app;
  - failures offer **Restart runtime** and a **Runtime log** disclosure;
  - duplicate, mismatch and failure each state their cause and remedy.
- **CI:** e2e modes `crash` (kill the runtime under a ready app; require the
  exit to be reported and a restart) and `duplicate` (second launch; require
  no second runtime and the first still running), both with
  `E2E_REQUIRE_NO_ORPHAN=1`. Linux runs under a session bus by default.
- **Deferred to D4a:** a close with active runs (warn, or cancel then quit),
  which needs the active-runs operation.

## D4a — Profiles: shared operations and API

- **Shared operations** (`crates/lightagent/src/ops.rs`, `ops::Profiles`):
  list, show, create, preset, use (an atomic active-profile write), and
  delete (an active profile hands over to `default`; `default` is
  protected). Each returns data. `lightagent profile …` now calls them with
  **unchanged output**, and `lightagent serve` performs the same operations
  for the API through the existing `RunFactory` seam.
- **Seam decision:** the operations live in the CLI crate and reach the API
  through `RunFactory`, not through a new `lightagent-ops` crate. A new crate
  would change the root `Cargo.lock`, which cannot be regenerated locally under
  the execution policy, and the CLI crate already owns every store. The
  architecture's §12 crate name is superseded by this seam (recorded in D4b's
  architecture update).
- **API** (operator scope, `Scope::Admin`; every other caller is refused):
  `GET /runs` (active runs), `POST /profiles`, `GET|DELETE /profiles/{id}`,
  `PUT /profiles/{id}/active`, `POST /profiles/{id}/preset`. Errors are
  `{"error":{"category","message"}}` with stable categories (`not_found`
  404, `invalid` 400, `conflict` 409, `denied` 403, `unavailable` 503,
  `internal` 500).
- **Contract:** 29 operations, including `profiles.delete`, which is marked
  destructive. **Inventory:** the six new API rows; the CLI profile rows now
  name their shared operation and route. The denominator is unchanged at
  226: these rows count once the desktop UI exists (D4b).
- **Evidence test** (`crates/lightagent/tests/desktop_profiles.rs`): the
  real runtime in desktop mode creates, shows, presets, activates and
  deletes profiles over HTTP, and `lightagent profile list --json` on the
  same home reports exactly that state (CLI–desktop equivalence). It also
  checks the five error categories and the hand-over to `default`.

## D4b — Profile switch, close with active runs, Profiles screen

- **Host profile switch** (`desktop/src-tauri/src/switch.rs`, command
  `profile_switch`): start the new runtime, then commit (architecture §7).
  - While runs are working, the switch is blocked and the UI lists them.
  - If the person chooses to stop them, pending approvals are rejected
    explicitly (recorded) and the runs cancelled, with a 10 s drain;
    otherwise nothing changes.
  - The new runtime starts on the profile with its own key. If it is not
    ready, it is stopped and nothing has changed.
  - Commit: the profile is made active through the new runtime (the CLI sees
    it), the app is routed to the new runtime, old streams end with
    `profile_switch`, and the old runtime is stopped off the UI thread.
  - Only one switch runs at a time. The planning decision is unit-tested.
- **Close with active runs:** the host holds the close and the UI asks
  (**Keep working**, focused by default, or **Stop runs and quit**). With no
  runs, the window closes as before.
- **Profiles screen** (`#/profiles`, from the account menu):
  - list, details, preset, create (labelled fields, the id rule shown, errors next to the form);
  - **Use**: in the desktop this switches the runtime and reloads; in the browser it sets the CLI's active profile and says this server keeps serving its own until it restarts;
  - **Delete** behind typed confirmation; `default` is protected.
- **Evidence planned:** `e2e/profiles.mjs` (in the existing `standalone Web UI
  render` job) drives create, invalid id, preset, use and delete through the
  real UI against real `lightagent serve`, and checks the server's profile
  store after each step.
- **Gap, stated rather than claimed:** the desktop-only switch and
  close-with-runs dialogs have **no hosted native-UI automation** yet. They
  are covered by unit tests of the decision, by the API-level D4a test, and
  by the host compile and clippy gates. Native WebView automation
  (`tauri-driver` on Linux and Windows) is planned for D8; until then those
  two flows are not claimed as end-to-end verified.
- **Architecture §7 and §12** now describe the operations seam that was built.

# Lightagent Desktop — architecture decision (D1)

Status: **decided for implementation; not yet validated by any desktop build.**
Inputs:
- the D0 audit ([DESKTOP_PROGRESS.md](DESKTOP_PROGRESS.md),
  [desktop-capabilities.json](desktop-capabilities.json));
- the governing plan ([DESKTOP_BUILD_PLAN.md](DESKTOP_BUILD_PLAN.md));
- the hosted-runner spike in draft PR #91 (evidence only, never merged).

Every number below is a **GitHub-hosted runner measurement**, named by its run.

## 1. Decision summary

| Question | Decision |
|---|---|
| Desktop framework | **Tauri 2** (`tauri` 2.12.x, CLI 2.12.x), in its own Cargo workspace under `desktop/` |
| Runtime placement | **Owned sidecar process**: the bundled `lightagent` binary of the same version and commit, started as `lightagent serve --desktop …` |
| Renderer ↔ runtime boundary | **Typed Tauri IPC to the host only.** The host is the sole HTTP client of the sidecar. The renderer never holds the launch credential and never opens a socket |
| UI | The existing React WebUI (`frontend/`) behind a transport interface. One codebase, built in `web` and `desktop` modes |
| Shared service layer | Typed operations in the CLI crate (`crates/lightagent/src/ops.rs`). The CLI prints their results, and `lightagent serve` performs them for the API through the `RunFactory` seam, so the CLI, Web UI and desktop change the same stores the same way. Extracted incrementally (D4a onwards; §12) |
| Profile switching | **Start-new-then-commit** sidecar replacement (§7) |
| Local-server security | Per-launch key over stdin, exact Host check, Origin refusal, and Bearer auth on every route including SSE. A key alone is not the design (§6) |
| Updates | Manual only. The desktop never runs `lightagent update` (§11, D1-Q1) |

## 2. Framework comparison

| Criterion | Tauri 2 | Electron | Weight / evidence |
|---|---|---|---|
| **macOS minimum** | Bundle default "macOS 10.13 and above"; `bundle.macOS.minimumSystemVersion` sets `LSMinimumSystemVersion` and `MACOSX_DEPLOYMENT_TARGET` (v2.tauri.app, macOS application bundle). The spike sets **10.14** and records the binary's `LC_BUILD_VERSION minos` (§3) | Electron 45 README: "macOS (Ventura and up)". Electron 42.11.14 (newest release at least 14 days old) declares the `LSMinimumSystemVersion` recorded in §3. No supported Electron reaches 10.14 | **Decisive.** The CLI's Intel floor is 10.14 (README, `release.yml`) |
| Intel architecture | x86_64 build is native on `macos-15-intel` | x64 binaries are published | Both fine. Architecture ≠ OS floor (§9) |
| Repository dependency policy | Own workspace (`desktop/`), so the root `Cargo.lock`, `check-deps.sh`, and "no sudo" root build are untouched. The spike applies the banned-crate list (`aws-lc-*`, `openssl-sys`, `native-tls`, `bindgen`, `cmake`) to the desktop graph for all targets (§3). Linux builds need `libwebkit2gtk-4.1-dev` (sudo apt): allowed on hosted runners, documented as a desktop-only build dependency | No Rust in the host. npm graph instead | Tauri keeps one language for host and runtime and passes the policy (§3) |
| Existing React frontend | Loads the Vite build from bundled assets | Same | Equal. WebView engines differ (WKWebView, WebView2, WebKitGTK), so render tests run on every OS |
| WebView engine risk | The system WebKit on macOS 10.14 is Safari 12–14 era. Vite 6 builds for `safari14` by default. **10.14 therefore also requires the Safari 14 system-WebKit update** (unverified, §9) | Bundled Chromium: one engine | Tauri risk, made explicit |
| Process lifecycle | The Rust host spawns and owns the sidecar (job object on Windows, process group on Unix) | The Node main process does the same | Equal |
| Package size / idle memory | Runner measurements, §3 | Runner measurements, §3 | Measured, not assumed |
| Security surface | Capability-scoped IPC; no Node in the renderer; strict CSP | Context isolation + sandbox are available; Node in the main process | Both adequate when configured; Tauri's default is narrower |
| Accessibility | Platform WebView accessibility trees (NSAccessibility, UIA, ATK) | Chromium accessibility on all OSes | Equal for an ARIA-correct React app; checked in D8 |
| Maintenance | Tauri 2 stable major; a 3.0 alpha exists, so a pinned 2.x line is required | 8-week majors; old majors lose support quickly | Tauri |

  (30.2 s and 30.3 s). It is consistent with WebKitGTK waiting on GPU or

  the product. D2 must investigate (for example
  `WEBKIT_DISABLE_DMABUF_RENDERER=1`) and either fix it or record why it is
  specific to Xvfb, before Linux P1 baselines are set.

**How the decision uses the evidence:**
- Electron starts faster on Windows and Linux (Xvfb).
- Tauri uses less memory on macOS. It ships a package about 60–150× smaller,
  except the AppImage, which carries WebKitGTK.
- Tauri passes the repository dependency policy.
- Tauri is the only option that can declare the 10.14 Intel floor, which is
  decisive.

The Windows and Linux startup gap and the Linux anomaly are tracked as D2
risks with acceptance thresholds (DESKTOP_ACCEPTANCE.md P1–P4).

## 4. Process model and ownership

```
┌──────────── Lightagent Desktop (Tauri host, Rust) ─────────────┐
│ window + WebView ──IPC──▶ host commands ──HTTP(Bearer)──▶ sidecar│
│  (React UI, CSP,          (allowlisted ops,         lightagent   │
│   no network)              SSE relay, dialogs)       serve        │
│                                                     --desktop    │
└───────────── owns: sidecar pid/job, key, logs ───────────────────┘
```

- The host is a single instance (`tauri-plugin-single-instance`); a second launch focuses the first window.
- **Spawn**:
  - the bundled `lightagent` from the app bundle only (Tauri `externalBin`), never `PATH`;
  - argv: `serve --desktop --host 127.0.0.1 --port 0 [--profile <id>]`;
  - the key is the **first line of stdin**. The host then keeps stdin open as the runtime's **lifeline**: however the host ends, even killed, the OS closes the pipe and the runtime exits (graceful shutdown, at most 5 s);
  - the home is the user's normal Lightagent home, so CLI, WebUI, and desktop share state.
- **Readiness**:
  - the sidecar writes exactly one JSON line to stdout:
    `{"lightagent":"serve","protocol":1,"version":"0.5.x","commit":"<sha>","addr":"127.0.0.1:<port>","profile":"<id>","capabilities":[…]}`;
  - the host waits up to 20 s, then requires `version` and `commit` to equal its own bundle metadata;
  - a mismatch is an explicit error, and the host never falls back to another binary.
- **Ownership**:
  - the host records the child's pid and process start time;
  - on Windows the child runs in a Job Object (kill-on-close), on Unix in its own process group;
  - the stdin lifeline makes the sidecar exit when the host disappears, so a crashed host leaves no orphan. This replaced the earlier `--parent-pid` polling design in D3a: it needs no polling and no platform process APIs, and it cannot be fooled by PID reuse;
  - the host never discovers or stops any process it did not spawn: no port scans and no name matching.
- **Shutdown**:
  - app close with active runs shows a dialog listing them, with *Cancel runs and quit* or *Keep window open*;
  - with no runs, or after cancellation, the host sends SIGTERM (CTRL_BREAK on Windows), waits 5 s, then kills the owned job or group.
- **Crash**:
  - an unexpected sidecar exit marks every open run stream `runtime_exited`;
  - the UI shows the last 200 log lines (redacted) and offers *Restart runtime*;
  - restart is automatic at most once per 60 s.
  - Runs are not resumed blindly. After restart, the UI reconciles from `GET /sessions`, `GET /runs/{id}`, and `GET /approvals`.
- **Logs**:
  - sidecar stdout and stderr go to a bounded ring buffer (2 MiB) and a rotated file, 5 × 2 MiB, under the OS app-log directory;
  - the key never appears in them because it is never printed, never in argv, and never in env.

## 5. Renderer ↔ host contract

- The renderer has **no** `fs`, `shell`, `http`, or `process` plugin
  permissions. Its Tauri capability file grants only the host's own commands:
  - `api_call(op, params, body)`;
  - `api_subscribe(run, after) → Channel`, plus `api_unsubscribe`;
  - `upload_attachment`;
  - `pick_path(kind)`;
  - `runtime_status`;
  - `runtime_restart`;
  - `open_external(url)`.
- `op` is an id from the **operation contract**
  `docs/desktop/contract/operations.json` (D2). Each op lists its method, its
  path template, a parameter pattern for every path segment, and whether it
  is destructive. The host refuses unknown ops and malformed parameters before
  any HTTP request. CI keeps the contract equal to the server routes and the
  frontend client.
- `api_subscribe` relays SSE events `{id, event, data}` unchanged, so the
  WebUI's existing sequencing (`id` order, gap detection, `?after=` replay in
  `frontend/src/hooks/useRunEvents.ts`) works as it does in the browser. The
  host adds the terminal pseudo-events `stream_closed{reason}` (for
  `profile_switch`, `runtime_exited`, or `shutdown`).
- **CSP**:
  `default-src 'self'; connect-src ipc: http://ipc.localhost; img-src 'self' data: asset:; style-src 'self' 'unsafe-inline'; script-src 'self'`.
  Navigation away from the app origin is blocked. External links go through
  `open_external`, which allows only `https:` and only after a user gesture.
- The frontend selects its transport at build time (`vite build --mode desktop`).
  `web` mode is unchanged (same-origin `fetch` and `EventSource`).

## 6. Local-server security design (desktop mode)

All of the following apply together. Each closes a different attack path.

| Layer | Rule | Closes |
|---|---|---|
| Bind | `127.0.0.1`, ephemeral port; `--desktop` refuses any non-loopback `--host` | LAN exposure |
| Launch credential | 256-bit key from the OS CSPRNG, generated by the host per sidecar launch. Written once as the first line of the sidecar's **stdin**, which then stays open as the lifeline and carries nothing else. Held only in host memory and overwritten on drop. **Never** in argv, env, a URL, a file, browser storage, the renderer, logs, crash reports, diagnostics, or artifacts | Other local users and processes reading argv, env, or files; leaks into the UI or artifacts |
| Authentication | Every route, **including `GET /runs/{id}/events` and `/health`**, requires `Authorization: Bearer <key>` (the existing `AuthConfig::keyed` path; constant-time comparison verified in D3). No query-string tokens. No cookies | Unauthenticated local callers; token leakage through URLs |
| Host validation | `Host` must equal exactly `127.0.0.1:<bound port>`, otherwise `421`/`403 host_rejected` | DNS rebinding: a remote page resolving its own name to 127.0.0.1 |
| Origin validation | Any request carrying an `Origin` header is refused (`403 origin_rejected`): the only legitimate client is the host's `reqwest`, which sends none. `Sec-Fetch-Site` other than `none`/absent is refused too | Browser-driven CSRF and cross-origin reads, even with a guessed port |
| Event stream | Same three checks as above. The host relays it, so the renderer never needs `EventSource` credentials | SSE token-in-URL leakage |
| No static UI | `--desktop` disallows `--web-root`; the fallback returns 404 | Serving content to a browser at all |
| CORS | No CORS headers, ever | Cross-origin readable responses |

The browser WebUI mode (`serve` without `--desktop`) gets the same Host and
Origin layers in D8: loopback host names plus the bound port, `--allow-host`
for reverse proxies, and same-origin requirements for state-changing methods.
That is a documented behaviour change with regression tests. It closes the D0
finding that loopback is unauthenticated.

**Verification (D3/D8):**
- contract tests for each rejection (missing or wrong key, wrong Host, any Origin, SSE without auth);
- a packaged end-to-end test that scans process argv, environment, log files, diagnostics bundles, and every uploaded artifact for the key (the test harness receives the key through a test-only fd) and for any 64-hex run.

## 7. Profile switching (decided)

The sidecar serves one profile: `AppState.session_profile`
(`crates/lightagent/src/serve.rs:739-760`). Switching therefore replaces the
sidecar using **start-new-then-commit**:

1. **Preconditions.** The host asks the current sidecar for active runs
   (`GET /runs`, which lists only unfinished runs; added in D4a) and pending
   approvals (`GET /approvals`). Implemented in `desktop/src-tauri/src/switch.rs` (D4b).
   - If any exist, the switch is refused with a dialog listing them. The user may choose
     *Cancel these runs and switch*: the host rejects each pending approval
     explicitly (`approve:false`, so it is recorded, not lost) and cancels each
     run.
   - It then waits up to 10 s for terminal states. If any run is still not terminal, the
     switch is aborted and nothing has changed.
2. **Start new.** The host spawns a second sidecar with `--profile <new>` and
   a fresh key, and waits for its handshake. The old sidecar is still running
   and authoritative.
3. **Failure before commit.** If the handshake fails or times out, the host
   kills the new owned process. The old sidecar, its key, and its streams are
   untouched, and the UI shows the redacted log excerpt. Nothing is persisted.
4. **Commit.**
   - The host switches IPC routing to the new sidecar.
   - It ends every relayed stream of the old one with `stream_closed{reason:"profile_switch"}`.
   - It persists the choice through the new sidecar's typed `profile use` operation (`ProfileStore::set_active`, already atomic: temp file plus rename, `paths.rs:173-185`). The CLI and WebUI then see the same active profile.
   - It stops the old sidecar (§4 shutdown).
5. **Failure after commit.** If persisting fails, the new sidecar stays in
   use and the UI warns that the CLI's active profile was not updated, with a
   retry. Old streams are not resumed across profiles. The UI reloads
   sessions, approvals, tools, and skills from the new sidecar.
6. **Credentials.** Each sidecar has its own key, and the old key dies with
   its process. Provider and MCP secrets are resolved by the sidecar from the
   profile's `SecretRef`s; the host and the renderer never read them.
7. **Locks and concurrent writers.**
   - The active-profile file is written atomically.
   - The CLI may switch profile while the desktop runs. The desktop sidecar stays bound to its `--profile`, and the status bar shows a *CLI active profile differs* notice.
   - Session transcripts get a cross-process advisory lock in D5, shared by the CLI chat and `serve`.
   - Configuration writes get a lock plus a generation/ETag check in D4. A stale `PUT /settings` returns `409`, and the UI reloads.
8. **Remote mode.** Switching is offered only if the remote server's
   capability list includes `profiles.scoped`. Otherwise the control is
   disabled with that reason.

## 8. Configuration, data paths, and credentials

- **Home and paths:**
  - one home, resolved by `LightagentPaths::resolve()` (honours `LIGHTAGENT_HOME`);
  - the desktop adds only `desktop/settings.json` (window state, last remote endpoints, UI preferences), written atomically, and holds no secrets.
- **Secret authority:**
  - provider keys are entered in the desktop, sent host → sidecar → the existing owner-only `~/.lightagent/secrets` store, and referenced by `SecretRef`;
  - the remote-credential vault and identity store are unchanged;
  - the renderer sees only redacted status (*set*, *env reference*, *missing*, or *unreadable*);
  - a secret-store read failure fails closed: the operation errors and no secret is substituted;
  - there is no second credential store.
- **Remote endpoints:** the key is saved through the same secrets store as a
  `SecretRef` named `desktop-remote-<id>`.
- **Migration:** none needed in D1. Any later schema change follows the plan:
  backup, atomic update, recovery, documented compatibility.

## 12. Shared service extraction and refined slices

- **Shared operations (as built, D4a):** typed operations live in the CLI
  crate (`crates/lightagent/src/ops.rs`), which already owns every store they
  change. They return data, so CLI handlers only print it (with their output
  unchanged) and `lightagent serve` performs the same operations for the API
  through the existing `RunFactory` seam. `lightagent-api` defines the
  request and error types (`lightagent_api::ops`), with stable categories
  (`not_found`, `invalid`, `conflict`, `denied`, `unavailable`, `internal`).
  This replaces the planned `lightagent-ops` crate, for three reasons:
  - it needs no root `Cargo.lock` change, which cannot be regenerated under
    the execution policy;
  - it creates no dependency cycle;
  - embedders and test fixtures keep the factory's `unavailable` defaults.
- No desktop feature scrapes CLI output or builds shell strings.

## 13. Threat model

| Asset | Threat | Mitigation | Verified by |
|---|---|---|---|
| Tool execution (files, terminal) | A local process or web page drives the sidecar | §6: key + Host + Origin + loopback; host-only client | D3 contract tests; D8 |
| Launch key | Leak via argv, env, logs, URL, renderer, or artifacts | Stdin delivery; never printed; renderer never receives it | D3/D9 scans for the key and any 64-hex run |
| Provider and MCP secrets | Exposure to the renderer or diagnostics | Backend-only `SecretRef` resolution; redacted status only | D4 tests; diagnostics redaction test |
| Session and config data | Corruption by concurrent CLI and desktop writers | Atomic writes; config lock + ETag (D4); session lock (D5) | Concurrency tests with isolated homes |
| Approvals | Lost or duplicated across reconnect or switch | Explicit reject on switch; reconciliation via `GET /approvals` | D5 end-to-end |
| Remote mode | A failed remote silently falls back to local execution | No fallback; the scope badge; capability negotiation | D8 tests |
| Extensions and skills | Path traversal, untrusted text granting permissions | Existing install validation plus fixtures; permissions stay in the runtime | D6 fixtures |
| Renderer | Remote content or script injection | Bundled assets only; strict CSP; no Node; allowlisted IPC | D2 config test; D8 |
| Updates | Unattended binary replacement | Manual only; no updater plugin | Inventory D1-Q1; D9 |
| Other Lightagent servers | The desktop kills or uses a server it does not own | Owns only its own pid/job; no discovery | D3 lifecycle test |
