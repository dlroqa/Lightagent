Remote mode: the desktop window served by a keyed Lightagent server (D8)

The desktop's parity for LIGHTAGENT_API_KEY: the window uses a server
started elsewhere with `lightagent serve --key-env LIGHTAGENT_API_KEY`.

- The host holds the key: from its own LIGHTAGENT_API_KEY, or from a key
  file chosen in the native picker and read by the host. It is kept in
  memory for the connection, never written, logged or shown, and the
  bundled runtime never receives it (removed from its environment).
- The key goes only to the origin the person named: HTTPS, or plain HTTP
  to this computer; no proxy; a redirect is an error and is not followed.
- The server must describe itself (new `GET /api/lightagent/v1/server`,
  behind the key) with this app's version and protocol, or the
  connection is refused naming both.
- While connected the bundled runtime is stopped: nothing runs on this
  computer, a failing server's requests fail with its error, and there is
  no fallback. A scope badge says where the window is served from.
  Disconnect starts the bundled runtime again. Profile switching and the
  runtime restart refer to this computer and are refused while remote.
- The host gains TLS the way the runtime has it: reqwest with
  rustls-no-provider and the ring provider installed first (no cmake, no
  OpenSSL); the lockfile is resolved on the runner.
- Tests: the server route (backend), the address rules and key checks
  (host units), remote.py through the host (refusals, wrong key, closed
  port, redirect trap, version and service mismatch, cancel, connect,
  remote-only execution, server down without fallback, reconnect,
  disconnect, key isolation and leak scans), and the rendered Settings
  card and badge in native.py.

struct Runtime {
    child: Child,
    _lifeline: Option<ChildStdin>,
    addr: String,
    first: String,
}

impl Drop for Runtime {
    fn drop(&mut self) {
        let _ = self.child.kill();
        let _ = self.child.wait();
    }

fn scratch(label: &str) -> PathBuf {
    let nanos = SystemTime::now()
        .duration_since(UNIX_EPOCH)
        .map(|elapsed| elapsed.as_nanos())
        .unwrap_or(0);
    let dir = std::env::temp_dir().join(format!(
        "lightagent-server-info-{label}-{}-{nanos}",
        std::process::id()
    ));
    std::fs::create_dir_all(&dir).expect("scratch dir");
    dir
}

fn first_line(child: &mut Child) -> String {
    let stdout = child.stdout.take().expect("stdout");
    let (tx, rx) = mpsc::channel();
    std::thread::spawn(move || {
        let mut reader = BufReader::new(stdout);
        let mut line = String::new();
        let _ = reader.read_line(&mut line);
        let _ = tx.send(line);
        let mut rest = String::new();
        while reader.read_line(&mut rest).is_ok_and(|read| read > 0) {
            rest.clear();
        }
    });
    rx.recv_timeout(Duration::from_secs(60))
        .expect("first line")
}

fn start(home: &Path, desktop: bool) -> Runtime {
    let mut command = Command::new(env!("CARGO_BIN_EXE_lightagent"));
    command.env("LIGHTAGENT_HOME", home).stdout(Stdio::piped()).stderr(Stdio::null());
    if desktop {
        command
            .args(["serve", "--desktop", "--host", "127.0.0.1", "--port", "0"])
            .stdin(Stdio::piped());
    } else {
        command
            .args(["serve", "--host", "127.0.0.1", "--port", "0", "--key-env", "LIGHTAGENT_TEST_SERVER_INFO_KEY"])
            .env("LIGHTAGENT_TEST_SERVER_INFO_KEY", KEY)
            .stdin(Stdio::null());
    }
    let mut child = command.spawn().expect("start lightagent serve");
    let lifeline = if desktop {
        let mut lifeline = child.stdin.take().expect("stdin");
        writeln!(lifeline, "{LAUNCH}").expect("key");
        lifeline.flush().expect("flush");
        Some(lifeline)
    } else {
        None
    };
    let first = first_line(&mut child);
    let addr = if desktop {
        let handshake: Value = serde_json::from_str(&first).expect("handshake");
        handshake["addr"].as_str().expect("addr").to_owned()
    } else {
        first
            .split("http://")
            .nth(1)
            .and_then(|rest| rest.split('/').next())
            .unwrap_or_else(|| panic!("no address in {first:?}"))
            .trim()
            .to_owned()
    };
    Runtime {
        child,
        _lifeline: lifeline,
        addr,
        first,
    }
}

fn get(runtime: &Runtime, path: &str, key: Option<&str>) -> (u16, String) {
    let mut stream = TcpStream::connect(&runtime.addr).expect("connect");
    stream
        .set_read_timeout(Some(Duration::from_secs(60)))
        .expect("timeout");
    let auth = key.map(|key| format!("Authorization: Bearer {key}\r\n")).unwrap_or_default();
    let head = format!(
        "GET {path} HTTP/1.1\r\nHost: {}\r\n{auth}Connection: close\r\n\r\n",
        runtime.addr
    );
    stream.write_all(head.as_bytes()).expect("send");
    let mut response = String::new();
    let _ = stream.read_to_string(&mut response);
    let status = response
        .split_whitespace()
        .nth(1)
        .and_then(|code| code.parse().ok())
        .unwrap_or(0);
    let body = response
        .split_once("\r\n\r\n")
        .map(|(_, body)| body.to_owned())
        .unwrap_or_default();
    (status, body)
}

//! Remote mode (D8): the window served by a Lightagent server elsewhere,
//! started as `lightagent serve --key-env LIGHTAGENT_API_KEY`, instead of
//! the bundled runtime.
//!
//! - The key is the host's. It comes from this app's own
//!   `LIGHTAGENT_API_KEY` (the variable the server reads) or from a key file
//!   the person chooses in the native picker, read by the host as a
//!   credential's secret is (secret.rs). It is kept in memory for the
//!   connection and never written, logged or shown, and the bundled runtime
//!   never receives it (sidecar.rs removes the variable from its
//!   environment).
//! - It is sent only to the origin the person named: HTTPS, or plain HTTP to
//!   this computer. The client uses no proxy and follows no redirect; a
//!   redirect is an error.
//! - The server must describe itself (`GET /api/lightagent/v1/server`) with
//!   this app's version and protocol, or the connection is refused, naming
//!   both.
//! - While connected the bundled runtime is stopped: nothing runs on this
//!   computer, and when the server fails its requests fail with that error.
//!   Nothing falls back to local execution. Disconnecting starts the bundled
//!   runtime again.

use std::sync::Arc;
use std::sync::atomic::Ordering;
use std::time::Duration;

use serde::{Deserialize, Serialize};

use crate::{Host, Runtime, lock};

const PROBE_TIMEOUT: Duration = Duration::from_secs(15);
const SHAPE: &str =
    "give the server's address only, like https://agent.example.com or https://agent.example.com:8787";

/// A connected remote server. Its key is overwritten when it is dropped.
pub struct Endpoint {
    pub origin: String,
    key: String,
    pub version: String,
    pub profile: Option<String>,
    pub capabilities: Vec<String>,
}

impl Endpoint {
    pub fn authorization(&self) -> String {
        format!("Bearer {}", self.key)
    }
}

impl Drop for Endpoint {
    fn drop(&mut self) {
        let len = self.key.len();
        self.key.clear();
        self.key.extend(std::iter::repeat_n('0', len));
    }
}

/// What a connection attempt did. Never carries the key.
#[derive(Debug, Serialize)]
pub struct Outcome {
    /// `connected`, or `cancelled` (no key file was chosen; nothing sent).
    pub state: &'static str,
    pub origin: Option<String>,
    pub version: Option<String>,
}

#[derive(Deserialize)]
struct ServerInfo {
    service: String,
    version: String,
    protocol: u32,
    #[serde(default)]
    capabilities: Vec<String>,
    #[serde(default)]
    profile: Option<String>,
}

/// The origin a key may be sent to: `https://host[:port]`, or `http://` to
/// this computer only. Anything else is refused before a connection.
pub fn origin_of(address: &str) -> Result<String, String> {
    let address = address.trim();
    let (scheme, rest) = address.split_once("://").ok_or(SHAPE)?;
    let scheme = scheme.to_ascii_lowercase();
    if scheme != "https" && scheme != "http" {
        return Err("the address must start with https:// (or http:// for this computer)".to_owned());
    }
    let authority = rest.strip_suffix('/').unwrap_or(rest).to_ascii_lowercase();
    if authority.is_empty()
        || authority.contains(['/', '?', '#', '@', '\\'])
        || authority.chars().any(char::is_whitespace)
    {
        return Err(SHAPE.to_owned());
    }
    let (host, port) = if authority.starts_with('[') {
        let end = authority.find(']').ok_or(SHAPE)?;
        let after = &authority[end + 1..];
        let port = if after.is_empty() {
            None
        } else {
            Some(after.strip_prefix(':').ok_or(SHAPE)?)
        };
        (&authority[..=end], port)
    } else {
        match authority.split_once(':') {
            Some((host, port)) => (host, Some(port)),
            None => (authority.as_str(), None),
        }
    };
    if let Some(port) = port
        && !(!port.is_empty() && port.bytes().all(|byte| byte.is_ascii_digit())
            && matches!(port.parse::<u16>(), Ok(number) if number != 0))
    {
        return Err(SHAPE.to_owned());
    }
    let valid = if let Some(inside) = host.strip_prefix('[').and_then(|rest| rest.strip_suffix(']')) {
        !inside.is_empty() && inside.bytes().all(|byte| byte.is_ascii_hexdigit() || byte == b':' || byte == b'.')
    } else {
        !host.is_empty()
            && !host.starts_with(['.', '-'])
            && !host.ends_with('.')
            && host.bytes().all(|byte| byte.is_ascii_alphanumeric() || byte == b'-' || byte == b'.')
    };
    if !valid {
        return Err(SHAPE.to_owned());
    }
    if scheme == "http" && !matches!(host, "127.0.0.1" | "localhost" | "[::1]") {
        return Err(
            "plain http:// is only for this computer (127.0.0.1, localhost or [::1]); over a network the key is sent only by https://"
                .to_owned(),
        );
    }
    Ok(format!("{scheme}://{authority}"))
}

/// A key the server can be sent in a header: one token of printable
/// characters.
fn check_key(key: &str) -> Result<(), String> {
    if key.is_empty() || key.len() > 4096 || !key.bytes().all(|byte| byte.is_ascii_graphic()) {
        return Err("the key must be one line of printable characters, without spaces".to_owned());
    }
    Ok(())
}

/// Connect the window to the server at `address`, with the key from
/// `source` (`environment` or `file`).
pub async fn connect(
    host: &Arc<Host>,
    app: &tauri::AppHandle,
    address: String,
    source: String,
) -> Result<Outcome, String> {
    let origin = origin_of(&address)?;
    let key = match source.as_str() {
        "environment" => std::env::var("LIGHTAGENT_API_KEY")
            .ok()
            .map(|key| key.trim().to_owned())
            .filter(|key| !key.is_empty())
            .ok_or("LIGHTAGENT_API_KEY is not set in this app's environment")?,
        "file" => {
            let Some(path) = crate::secret::pick(app).await? else {
                return Ok(Outcome {
                    state: "cancelled",
                    origin: None,
                    version: None,
                });
            };
            crate::secret::read_secret(&path)
                .map_err(|reason| format!("the key file cannot be used: {reason}"))?
                .trim()
                .to_owned()
        }
        _ => return Err("the key comes from the environment or from a file".to_owned()),
    };
    check_key(&key)?;
    let info = probe(&host.http, &origin, &key).await?;
    let _lifecycle = host.lifecycle.lock().await;
    if host.switching.load(Ordering::Acquire) {
        return Err("a profile switch is in progress; try again when it has finished".to_owned());
    }
    if !host.is_remote() {
        let runs = crate::switch::active_runs(host).await.unwrap_or_default();
        if !runs.is_empty() {
            return Err(format!(
                "{} run(s) are working on this computer; stop them before using another server",
                runs.len()
            ));
        }
    }
    let version = info.version.clone();
    let previous = std::mem::replace(
        &mut *lock(&host.runtime),
        Runtime::Remote(Endpoint {
            origin: origin.clone(),
            key,
            version: info.version,
            profile: info.profile,
            capabilities: info.capabilities,
        }),
    );
    // Stopping the bundled runtime may take its grace period.
    let _ = tauri::async_runtime::spawn_blocking(move || drop(previous)).await;
    host.e2e
        .note_once("REMOTE_CONNECTED", &format!("origin={origin} version={version}"));
    Ok(Outcome {
        state: "connected",
        origin: Some(origin),
        version: Some(version),
    })
}

/// Leave remote mode and start the bundled runtime again.
pub async fn disconnect(host: &Arc<Host>) -> Result<(), String> {
    let host = Arc::clone(host);
    tauri::async_runtime::spawn_blocking(move || {
        let _lifecycle = host.lifecycle.blocking_lock();
        if !host.is_remote() {
            return;
        }
        // Dropping the endpoint overwrites its key.
        *lock(&host.runtime) = Runtime::Starting;
        host.e2e.note_once("REMOTE_DISCONNECTED", "");
        crate::start_runtime(&host, false);
    })
    .await
    .map_err(|error| error.to_string())
}

/// Ask the server what it is, with the key, and check it can serve this app.
async fn probe(http: &reqwest::Client, origin: &str, key: &str) -> Result<ServerInfo, String> {
    let response = http
        .get(format!("{origin}/api/lightagent/v1/server"))
        .header(reqwest::header::AUTHORIZATION, format!("Bearer {key}"))
        .timeout(PROBE_TIMEOUT)
        .send()
        .await
        .map_err(|error| format!("could not reach {origin}: {}", reason(&error)))?;
    let status = response.status();
    if status.is_redirection() {
        return Err(format!(
            "{origin} answered with a redirect ({}); the key is sent only to the address you gave, so it was not followed",
            status.as_u16()
        ));
    }
    if status == reqwest::StatusCode::UNAUTHORIZED || status == reqwest::StatusCode::FORBIDDEN {
        return Err(format!(
            "{origin} refused the key ({}): check LIGHTAGENT_API_KEY, or the key file, against the key the server was started with",
            status.as_u16()
        ));
    }
    if status == reqwest::StatusCode::NOT_FOUND {
        return Err(format!(
            "{origin} does not describe itself: it is not a Lightagent server, or one older than {}",
            crate::VERSION
        ));
    }
    if !status.is_success() {
        return Err(format!("{origin} answered {}", status.as_u16()));
    }
    let body = response
        .text()
        .await
        .map_err(|error| format!("{origin}'s answer was cut off: {}", reason(&error)))?;
    let info: ServerInfo = serde_json::from_str(&body)
        .map_err(|_| format!("{origin} answered, but not as a Lightagent server"))?;
    if info.service != "lightagent" {
        return Err(format!("{origin} is not a Lightagent server"));
    }
    if info.protocol != crate::contract::PROTOCOL {
        return Err(format!(
            "{origin} speaks protocol {}, and this app speaks {}",
            info.protocol,
            crate::contract::PROTOCOL
        ));
    }
    if info.version != crate::VERSION {
        return Err(format!(
            "{origin} runs Lightagent {}, and this app is {}: they must be the same version",
            info.version,
            crate::VERSION
        ));
    }
    Ok(info)
}

fn reason(error: &reqwest::Error) -> String {
    if error.is_timeout() {
        "it did not answer in time".to_owned()
    } else if error.is_connect() {
        "the connection failed (nothing is listening there, or there is no network)".to_owned()
    } else {
        "the request failed".to_owned()
    }
}

function errorText(cause: unknown): string {

/**
 * Remote mode (desktop only): use a Lightagent server elsewhere instead of
 * this computer's runtime. The host reads the key itself (its own

 * it; this window only names the server (desktop/src-tauri/src/remote.rs).

export function RemoteModeCard() {
  const [status, setStatus] = useState<HostStatus | null>(null);
  const [address, setAddress] = useState("");
  const [source, setSource] = useState<"environment" | "file">("environment");
  const [busy, setBusy] = useState(false);
  const [message, setMessage] = useState<string | null>(null);
  const [error, setError] = useState<string | null>(null);
  const keyGroup = useId();

  useEffect(() => {
    if (!isDesktop) return;
    void invoke<HostStatus>("runtime_status").then(setStatus).catch(() => undefined);
  }, []);
  if (!isDesktop) return null;

  async function connect() {
    setBusy(true);
    setError(null);
    setMessage(null);
    try {
      const outcome = await invoke<RemoteOutcome>("remote_connect", { url: address.trim(), source });
      if (outcome.state === "cancelled") {
        setMessage("No key file was chosen, so nothing was sent.");
        return;
      }
      // Everything on screen came from this computer's runtime: start fresh.
      window.location.reload();

      setError(`${errorText(cause)}. This computer's runtime is still in use.`);
    } finally {
      setBusy(false);
    }
  }

    setError(null);

      setBusy(false);

            <legend className="field__label">The server&apos;s key</legend>

ran. Stubs stand in for a redirecting server (with a trap that records any

def free_port():
    with socket.socket() as sock:
        sock.bind(("127.0.0.1", 0))
        return sock.getsockname()[1]


def listening(port):
    try:
        socket.create_connection(("127.0.0.1", port), timeout=1).close()
        return True
    except OSError:
        return False


def cli_in(cli, home, *args):
    env = dict(os.environ, LIGHTAGENT_HOME=home)
    out = subprocess.run([cli, *args], env=env, capture_output=True, text=True, timeout=60)
    if out.returncode != 0:
        raise RuntimeError(f"lightagent {' '.join(args)}: {out.stderr.strip()[:300]}")
    return out.stdout


class RemoteServer:
    """`lightagent serve --key-env LIGHTAGENT_API_KEY` with its own home."""

    def __init__(self, cli, home, key, port, log):
        self.args = [cli, "serve", "--host", "127.0.0.1", "--port", str(port), "--key-env", "LIGHTAGENT_API_KEY"]
        self.env = dict(os.environ, LIGHTAGENT_HOME=home, LIGHTAGENT_API_KEY=key)
        for name in ("LIGHTAGENT_DESKTOP_E2E_DRIVER", "LIGHTAGENT_DESKTOP_E2E_REPORT"):
            self.env.pop(name, None)
        self.port, self.log, self.process = port, log, None

    def start(self):
        self.process = subprocess.Popen(self.args, env=self.env, stdin=subprocess.DEVNULL,
                                        stdout=open(self.log, "a"), stderr=subprocess.STDOUT)
        if not harness.wait_until(lambda: listening(self.port), timeout=60):
            raise RuntimeError("the remote server did not start")

    def stop(self):
        if self.process and self.process.poll() is None:
            self.process.terminate()
            try:
                self.process.wait(timeout=15)
            except subprocess.TimeoutExpired:
                self.process.kill()
                self.process.wait()
        harness.wait_until(lambda: not listening(self.port), timeout=15)


def stub(answer):
    """A loopback HTTP server answering every GET with `answer(handler)`;
    it records the headers of every request it receives."""
    seen = []

    class Handler(BaseHTTPRequestHandler):
        def log_message(self, *args):
            pass

        def do_GET(self):
            seen.append({"path": self.path, "authorization": self.headers.get("Authorization")})
            code, headers, body = answer(self)
            self.send_response(code)
            for name, value in headers.items():
                self.send_header(name, value)
            data = json.dumps(body).encode() if body is not None else b""
            self.send_header("Content-Type", "application/json")
            self.send_header("Content-Length", str(len(data)))
            self.end_headers()
            self.wfile.write(data)

    server = ThreadingHTTPServer(("127.0.0.1", 0), Handler)
    server.daemon_threads = True
    threading.Thread(target=server.serve_forever, daemon=True).start()
    return server, f"http://127.0.0.1:{server.server_address[1]}", seen

def scenario_remote(view, host, provider, cli):
    """Remote mode in the rendered window: the Settings card refuses a plain
    http address elsewhere and says why; connects to a keyed server on this

    returns to this computer's runtime and the badge goes."""
    import tempfile

    check = Checks("remote_mode")
    remote_home = tempfile.mkdtemp(prefix="lightagent-native-remote-")
    env = dict(os.environ, LIGHTAGENT_HOME=remote_home, LIGHTAGENT_API_KEY=REMOTE_KEY)
    for name in ("LIGHTAGENT_DESKTOP_E2E_DRIVER", "LIGHTAGENT_DESKTOP_E2E_REPORT"):
        env.pop(name, None)
    subprocess.run([cli, "config", "set", "inference.base_url", provider.url], env=env, check=True,
                   capture_output=True, timeout=60)
    with socket.socket() as sock:
        sock.bind(("127.0.0.1", 0))
        port = sock.getsockname()[1]
    server = subprocess.Popen([cli, "serve", "--host", "127.0.0.1", "--port", str(port), "--key-env", "LIGHTAGENT_API_KEY"],
                              env=env, stdin=subprocess.DEVNULL, stdout=subprocess.DEVNULL, stderr=subprocess.DEVNULL)
    origin = f"http://127.0.0.1:{port}"

    def up():
        try:
            socket.create_connection(("127.0.0.1", port), timeout=1).close()
            return True
        except OSError:
            return False

    try:
        harness.wait_until(up, timeout=60)
        view.go("Settings", "#/settings")
        refused = view.js("""
          fill(await need("the server address", () => document.querySelector("#remote-url")), "http://192.0.2.10:8787");
          press(await need("Connect", () => one("button", "Connect")));
          return await until(() => all("alert").map((el) => nameOf(el)).find((s) => /only for this computer/.test(s)) || null, 20000);""", 30)
        check.expect("a plain-http address elsewhere is refused before anything is sent, and the card says why", bool(refused), refused)
        view.js(T('fill(document.querySelector("#remote-url"), $ORIGIN); await sleep(200); press(one("button", "Connect")); return true;', ORIGIN=origin))
        connected = harness.wait_until(lambda: host.call("status").get("mode") == "remote", timeout=60)
        time.sleep(1.5)
        view.wait_ui()
        badge = view.js('return await until(() => all("status").map((el) => nameOf(el)).find((s) => s.startsWith("Remote ·")) || null, 15000);')
        view.shot("remote-connected")
        check.expect("Connect serves the window from the remote and the scope badge names it",
                     bool(connected) and f"127.0.0.1:{port}" in (badge or ""), (host.call("status").get("remote"), badge))
        check.expect("the bundled runtime is stopped while remote",
                     harness.wait_until(lambda: not harness.runtimes(host.process.pid), timeout=20))
        view.new_chat()

        answered = view.wait_text("Answered remotely.") and view.idle()
        check.expect("a conversation in the window runs on the remote server",
                     answered and saved_with(remote_home, "[e2e:remote]") is not None)
        check.expect("nothing of it is kept on this computer", saved_with(os.environ["LIGHTAGENT_HOME"], "[e2e:remote]") is None)

        back = harness.wait_until(lambda: (lambda s: s.get("mode") == "local" and s["state"] == "ready")(host.call("status")), timeout=90)
