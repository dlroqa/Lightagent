=== message
Release workflow: build, verify and checksum Lightagent Desktop with the CLI

A release now carries the Desktop installers next to the CLI archives, from
the same tag and the same run. A manual run stays a dry run (the gate is
unchanged): every package is built, verified and checksummed, and kept only
as run artifacts.

- desktop (4 targets, native, the architecture checked as for the CLI):
  the installers (.dmg arm64 and x64, NSIS and MSI, .deb and AppImage)
  with the third-party notices bundled; the test-only driver absent; each
  installer installed, launched and uninstalled on its runner with the
  user data checks (the reinstall is a same-version reinstall, not an
  upgrade); the notices checked against the crate versions found in the
  shipped binaries themselves; the signature state of what ships.
- Signing uses the signing secrets when they are configured, and only then:
  macOS through Tauri's Developer ID signing and notarization
  (APPLE_CERTIFICATE, APPLE_CERTIFICATE_PASSWORD, APPLE_SIGNING_IDENTITY;
  APPLE_API_ISSUER, APPLE_API_KEY, APPLE_API_KEY_P8), Windows through
  Tauri's signCommand (repository variable WINDOWS_SIGN_COMMAND, its tool
  WINDOWS_SIGN_TOOL as crate@version, AZURE_* secrets for it). An
  incomplete set fails; none at all builds unsigned and says so.
- desktop-sources: SBOMs and notices per target, the vulnerability audits,
  and desktop-mpl-sources-<version>.tar.gz, packed reproducibly and checked
  (its SHA256SUMS, its MPL-2.0 text, exactly the archives the notices name).
- desktop-older-os: the macOS arm64 and Windows installers installed,
  launched and uninstalled on the oldest hosted images (macos-14,
  windows-2022).
- checksums and publish wait for all of it; SHA256SUMS covers the CLI
  archives, the installers and the MPL-2.0 sources, so `lightagent update`
  keeps reading its archive's line from the same file.

scripts/desktop-notices-shipped.py is the shipped-version check;
scripts/test-release-gate.py covers the new jobs' shape.

name: release

# A pushed tag `v<version>` builds, smoke-tests and checksums every package,
# the CLI archives and the Lightagent Desktop installers, then drafts a GitHub
# release. A manual run is a dry run unless `dry_run` is turned off: it builds
# the same packages and uploads them with SHA256SUMS as run artifacts, and
# creates no release or tag. Turning `dry_run` off is only accepted on the tag
# itself. scripts/release-gate.sh makes the decision and
# scripts/test-release-gate.py tests it, and the shape of this file, in check.
#
# The Desktop installers are signed only when the signing secrets are
# configured (docs/desktop/DESKTOP_RELEASE_ACCEPTANCE.md); without them they
# are built unsigned, and every run reports the signature state of what ships.

on:
  push:
    tags: ['v*']
  workflow_dispatch:
    inputs:
      dry_run:
        description: Build, smoke-test and checksum only; create no release
        type: boolean
        default: true

permissions:
  contents: read

jobs:
  gate:
    name: release gate
    runs-on: ubuntu-24.04
    timeout-minutes: 5
    outputs:
      publish: ${{ steps.decide.outputs.publish }}
    steps:
      - uses: actions/checkout@v7
      - id: decide
        # `shell: bash` adds pipefail, so a refusal fails the step through tee.
        shell: bash
        env:
          EVENT: ${{ github.event_name }}
          REF_TYPE: ${{ github.ref_type }}
          REF_NAME: ${{ github.ref_name }}
          DRY_RUN: ${{ toJSON(inputs.dry_run) }}
        run: bash ./scripts/release-gate.sh | tee -a "$GITHUB_OUTPUT"

  build:
    name: build (${{ matrix.label }})
    needs: gate
    runs-on: ${{ matrix.os }}
    timeout-minutes: 60
    # Every package is built and smoke-tested natively on its own architecture;
    # `arch` is what the runner must report, checked below, not inferred from
    # the label. `macos_min` is the package's minimum macOS (its deployment
    # target): 11.0 is the floor for Apple silicon, and 10.14 for Intel is the
    # floor of rustls-platform-verifier (Security.framework), above Rust's 10.12.
    strategy:
      fail-fast: false
      matrix:
        include:
          - { os: macos-15, label: macos-arm64, triple: aarch64-apple-darwin, arch: arm64, macos_min: '11.0' }
          - { os: macos-15-intel, label: macos-x64, triple: x86_64-apple-darwin, arch: x86_64, macos_min: '10.14' }
          - { os: windows-latest, label: windows-x64, triple: x86_64-pc-windows-msvc, arch: x86_64 }
          - { os: ubuntu-22.04, label: linux-x64, triple: x86_64-unknown-linux-gnu, arch: x86_64 }
    defaults:
      run:
        shell: bash
    steps:
      - uses: actions/checkout@v7
      - run: rustup show
      - name: Runner architecture
        env:
          WANT_ARCH: ${{ matrix.arch }}
          WANT_TRIPLE: ${{ matrix.triple }}
        run: |
          arch="$(uname -m)"
          host="$(rustc -vV | sed -n 's/^host: //p')"
          {
            echo "### ${{ matrix.label }} runner"
            echo "- uname: $(uname -srm)"
            echo "- rustc host: $host"
            if [ "$(uname -s)" = Darwin ]; then
              echo "- macOS: $(sw_vers -productVersion) ($(sw_vers -buildVersion))"
              echo "- CPU: $(sysctl -n machdep.cpu.brand_string)"
              echo "- hw.optional.arm64: $(sysctl -n hw.optional.arm64 2>/dev/null || echo absent)"
              echo "- sysctl.proc_translated: $(sysctl -n sysctl.proc_translated 2>/dev/null || echo absent)"
            fi
          } | tee -a "$GITHUB_STEP_SUMMARY"
          [ "$arch" = "$WANT_ARCH" ] || { echo "runner is $arch, expected $WANT_ARCH" >&2; exit 1; }
          [ "$host" = "$WANT_TRIPLE" ] || { echo "rustc host is $host, expected $WANT_TRIPLE" >&2; exit 1; }
          if [ "$(uname -s)" = Darwin ]; then
            # Rosetta reports x86_64 from uname on Apple silicon; neither may
            # stand in for the real hardware.
            [ "$(sysctl -n sysctl.proc_translated 2>/dev/null || echo 0)" = 0 ] \
              || { echo "running under Rosetta translation" >&2; exit 1; }
            if [ "$WANT_ARCH" = x86_64 ] && [ "$(sysctl -n hw.optional.arm64 2>/dev/null || echo 0)" = 1 ]; then
              echo "an Apple silicon machine, not an Intel Mac" >&2; exit 1
            fi
          fi
      - name: macOS deployment target
        if: matrix.macos_min
        run: echo "MACOSX_DEPLOYMENT_TARGET=${{ matrix.macos_min }}" >> "$GITHUB_ENV"
      - uses: Swatinem/rust-cache@v2
        with:
          key: release-${{ matrix.label }}
      - uses: actions/setup-node@v7
        with:
          node-version: '22'
          cache: npm
          cache-dependency-path: frontend/package-lock.json
      - run: bash ./scripts/check-versions.sh
      - run: npm ci --prefix frontend
      - run: npm run build --prefix frontend
      - run: cargo build --release -p lightagent --target ${{ matrix.triple }}
      - run: bash ./scripts/package-lightagent.sh ${{ matrix.triple }}
      - run: bash ./scripts/smoke-lightagent-package.sh
      - uses: actions/upload-artifact@v7
        with:
          name: lightagent-${{ matrix.label }}
          if-no-files-found: error
          path: dist-cli/lightagent-*

          - { os: ubuntu-22.04, label: linux-x64, triple: x86_64-unknown-linux-gnu, arch: x86_64, bundles: 'deb,appimage', installers: 2 }
    defaults:
      run:
        shell: bash
    env:

      TAURI_CLI: '@tauri-apps/cli@2.12.3'
      APPIMAGE_EXTRACT_AND_RUN: '1'
    steps:

      - run: rustup show

          [ "$host" = "$WANT_TRIPLE" ] || { echo "rustc host is $host, expected $WANT_TRIPLE" >&2; exit 1; }
          if [ "$(uname -s)" = Darwin ]; then
            [ "$(sysctl -n sysctl.proc_translated 2>/dev/null || echo 0)" = 0 ] \
              || { echo "running under Rosetta translation" >&2; exit 1; }
            if [ "$WANT_ARCH" = x86_64 ] && [ "$(sysctl -n hw.optional.arm64 2>/dev/null || echo 0)" = 1 ]; then
              echo "an Apple silicon machine, not an Intel Mac" >&2; exit 1
            fi
          fi
      - name: macOS deployment target
        if: matrix.macos_min
        run: echo "MACOSX_DEPLOYMENT_TARGET=${{ matrix.macos_min }}" >> "$GITHUB_ENV"
      - name: Linux system packages (webview, display)
        if: runner.os == 'Linux'
        run: |
          sudo apt-get update
          sudo apt-get install -y libwebkit2gtk-4.1-dev libayatana-appindicator3-dev librsvg2-dev patchelf file xvfb imagemagick xdotool
      - uses: Swatinem/rust-cache@v2
        with:
          workspaces: |
            . -> target
            desktop -> target

      - uses: actions/setup-node@v7

          node-version: '22'
          cache: npm

          mkdir -p desktop/src-tauri/binaries

      - name: Signing configuration (the signing secrets only; otherwise unsigned)
        id: signing
        env:
          APPLE_CERTIFICATE: ${{ secrets.APPLE_CERTIFICATE }}
          APPLE_CERTIFICATE_PASSWORD: ${{ secrets.APPLE_CERTIFICATE_PASSWORD }}
          APPLE_SIGNING_IDENTITY: ${{ secrets.APPLE_SIGNING_IDENTITY }}
          APPLE_API_ISSUER: ${{ secrets.APPLE_API_ISSUER }}
          APPLE_API_KEY: ${{ secrets.APPLE_API_KEY }}
          APPLE_API_KEY_P8: ${{ secrets.APPLE_API_KEY_P8 }}
          WINDOWS_SIGN_COMMAND: ${{ vars.WINDOWS_SIGN_COMMAND }}
          WINDOWS_SIGN_TOOL: ${{ vars.WINDOWS_SIGN_TOOL }}
        run: |
          set -euo pipefail
          put() { { echo "$1<<LA_SIGNING_END"; printf '%s\n' "$2"; echo "LA_SIGNING_END"; } >> "$GITHUB_ENV"; }
          given() { n=0; for v in "$@"; do [ -n "${!v}" ] && n=$((n + 1)); done; echo "$n"; }
          mode=unsigned; config='{}'
          case "$RUNNER_OS" in
            macOS)
              case "$(given APPLE_CERTIFICATE APPLE_CERTIFICATE_PASSWORD APPLE_SIGNING_IDENTITY)" in
                0) ;;
                3)
                  for v in APPLE_CERTIFICATE APPLE_CERTIFICATE_PASSWORD APPLE_SIGNING_IDENTITY; do put "$v" "${!v}"; done
                  mode="signed (Developer ID), not notarized"
                  case "$(given APPLE_API_ISSUER APPLE_API_KEY APPLE_API_KEY_P8)" in
                    0) ;;
                    3)
                      key="$RUNNER_TEMP/notary/AuthKey_$APPLE_API_KEY.p8"; mkdir -p "$(dirname "$key")"

                      put APPLE_API_ISSUER "$APPLE_API_ISSUER"; put APPLE_API_KEY "$APPLE_API_KEY"; put APPLE_API_KEY_PATH "$key"
                      mode="signed (Developer ID), notarized" ;;
                    *) echo "::error::notarization needs APPLE_API_ISSUER, APPLE_API_KEY and APPLE_API_KEY_P8 together"; exit 1 ;;

                mode="signed (signCommand)"
              elif [ -n "$WINDOWS_SIGN_TOOL" ]; then
                echo "::error::WINDOWS_SIGN_TOOL is set without WINDOWS_SIGN_COMMAND"; exit 1
              fi ;;
          esac
          echo "mode=$mode" >> "$GITHUB_OUTPUT"
          printf 'config=%s\n' "$config" >> "$GITHUB_OUTPUT"
          echo "- signing (${{ matrix.label }}): $mode" | tee -a "$GITHUB_STEP_SUMMARY"
      - name: Build and bundle
        working-directory: desktop
        env:
          SIGN_CONFIG: ${{ steps.signing.outputs.config }}
          AZURE_CLIENT_ID: ${{ secrets.AZURE_CLIENT_ID }}
          AZURE_CLIENT_SECRET: ${{ secrets.AZURE_CLIENT_SECRET }}

                echo "- codesign --verify --deep --strict: $verified"

"""Whether the bundled third-party notices name every crate, at its exact
version, that the shipped Desktop binaries were built from (release.yml).

  python desktop-notices-shipped.py <THIRD_PARTY_NOTICES.txt> <binary>...

A release build keeps the source paths of code that can panic, under the
cargo registry: .../index.crates.io-<hash>/<crate>-<version>/... Those paths
are a record taken from the shipped binaries themselves, not from cargo
metadata (which the notices are generated from), of which crates and versions
the binaries contain. Every one must be in the notices with the same version.
Crates the notices list that the binaries do not show are counted, not
failed: a crate with no code that can panic leaves no path, and procedural
macros leave none.

Prints a Markdown summary. Exits 1 when a crate or version found in a binary
is not in the notices, or when a binary shows no registry path at all (the
check would prove nothing).
"""

import re
import sys

PATH = re.compile(rb"index\.crates\.io-[0-9a-f]{16}[/\\]([A-Za-z0-9_.+-]+?)-(\d+\.\d+\.\d+(?:-[0-9A-Za-z.-]+)?(?:\+[0-9A-Za-z.-]+)?)[/\\]")
LISTED = re.compile(r"^-- (\S+) (\S+) \(cargo\);")


def main():
    notices, binaries = sys.argv[1], sys.argv[2:]
    listed = {}
    with open(notices, encoding="utf-8") as handle:
        for line in handle:
            match = LISTED.match(line)
            if match:
                listed.setdefault(match.group(1), set()).add(match.group(2))
    lines = [f"### notices against the shipped binaries: {sum(len(v) for v in listed.values())} crate versions listed"]
    problems, seen = [], set()
    for binary in binaries:
        with open(binary, "rb") as handle:
            data = handle.read()
        found = {(m.group(1).decode(), m.group(2).decode()) for m in PATH.finditer(data)}
        missing = sorted(f"{name}@{version}" for name, version in found if version not in listed.get(name, ()))
        seen |= found
        lines.append(f"- {binary}: {len(found)} crate versions found in the binary, {len(missing)} not in the notices"
                     + (f": {', '.join(missing[:40])}" if missing else ""))
        if not found:
            problems.append(f"{binary}: no registry path found, nothing was checked")
        if missing:
            problems.append(f"{binary}: {len(missing)} shipped crate versions are not in the notices")
    unseen = sum(1 for name, versions in listed.items() for version in versions if (name, version) not in seen)
    lines.append(f"- listed but not visible in any binary (no panicking code, or build time only): {unseen}")
    lines += [f"- FAILED {problem}" for problem in problems]
    print("\n".join(lines))
    return 1 if problems else 0


if __name__ == "__main__":
    sys.exit(main())
