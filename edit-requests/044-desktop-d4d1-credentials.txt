//! Provider API keys in Lightagent's owner-only secrets store, the one
//! credential store (DESKTOP_ARCHITECTURE.md §8). `lightagent setup`, the
//! HTTP API and the desktop store, replace and remove a key through these
//! functions, so all of them write the same file and the same reference.
//! Nothing here returns a stored key.

use crate::config::{Config, SecretRef};
use crate::paths::{self, LightagentPaths};

/// The local Lightweight endpoint's provider id.
pub const LOCAL_PROVIDER_ID: &str = "lightweight-local";

const LOCAL_URL: &str = "http://127.0.0.1:11434";

/// A filename-safe provider id from a provider's name: lowercase ASCII
/// letters and digits, with anything else as `-`.
pub fn key_id(name: &str) -> String {
    let id = name
        .chars()
        .map(|character| {
            if character.is_ascii_alphanumeric() {
                character.to_ascii_lowercase()
            } else {
                '-'
            }
        })
        .collect::<String>();
    let id = id.trim_matches('-');
    if id.is_empty() {
        "provider".to_owned()
    } else {
        id.to_owned()
    }
}

fn same_url(left: &str, right: &str) -> bool {
    left.trim_end_matches('/') == right.trim_end_matches('/')
}

/// The provider id whose key the active endpoint (`inference.base_url`)
/// uses: its saved provider's, the local endpoint's, or `custom`.
pub fn active_key_id(config: &Config) -> String {
    let base_url = &config.inference.base_url;
    if let Some(saved) = config
        .inference
        .saved_providers
        .iter()
        .find(|saved| same_url(&saved.base_url, base_url))
    {
        return key_id(&saved.name);
    }
    if same_url(base_url, LOCAL_URL) {
        LOCAL_PROVIDER_ID.to_owned()
    } else {
        "custom".to_owned()
    }
}

/// A key typed by a person, checked before it is stored: trimmed, not
/// empty, and something an HTTP header can carry. The error never repeats
/// the key.
pub fn validate_key(key: &str) -> Result<&str, &'static str> {
    let key = key.trim();
    if key.is_empty() {
        return Err("the provider API key cannot be empty");
    }
    if key.chars().any(char::is_control) {
        return Err("the provider API key contains a line break or control character");
    }
    Ok(key)
}

/// Store `key` for `provider_id`, owner-only and atomically, replacing any
/// key stored for it before. Returns the reference to put in the config.
pub fn store_key(
    paths: &LightagentPaths,
    provider_id: &str,
    key: &str,
) -> Result<SecretRef, String> {
    let key = validate_key(key).map_err(str::to_owned)?;
    let path = paths.provider_key_file(&key_id(provider_id));
    paths::write_private(&path, key.as_bytes())
        .map_err(|error| format!("the provider API key could not be saved: {}", error.kind()))?;
    Ok(SecretRef::file(path))
}

/// The reference `store_key` gives for `provider_id`.
pub fn stored_reference(paths: &LightagentPaths, provider_id: &str) -> SecretRef {
    SecretRef::file(paths.provider_key_file(&key_id(provider_id)))
}

/// Point the active endpoint, and its saved provider if it has one, at
/// `secret`; `None` sends no key.
pub fn use_key(config: &mut Config, secret: Option<SecretRef>) {
    let base_url = config.inference.base_url.clone();
    for saved in config
        .inference
        .saved_providers
        .iter_mut()
        .filter(|saved| same_url(&saved.base_url, &base_url))
    {
        saved.api_key = secret.clone();
    }
    config.inference.api_key = secret;
}

/// Remove the key stored for `provider_id`. Returns whether a file was
/// removed. The config's references are the caller's to clear first.
pub fn remove_key(paths: &LightagentPaths, provider_id: &str) -> Result<bool, String> {
    match std::fs::remove_file(paths.provider_key_file(&key_id(provider_id))) {
        Ok(()) => Ok(true),
        Err(error) if error.kind() == std::io::ErrorKind::NotFound => Ok(false),
        Err(error) => Err(format!(
            "the provider API key could not be removed: {}",
            error.kind()
        )),
    }
}

/// `lightagent config jev-routing-report`: the same dry run, from the same

/// The active endpoint's provider API key as status only: whether one is
/// configured and usable, and where it comes from, never the key itself or
/// the file that holds it.
fn credential_view(
    config: &lightagent_core::Config,
    revision: &ConfigRevision,
) -> serde_json::Value {
    let source = config.inference.api_key.as_ref().map(|secret| match secret {
        SecretRef::Env { .. } => "env",
        SecretRef::File { .. } => "file",
    });
    json!({
        "provider_key_id": lightagent_core::provider_keys::active_key_id(config),
        "base_url": config.inference.base_url,
        "state": lightagent_core::secret_state(config.inference.api_key.as_ref()),
        "source": source,
        "revision": revision.as_str(),
    })
}

/// A provider key typed into Settings. Write-only: no `Debug`, and never
/// returned or logged.
#[derive(Deserialize)]
struct CredentialBody {
    key: String,
    #[serde(default)]
    expected_revision: Option<String>,
}

async fn get_provider_credential(State(state): State<Arc<AppState>>, headers: HeaderMap) -> Response {
    if let Some(rejection) = deny(&state, &headers, Scope::Admin) {
        return rejection;
    }
    let Some(store) = &state.config_store else {
        return internal("Lightagent settings are unavailable in this embedding");
    };
    match store.load_with_revision() {
        Ok((config, revision)) => Json(credential_view(&config, &revision)).into_response(),
        Err(error) => config_error(error),
    }
}

/// Save or replace the active endpoint's key in the shared secrets store
/// and point the config at it, as `lightagent setup provider` does. With
/// `expected_revision`, a stale save gets 409 and writes nothing.
async fn put_provider_credential(
    State(state): State<Arc<AppState>>,
    headers: HeaderMap,
    Json(body): Json<CredentialBody>,
) -> Response {
    if let Some(rejection) = deny(&state, &headers, Scope::Admin) {
        return rejection;
    }
    let (Some(store), Some(root)) = (&state.config_store, home(&state)) else {
        return internal("Lightagent settings are unavailable in this embedding");
    };
    if let Err(message) = lightagent_core::provider_keys::validate_key(&body.key) {
        return bad_request(message);
    }
    let paths = LightagentPaths::rooted_at(root);
    let expected = body.expected_revision.as_deref().map(ConfigRevision::parse);
    // The key is written inside the config lock, after the revision check, so
    // a refused save changes nothing.
    let saved = store.update(expected.as_ref(), |config| {
        let provider_id = lightagent_core::provider_keys::active_key_id(config);
        let secret = lightagent_core::provider_keys::store_key(&paths, &provider_id, &body.key)?;
        lightagent_core::provider_keys::use_key(config, Some(secret));
        Ok::<(), String>(())
    });
    match saved {
        Ok(((), revision)) => match store.load() {
            Ok(config) => Json(credential_view(&config, &revision)).into_response(),
            Err(error) => config_error(error),
        },
        Err(UpdateError::Change(message)) => internal(&message),
        Err(UpdateError::Store(error)) => config_error(error),
    }
}

/// Remove the active endpoint's key: the config stops referring to it, and
/// the stored file is deleted when it is the shared store's own.
async fn delete_provider_credential(
    State(state): State<Arc<AppState>>,
    headers: HeaderMap,
) -> Response {
    if let Some(rejection) = deny(&state, &headers, Scope::Admin) {
        return rejection;
    }
    let (Some(store), Some(root)) = (&state.config_store, home(&state)) else {
        return internal("Lightagent settings are unavailable in this embedding");
    };
    let paths = LightagentPaths::rooted_at(root);
    let removed = store.update(None, |config| {
        let provider_id = lightagent_core::provider_keys::active_key_id(config);
        let ours = lightagent_core::provider_keys::stored_reference(&paths, &provider_id);
        let previous = config.inference.api_key.clone();
        lightagent_core::provider_keys::use_key(config, None);
        Ok::<_, std::convert::Infallible>((previous == Some(ours)).then_some(provider_id))
    });
    match removed {
        Ok((stored, revision)) => {
            if let Some(provider_id) = stored
                && let Err(message) = lightagent_core::provider_keys::remove_key(&paths, &provider_id)
            {
                return internal(&message);
            }
            match store.load() {
                Ok(config) => Json(credential_view(&config, &revision)).into_response(),
                Err(error) => config_error(error),
            }
        }
        Err(error) => config_error(error.into_store()),
    }
}

    _lifeline: ChildStdin,
    addr: String,
}

impl Drop for Runtime {
    fn drop(&mut self) {
        let _ = self.child.kill();
        let _ = self.child.wait();
    }
}

fn scratch_home(label: &str) -> PathBuf {
    let nanos = SystemTime::now()
        .duration_since(UNIX_EPOCH)
        .map(|elapsed| elapsed.as_nanos())
        .unwrap_or(0);
    let home = std::env::temp_dir().join(format!(
        "lightagent-desktop-credentials-{label}-{}-{nanos}",
        std::process::id()
    ));
    std::fs::create_dir_all(&home).expect("home");
    home
}

fn start(home: &Path) -> Runtime {
    let mut child = Command::new(env!("CARGO_BIN_EXE_lightagent"))
        .args(["serve", "--desktop", "--host", "127.0.0.1", "--port", "0"])
        .env("LIGHTAGENT_HOME", home)
        .env_remove(UNSET)
        .stdin(Stdio::piped())
        .stdout(Stdio::piped())
        .stderr(Stdio::null())
        .spawn()
        .expect("start lightagent serve --desktop");
    let mut lifeline = child.stdin.take().expect("stdin");
    writeln!(lifeline, "{KEY}").expect("key");
    lifeline.flush().expect("flush");
    let stdout = child.stdout.take().expect("stdout");
    let (tx, rx) = mpsc::channel();
    std::thread::spawn(move || {
        let mut line = String::new();
        let _ = BufReader::new(stdout).read_line(&mut line);
        let _ = tx.send(line);
    });
    let line = rx.recv_timeout(Duration::from_secs(60)).expect("handshake");
    let handshake: serde_json::Value = serde_json::from_str(&line).expect("handshake JSON");
    let addr = handshake["addr"].as_str().expect("addr").to_owned();
    Runtime {
        child,
        _lifeline: lifeline,
        addr,
    }
}

/// One authenticated request: the status, the raw body and its JSON.
fn call(
    runtime: &Runtime,
    method: &str,
    path: &str,
    body: Option<&str>,
) -> (u16, String, serde_json::Value) {
    let mut stream = TcpStream::connect(&runtime.addr).expect("connect");
    stream
        .set_read_timeout(Some(Duration::from_secs(20)))
        .expect("timeout");
    let body = body.unwrap_or("");
    let head = format!(
        "{method} {path} HTTP/1.1\r\nHost: {}\r\nAuthorization: Bearer {KEY}\r\nContent-Type: application/json\r\nContent-Length: {}\r\nConnection: close\r\n\r\n{body}",
        runtime.addr,
        body.len()
    );
    stream.write_all(head.as_bytes()).expect("send");
    let mut response = String::new();
    let _ = stream.read_to_string(&mut response);
    let status = response
        .split_whitespace()
        .nth(1)
        .and_then(|code| code.parse().ok())
        .unwrap_or(0);
    let raw = response
        .split_once("\r\n\r\n")
        .map(|(_, body)| body.to_owned())
        .unwrap_or_default();
    let json = serde_json::from_str(&raw).unwrap_or(serde_json::Value::Null);
    (status, raw, json)
}

        .args(args)
        .env("LIGHTAGENT_HOME", home)
        .env_remove(UNSET)
        .stdin(Stdio::null())
        .output()
        .expect("run lightagent")
}

        "{args:?}: {}",

    );
    String::from_utf8_lossy(&output.stdout).trim().to_owned()
}

    let reference = cli_ok(&home, &["config", "get", "inference.api_key"]);
    assert!(reference.starts_with("${file:"), "{reference}");
    assert!(reference.contains("provider-lightweight-local.key"), "{reference}");
    for args in [
        vec!["config", "show"],
        vec!["config", "get", "inference.api_key"],
    ] {
        let output = cli(&home, &args);
        assert!(!String::from_utf8_lossy(&output.stdout).contains(FIRST), "{args:?}");
        assert!(!String::from_utf8_lossy(&output.stderr).contains(FIRST), "{args:?}");
    }
    for path in [
        CREDENTIAL,
        "/api/lightagent/v1/config",
        "/api/lightagent/v1/settings",
        "/api/lightagent/v1/config/inference.api_key",
    ] {
        let (_, raw, _) = call(&runtime, "GET", path, None);
        assert!(!raw.contains(FIRST), "{path}: {raw}");
    }

    // A save against the revision read before it is refused and writes nothing.
    let (status, raw) = save(&runtime, SECOND, &absent["revision"]);
    assert_eq!(status, 409, "{raw}");
    assert_eq!(std::fs::read_to_string(&key_file).expect("key file"), FIRST);

    // Entries that cannot be sent are refused with nothing written.
    for bad in ["   ", "two\nlines"] {
        let (status, raw) = save(&runtime, bad, &saved["revision"]);
        assert_eq!(status, 400, "{raw}");
        assert!(!raw.contains("two"), "{raw}");
        assert_eq!(std::fs::read_to_string(&key_file).expect("key file"), FIRST);
    }

    // Replaced in place.
    let (status, raw) = save(&runtime, SECOND, &saved["revision"]);
    assert_eq!(status, 200, "{raw}");
    assert!(!raw.contains(SECOND), "{raw}");
    assert_eq!(std::fs::read_to_string(&key_file).expect("key file"), SECOND);

    // Removed: no reference, no file, and the CLI agrees.
    let (status, raw, removed) = call(&runtime, "DELETE", CREDENTIAL, None);
    assert_eq!(status, 200, "{raw}");
    assert_eq!(removed["state"], "absent");
    assert!(!key_file.exists());
    assert_eq!(cli_ok(&home, &["config", "get", "inference.api_key"]), "");
}

/// A configured provider endpoint that only records whether anything
/// connected to it.
fn endpoint() -> (TcpListener, String) {
    let listener = TcpListener::bind("127.0.0.1:0").expect("listener");
    listener.set_nonblocking(true).expect("non-blocking");
    let port = listener.local_addr().expect("address").port();
    (listener, format!("http://127.0.0.1:{port}/v1"))
}

fn contacted(listener: &TcpListener) -> bool {
    match listener.accept() {
        Ok(_) => true,
        Err(error) if error.kind() == std::io::ErrorKind::WouldBlock => false,
        Err(error) => panic!("accept: {error}"),
    }
}

fn set_reference(home: &Path, reference: serde_json::Value) {
    let path = home.join("config.json");
    let mut value: serde_json::Value =
        serde_json::from_slice(&std::fs::read(&path).expect("read")).expect("config JSON");
    value["inference"]["api_key"] = reference;
    std::fs::write(&path, serde_json::to_vec_pretty(&value).expect("JSON")).expect("write");
}

#[test]
fn a_configured_key_that_cannot_be_used_fails_closed_and_an_absent_one_does_not() {
    let home = scratch_home("fail-closed");
    let (listener, base_url) = endpoint();
    cli_ok(&home, &["config", "set", "inference.base_url", &base_url]);
    let empty = home.join("empty-provider-key");
    std::fs::write(&empty, " \n").expect("empty key");
    let control = home.join("control-provider-key");
    std::fs::write(&control, "abc\u{7}def").expect("control key");
    let runtime = start(&home);

    let cases = [
        (
            serde_json::json!({ "kind": "file", "path": home.join("missing-provider-key") }),
            "unreadable",
            "cannot be read",
        ),
        (
            serde_json::json!({ "kind": "env", "var": UNSET }),
            "unresolved",
            "environment variable is not set",
        ),
        (
            serde_json::json!({ "kind": "file", "path": empty }),
            "empty",
            "is empty",
        ),
        (
            serde_json::json!({ "kind": "file", "path": control }),
            "invalid",
            "control character",
        ),
    ];
    for (reference, state, words) in cases {
        set_reference(&home, reference.clone());
        let (_, _, status) = call(&runtime, "GET", CREDENTIAL, None);
        assert_eq!(status["state"], state, "{reference}");
        for args in [vec!["models"], vec!["chat"]] {
            let output = cli(&home, &args);
            let stderr = String::from_utf8_lossy(&output.stderr);
            assert!(!output.status.success(), "{args:?} {reference}: {stderr}");
            assert!(stderr.contains(words), "{args:?} {reference}: {stderr}");
            assert!(!stderr.contains("provider-key"), "{args:?} names the key file: {stderr}");
            assert!(!stderr.contains(UNSET), "{args:?} names the variable: {stderr}");
        }
        // The desktop runtime's run fails the same way.
        let (status, raw, run) = call(
            &runtime,
            "POST",
            "/api/lightagent/v1/runs",
            Some(r#"{"message":"hello"}"#),
        );
        assert_eq!(status, 202, "{raw}");
        let id = run["id"].as_str().expect("run id").to_owned();
        let deadline = Instant::now() + Duration::from_secs(30);
        loop {
            let (_, _, run) = call(&runtime, "GET", &format!("/api/lightagent/v1/runs/{id}"), None);
            if run["status"] == "failed" {
                break;
            }
            assert_ne!(run["status"], "completed", "{run}");
            assert!(Instant::now() < deadline, "the run never failed: {run}");
            std::thread::sleep(Duration::from_millis(50));
        }
        let (_, events, _) = call(&runtime, "GET", &format!("/api/lightagent/v1/runs/{id}/events"), None);
        assert!(events.contains(words), "{reference}: {events}");
        assert!(!contacted(&listener), "the endpoint was contacted with {reference}");
    }

    // No key configured is intentionally absent: the request is made, keyless.
    set_reference(&home, serde_json::Value::Null);
    let (_, _, status) = call(&runtime, "GET", CREDENTIAL, None);
    assert_eq!(status["state"], "absent");
    let mut models = Command::new(env!("CARGO_BIN_EXE_lightagent"))
        .arg("models")
        .env("LIGHTAGENT_HOME", &home)
        .stdin(Stdio::null())
        .stdout(Stdio::null())
        .stderr(Stdio::null())
        .spawn()
        .expect("start lightagent models");
    let deadline = Instant::now() + Duration::from_secs(30);
    let mut reached = false;
    while Instant::now() < deadline {
        if contacted(&listener) {
            reached = true;
            break;
        }
        std::thread::sleep(Duration::from_millis(20));
    }
    let _ = models.kill();
    let _ = models.wait();
    assert!(reached, "with no key configured, `lightagent models` never reached the endpoint");
}
