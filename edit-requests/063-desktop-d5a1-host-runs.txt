D5a-1: the host's relay resumes a broken stream; runs through the host

The host's relay of a run's event stream used to end at the first break
(the connection failing, or ending while the run was still working) and
leave the UI to resume it. It now resumes the stream itself, from the last
event it forwarded, with `Last-Event-ID`. The runtime numbers each event by
its position in the run, so nothing is skipped, and an event at or before
the last one forwarded is never forwarded again (a server that replays
from the start is deduplicated). Five resumes in a row that forward nothing
report the break; a runtime that no longer has the run (a restart) ends the
relay at once with that reason. A stream that ends is finished only when
the runtime says the run is (a run cancelled during an approval wait ends
with no terminal event).

Hosted run scenarios through the host and the real runtime, with a
deterministic fake provider:
- desktop/src-tauri/src/e2e_driver.rs (test builds only, `e2e-hooks`):
  JSON commands on stdin, each calling the same function as the UI's IPC
  command (`fetch`, `subscribe`, `poll`, `unsubscribe`, the switch), plus
  two test hooks: drop a relay's connection, and hold a switch at its gate.
  The commands' bodies moved out of the `#[tauri::command]` wrappers into
  plain functions; behavior is unchanged.
- desktop/e2e/harness.py: the fake provider (scripted turns, named gates,
  every request recorded) and the driver client.
- desktop/e2e/runs.py: streaming and a tool, approval granted and denied,
  cancel, two relay drops mid-run, and regenerate (R1: replacement; a
  provider failure, a refusal, an empty answer and a cancel keep the saved
  answer; a stale answer id and a busy session are refused).
- desktop/e2e/switch.py: a pending approval during a switch (refused while
  the switch holds its gate; blocked without stopping runs; rejected
  explicitly when runs are stopped; the new runtime knows nothing of it).
- desktop.yml: the switch job runs both (`if: success() || failure()`).

/// test driver) call exactly this.
fn subscribe(host: &Arc<Host>, target: String) -> Result<u64, String> {

    }

/// How one connection of a relay ended.
enum Leg {
    /// The runtime ended the stream after the run's last event.
    Done,
    /// The connection failed, was dropped, or ended while the run was still
    /// working. `progress`: it forwarded at least one new event.
    Broken { reason: String, progress: bool },
}

/// What the runtime says about a relayed run once its stream has ended.
enum RunCheck {
    Finished,
    Working,
    Gone,
    Unknown(String),
}

/// Relay a run's event stream into `subscription`. A break is resumed from
/// the last event forwarded, with `Last-Event-ID`: the runtime numbers each
/// event by its position in the run, so nothing is skipped, and an event at
/// or before the last one forwarded is never forwarded again. After
/// [`RELAY_ATTEMPTS`] resumes in a row that forward nothing, the break is
/// reported.
async fn relay_stream(host: &Host, target: &str, subscription: &Subscription) -> Result<(), String> {
    let mut last = None;
    let mut failures = 0;
    let (mut resumes, mut skipped) = (0u32, 0usize);
    let outcome = loop {
        match relay_leg(host, target, subscription, &mut last, &mut skipped).await {
            Ok(Leg::Done) => break Ok(()),
            Err(reason) => break Err(reason),
            Ok(Leg::Broken { reason, progress }) => {
                failures = if progress { 1 } else { failures + 1 };
                if failures > RELAY_ATTEMPTS {
                    break Err(format!("{reason}; the stream could not be resumed"));
                }
                resumes += 1;
                tokio::time::sleep(RELAY_BACKOFF * 2u32.pow(failures - 1)).await;
            }
        }
    };
    let path = target.split('?').next().unwrap_or(target);
    let last = last.map_or_else(|| "none".to_owned(), |id: u64| id.to_string());
    host.e2e.note_once(
        "RELAY_END",
        &format!(
            "target={path} resumes={resumes} skipped={skipped} last={last} error={}",
            outcome.is_err()
        ),
    );
    outcome
}

/// One connection of a relay, resumed after `last` when there is one.
async fn relay_leg(
    host: &Host,
    target: &str,
    subscription: &Subscription,
    last: &mut Option<u64>,
    skipped: &mut usize,
) -> Result<Leg, String> {
    let broken = |reason: String, progress: bool| -> Result<Leg, String> {
        Ok(Leg::Broken { reason, progress })
    };
    let (base, authorization) = match host.connection() {
        Ok(connection) => connection,
        Err(reason) => return broken(reason, false),
    };
    let mut request = host
        .http
        .get(format!("{base}{target}"))
        .header(reqwest::header::AUTHORIZATION, authorization)
        .header(reqwest::header::ACCEPT, "text/event-stream");
    if let Some(last) = *last {
        request = request.header("Last-Event-ID", last.to_string());
    }
    let response = match request.send().await {
        Ok(response) => response,
        Err(error) => {
            return broken(
                format!("the runtime did not answer: {}", error.without_url()),
                false,
            );
        }
    };
    if response.status() == reqwest::StatusCode::NOT_FOUND {
        return Err(RUN_GONE.to_owned());
    }
    if !response.status().is_success() {
        return Err(format!(
            "the runtime refused the stream ({})",
            response.status().as_u16()
        ));
    }
    let mut parser = sse::SseParser::default();
    let mut stream = response.bytes_stream();
    let mut progress = false;
    loop {
        let next = std::pin::pin!(stream.next());
        let dropped = std::pin::pin!(subscription.interrupt.notified());
        let chunk = match futures_util::future::select(next, dropped).await {
            futures_util::future::Either::Left((chunk, _)) => chunk,
            futures_util::future::Either::Right(((), _)) => {
                return broken("the relay's connection was dropped".to_owned(), progress);
            }
        };
        match chunk {
            Some(Ok(chunk)) => {
                let events = parser.push(&chunk);
                let total = events.len();
                let fresh: Vec<_> = events
                    .into_iter()
                    .filter(|event| advance(last, event))
                    .collect();
                *skipped += total - fresh.len();
                if !fresh.is_empty() {
                    progress = true;
                    lock(&subscription.state).events.extend(fresh);
                    subscription.notify.notify_waiters();
                }
            }
            Some(Err(error)) => {
                return broken(
                    format!("the stream broke: {}", error.without_url()),
                    progress,
                );
            }
            None => break,
        }
    }
    match check_run(host, target).await {
        RunCheck::Finished => Ok(Leg::Done),
        RunCheck::Working => broken(
            "the stream ended while the run was still working".to_owned(),
            progress,
        ),
        RunCheck::Gone => Err(RUN_GONE.to_owned()),
        RunCheck::Unknown(reason) => broken(reason, progress),
    }
}

/// Whether `event` follows the last event forwarded, recording it as the
/// last if so. An event without a numeric id cannot repeat one.
fn advance(last: &mut Option<u64>, event: &sse::SseEvent) -> bool {
    match event.id.as_deref().and_then(|id| id.parse::<u64>().ok()) {
        Some(id) if last.is_some_and(|seen| id <= seen) => false,
        Some(id) => {
            *last = Some(id);
            true
        }
        None => true,
    }
}

/// Ask the runtime about the run a stream relayed, once the stream ended.
async fn check_run(host: &Host, target: &str) -> RunCheck {
    let path = target.split('?').next().unwrap_or(target);
    let Some(run) = path.strip_suffix("/events") else {
        return RunCheck::Finished;
    };
    let (base, authorization) = match host.connection() {
        Ok(connection) => connection,
        Err(reason) => return RunCheck::Unknown(reason),
    };
    let response = match host
        .http
        .get(format!("{base}{run}"))
        .header(reqwest::header::AUTHORIZATION, authorization)
        .send()
        .await
    {
        Ok(response) => response,
        Err(error) => {
            return RunCheck::Unknown(format!(
                "the runtime did not answer: {}",
                error.without_url()
            ));
        }
    };
    if response.status() == reqwest::StatusCode::NOT_FOUND {
        return RunCheck::Gone;
    }
    let body = response.text().await.unwrap_or_default();
    let run: serde_json::Value = serde_json::from_str(&body).unwrap_or_default();
    match run["status"].as_str() {
        Some("completed" | "cancelled" | "failed") => RunCheck::Finished,
        Some(_) => RunCheck::Working,
        None => RunCheck::Unknown("the runtime's answer about the run was unreadable".to_owned()),
    }
}

//! A test-only driver for hosted end-to-end runs. Compiled only into test
//! builds (`--features e2e-hooks`), never into the bundled app.
//!
//! With `LIGHTAGENT_DESKTOP_E2E_DRIVER=1` the host reads one JSON command
//! per line on stdin and answers each on stdout as `E2E_DRIVER {json}`,
//! carrying the command's `id`. Each command calls the function the UI's IPC
//! command calls (`fetch`, `subscribe`, `poll`, `unsubscribe`, the switch),
//! so a harness drives the host's own contract check, transport, relay and
//! switch against the real runtime, without a WebView. Commands run
//! concurrently.
//!
//! Two hooks exist only here: `drop_relay` drops a relay's connection to the
//! runtime (the relay must resume it); and with
//! `LIGHTAGENT_DESKTOP_E2E_SWITCH_GATE=1` every switch, once it has raised
//! its run gate, announces `{"event":"switch_held"}` and waits for
//! `release_switch`, so operations during a switch are tried at a known
//! moment.

use std::io::{BufRead as _, Write as _};
use std::sync::Arc;

use serde::Deserialize;
use serde_json::{Value, json};
use tokio::sync::Notify;

use crate::{Host, lock};

static SWITCH_GATE: Notify = Notify::const_new();

#[derive(Deserialize)]
#[serde(tag = "op", rename_all = "snake_case")]
enum Op {
    Status,
    Fetch {
        method: String,
        target: String,
        #[serde(default)]
        body: Option<String>,
    },
    Subscribe {
        target: String,
    },
    Poll {
        subscription: u64,
        wait_ms: u64,
    },
    Unsubscribe {
        subscription: u64,
    },
    DropRelay {
        subscription: u64,
    },
    Switch {
        profile: String,
        cancel_runs: bool,
    },
    ReleaseSwitch,
}

/// Start reading commands, when the driver is asked for.
pub fn start(host: &Arc<Host>) {
    if std::env::var_os("LIGHTAGENT_DESKTOP_E2E_DRIVER").is_none() {
        return;
    }
    let host = Arc::clone(host);
    std::thread::spawn(move || {
        for line in std::io::stdin().lock().lines() {
            let Ok(line) = line else { break };
            let command: Value = match serde_json::from_str(&line) {
                Ok(command) => command,
                Err(error) => {
                    say(&json!({ "ok": false, "error": format!("unreadable command: {error}") }));
                    continue;
                }
            };
            let id = command["id"].clone();
            let host = Arc::clone(&host);
            tauri::async_runtime::spawn(async move {
                let answer = match serde_json::from_value::<Op>(command) {
                    Ok(op) => run(&host, op).await,
                    Err(error) => Err(format!("unknown command: {error}")),
                };
                say(&match answer {
                    Ok(result) => json!({ "id": id, "ok": true, "result": result }),
                    Err(error) => json!({ "id": id, "ok": false, "error": error }),
                });
            });
        }
    });
    say(&json!({ "event": "driver_ready" }));
}

async fn run(host: &Arc<Host>, op: Op) -> Result<Value, String> {
    Ok(match op {
        Op::Status => json!(crate::status(host)),
        Op::Fetch {
            method,
            target,
            body,
        } => json!(crate::fetch(host, method, target, body, None, None).await?),
        Op::Subscribe { target } => json!(crate::subscribe(host, target)?),
        Op::Poll {
            subscription,
            wait_ms,
        } => json!(crate::poll(host, subscription, wait_ms).await?),
        Op::Unsubscribe { subscription } => {
            crate::unsubscribe(host, subscription);
            Value::Null
        }
        Op::DropRelay { subscription } => {
            let entry = lock(&host.subscriptions)
                .get(&subscription)
                .cloned()
                .ok_or("unknown subscription")?;
            entry.interrupt.notify_one();
            Value::Null
        }
        Op::Switch {
            profile,
            cancel_runs,
        } => json!(crate::switch::switch(host, profile, cancel_runs).await?),
        Op::ReleaseSwitch => {
            SWITCH_GATE.notify_one();
            Value::Null
        }
    })
}

/// Hold a switch, its run gate raised, until `release_switch`, when asked to.
pub async fn switch_gate(profile: &str) {
    if std::env::var_os("LIGHTAGENT_DESKTOP_E2E_SWITCH_GATE").is_none() {
        return;
    }
    say(&json!({ "event": "switch_held", "profile": profile }));
    SWITCH_GATE.notified().await;
}

fn say(message: &Value) {
    let mut out = std::io::stdout().lock();
    let _ = writeln!(out, "E2E_DRIVER {message}");
    let _ = out.flush();
}

        self.server.daemon_threads = True
        threading.Thread(target=self.server.serve_forever, daemon=True).start()
        self.url = f"http://127.0.0.1:{self.server.server_address[1]}"

    def record(self, **entry):
        with self.condition:
            entry["t"] = round(time.time(), 3)
            self.log.append(entry)
            self.condition.notify_all()

    def requests(self, scenario):
        with self.condition:
            return [e for e in self.log if e["event"] == "request" and e["scenario"] == scenario]

    def wait_for(self, event, scenario, timeout=WAIT, **match):
        """Wait until the log holds `event` for `scenario` (matching `match`)."""
        deadline = time.time() + timeout

        def found():
            return any(
                e["event"] == event and e["scenario"] == scenario and all(e.get(k) == v for k, v in match.items())
                for e in self.log
            )

        with self.condition:
            while not found():
                left = deadline - time.time()
                if left <= 0:
                    return False
                self.condition.wait(left)
            return True

    def gate(self, name):
        with self.condition:
            return self.gates.setdefault(name, threading.Event())

    def open(self, name):
        self.gate(name).set()

    def close(self):
        with self.condition:
            gates = list(self.gates.values())
        for gate in gates:
            gate.set()
        self.server.shutdown()

    def reply(self, handler, code, value):
        body = json.dumps(value).encode()
        handler.send_response(code)
        handler.send_header("Content-Type", "application/json")
        handler.send_header("Content-Length", str(len(body)))
        handler.end_headers()
        handler.wfile.write(body)

    def chunk(self, handler, delta, finish=None):
        choice = {"index": 0, "delta": delta}
        if finish:
            choice["finish_reason"] = finish
        handler.wfile.write(f"data: {json.dumps({'choices': [choice]})}\n\n".encode())
        handler.wfile.flush()

    def serve(self, handler, request):
        messages = request.get("messages") or []
        users = [i for i, m in enumerate(messages) if m.get("role") == "user"]
        last = users[-1] if users else -1
        found = re.search(r"\[e2e:([a-z0-9-]+)\]", content_of(messages[last]) if users else "")
        scenario = found.group(1) if found else "none"
        turn = sum(1 for m in messages[last + 1:] if m.get("role") == "assistant")
        tools = {}
        for tool in request.get("tools") or []:
            name = (tool.get("function") or {}).get("name")
            if name:
                tools[normal(name)] = name
        index = len(self.requests(scenario))
        self.record(event="request", scenario=scenario, turn=turn, index=index, tools=sorted(tools.values()))
        script = self.scripts.get(scenario)
        steps = script(self, turn, index) if script else [("text", "ok"), ("finish", "stop")]
        if steps and steps[0][0] == "status":
            self.reply(handler, steps[0][1], {"error": {"message": "the fake provider failed on purpose"}})
            self.record(event="failed", scenario=scenario, turn=turn)
            return
        handler.send_response(200)
        handler.send_header("Content-Type", "text/event-stream")
        handler.send_header("Connection", "close")
        handler.end_headers()
        handler.close_connection = True
        calls = 0
        try:
            self.chunk(handler, {"role": "assistant"})
            for step in steps:
                kind = step[0]
                if kind == "text":
                    self.chunk(handler, {"content": step[1]})
                elif kind == "refusal":
                    self.chunk(handler, {"refusal": step[1]})
                elif kind == "tool":
                    call = {
                        "index": calls,
                        "id": f"call_{scenario}_{index}_{calls}",
                        "type": "function",
                        "function": {"name": tools.get(normal(step[1]), step[1]), "arguments": json.dumps(step[2])},
                    }
                    self.chunk(handler, {"tool_calls": [call]})
                    calls += 1
                elif kind == "gate":
                    self.record(event="gate", scenario=scenario, gate=step[1])
                    if not self.gate(step[1]).wait(WAIT):
                        self.record(event="gate-timeout", scenario=scenario, gate=step[1])
                        return
                elif kind == "finish":
                    self.chunk(handler, {}, finish=step[1])
            handler.wfile.write(b"data: [DONE]\n\n")
            handler.wfile.flush()
            self.record(event="finished", scenario=scenario, turn=turn)
        except OSError:
            self.record(event="disconnected", scenario=scenario, turn=turn)


class HostError(Exception):

class Host:
    """A host test build, driven over its stdin (e2e_driver.rs)."""

    def __init__(self, command, env, log_path):
        self.log = open(log_path, "w", encoding="utf-8")
        self.logging = threading.Lock()
        self.process = subprocess.Popen(
            command, env=env, stdin=subprocess.PIPE, stdout=subprocess.PIPE, text=True, bufsize=1
        )
        self.answers = {}
        self.events = []
        self.condition = threading.Condition()
        self.sending = threading.Lock()
        self.next_id = 0
        threading.Thread(target=self._read, daemon=True).start()

    def _note(self, direction, text):
        with self.logging:
            if not self.log.closed:
                self.log.write(f"{time.time():.3f} {direction} {text}\n")
                self.log.flush()

    def _read(self):
        for line in self.process.stdout:
            if not line.startswith("E2E_DRIVER "):
                continue
            text = line[len("E2E_DRIVER "):].strip()
            self._note("<", text)
            try:
                message = json.loads(text)
            except ValueError:
                continue
            with self.condition:
                if message.get("id") is not None:
                    self.answers[message["id"]] = message
                else:
                    self.events.append(message)
                self.condition.notify_all()
        with self.condition:
            self.events.append({"event": "exited"})
            self.condition.notify_all()

    def send(self, op, **args):
        with self.sending:
            self.next_id += 1
            ident = self.next_id
            line = json.dumps({"id": ident, "op": op, **args})
            self._note(">", line)
            self.process.stdin.write(line + "\n")
            self.process.stdin.flush()
        return ident

    def answer(self, ident, timeout=WAIT):
        deadline = time.time() + timeout
        with self.condition:
            while ident not in self.answers:
                left = deadline - time.time()
                if left <= 0 or self.process.poll() is not None:
                    raise HostError(f"no answer to command {ident}")
                self.condition.wait(min(left, 1.0))
            message = self.answers.pop(ident)
        if not message.get("ok"):
            raise HostError(message.get("error") or "the command failed")
        return message.get("result")

    def call(self, op, timeout=WAIT, **args):
        return self.answer(self.send(op, **args), timeout)

    def wait_event(self, name, timeout=WAIT):
        """Wait for, and take, the oldest unsolicited driver event `name`."""
        deadline = time.time() + timeout
        with self.condition:
            while True:
                for position, event in enumerate(self.events):
                    if event.get("event") == name:
                        return self.events.pop(position)
                left = deadline - time.time()
                if left <= 0 or self.process.poll() is not None:
                    raise HostError(f"no {name} event")
                self.condition.wait(min(left, 1.0))

    def ready(self, timeout=WAIT):
        self.wait_event("driver_ready", timeout)
        deadline = time.time() + timeout
        while time.time() < deadline:
            status = self.call("status")
            if status["state"] == "ready":
                return status
            if status["state"] in ("failed", "mismatch", "duplicate"):
                raise HostError(f"the runtime is {status['state']}: {status.get('detail')}")
            time.sleep(0.2)
        raise HostError("the runtime did not become ready")

    def fetch(self, method, target, body=None):
        result = self.call("fetch", method=method, target=target, body=None if body is None else json.dumps(body))
        try:
            value = json.loads(result["body"]) if result["body"] else None
        except ValueError:
            value = result["body"]
        return result["status"], value

    def try_fetch(self, method, target, body=None):
        """A fetch, or ("refused", message) when the host itself refuses it."""
        try:
            return self.fetch(method, target, body)
        except HostError as error:
            return "refused", str(error)

    def new_session(self):
        status, body = self.fetch("POST", f"{API}/sessions", {})
        if status != 201:
            raise HostError(f"sessions.create: {status} {body}")
        return body["id"]

    def start_run(self, session, message):
        status, body = self.fetch("POST", f"{API}/runs", {"message": message, "session_id": session})
        if status != 202:
            raise HostError(f"runs.create: {status} {body}")
        return body["id"]

    def session(self, session):
        status, body = self.fetch("GET", f"{API}/sessions/{session}")
        if status != 200:
            raise HostError(f"sessions.get: {status} {body}")
        return body

    def workspace(self):
        status, body = self.fetch("GET", f"{API}/workspace")
        if status != 200:
            raise HostError(f"workspace.get: {status} {body}")
        return body["workspace"]["effective"]

    def stop(self):
        try:
            self.process.stdin.close()
        except OSError:
            pass
        self.process.terminate()
        try:
            self.process.wait(timeout=15)
        except subprocess.TimeoutExpired:
            self.process.kill()
            self.process.wait()
        with self.logging:
            self.log.close()


class Stream:
    """A run's events as the host relays them (subscribe, then poll)."""

    def __init__(self, host, run):
        self.host = host
        self.run = run
        self.subscription = host.call("subscribe", target=f"{API}/runs/{run}/events")
        self.events = []
        self.closed = False
        self.error = None

    def poll(self, wait_ms=2000):
        batch = self.host.call("poll", subscription=self.subscription, wait_ms=wait_ms)
        self.events.extend(batch["events"])
        if batch["closed"]:
            self.closed, self.error = True, batch["error"]

    def until(self, predicate, timeout=WAIT):
        deadline = time.time() + timeout
        while not predicate(self) and not self.closed and time.time() < deadline:
            self.poll()
        return predicate(self)

    def finish(self, timeout=WAIT):
        self.until(lambda stream: stream.closed, timeout)
        return self

    def drop(self):
        self.host.call("drop_relay", subscription=self.subscription)

    def names(self):
        return [event["event"] for event in self.events]

    def has(self, name):
        return name in self.names()

    def ids(self):
        return [int(event["id"]) for event in self.events if event.get("id") not in (None, "")]

    def data(self, name):
        found = []
        for event in self.events:
            if event["event"] == name:
                try:
                    found.append(json.loads(event["data"]))
                except ValueError:
                    found.append(event["data"])
        return found

    def text(self):
        return "".join(item.get("content") or "" for item in self.data("model.delta") if isinstance(item, dict))


class Checks:
    """Named expectations for one scenario; a failure says what was seen."""

    def __init__(self, scenario):
        self.scenario = scenario
        self.results = {}
        self.failures = []

    def expect(self, name, ok, seen=None):
        self.results[name] = {"ok": bool(ok), "seen": seen}
        if not ok:
            self.failures.append(f"{self.scenario}: {name} (seen: {seen!r})")

"""Run scenarios through the desktop host, on a GitHub-hosted runner (D5a).

Usage: python runs.py <out-dir> <lightagent-cli> -- <host command...>

The host is a test build (`--features e2e-hooks`) driven over its stdin
(harness.Host). Its runtime is the real `lightagent serve --desktop`; its
provider is harness.FakeProvider, so every turn is deterministic. Scenarios
synchronize on what the host relayed or the provider received, not on time.
LIGHTAGENT_HOME, set by the caller, is isolated.

  stream      deltas stream, a tool that needs no approval runs, the answer is saved
  approve     a file write waits for approval; granted, it runs exactly once
  deny        denied, it never runs, and the model is told so
  cancel      a run cancelled mid-stream ends cancelled; the session takes the next run
  relay       the host's relay is dropped twice mid-run; it resumes, and the
              events it delivers are every event of the run, once each, in order
  regenerate  R1 through the host: a finished answer replaces the saved one;
              a provider failure, a refusal, an empty answer and a cancel keep
              it; a stale answer id and a busy session are refused

This is host-level evidence (host transport and relay plus the real
runtime), not browser or native WebView evidence. Writes runs.json; exits
non-zero when any expectation fails.
"""

def scripts(provider):
    def write(name, content):
        return ("tool", "fs.write", {"path": name, "content": content, "append": True})

    provider.scripts.update({
        "stream": lambda p, turn, index: (
            [("text", "Checking "), ("text", "the time. "), ("tool", "datetime.now", {}), ("finish", "tool_calls")]
            if turn == 0 else [("text", "It is "), ("text", "now."), ("finish", "stop")]
        ),
        "approve": lambda p, turn, index: (
            [write("approved.txt", "granted\n"), ("finish", "tool_calls")]

        ),
        "deny": lambda p, turn, index: (
            [write("denied.txt", "should not exist\n"), ("finish", "tool_calls")]
            if turn == 0 else [("text", "Not written."), ("finish", "stop")]
        ),
        "cancel": lambda p, turn, index: [("text", "Working"), ("gate", "cancel"), ("text", " never"), ("finish", "stop")],
        "relay": lambda p, turn, index: (
            [("text", "one "), ("gate", "relay"), ("text", "two "), ("text", "three "),
             ("tool", "datetime.now", {}), ("finish", "tool_calls")]
            if turn == 0 else [("gate", "relay-2"), ("text", "four."), ("finish", "stop")]
        ),
        "regen": lambda p, turn, index: {
            "first": [("text", "First answer."), ("finish", "stop")],
            "second": [("text", "Second answer."), ("finish", "stop")],
            "fail": [("status", 500)],
            "refusal": [("refusal", "I can't help with that."), ("finish", "stop")],
            "empty": [("finish", "stop")],
            "hold": [("text", "Partial"), ("gate", "regen-hold"), ("text", " never"), ("finish", "stop")],
        }[p.modes.get("regen", "first")],
    })


def recorded(host, session, runs):

def scenario_approval(host, provider, workspace, approve):
    name = "approve" if approve else "deny"
    path = os.path.join(workspace, "approved.txt" if approve else "denied.txt")

    status, listing = host.fetch("GET", f"{API}/approvals")

def scenario_cancel(host, provider, workspace):
    check = Checks("cancel")
    session = host.new_session()
    run = host.start_run(session, "[e2e:cancel] Work for a while.")
    stream = Stream(host, run)
    stream.until(lambda s: "Working" in s.text())

    status, _ = host.fetch("POST", f"{API}/runs/{run}/cancel", {})
    stream.finish()
    _, info = host.fetch("GET", f"{API}/runs/{run}")
    saved = recorded(host, session, 1)

    deadline, follow = time.time() + 10, None

        follow = host.fetch("POST", f"{API}/runs", {"message": "[e2e:after-cancel] Next.", "session_id": session})
        if follow[0] != 409:
            break
        time.sleep(0.1)
    if follow and follow[0] == 202:
        Stream(host, follow[1]["id"]).finish()
    check.expect("the next run on the session is admitted", follow and follow[0] == 202, follow)
    provider.open("cancel")
    check.expect("the cancelled turn was not sent again", len(provider.requests("cancel")) == 1, provider.requests("cancel"))
    return check

def scenario_regenerate(host, provider, workspace):
    check = Checks("regenerate")
    session = host.new_session()
    provider.modes["regen"] = "first"

    def regenerate(mode, answer):
        provider.modes["regen"] = mode
        return host.fetch("POST", f"{API}/sessions/{session}/regenerate", {"run_id": answer})

    def settled():

        # session is free again (a busy one is refused as busy first).
        deadline = time.time() + 30
        while time.time() < deadline:
            status, body = host.fetch("POST", f"{API}/sessions/{session}/regenerate", {"run_id": "r-stale"})
            if status == 409 and "latest" in json.dumps(body):
                return True
            time.sleep(0.1)
        return False

    status, created = regenerate("second", saved["runs"][-1]["run_id"])
    replaced = Stream(host, created["id"]).finish() if status == 202 else None
    current = harness.wait_until(lambda: (lambda s: s if harness.answers(s) == ["Second answer."] else None)(host.session(session)), timeout=30)
    check.expect("a finished answer replaces the saved one", bool(current), host.session(session))
    if not current:
        return check
    check.expect("one run record, the new one", len(current["runs"]) == 1 and current["runs"][-1]["run_id"] != saved["runs"][-1]["run_id"], current["runs"])
    check.expect("the replacement streamed", replaced is not None and replaced.text() == "Second answer.", replaced and replaced.text())
    check.expect("a stale answer id is refused", settled(), None)
    answer = current["runs"][-1]["run_id"]

    for mode in ("fail", "refusal", "empty"):
        before = host.session(session)
        status, created = regenerate(mode, answer)
        stream = Stream(host, created["id"]).finish() if status == 202 else None
        free = settled()
        after = host.session(session)
        check.expect(f"{mode}: accepted and run", status == 202 and stream is not None, (status, created))
        check.expect(f"{mode}: the saved answer is kept", free and after["messages"] == before["messages"] and after["runs"] == before["runs"], harness.answers(after))

    before = host.session(session)
    status, created = regenerate("hold", answer)
    stream = Stream(host, created["id"])
    stream.until(lambda s: "Partial" in s.text())
    busy = host.fetch("POST", f"{API}/sessions/{session}/regenerate", {"run_id": answer})
    host.fetch("POST", f"{API}/runs/{created['id']}/cancel", {})
    stream.finish()

            ("deny", lambda: scenario_approval(host, provider, workspace, False)),

                check = scenario()
                results[name] = check.results
                failures.extend(check.failures)
            except Exception as error:  # a scenario that breaks is a failure, and the rest still run

        failures.append(f"setup: {error!r}")
    finally:
        host.stop()
        provider.close()

    results["failures"] = failures
    with open(os.path.join(out, "runs.json"), "w", encoding="utf-8") as handle:
        json.dump(results, handle, indent=1)
    print(json.dumps({"failures": failures}, indent=1))
