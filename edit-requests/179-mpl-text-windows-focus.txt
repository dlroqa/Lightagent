- desktop-sbom.py: selectors 0.38.0 is published on crates.io without a
  license file (found by the notices gate on every target). An MPL-2.0
  component without its own license file now gets the full MPL-2.0 text
  from another MPL-2.0 component's published file, and the notices say
  so; the gate fails only when no MPL-2.0 text is available at all. The
  MPL source set adds MPL-2.0.txt and names the archives without one.
- native.py (Windows, recorded): real Tab presses at launch, while the
  page holds focus; after another window was in front and this one is
  brought back; and after a direct Win32 SetFocus on the WebView's
  container window (WRY_WEBVIEW), the mechanism a host fix would use.
  The post-scenario probe gains the same container stage.

def windows_away_and_back(host):
    """Another window brought in front of the app, then the app brought back."""
    away = subprocess.Popen([sys.executable, "-c",
                             "import tkinter as t; r = t.Tk(); r.title('la-focus-away'); r.after(30000, r.destroy); r.mainloop()"])
    try:
        time.sleep(3)
        front = windows_input("[Wi]::Front('la-focus-away')")
        time.sleep(0.6)
        focus_window(host)
        return front
    finally:
        away.kill()


def windows_launch_keyboard(view, host):
    """Real key presses right after launch, while the page holds focus; after
    another window was in front; and, if focus did not come back, after a
    direct Win32 SetFocus on the WebView's container window."""
    stages = {"launch": windows_stage(view, host)}
    stages["away_and_back"] = {"away": windows_away_and_back(host), **windows_stage(view, host)}
    if not stages["away_and_back"]["tab"]["moved"]:
        stages["container_focus"] = {"set": windows_input(f"[Wi]::FocusContainer({host.process.pid})"), **windows_stage(view, host)}
    return stages
