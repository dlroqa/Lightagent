name: desktop spike

# D1 evidence only (branch desktop/d1-spike; never merged). Builds the smallest
# Tauri 2 and Electron windows on every declared target and records runner
# measurements: build result, binary deployment target, dependency policy,
# package size, time to first page load, idle memory, and a screenshot.
# Numbers are GitHub-hosted runner measurements, not user-machine claims.

on:
  pull_request:
    paths:
      - 'desktop-spike/**'
      - '.github/workflows/desktop-spike.yml'
  workflow_dispatch:

concurrency:
  group: desktop-spike-${{ github.ref }}
  cancel-in-progress: true

permissions:
  contents: read

env:
  CARGO_INCREMENTAL: '0'
  APPIMAGE_EXTRACT_AND_RUN: '1'

jobs:
  tauri:
    name: tauri (${{ matrix.label }})
    runs-on: ${{ matrix.os }}
    timeout-minutes: 60
    strategy:
      fail-fast: false
      matrix:
        include:
          - { os: macos-15-intel, label: macos-x64, arch: x86_64, macos_min: '10.14', bundles: 'app,dmg' }
          - { os: macos-15, label: macos-arm64, arch: arm64, macos_min: '11.0', bundles: 'app,dmg' }
          - { os: windows-latest, label: windows-x64, arch: x86_64, bundles: 'nsis,msi' }
          - { os: ubuntu-22.04, label: linux-x64, arch: x86_64, bundles: 'deb,appimage' }
    defaults:
      run:
        shell: bash
    steps:
      - uses: actions/checkout@v7
        with:
          ref: ${{ github.event.pull_request.head.sha || github.sha }}
      - name: Runner identity
        env:
          WANT_ARCH: ${{ matrix.arch }}
        run: |
          arch=$(uname -m); case "$arch" in x86_64|amd64|AMD64) arch=x86_64;; aarch64|arm64) arch=arm64;; esac
          {
            echo "### tauri ${{ matrix.label }} runner"
            echo "- image: ${ImageOS:-unknown} ${ImageVersion:-}"
            echo "- head: \`$(git rev-parse HEAD)\` tree \`$(git rev-parse 'HEAD^{tree}')\`"
            echo "- uname -m: $arch"
            if [ "$(uname -s)" = Darwin ]; then
              echo "- macOS: $(sw_vers -productVersion); translated: $(sysctl -n sysctl.proc_translated 2>/dev/null || echo 0)"
            fi
          } | tee -a "$GITHUB_STEP_SUMMARY"
          [ "$arch" = "$WANT_ARCH" ] || { echo "runner is $arch, expected $WANT_ARCH" >&2; exit 1; }
      - name: macOS deployment target
        if: matrix.macos_min
        run: echo "MACOSX_DEPLOYMENT_TARGET=${{ matrix.macos_min }}" >> "$GITHUB_ENV"
      - name: Linux system packages (webview)
        if: runner.os == 'Linux'
        run: |
          sudo apt-get update
          sudo apt-get install -y libwebkit2gtk-4.1-dev libayatana-appindicator3-dev librsvg2-dev patchelf file xvfb imagemagick
      - run: rustup show
      - uses: Swatinem/rust-cache@v2
        with:
          workspaces: desktop-spike/tauri/src-tauri
          key: spike-${{ matrix.label }}
      - uses: actions/setup-node@v7
        with:
          node-version: '22'
      - uses: actions/setup-python@v6
        with:
          python-version: '3.12'
      - run: python -m pip install psutil==7.1.0 pillow==11.3.0
      - name: Icons from the repository logo
        working-directory: desktop-spike/tauri
        run: npx --yes @tauri-apps/cli@2.12.3 icon ../../icon/LightagentLogo.png -o src-tauri/icons
      - name: Build and bundle
        working-directory: desktop-spike/tauri
        run: |
          start=$(date +%s)
          npx --yes @tauri-apps/cli@2.12.3 build --ci --bundles ${{ matrix.bundles }}
          echo "- build+bundle wall time: $(( $(date +%s) - start ))s (cold or cached per rust-cache)" | tee -a "$GITHUB_STEP_SUMMARY"
      - name: Dependency policy (repository rules applied to the desktop crate)
        working-directory: desktop-spike/tauri/src-tauri
        run: |
          status=0
          for crate in aws-lc-sys aws-lc-rs openssl-sys native-tls bindgen cmake; do
            if cargo tree --locked --target all -e normal,build -i "$crate" >/dev/null 2>&1; then
              echo "- POLICY: \`$crate\` is in the graph" | tee -a "$GITHUB_STEP_SUMMARY"; status=1
              cargo tree --locked --target all -e normal,build -i "$crate" | head -15
            fi
          done
          echo "- crates in Cargo.lock: $(grep -c '^name = ' Cargo.lock)" | tee -a "$GITHUB_STEP_SUMMARY"
          [ $status = 0 ] && echo "- dependency policy: clean (no cmake/OpenSSL/libclang crates, all targets)" | tee -a "$GITHUB_STEP_SUMMARY"
          exit $status
      - name: Binary facts
        working-directory: desktop-spike/tauri/src-tauri/target/release
        run: |
          {
            echo "#### binary"
            if [ "$(uname -s)" = Darwin ]; then
              bin=$(ls -d bundle/macos/*.app)/Contents/MacOS/lightagent-desktop-spike
              echo "- archs: $(lipo -archs "$bin")"
              echo "- load command (deployment target):"
              echo '```'; otool -l "$bin" | grep -A4 -E 'LC_BUILD_VERSION|LC_VERSION_MIN_MACOSX' | head -12; echo '```'
              echo "- Info.plist LSMinimumSystemVersion: $(/usr/libexec/PlistBuddy -c 'Print :LSMinimumSystemVersion' "$(ls -d bundle/macos/*.app)/Contents/Info.plist" 2>/dev/null || echo unset)"
            fi
            echo "#### packages"
            find bundle -maxdepth 2 -type f \( -name '*.dmg' -o -name '*.msi' -o -name '*.exe' -o -name '*.deb' -o -name '*.AppImage' \) -exec ls -l {} \; | awk '{printf "- %s: %.1f MB\n", $NF, $5/1048576}'
          } | tee -a "$GITHUB_STEP_SUMMARY"
      - name: Launch and measure (runner measurement)
        run: |
          out="$RUNNER_TEMP/spike-out"
          r=desktop-spike/tauri/src-tauri/target/release
          case "$(uname -s)" in
            Darwin) bin="$(ls -d $r/bundle/macos/*.app)/Contents/MacOS/lightagent-desktop-spike"; python desktop-spike/measure.py "$out" -- "$bin" ;;
            Linux) xvfb-run -a -s '-screen 0 1280x800x24' python desktop-spike/measure.py "$out" -- "$r/lightagent-desktop-spike" ;;
            *) python desktop-spike/measure.py "$out" -- "$r/lightagent-desktop-spike.exe" ;;
          esac
          python -c "import json,sys; d=json.load(open(sys.argv[1])); print('- page loaded (wall, from spawn): %ss; idle RSS after 10s: %s MB' % (d.get('page_loaded_wall_s'), d.get('idle_rss_mb_after_10s')))" "$out/measure.json" | tee -a "$GITHUB_STEP_SUMMARY"
      - uses: actions/upload-artifact@v7
        if: always()
        with:
          name: spike-tauri-${{ matrix.label }}
          retention-days: 30
          path: |
            ${{ runner.temp }}/spike-out/
            desktop-spike/tauri/src-tauri/Cargo.lock
            desktop-spike/tauri/src-tauri/target/release/bundle/**/*.dmg
            desktop-spike/tauri/src-tauri/target/release/bundle/**/*.msi
            desktop-spike/tauri/src-tauri/target/release/bundle/nsis/*.exe
            desktop-spike/tauri/src-tauri/target/release/bundle/**/*.deb
            desktop-spike/tauri/src-tauri/target/release/bundle/**/*.AppImage

  electron:
    name: electron (${{ matrix.label }})
    runs-on: ${{ matrix.os }}
    timeout-minutes: 30
    strategy:
      fail-fast: false
      matrix:
        include:
          - { os: macos-15-intel, label: macos-x64, arch: x86_64 }
          - { os: macos-15, label: macos-arm64, arch: arm64 }
          - { os: windows-latest, label: windows-x64, arch: x86_64 }
          - { os: ubuntu-22.04, label: linux-x64, arch: x86_64 }
    defaults:
      run:
        shell: bash
    steps:
      - uses: actions/checkout@v7
        with:
          ref: ${{ github.event.pull_request.head.sha || github.sha }}
      - name: Runner identity
        env:
          WANT_ARCH: ${{ matrix.arch }}
        run: |
          arch=$(uname -m); case "$arch" in x86_64|amd64|AMD64) arch=x86_64;; aarch64|arm64) arch=arm64;; esac
          echo "### electron ${{ matrix.label }} runner: $arch, head \`$(git rev-parse HEAD)\`" | tee -a "$GITHUB_STEP_SUMMARY"
          [ "$arch" = "$WANT_ARCH" ] || { echo "runner is $arch, expected $WANT_ARCH" >&2; exit 1; }
      - name: Linux display
        if: runner.os == 'Linux'
        run: sudo apt-get update && sudo apt-get install -y xvfb imagemagick
      - uses: actions/setup-node@v7
        with:
          node-version: '22'
      - uses: actions/setup-python@v6
        with:
          python-version: '3.12'
      - run: python -m pip install psutil==7.1.0 pillow==11.3.0
      - name: Install Electron 42.11.14
        working-directory: desktop-spike/electron
        run: |
          npm install --no-audit --no-fund
          # The binary arrives through the package's own install script; run it
          # explicitly so an npm that skips lifecycle scripts cannot hide it.
          [ -d node_modules/electron/dist ] || node node_modules/electron/install.js
          ls node_modules/electron
          cat node_modules/electron/path.txt; echo
      - name: Runtime size (proxy for package size)
        working-directory: desktop-spike/electron
        run: |
          dist=node_modules/electron/dist
          python -c "import shutil,sys; shutil.make_archive(sys.argv[2], 'zip', sys.argv[1])" "$dist" "$RUNNER_TEMP/electron-runtime"
          python -c "import os,sys; print('- Electron runtime, zipped (package-size proxy, no app code or installer): %.1f MB' % (os.path.getsize(sys.argv[1])/1048576))" "$RUNNER_TEMP/electron-runtime.zip" | tee -a "$GITHUB_STEP_SUMMARY"
          echo "- macOS minimum declared by this runtime: $( [ -f $dist/Electron.app/Contents/Info.plist ] && /usr/libexec/PlistBuddy -c 'Print :LSMinimumSystemVersion' $dist/Electron.app/Contents/Info.plist || echo n/a)" | tee -a "$GITHUB_STEP_SUMMARY"
      - name: Launch and measure (runner measurement)
        run: |
          out="$RUNNER_TEMP/spike-out"
          bin=$(node -p "require('./desktop-spike/electron/node_modules/electron')")
          case "$(uname -s)" in
            Linux) xvfb-run -a -s '-screen 0 1280x800x24' python desktop-spike/measure.py "$out" -- "$bin" desktop-spike/electron ;;
            *) python desktop-spike/measure.py "$out" -- "$bin" desktop-spike/electron ;;
          esac
          python -c "import json,sys; d=json.load(open(sys.argv[1])); print('- page loaded (wall, from spawn): %ss; idle RSS after 10s: %s MB' % (d.get('page_loaded_wall_s'), d.get('idle_rss_mb_after_10s')))" "$out/measure.json" | tee -a "$GITHUB_STEP_SUMMARY"
      - uses: actions/upload-artifact@v7
        if: always()
        with:
          name: spike-electron-${{ matrix.label }}
          retention-days: 30
          path: ${{ runner.temp }}/spike-out/
=== end
