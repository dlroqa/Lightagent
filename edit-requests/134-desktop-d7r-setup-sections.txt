/// The built-in, account-free search endpoint offered by `lightagent setup web`.
pub const DUCKDUCKGO_SEARCH_ENDPOINT: &str = "https://html.duckduckgo.com/html/";

/// What `lightagent setup` → web offers, in its order: the web tools off,
/// fetching pages only, or agentic search through DuckDuckGo or a SearXNG /
/// custom JSON endpoint. Setup and the desktop's Settings both apply it with
/// [`WebMode::apply`], so they write the same keys.
#[derive(Clone, Copy, Debug, PartialEq, Eq, Serialize, Deserialize)]
#[serde(rename_all = "snake_case")]
pub enum WebMode {
    Off,
    Fetch,
    Duckduckgo,
    Custom,
}

impl WebMode {
    /// Setup's menu order.
    pub const ALL: [Self; 4] = [Self::Off, Self::Fetch, Self::Duckduckgo, Self::Custom];

    /// The mode `config` is in.
    pub fn of(config: &Config) -> Self {
        if !config.web.enabled {
            Self::Off
        } else if config.web.search.endpoint.as_deref() == Some(DUCKDUCKGO_SEARCH_ENDPOINT) {
            Self::Duckduckgo
        } else if config.web.search.endpoint.is_some() {
            Self::Custom
        } else {
            Self::Fetch
        }
    }

    /// Write the mode as setup does. Searching also turns on realtime
    /// knowledge; DuckDuckGo needs no key. `endpoint` is the custom search
    /// URL, used by `Custom` only; trimmed, and empty means none.
    pub fn apply(self, config: &mut Config, endpoint: Option<&str>) {
        config.web.enabled = self != Self::Off;
        match self {
            Self::Duckduckgo => {
                config.web.search.endpoint = Some(DUCKDUCKGO_SEARCH_ENDPOINT.to_owned());
                config.web.search.query_param = "q".to_owned();
                config.web.search.api_key = None;
            }
            Self::Custom => {
                config.web.search.endpoint = endpoint
                    .map(str::trim)
                    .filter(|url| !url.is_empty())
                    .map(str::to_owned);
            }
            Self::Off | Self::Fetch => config.web.search.endpoint = None,
        }
        config.rag.realtime_enabled = matches!(self, Self::Duckduckgo | Self::Custom);
    }
}

const WEB_MODES: { mode: WebMode; label: string }[] = [

  { mode: "duckduckgo", label: "Agentic search — DuckDuckGo (no account)" },

/**
 * `lightagent setup` → web, on the same keys: the web tools off, fetching
 * pages only, or agentic search, which also turns on realtime knowledge. A
 * custom search saves only with its URL.
 */
function WebAccessField({ settings, disabled, onSave }: {
  settings: LightagentSettings | null;
  disabled: boolean;
  onSave: (patch: Partial<LightagentSettings>) => Promise<void> | void;
}) {
  const mode: WebMode = settings?.web_mode ?? (settings?.web_enabled ? "fetch" : "off");
  const savedEndpoint = mode === "custom" ? settings?.web_search_endpoint ?? "" : "";
  const [choice, setChoice] = useState<WebMode>(mode);
  const [endpoint, setEndpoint] = useState(savedEndpoint);
  const [error, setError] = useState<string | null>(null);
  useEffect(() => {
    setChoice(mode);
    setEndpoint(savedEndpoint);
    setError(null);
  }, [mode, savedEndpoint]);
  const save = (next: WebMode, url: string) => {
    if (next === "custom" && !url.trim()) {
      setError("Enter the search endpoint URL, such as http://127.0.0.1:8080/search?format=json.");
      return;
    }
    setError(null);
    void onSave({
      web_mode: next,
      web_enabled: next !== "off",
      web_search_endpoint: next === "custom" ? url.trim() : null,
    });
  };
  return (
    <div className="field" style={{ padding: "12px 0", borderBottom: "1px solid var(--rule)" }}>
      <label className="field__label" htmlFor="web-access">Web access</label>
      <select id="web-access" className="select" value={choice} disabled={disabled}
        onChange={(event) => {
          const next = event.target.value as WebMode;
          setChoice(next);
          if (next !== "custom") save(next, "");
        }}>
        {WEB_MODES.map((option) => <option key={option.mode} value={option.mode}>{option.label}</option>)}
      </select>
      {choice === "custom" && (
        <form style={{ display: "flex", gap: 8, alignItems: "flex-end", flexWrap: "wrap" }}
          onSubmit={(event) => {
            event.preventDefault();
            save("custom", endpoint);
          }}>
          <div className="field" style={{ flex: "1 1 220px" }}>
            <label className="field__label" htmlFor="web-access-endpoint">Search endpoint (JSON)</label>
            <input id="web-access-endpoint" className="input" value={endpoint} disabled={disabled}
              placeholder="http://127.0.0.1:8080/search?format=json" autoComplete="off" spellCheck={false}
              onChange={(event) => setEndpoint(event.target.value)} />
          </div>
          <button type="submit" className="btn" disabled={disabled}>Save endpoint</button>
        </form>
      )}
      {error && <p className="config-editor__error" role="alert">{error}</p>}
      <div style={{ fontSize: 11.5, color: "var(--text-muted)" }}>
        The same choice as <code>lightagent setup</code> → web. Searching also turns on realtime knowledge.
      </div>
    </div>
  );
}

// The Settings controls that mirror `lightagent setup` (inventory rows
// setup.tools.mode, setup.web.mode, setup.approvals.policy and
// setup.tui.reasoning) against the real harness. Each choice is made on the
// Settings screen and read back from the API and the config keys. Web access
// offers setup's four modes, and a custom search needs its URL first.
// Filesystem tools and Terminal tool give setup's three tool choices.
// Approval policy and Show reasoning in terminal save what setup saves.
// Everything is restored.
// crates/lightagent/tests/desktop_setup_sections.rs proves each choice
// writes what `lightagent setup` writes; desktop/e2e/runs.py
// `setup_sections` covers the host.

const ENDPOINT = "http://127.0.0.1:9/search?format=json";
const DUCKDUCKGO = "https://html.duckduckgo.com/html/";

export async function checkSetupSections(browser, base, outDir) {
  const failures = [];
  const page = await browser.newPage({ viewport: { width: 1280, height: 1400 } });
  const errors = [];
  page.on("pageerror", (error) => errors.push(error.message));
  const api = `${base}/api/lightagent/v1`;
  const settings = async () => (await fetch(`${api}/settings`)).json();
  const key = async (name) => {
    const response = await fetch(`${api}/config/${name}`);
    return response.status === 200 ? (await response.json()).value : null;
  };
  const until = async (what, probe, timeout = 10000) => {
    const deadline = Date.now() + timeout;
    while (Date.now() < deadline) {
      if (await probe()) return;
      await new Promise((resolve) => setTimeout(resolve, 150));
    }
    throw new Error(`timed out waiting for ${what}`);
  };
  const step = async (name, body) => {
    try {
      await body();
    } catch (err) {
      failures.push(`setup sections: ${name}: ${err instanceof Error ? err.message.split("\n")[0] : String(err)}`);
      await page.screenshot({ path: `${outDir}/setup-sections-failed-${name.replace(/\W+/g, "-")}.png` }).catch(() => {});
    }
  };
  const toggle = async (name, on) => {
    const control = page.getByRole("switch", { name, exact: true });
    if ((await control.getAttribute("aria-checked")) !== String(on)) {
      await control.click({ timeout: 10000 });
    }
    await until(`${name} ${on ? "on" : "off"}`, async () => (await control.getAttribute("aria-checked")) === String(on));
  };
  const original = await settings();
  try {
    await page.goto(`${base}/#/settings`);
    const web = page.locator("#web-access");
    await web.waitFor({ timeout: 15000 });

    await step("web access: setup's modes", async () => {
      for (const [mode, label, endpoint, realtime] of [
        ["fetch", "Fetch web pages only", "", "false"],
        ["duckduckgo", "Agentic search — DuckDuckGo (no account)", DUCKDUCKGO, "true"],
        ["off", "Off", "", "false"],
      ]) {
        await web.selectOption({ label });
        await until(`web mode ${mode}`, async () => (await settings()).web_mode === mode);
        const seen = [await key("web.enabled"), await key("web.search.endpoint"), await key("rag.realtime_enabled")];
        const expected = [String(mode !== "off"), endpoint, realtime];
        if (JSON.stringify(seen) !== JSON.stringify(expected)) {
          throw new Error(`${mode}: the config reads ${JSON.stringify(seen)}, not ${JSON.stringify(expected)}`);
        }
      }
    });

    await step("web access: a custom search needs its URL, then saves it", async () => {
      await web.selectOption({ label: "Agentic search — SearXNG or custom JSON endpoint" });
      const url = page.locator("#web-access-endpoint");
      await url.fill("");
      await page.getByRole("button", { name: "Save endpoint", exact: true }).click({ timeout: 10000 });
      await page.getByRole("alert").filter({ hasText: "Enter the search endpoint URL" }).waitFor({ timeout: 5000 });
      if ((await settings()).web_mode === "custom") throw new Error("a custom search was saved without its URL");
      await url.fill(ENDPOINT);
      await page.screenshot({ path: `${outDir}/setup-sections-web-custom.png` });
      await page.getByRole("button", { name: "Save endpoint", exact: true }).click({ timeout: 10000 });
      await until("the custom search", async () => (await settings()).web_mode === "custom");
      const seen = [await key("web.search.endpoint"), await key("rag.realtime_enabled")];
      if (seen[0] !== ENDPOINT || seen[1] !== "true") throw new Error(`the config reads ${JSON.stringify(seen)}`);
    });

    await step("local tools: setup's three choices", async () => {
      for (const [files, terminal] of [[true, true], [true, false], [false, false]]) {
        await toggle("Filesystem tools", files);
        if (files) await toggle("Terminal tool", terminal);
        await until(`tools ${files}/${terminal}`, async () => {
          const saved = await settings();
          return saved.filesystem_tools_enabled === files && saved.terminal_enabled === terminal;
        });
      }
    });

    await step("approval policy and reasoning: what setup saves", async () => {
      const policy = page.locator("#agent-approval");
      for (const value of ["strict", "permissive", "balanced"]) {
        await policy.selectOption(value);
        await until(`approval ${value}`, async () => (await settings()).approval_policy === value);
      }
      for (const on of [false, true]) {
        await toggle("Show reasoning in terminal", on);
        await until(`reasoning ${on}`, async () =>
          (await settings()).show_reasoning_in_tui === on && (await key("tui.show_reasoning")) === String(on));
      }
    });
    if (errors.length) failures.push(`setup sections: page errors: ${errors.join("; ")}`);
  } catch (err) {
    failures.push(`setup sections: ${err instanceof Error ? err.message.split("\n")[0] : String(err)}`);
  } finally {
    await page.close();
    const { revision: _revision, ...restore } = original;
    await fetch(`${api}/settings`, {
      method: "PUT",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify(restore),
    }).catch(() => {});
  }
  return failures;
}

use serde_json::{Value, json};

const KEY: &str = "b4c5d6e7f8091a2b3c4d5e6f708192a3b4c5d6e7f8091a2b3c4d5e6f708192a3";
const SETTINGS: &str = "/api/lightagent/v1/settings";
const CUSTOM: &str = "http://127.0.0.1:9/search?format=json";

struct Runtime {
    child: Child,
    _lifeline: ChildStdin,
    addr: String,
}

impl Drop for Runtime {
    fn drop(&mut self) {
        let _ = self.child.kill();
        let _ = self.child.wait();
    }
}

fn scratch_home(label: &str) -> PathBuf {
    let nanos = SystemTime::now()
        .duration_since(UNIX_EPOCH)
        .map(|elapsed| elapsed.as_nanos())
        .unwrap_or(0);
    let home = std::env::temp_dir().join(format!(
        "lightagent-desktop-setup-sections-{label}-{}-{nanos}",
        std::process::id()
    ));
    std::fs::create_dir_all(&home).expect("home");
    home
}

fn start(home: &Path) -> Runtime {
    let mut child = Command::new(env!("CARGO_BIN_EXE_lightagent"))
        .args(["serve", "--desktop", "--host", "127.0.0.1", "--port", "0"])
        .env("LIGHTAGENT_HOME", home)
        .stdin(Stdio::piped())
        .stdout(Stdio::piped())
        .stderr(Stdio::null())
        .spawn()
        .expect("start lightagent serve --desktop");
    let mut lifeline = child.stdin.take().expect("stdin");
    writeln!(lifeline, "{KEY}").expect("key");
    lifeline.flush().expect("flush");
    let stdout = child.stdout.take().expect("stdout");
    let (tx, rx) = mpsc::channel();
    std::thread::spawn(move || {
        let mut line = String::new();
        let _ = BufReader::new(stdout).read_line(&mut line);
        let _ = tx.send(line);
    });
    let line = rx.recv_timeout(Duration::from_secs(60)).expect("handshake");
    let handshake: Value = serde_json::from_str(&line).expect("handshake JSON");
    let addr = handshake["addr"].as_str().expect("addr").to_owned();
    Runtime {
        child,
        _lifeline: lifeline,
        addr,
    }
}

fn call(runtime: &Runtime, method: &str, path: &str, body: Option<&str>) -> (u16, String) {
    let mut stream = TcpStream::connect(&runtime.addr).expect("connect");
    stream
        .set_read_timeout(Some(Duration::from_secs(60)))
        .expect("timeout");
    let body = body.unwrap_or("");
    let head = format!(
        "{method} {path} HTTP/1.1\r\nHost: {}\r\nAuthorization: Bearer {KEY}\r\nContent-Type: application/json\r\nContent-Length: {}\r\nConnection: close\r\n\r\n{body}",
        runtime.addr,
        body.len()
    );
    stream.write_all(head.as_bytes()).expect("send");
    let mut response = String::new();
    let _ = stream.read_to_string(&mut response);
    let status = response
        .split_whitespace()
        .nth(1)
        .and_then(|code| code.parse().ok())
        .unwrap_or(0);
    let raw = response
        .split_once("\r\n\r\n")
        .map(|(_, body)| body.to_owned())
        .unwrap_or_default();
    (status, raw)
}

fn parsed(raw: &str) -> Value {
    serde_json::from_str(raw).unwrap_or_else(|error| panic!("{error}: {raw}"))
}

fn cli(home: &Path, args: &[&str]) -> Output {
    Command::new(env!("CARGO_BIN_EXE_lightagent"))
        .args(args)
        .env("LIGHTAGENT_HOME", home)
        .stdin(Stdio::null())
        .output()
        .expect("run lightagent")
}

fn cli_ok(home: &Path, args: &[&str]) -> Output {
    let output = cli(home, args);
    assert!(
        output.status.success(),
        "lightagent {args:?}: {}",
        String::from_utf8_lossy(&output.stderr)
    );
    output
}

/// `lightagent setup <section>` with `input` on its standard input, which is
/// not a terminal, so setup reads numbered choices. Two minutes at most.
fn setup(home: &Path, section: &str, input: &str) {
    let mut child = Command::new(env!("CARGO_BIN_EXE_lightagent"))
        .args(["setup", section])
        .env("LIGHTAGENT_HOME", home)
        .stdin(Stdio::piped())
        .stdout(Stdio::null())
        .stderr(Stdio::piped())
        .spawn()
        .expect("start lightagent setup");
    child
        .stdin
        .take()
        .expect("stdin")
        .write_all(input.as_bytes())
        .expect("input");
    let deadline = Instant::now() + Duration::from_secs(120);
    let status = loop {
        if let Some(status) = child.try_wait().expect("setup status") {
            break status;
        }
        if Instant::now() >= deadline {
            let _ = child.kill();
            panic!("lightagent setup {section} did not finish");
        }
        std::thread::sleep(Duration::from_millis(50));
    };
    let mut stderr = String::new();
    if let Some(mut pipe) = child.stderr.take() {
        let _ = pipe.read_to_string(&mut stderr);
    }
    assert!(status.success(), "lightagent setup {section} {input:?}: {stderr}");
}

/// What the four setup sections write, from `lightagent --json config show`.
/// Every field must be there with its type: a missing one never compares
/// equal to another missing one.
fn sections(home: &Path) -> Value {
    let output = cli_ok(home, &["--json", "config", "show"]);
    let shown: Value = serde_json::from_slice(&output.stdout).expect("config JSON");
    let picked = json!({
        "tools_enabled": shown["tools"]["enabled"],
        "allow_terminal": shown["tools"]["allow_terminal"],
        "web": shown["web"],
        "realtime": shown["rag"]["realtime_enabled"],
        "approval_policy": shown["security"]["approval_policy"],
        "show_reasoning": shown["tui"]["show_reasoning"],
    });
    for field in ["tools_enabled", "allow_terminal", "realtime", "show_reasoning"] {
        assert!(picked[field].is_boolean(), "{field}: {picked}");
    }
    assert!(picked["approval_policy"].is_string(), "{picked}");
    assert!(
        picked["web"]["enabled"].is_boolean() && picked["web"]["search"].is_object(),
        "{picked}"
    );
    picked
}

#[test]
fn settings_write_what_each_setup_choice_writes() {
    let by_setup = scratch_home("setup");
    let by_settings = scratch_home("settings");
    for home in [&by_setup, &by_settings] {
        cli_ok(
            home,
            &["config", "set", "inference.base_url", "http://127.0.0.1:9"],
        );
    }
    let desktop = start(&by_settings);
    assert_eq!(sections(&by_setup), sections(&by_settings), "they start alike");
    for (section, input, patch, mode) in steps() {
        let before = sections(&by_setup);
        setup(&by_setup, section, &input);
        let (status, raw) = call(&desktop, "GET", SETTINGS, None);
        assert_eq!(status, 200, "{raw}");
        let mut settings = parsed(&raw);
        for (field, value) in patch.as_object().expect("patch") {
            settings[field] = value.clone();
        }
        let (status, raw) = call(&desktop, "PUT", SETTINGS, Some(&settings.to_string()));
        assert_eq!(status, 200, "{section} {input:?}: {raw}");
        let after = sections(&by_setup);
        assert_ne!(before, after, "{section} {input:?} changed nothing");
        assert_eq!(after, sections(&by_settings), "{section} {input:?}");
        if let Some(mode) = mode {
            assert_eq!(parsed(&raw)["web_mode"], mode, "{raw}");
        }
    }

    // The searches as setup writes them.
    setup(&by_setup, "web", "3\n");
    let shown = sections(&by_setup);
    assert_eq!(
        shown["web"]["search"]["endpoint"],
        "https://html.duckduckgo.com/html/"
    );
    assert_eq!(shown["realtime"], true);

    // The runtime reads the web mode the CLI's setup wrote.
    let reader = start(&by_setup);
    let (status, raw) = call(&reader, "GET", SETTINGS, None);
    assert_eq!(status, 200, "{raw}");
    assert_eq!(parsed(&raw)["web_mode"], "duckduckgo", "{raw}");

    // A custom search without its URL is refused, and nothing is written.
    let before = sections(&by_settings);
    let (_, raw) = call(&desktop, "GET", SETTINGS, None);
    let mut settings = parsed(&raw);
    settings["web_mode"] = json!("custom");
    settings["web_search_endpoint"] = json!("  ");
    let (status, raw) = call(&desktop, "PUT", SETTINGS, Some(&settings.to_string()));
    assert_eq!(status, 400, "{raw}");
    assert_eq!(sections(&by_settings), before);
}
