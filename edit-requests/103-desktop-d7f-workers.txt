pub(crate) fn worker_assignment(
    root: &std::path::Path,
    config: &Config,
    request: lightagent_api::ops::WorkerRequest,
) -> Result<serde_json::Value, String> {
    let profiles = ProfileStore::new(root);
    match request {
        lightagent_api::ops::WorkerRequest::Assign {
            role,
            worker,
            profile,
        } => {
            let lead = chat::resolve_profile(&profiles, config, profile)?;
            let role = worker_role(&role)?;
            let target = match worker {
                Some(id) => lightagent_core::AssignmentTarget::Profile {
                    profile: ProfileId::new(&id).map_err(|error| error.to_string())?,
                },
                None => lightagent_core::AssignmentTarget::Builtin,
            };
            let assignment = workers::assign(root, &lead.id, role, target)?;
            serde_json::to_value(&assignment).map_err(|error| error.to_string())
        }
        lightagent_api::ops::WorkerRequest::Revoke { role, profile } => {
            let lead = chat::resolve_profile(&profiles, config, profile)?;
            let role = worker_role(&role)?;
            let dir = profiles.handle(&lead.id).dir().to_path_buf();
            let removed = lightagent_core::WorkerAssignments::new(lead.id.clone(), &dir)
                .revoke(role)
                .map_err(|error| error.to_string())?;
            Ok(serde_json::json!({ "role": role, "revoked": removed }))
        }
    }
}

/// `lightagent workers assign <role>`: to the profile `worker`, or the
/// built-in worker; for the active profile or the one named. Operator only.
async fn assign_worker(
    State(state): State<Arc<AppState>>,
    headers: HeaderMap,
    Path(role): Path<String>,
    Json(body): Json<AssignWorkerBody>,
) -> Response {
    if let Some(rejection) = deny(&state, &headers, Scope::Admin) {
        return rejection;
    }
    let request = ops::WorkerRequest::Assign {
        role,
        worker: body.worker,
        profile: body.profile,
    };
    match state.manager.workers_change(request).await {
        Ok(value) => Json(value).into_response(),
        Err(error) => error.into_response(),
    }
}

/**
 * Each specialist role's readiness, and its assignment, through the
 * operations `lightagent workers` shares: whose workers (when there is more
 * than one profile), and per role Assign (the built-in worker or another
 * profile) and Revoke.
 */
function Specialists() {
  const [workers, setWorkers] = useState<WorkerReadiness[] | null>(null);
  const [error, setError] = useState<string | null>(null);
  const [profiles, setProfiles] = useState<ProfileSummary[]>([]);
  const [lead, setLead] = useState("");
  const [targets, setTargets] = useState<Record<string, string>>({});
  const [busy, setBusy] = useState(false);
  const [message, setMessage] = useState<string | null>(null);

  async function load(profile: string) {
    try {
      setWorkers((await agentApi.workersFor(profile || undefined)).workers);
      setError(null);
    } catch (cause) {
      setError(causeText(cause));
    }
  }

  useEffect(() => {
    agentApi
      .profiles()
      .then((catalog) => {
        setProfiles(catalog.profiles);
        setLead(catalog.active_profile);
        void load(catalog.active_profile);
      })
      .catch(() => void load(""));
  }, []);

  async function change(action: () => Promise<string>) {
    setBusy(true);
    setMessage(null);
    try {
      setMessage(await action());
    } catch (cause) {
      setMessage(causeText(cause));
    } finally {
      setBusy(false);
      void load(lead);
    }
  }

  if (error) return <p className="muted">Specialist readiness is unavailable: {error}</p>;
  if (!workers) return <span className="muted">Loading specialists…</span>;
  return (
    <div data-testid="specialists" style={{ display: "flex", flexDirection: "column", gap: 8 }}>
      {profiles.length > 1 && (

          <label className="field__label" htmlFor="workers-lead">Workers of</label>

            onChange={(event) => { setLead(event.target.value); setWorkers(null); void load(event.target.value); }}>
            {profiles.map((profile) => <option key={profile.id} value={profile.id}>{profile.name || profile.id}</option>)}
          </select>
        </div>
      )}
      {message && <div className="notice notice--info" role="status">{message}</div>}
      {workers.map((worker) => {

        const choice = targets[worker.role] ?? "builtin";
        return (
          <div key={worker.role} data-testid={`worker-${worker.role}`}

            <div style={{ display: "flex", alignItems: "center", gap: 10, flexWrap: "wrap" }}>
              <strong style={{ minWidth: 90, textTransform: "capitalize" }}>{worker.role}</strong>

fn scratch_home(label: &str) -> PathBuf {
    let nanos = SystemTime::now()
        .duration_since(UNIX_EPOCH)
        .map(|elapsed| elapsed.as_nanos())
        .unwrap_or(0);
    let home = std::env::temp_dir().join(format!(
        "lightagent-desktop-workers-{label}-{}-{nanos}",
        std::process::id()
    ));
    std::fs::create_dir_all(&home).expect("home");
    home
}

fn start(home: &Path) -> Runtime {
    let mut child = Command::new(env!("CARGO_BIN_EXE_lightagent"))
        .args(["serve", "--desktop", "--host", "127.0.0.1", "--port", "0"])
        .env("LIGHTAGENT_HOME", home)
        .stdin(Stdio::piped())
        .stdout(Stdio::piped())
        .stderr(Stdio::null())
        .spawn()
        .expect("start lightagent serve --desktop");
    let mut lifeline = child.stdin.take().expect("stdin");
    writeln!(lifeline, "{KEY}").expect("key");
    lifeline.flush().expect("flush");
    let stdout = child.stdout.take().expect("stdout");
    let (tx, rx) = mpsc::channel();
    std::thread::spawn(move || {
        let mut line = String::new();
        let _ = BufReader::new(stdout).read_line(&mut line);
        let _ = tx.send(line);
    });
    let line = rx.recv_timeout(Duration::from_secs(60)).expect("handshake");
    let handshake: Value = serde_json::from_str(&line).expect("handshake JSON");
    let addr = handshake["addr"].as_str().expect("addr").to_owned();
    Runtime {
        child,
        _lifeline: lifeline,
        addr,
    }
}

/// One authenticated request: the status and the raw body.
fn call(runtime: &Runtime, method: &str, path: &str, body: Option<&str>) -> (u16, String) {
    let mut stream = TcpStream::connect(&runtime.addr).expect("connect");
    stream
        .set_read_timeout(Some(Duration::from_secs(60)))
        .expect("timeout");
    let body = body.unwrap_or("");
    let head = format!(
        "{method} {path} HTTP/1.1\r\nHost: {}\r\nAuthorization: Bearer {KEY}\r\nContent-Type: application/json\r\nContent-Length: {}\r\nConnection: close\r\n\r\n{body}",
        runtime.addr,
        body.len()
    );
    stream.write_all(head.as_bytes()).expect("send");
    let mut response = String::new();
    let _ = stream.read_to_string(&mut response);
    let status = response
        .split_whitespace()
        .nth(1)
        .and_then(|code| code.parse().ok())
        .unwrap_or(0);
    let raw = response
        .split_once("\r\n\r\n")
        .map(|(_, body)| body.to_owned())
        .unwrap_or_default();
    (status, raw)
}

fn cli_ok(home: &Path, args: &[&str]) {
    let output: Output = Command::new(env!("CARGO_BIN_EXE_lightagent"))
        .args(args)
        .env("LIGHTAGENT_HOME", home)
        .stdin(Stdio::null())

    );

fn parsed(raw: &str) -> Value {
    serde_json::from_str(raw).unwrap_or_else(|error| panic!("not JSON ({error}): {raw}"))
}

fn cli_json(home: &Path, args: &[&str]) -> Value {
    let mut all = vec!["--json"];
    all.extend_from_slice(args);
    let output: Output = Command::new(env!("CARGO_BIN_EXE_lightagent"))
        .args(&all)
        .env("LIGHTAGENT_HOME", home)
        .stdin(Stdio::null())
        .output()
        .expect("run lightagent");
    assert!(
        output.status.success(),
        "lightagent {args:?}: {}",
        String::from_utf8_lossy(&output.stderr)
    );
    serde_json::from_slice(&output.stdout).expect("JSON output")
}

/// The API's readiness for `profile` (the active one when `None`).
fn api_status(runtime: &Runtime, profile: Option<&str>) -> Value {
    let path = match profile {
        Some(profile) => format!("{WORKERS}?profile={profile}"),
        None => WORKERS.to_owned(),
    };
    let (status, raw) = call(runtime, "GET", &path, None);
    assert_eq!(status, 200, "{raw}");
    parsed(&raw)["workers"].clone()
}

#[test]
fn the_cli_and_the_api_keep_one_set_of_assignments() {
    let home = scratch_home("one");
    cli_ok(&home, &["config", "set", "extensions.enabled", "true"]);
    cli_ok(&home, &["profile", "create", "work"]);
    cli_ok(&home, &["profile", "create", "helper"]);
    cli_ok(&home, &["profile", "use", "work"]);
    let runtime = start(&home);
    assert_eq!(api_status(&runtime, None), cli_json(&home, &["workers", "status"]));

    // Assigned over the API to a profile: the CLI reports the same.
    let (status, raw) = call(
        &runtime,
        "PUT",
        &format!("{WORKERS}/coder"),
        Some(r#"{"worker":"helper"}"#),
    );
    assert_eq!(status, 200, "{raw}");
    assert!(parsed(&raw)["id"].is_string(), "{raw}");
    let after = api_status(&runtime, None);
    assert_eq!(after, cli_json(&home, &["workers", "status"]));
    assert!(after.to_string().contains("helper"), "{after}");

    // Revoked by the CLI: the API reports the same.
    let revoked = cli_json(&home, &["workers", "revoke", "coder"]);
    assert_eq!(revoked["revoked"], true, "{revoked}");
    assert_eq!(api_status(&runtime, None), cli_json(&home, &["workers", "status"]));

    // The built-in worker over the API; revoked over the API, then nothing left.
    let (status, raw) = call(&runtime, "PUT", &format!("{WORKERS}/reviewer"), Some("{}"));
    assert_eq!(status, 200, "{raw}");
    let (status, raw) = call(&runtime, "DELETE", &format!("{WORKERS}/reviewer"), None);
    assert_eq!(status, 200, "{raw}");
    assert_eq!(parsed(&raw)["revoked"], true, "{raw}");
    let (_, raw) = call(&runtime, "DELETE", &format!("{WORKERS}/reviewer"), None);
    assert_eq!(parsed(&raw)["revoked"], false, "{raw}");

    // Another profile's workers, named.
    let body = r#"{"worker":"work","profile":"helper"}"#;
    let (status, raw) = call(&runtime, "PUT", &format!("{WORKERS}/tester"), Some(body));
    assert_eq!(status, 200, "{raw}");
    assert_eq!(
        api_status(&runtime, Some("helper")),
        cli_json(&home, &["workers", "status", "--profile", "helper"])
    );
    let (status, raw) = call(
        &runtime,
        "DELETE",
        &format!("{WORKERS}/tester?profile=helper"),
        None,
    );
    assert_eq!(status, 200, "{raw}");
    assert_eq!(parsed(&raw)["revoked"], true, "{raw}");

    // Refused as the CLI refuses them.
    for (role, body) in [
        ("wizard", "{}"),
        ("coder", r#"{"worker":"work"}"#),
        ("coder", r#"{"worker":"nobody"}"#),
    ] {
        let (status, raw) = call(&runtime, "PUT", &format!("{WORKERS}/{role}"), Some(body));
        assert_eq!(status, 400, "{role} {body}: {raw}");
    }
}

export async function checkWorkers(browser, base, outDir) {
  const failures = [];
  const page = await browser.newPage({ viewport: { width: 1280, height: 1500 } });
  const errors = [];
  page.on("pageerror", (error) => errors.push(error.message));
  try {
    await page.goto(`${base}/#/tools`);
    const card = page.getByTestId("specialists");
    await card.waitFor({ timeout: 15000 });
    const row = page.getByTestId("worker-coder");
    await row.waitFor({ timeout: 10000 });
    await row.getByRole("combobox").selectOption("builtin");
    await row.getByRole("button", { name: "Assign coder" }).click();
    await card.getByRole("status").filter({ hasText: "Assigned coder to the built-in worker." }).waitFor({ timeout: 10000 });
    await page.screenshot({ path: `${outDir}/workers-assigned.png`, fullPage: true });
    await row.getByRole("button", { name: "Revoke coder" }).click();
    await card.getByRole("status").filter({ hasText: "Revoked the coder assignment." }).waitFor({ timeout: 10000 });
    await row.getByRole("button", { name: "Revoke coder" }).click();
    await card.getByRole("status").filter({ hasText: "There was no coder assignment." }).waitFor({ timeout: 10000 });
    if (errors.length) failures.push(`workers: page errors: ${errors.join("; ")}`);
  } catch (err) {
    failures.push(`workers: ${err instanceof Error ? err.message.split("\n")[0] : String(err)}`);
    await page.screenshot({ path: `${outDir}/workers-failed.png` }).catch(() => {});
  } finally {
    await page.close();
    await fetch(`${base}/api/lightagent/v1/workers/coder`, { method: "DELETE" }).catch(() => {});
  }
  return failures;
}
