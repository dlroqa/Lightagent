D6-1: extension operations shared by the CLI, the API and the desktop

`lightagent extensions` kept its logic in the CLI crate, so the desktop
could not act on extensions without a second implementation.

- lightagent_extensions::manage: list, show, set_enabled, install and
  uninstall over the same roots (global and the active profile's) and the
  same configuration, with typed errors (not found, invalid, conflict,
  unavailable, store). The CLI now prints their results; its text output is
  unchanged and its JSON gains `scope` (and `show` gains `mcp` and
  `source`).
- An install records its source in `.lightagent-source.json`: the local
  folder, or the HTTPS repository, sub-folder and the commit it installed
  (`git rev-parse HEAD`). If that record or the settings cannot be saved,
  the install is undone. git runs with GIT_ALLOW_PROTOCOL=https and no
  prompt; nothing fetched is executed.
- MCP is switched on by an install only when the caller asks
  (`activate_mcp`). The CLI asks, as before; the API defaults to no, so an
  install never silently lets an extension's servers run.
- `show` lists each MCP server's command and args or URL, and only the
  names of its environment variables and headers, never their values.
- API (operator): GET|POST /extensions, GET|DELETE /extensions/{name}
  (`?scope=profile`), PUT /extensions/{name}/enabled. The API accepts an
  absolute folder or an https:// Git URL only. Desktop contract operations
  extensions.*; the host refuses the three that change state during a
  profile switch. Inventory: five backend rows (not counted).
- Tests (separate processes): CLI and API installs give the same files,
  settings and listing, and toggle and remove alike; a failed install
  (already installed, unusable source, unreachable host, a link out of the
  bundle) or removal leaves what was installed; a profile install stays
  with its profile; MCP waits to be asked; no env value is shown.

//! The extension operations behind `lightagent extensions …`, shared by the
//! CLI, the HTTP API and the desktop app. Each acts on the same extension
//! roots (the global `<home>/extensions/` and the active profile's own) and
//! the same configuration, with the same checks and the same errors.
//!
//! An install grants no more than the CLI always has: it switches the
//! extension system on and takes the extension off the disabled list. The
//! MCP servers an extension declares are contacted only while `mcp.enabled`
//! is set, and an install sets it only when the caller asks
//! ([`InstallOptions::activate_mcp`]).

use std::path::{Component, Path, PathBuf};
use std::process::Command;

use lightagent_core::{
    Config, ConfigStore, LightagentPaths, McpServerEntry, ProfileStore, RunId, SkillStore,
};
use serde::{Deserialize, Serialize};

use crate::{Extension, ExtensionStore, UninstallError, extension_dirs};

/// The file an install writes into the installed extension to record where
/// it came from. One the source itself holds is replaced.
pub const SOURCE_FILE: &str = ".lightagent-source.json";

/// Where an extension is installed.
#[derive(Clone, Copy, Debug, PartialEq, Eq, Serialize, Deserialize)]
#[serde(rename_all = "snake_case")]
pub enum Scope {
    /// `<home>/extensions/`, loaded by every profile.
    Global,
    /// The active profile's own `extensions/`, which overrides the global set.
    Profile,
}

/// Why an extension operation did nothing.
#[derive(Clone, Debug, PartialEq, Eq)]
pub enum ManageError {
    /// No installed extension has the name.
    NotFound(String),
    /// The request cannot work as given: a bad name, source or manifest.
    Invalid(String),
    /// What is installed does not allow it now: already installed, in use,
    /// or left by an interrupted removal.
    Conflict(String),
    /// Something outside Lightagent failed: git could not start or fetch.
    Unavailable(String),
    /// The configuration or the profile store could not be read or saved.
    Store(String),
}

impl std::fmt::Display for ManageError {
    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
        match self {
            Self::NotFound(message)
            | Self::Invalid(message)
            | Self::Conflict(message)
            | Self::Unavailable(message)
            | Self::Store(message) => f.write_str(message),
        }
    }
}

impl std::error::Error for ManageError {}

/// Where an installed extension came from, as its install recorded it.
#[derive(Clone, Debug, PartialEq, Eq, Serialize, Deserialize)]
#[serde(tag = "kind", rename_all = "snake_case")]

    /// A local folder.
    Local { path: PathBuf },
    /// An HTTPS Git repository: the folder in it, and the commit installed.
    Git {

        subdir: String,

/// An MCP server an extension declares, as shown before it is trusted: what
/// it runs or where it connects. Of its environment and headers only the

pub enum McpServerView {
    Stdio {
        name: String,
        command: String,
        args: Vec<String>,
        env: Vec<String>,
    },
    Http {
        name: String,
        url: String,
        headers: Vec<String>,
        auth: bool,
    },
}

    pub name: String,
    pub version: String,
    pub description: String,

    pub instructions: String,
    pub mcp_servers: Vec<String>,
    pub mcp: Vec<McpServerView>,
    pub skills: Vec<String>,
    pub source: Option<Source>,
    /// Whether MCP is on: the extension's servers are contacted only then.
    pub mcp_enabled: bool,
}

/// How to install.
#[derive(Clone, Copy, Debug)]

    pub name: String,
    pub dir: PathBuf,
    pub scope: Scope,
    pub mcp_servers: Vec<String>,

    /// Whether MCP is on after the install.
    pub mcp_enabled: bool,
    pub source: Source,

/// An extension switched on or off.
#[derive(Clone, Debug, Serialize)]
pub struct Toggled {

    pub active: bool,
    /// The master switch, `extensions.enabled`.
    pub extensions_enabled: bool,

#[derive(Clone, Debug, Serialize)]

/// The extension root for `scope`.
pub fn root(paths: &LightagentPaths, scope: Scope) -> Result<PathBuf, ManageError> {
    match scope {
        Scope::Global => Ok(paths.extensions_dir()),
        Scope::Profile => profile_root(paths)?.ok_or_else(|| {
            ManageError::Invalid("no active profile — run `lightagent init` first".to_owned())
        }),
    }
}

/// The active profile's own extension root, if a profile is active.
fn profile_root(paths: &LightagentPaths) -> Result<Option<PathBuf>, ManageError> {
    let profiles = ProfileStore::new(paths.root());
    let active = profiles
        .active()
        .map_err(|error| ManageError::Store(error.to_string()))?;
    Ok(active.map(|active| profiles.handle(&active).dir().join("extensions")))
}

/// The extensions the active profile loads, and that profile's own root.
fn installed(paths: &LightagentPaths) -> Result<(ExtensionStore, Option<PathBuf>), ManageError> {
    let profiles = ProfileStore::new(paths.root());
    let active = profiles
        .active()
        .map_err(|error| ManageError::Store(error.to_string()))?;
    Ok(match active {
        Some(active) => {
            let dir = profiles.handle(&active).dir().to_path_buf();
            (
                ExtensionStore::load(&extension_dirs(paths.root(), &dir)),
                Some(dir.join("extensions")),
            )
        }
        None => (ExtensionStore::load(&[paths.extensions_dir()]), None),
    })
}

fn scope_of(ext: &Extension, own: Option<&Path>) -> Scope {
    if own.is_some_and(|own| ext.dir.starts_with(own)) {
        Scope::Profile
    } else {
        Scope::Global
    }
}

fn not_installed(name: &str) -> ManageError {
    ManageError::NotFound(format!("no installed extension named '{name}'"))
}

fn server_names(ext: &Extension) -> Vec<String> {
    ext.mcp_servers
        .iter()
        .map(|server| server.name().to_owned())
        .collect()
}

fn server_view(server: &McpServerEntry) -> McpServerView {
    match server {
        McpServerEntry::Stdio {
            name,
            command,
            args,
            env,
        } => McpServerView::Stdio {
            name: name.clone(),
            command: command.clone(),
            args: args.clone(),
            env: env.keys().cloned().collect(),
        },
        McpServerEntry::Http {
            name,
            url,
            headers,
            auth,
            ..
        } => McpServerView::Http {
            name: name.clone(),
            url: url.clone(),
            headers: headers.keys().cloned().collect(),
            auth: auth.is_some(),
        },
    }
}

/// Where `ext` came from, if its install recorded it.
pub fn source_of(ext: &Extension) -> Option<Source> {
    let text = std::fs::read_to_string(ext.dir.join(SOURCE_FILE)).ok()?;
    serde_json::from_str(&text).ok()
}

/// `extensions list`: every installed extension the active profile loads.
pub fn list(paths: &LightagentPaths, config: &Config) -> Result<Listing, ManageError> {
    let (store, own) = installed(paths)?;
    let extensions = store
        .all()
        .map(|ext| Summary {
            name: ext.name.clone(),
            version: ext.version.clone(),
            description: ext.description.clone(),
            active: store.is_active(&ext.name, &config.extensions),
            scope: scope_of(ext, own.as_deref()),
            skills_dir: ext.skills_dir(),
            mcp_servers: server_names(ext),
        })
        .collect();
    Ok(Listing {
        enabled: config.extensions.enabled,
        extensions,
    })
}

/// `extensions show <name>`: one extension's manifest and contributions.
pub fn show(paths: &LightagentPaths, config: &Config, name: &str) -> Result<Detail, ManageError> {
    let (store, own) = installed(paths)?;
    let ext = store.get(name).ok_or_else(|| not_installed(name))?;
    Ok(Detail {
        name: ext.name.clone(),
        version: ext.version.clone(),
        description: ext.description.clone(),
        active: store.is_active(name, &config.extensions),
        scope: scope_of(ext, own.as_deref()),
        dir: ext.dir.clone(),
        instructions: ext.instructions.clone(),
        mcp_servers: server_names(ext),
        mcp: ext.mcp_servers.iter().map(server_view).collect(),
        skills: SkillStore::load(&[ext.skills_dir()]).names(),
        source: source_of(ext),
        mcp_enabled: config.mcp.enabled,
    })
}

/// `extensions enable|disable <name>`: keep an installed extension active or
/// inactive. A name that is not installed is refused, so a typo is caught.
pub fn set_enabled(
    paths: &LightagentPaths,
    name: &str,
    enabled: bool,
) -> Result<Toggled, ManageError> {
    let (store, _) = installed(paths)?;
    if store.get(name).is_none() {
        return Err(not_installed(name));
    }
    let (extensions_enabled, _) = ConfigStore::at(paths)
        .update(None, |config| {
            if enabled {
                config.extensions.disabled.retain(|n| n != name);
            } else if !config.extensions.disabled.iter().any(|n| n == name) {
                config.extensions.disabled.push(name.to_owned());
                config.extensions.disabled.sort();
            }
            Ok::<_, String>(config.extensions.enabled)
        })
        .map_err(|error| ManageError::Store(error.to_string()))?;
    Ok(Toggled {
        name: name.to_owned(),
        active: extensions_enabled && enabled,
        extensions_enabled,
    })
}

/// `extensions install <source>`: a local folder, or an HTTPS Git repository
/// (`#subdir` selects a folder in it), into the root for `options.scope`.
/// An extension already installed there is refused and left as it is. If
/// the source record or the settings cannot be saved, the new copy is
/// removed again.
pub fn install(
    paths: &LightagentPaths,
    source: &Path,
    options: InstallOptions,
) -> Result<Installed, ManageError> {
    let root = root(paths, options.scope)?;
    let (ext, origin) = fetch_and_install(source, &root)?;
    let recorded = serde_json::to_vec_pretty(&origin)
        .map_err(|error| error.to_string())
        .and_then(|bytes| {
            std::fs::write(ext.dir.join(SOURCE_FILE), bytes).map_err(|error| error.to_string())
        });
    if let Err(error) = recorded {
        let _ = crate::uninstall(&ext.name, &root);
        return Err(ManageError::Store(format!(
            "could not record the extension's source: {error}"
        )));
    }
    let saved = ConfigStore::at(paths).update(None, |config| {
        config.extensions.enabled = true;
        config.extensions.disabled.retain(|name| name != &ext.name);
        if options.activate_mcp && !ext.mcp_servers.is_empty() {
            config.mcp.enabled = true;
        }
        Ok::<_, String>(config.mcp.enabled)
    });
    match saved {
        Ok((mcp_enabled, _)) => Ok(Installed {
            name: ext.name.clone(),
            dir: ext.dir.clone(),
            scope: options.scope,
            mcp_servers: server_names(&ext),
            active: true,
            mcp_enabled,
            source: origin,
        }),
        Err(error) => {
            let _ = crate::uninstall(&ext.name, &root);
            Err(ManageError::Store(format!(
                "could not save extension settings: {error}"
            )))
        }
    }
}

/// `extensions uninstall <name>`: remove exactly that installed copy from
/// the root for `scope`.
pub fn uninstall(
    paths: &LightagentPaths,
    name: &str,
    scope: Scope,
) -> Result<Removed, ManageError> {
    let root = root(paths, scope)?;
    crate::try_uninstall(name, &root).map_err(|error| match &error {
        UninstallError::InvalidName => ManageError::Invalid(error.to_string()),
        UninstallError::NotInstalled { .. } => ManageError::NotFound(error.to_string()),
        UninstallError::Io { .. } => ManageError::Store(error.to_string()),
        UninstallError::Refused { .. }
        | UninstallError::InUse { .. }
        | UninstallError::PartiallyRemoved { .. }
        | UninstallError::Unfinished { .. } => ManageError::Conflict(error.to_string()),
    })?;
    Ok(Removed {
        name: name.to_owned(),
        removed_from: root,
    })
}

/// A refusal from [`crate::install`]: a copy already there is a conflict,
/// anything else a source that cannot be installed.
fn install_refusal(message: String) -> ManageError {
    if message.contains("is already installed") || message.contains("staging directory") {
        ManageError::Conflict(message)
    } else {
        ManageError::Invalid(message)
    }
}

/// Install from a local folder, or from a shallow clone of an HTTPS Git
/// repository that is removed afterwards. Nothing fetched is run: the clone
/// is read and copied like a local folder.
fn fetch_and_install(source: &Path, root: &Path) -> Result<(Extension, Source), ManageError> {
    let Some((url, subdir)) = git_source(source)? else {
        let ext = crate::install(source, root).map_err(install_refusal)?;
        let path = source.canonicalize().unwrap_or_else(|_| source.to_path_buf());
        return Ok((ext, Source::Local { path }));
    };
    let checkout = std::env::temp_dir().join(format!(
        "lightagent-extension-fetch-{}",
        RunId::new().as_str()
    ));
    let result = (|| {
        let output = Command::new("git")
            .args(["clone", "--depth", "1", "--"])
            .arg(&url)
            .arg(&checkout)
            .env("GIT_TERMINAL_PROMPT", "0")
            .env("GIT_ALLOW_PROTOCOL", "https")
            .output()
            .map_err(|e| ManageError::Unavailable(format!("could not start git: {e}")))?;
        if !output.status.success() {
            let detail = String::from_utf8_lossy(&output.stderr);
            return Err(ManageError::Unavailable(format!(
                "git clone failed: {}",
                detail.trim()
            )));
        }
        let head = Command::new("git")
            .arg("-C")
            .arg(&checkout)
            .args(["rev-parse", "HEAD"])
            .output()
            .map_err(|e| ManageError::Unavailable(format!("could not start git: {e}")))?;
        let revision = String::from_utf8_lossy(&head.stdout).trim().to_owned();
        if !head.status.success() || revision.is_empty() {
            return Err(ManageError::Unavailable(
                "could not read the cloned commit".to_owned(),
            ));
        }
        let ext = crate::install(&checkout.join(&subdir), root).map_err(install_refusal)?;
        Ok((
            ext,
            Source::Git {
                url: url.clone(),
                subdir: subdir.to_string_lossy().replace('\\', "/"),
                revision,
            },
        ))
    })();
    let _ = std::fs::remove_dir_all(&checkout);
    result
}

/// An HTTPS Git source (`https://…` or `git+https://…`, `#subdir` optional),
/// or `None` for a local folder. A URL with credentials, and a sub-folder
/// that would leave the repository, are refused.
fn git_source(source: &Path) -> Result<Option<(String, PathBuf)>, ManageError> {
    let text = source.to_string_lossy();
    let raw = text.strip_prefix("git+").unwrap_or(&text);
    if !raw.starts_with("https://") {
        return Ok(None);
    }
    let (url, subdir) = raw.split_once('#').unwrap_or((raw, ""));
    if url.len() <= "https://".len() {
        return Err(ManageError::Invalid(
            "Git URL needs a host and repository path".to_owned(),
        ));
    }
    let authority = url
        .trim_start_matches("https://")
        .split('/')
        .next()
        .unwrap_or("");
    if authority.contains('@') {
        return Err(ManageError::Invalid(
            "Git URL must not contain credentials; use a Git credential helper".to_owned(),
        ));
    }
    let subdir = PathBuf::from(subdir);
    if !subdir.as_os_str().is_empty()
        && !subdir
            .components()
            .all(|part| matches!(part, Component::Normal(_)))
    {
        return Err(ManageError::Invalid(
            "Git extension subdirectory must stay within the repository".to_owned(),
        ));
    }
    Ok(Some((url.to_owned(), subdir)))
}

//! `lightagent extensions` — install, remove, list and toggle capability bundles.
//!
//! An extension is discovered under the global `<home>/extensions/` directory and
//! the active profile's own; installing one is dropping its directory in place.
//! Installation copies a local bundle into the selected store; uninstall removes
//! exactly that installed copy. The operations are
//! [`lightagent_extensions::manage`], shared with the HTTP API and the desktop
//! app; this module prints their results.

use std::path::Path;

use lightagent_core::{Config, ConfigStore, LightagentPaths};
use lightagent_extensions::manage::{self, InstallOptions, Scope};

fn paths() -> Result<LightagentPaths, String> {
    LightagentPaths::resolve().map_err(|error| error.to_string())
}

fn config(paths: &LightagentPaths) -> Result<Config, String> {
    ConfigStore::at(paths)
        .load()
        .map_err(|error| error.to_string())
}

fn scope(profile: bool) -> Scope {
    if profile {
        Scope::Profile
    } else {
        Scope::Global
    }
}

/// Install a local bundle and activate the extension system it needs. The
/// CLI switches MCP on for an extension that brings MCP servers, and says so.
pub fn install(source: &Path, profile: bool, json: bool) -> Result<(), String> {
    let options = InstallOptions {
        scope: scope(profile),
        activate_mcp: true,
    };
    let installed =
        manage::install(&paths()?, source, options).map_err(|error| error.to_string())?;
    if json {
        println!(
            "{}",
            serde_json::json!({
                "name": installed.name, "dir": installed.dir,
                "mcp_servers": installed.mcp_servers.len(), "active": true
            })
        );
    } else {
        println!(
            "Installed '{}' in {}.",
            installed.name,
            installed.dir.display()
        );
        if !installed.mcp_servers.is_empty() {
            println!(
                "MCP enabled; run `lightagent tools list` to verify the server and its tools."
            );
        }
    }
    Ok(())
}

/// Remove an installed bundle from the selected store.
pub fn uninstall(name: &str, profile: bool, json: bool) -> Result<(), String> {
    let removed =
        manage::uninstall(&paths()?, name, scope(profile)).map_err(|error| error.to_string())?;
    if json {
        println!(
            "{}",
            serde_json::json!({ "name": removed.name, "removed_from": removed.removed_from })
        );
    } else {
        println!(
            "Uninstalled '{}' from {}.",
            removed.name,
            removed.removed_from.display()
        );
    }
    Ok(())
}

/// `extensions list` — every installed extension, marking the active ones.
pub fn list(json: bool) -> Result<(), String> {
    let paths = paths()?;
    let config = config(&paths)?;
    let listing = manage::list(&paths, &config).map_err(|error| error.to_string())?;
    if json {
        let value = serde_json::json!({
            "enabled": listing.enabled,
            "extensions": listing.extensions.iter().map(|ext| serde_json::json!({
                "name": ext.name,
                "version": ext.version,
                "description": ext.description,
                "active": ext.active,
                "scope": ext.scope,
                "skills_dir": ext.skills_dir,
                "mcp_servers": ext.mcp_servers,
            })).collect::<Vec<_>>(),
        });
        println!("{value:#}");
        return Ok(());
    }
    if listing.extensions.is_empty() {
        println!(
            "No extensions installed. Use `lightagent extensions install <directory>` or add one under {}.",
            paths.extensions_dir().display()
        );
        return Ok(());
    }
    if !listing.enabled {
        println!("(extensions are disabled globally — `lightagent config set` extensions.enabled)");
    }
    for ext in &listing.extensions {
        let mark = if ext.active { "●" } else { "○" };
        if ext.description.is_empty() {
            println!("{mark} {}", ext.name);
        } else {
            println!("{mark} {} — {}", ext.name, ext.description);
        }
    }
    Ok(())
}

/// `extensions show <name>` — one extension's manifest and contributions.
pub fn show(name: &str, json: bool) -> Result<(), String> {
    let paths = paths()?;
    let config = config(&paths)?;
    let ext = manage::show(&paths, &config, name).map_err(|error| error.to_string())?;
    if json {
        let value = serde_json::json!({
            "name": ext.name,
            "version": ext.version,
            "description": ext.description,
            "active": ext.active,
            "scope": ext.scope,
            "dir": ext.dir,
            "instructions": ext.instructions,
            "mcp_servers": ext.mcp_servers,
            "mcp": ext.mcp,
            "skills": ext.skills,
            "source": ext.source,
        });
        println!("{value:#}");
        return Ok(());
    }
    println!("{}", ext.name);
    if !ext.version.is_empty() {
        println!("  version: {}", ext.version);
    }
    if !ext.description.is_empty() {

}

/// `lightagent extensions install`, from the same shared operation.
async fn install_extension(
    State(state): State<Arc<AppState>>,
    headers: HeaderMap,
    Json(body): Json<InstallExtensionBody>,
) -> Response {
    if let Some(rejection) = deny(&state, &headers, Scope::Admin) {
        return rejection;
    }
    let paths = match extension_paths(&state) {
        Ok(paths) => paths,
        Err(response) => return response,
    };
    // The server's working folder means nothing to a client, so a relative
    // path is refused rather than resolved against it.
    let remote = body.source.starts_with("https://") || body.source.starts_with("git+https://");
    if !remote && !std::path::Path::new(&body.source).is_absolute() {
        return bad_request("the source must be an absolute folder path or an https:// Git URL");
    }
    let options = extension_ops::InstallOptions {
        scope: body.scope.unwrap_or(extension_ops::Scope::Global),
        activate_mcp: body.activate_mcp,
    };
    let source = PathBuf::from(body.source);
    match tokio::task::spawn_blocking(move || extension_ops::install(&paths, &source, options))
        .await
    {
        Ok(Ok(installed)) => (StatusCode::CREATED, Json(installed)).into_response(),
        Ok(Err(error)) => extension_error(error),
        Err(error) => internal(&error.to_string()),
    }
}

fn start(home: &Path) -> Runtime {

/// One authenticated request: the status and the raw body.
fn call(runtime: &Runtime, method: &str, path: &str, body: Option<&str>) -> (u16, String) {
    let mut stream = TcpStream::connect(&runtime.addr).expect("connect");
    stream
        .set_read_timeout(Some(Duration::from_secs(120)))
        .expect("timeout");
    let body = body.unwrap_or("");
    let head = format!(
        "{method} {path} HTTP/1.1\r\nHost: {}\r\nAuthorization: Bearer {KEY}\r\nContent-Type: application/json\r\nContent-Length: {}\r\nConnection: close\r\n\r\n{body}",
        runtime.addr,
        body.len()
    );
    stream.write_all(head.as_bytes()).expect("send");
    let mut response = String::new();
    let _ = stream.read_to_string(&mut response);
    let status = response
        .split_whitespace()
        .nth(1)
        .and_then(|code| code.parse().ok())
        .unwrap_or(0);
    let raw = response
        .split_once("\r\n\r\n")
        .map(|(_, body)| body.to_owned())
        .unwrap_or_default();
    (status, raw)
}

    let mut pending = vec![dir.to_path_buf()];
    while let Some(next) = pending.pop() {
        for entry in std::fs::read_dir(&next).expect("read dir") {
            let path = entry.expect("entry").path();
            if path.is_dir() {
                pending.push(path);
            } else {
                let relative = path
                    .strip_prefix(dir)
                    .expect("relative")
                    .to_string_lossy()
                    .replace('\\', "/");
                found.insert(relative, std::fs::read(&path).expect("read"));
            }
        }
    }
    found
}

/// Hidden entries in an extension root: staging or removal leftovers.
fn leftovers(root: &Path) -> Vec<String> {
    std::fs::read_dir(root)
        .map(|entries| {
            entries
                .flatten()
                .map(|entry| entry.file_name().to_string_lossy().into_owned())
                .filter(|name| name.starts_with('.'))
                .collect()
        })
        .unwrap_or_default()
}

/// A listing's extensions without their home-specific folders.
fn summaries(listing: &Value) -> Vec<Value> {
    listing["extensions"]
        .as_array()
        .expect("extensions")
        .iter()
        .map(|ext| {
            let mut ext = ext.clone();
            if let Some(fields) = ext.as_object_mut() {
                fields.remove("skills_dir");
            }
            ext
        })
        .collect()
}

#[test]
fn the_cli_and_the_api_install_list_toggle_and_remove_alike() {
    let work = scratch("alike");
    let source = bundle(&work, "greeter", false);
    let (cli_home, api_home) = (work.join("cli"), work.join("api"));
    for home in [&cli_home, &api_home] {
        std::fs::create_dir_all(home).expect("home");
        cli_ok(home, &["config", "set", "extensions.enabled", "true"]);
    }
    let runtime = start(&api_home);

    let by_cli = cli_json(&cli_home, &["extensions", "install", arg(&source)]);
    assert_eq!(by_cli["name"], "greeter", "{by_cli}");
    let body = json!({ "source": source }).to_string();
    let (status, raw) = call(&runtime, "POST", EXTENSIONS, Some(&body));
    assert_eq!(status, 201, "{raw}");
    let by_api = parsed(&raw);
    assert_eq!(by_api["name"], "greeter", "{by_api}");
    assert_eq!(by_api["scope"], "global", "{by_api}");
    assert_eq!(by_api["source"]["kind"], "local", "{by_api}");
    assert_eq!(by_api["mcp_enabled"], false, "{by_api}");

    // The same files (the source record included), settings and listing.
    assert_eq!(
        files(&cli_home.join("extensions").join("greeter")),
        files(&api_home.join("extensions").join("greeter"))
    );
    assert_eq!(config(&cli_home)["extensions"], config(&api_home)["extensions"]);
    assert_eq!(config(&cli_home)["mcp"], config(&api_home)["mcp"]);
    let (status, raw) = call(&runtime, "GET", EXTENSIONS, None);
    assert_eq!(status, 200, "{raw}");
    let listed = parsed(&raw);
    assert_eq!(listed, cli_json(&api_home, &["extensions", "list"]));
    assert_eq!(
        summaries(&cli_json(&cli_home, &["extensions", "list"])),
        summaries(&listed)
    );
    let (status, raw) = call(&runtime, "GET", &format!("{EXTENSIONS}/greeter"), None);
    assert_eq!(status, 200, "{raw}");
    let shown = parsed(&raw);
    let shown_by_cli = cli_json(&api_home, &["extensions", "show", "greeter"]);
    for field in [
        "name",
        "version",
        "description",
        "active",
        "scope",
        "dir",
        "instructions",
        "mcp_servers",
        "mcp",
        "skills",
        "source",
    ] {
        assert_eq!(shown[field], shown_by_cli[field], "{field}");
    }
    assert_eq!(shown["skills"], json!(["greet"]), "{shown}");

    // Switched off through the API, on through the CLI: each sees the other.
    let (status, raw) = call(
        &runtime,
        "PUT",
        &format!("{EXTENSIONS}/greeter/enabled"),
        Some(r#"{"enabled":false}"#),
    );
    assert_eq!(status, 200, "{raw}");
    assert_eq!(parsed(&raw)["active"], false, "{raw}");
    assert_eq!(
        cli_json(&api_home, &["extensions", "list"])["extensions"][0]["active"],
        false
    );
    assert_eq!(config(&api_home)["extensions"]["disabled"], json!(["greeter"]));
    cli_json(&api_home, &["extensions", "enable", "greeter"]);
    let (_, raw) = call(&runtime, "GET", &format!("{EXTENSIONS}/greeter"), None);
    assert_eq!(parsed(&raw)["active"], true, "{raw}");

    // Removed through the API: gone for the CLI too, with nothing left over.
    let (status, raw) = call(&runtime, "DELETE", &format!("{EXTENSIONS}/greeter"), None);
    assert_eq!(status, 200, "{raw}");
    assert_eq!(
        cli_json(&api_home, &["extensions", "list"])["extensions"],
        json!([])
    );
    assert!(!api_home.join("extensions").join("greeter").exists());
    assert!(leftovers(&api_home.join("extensions")).is_empty());
}

#[test]
fn failed_installs_and_removals_leave_what_was_there() {
    let work = scratch("failures");
    let home = work.join("home");
    std::fs::create_dir_all(&home).expect("home");
    cli_ok(&home, &["config", "set", "extensions.enabled", "true"]);
    let runtime = start(&home);
    let install = |source: &str| {
        let body = json!({ "source": source }).to_string();
        call(&runtime, "POST", EXTENSIONS, Some(&body))
    };
    let installed_dir = home.join("extensions").join("keeper");

    let source = bundle(&work, "keeper", false);
    let (status, raw) = install(arg(&source));
    assert_eq!(status, 201, "{raw}");
    let before = files(&installed_dir);

    // Installing it again, even a newer version, is refused; the installed
    // copy stays exactly as it was.
    std::fs::write(
        source.join("extension.json"),
        r#"{"name":"keeper","version":"2.0.0"}"#,
    )
    .expect("manifest");
    let (status, raw) = install(arg(&source));
    assert_eq!(status, 409, "{raw}");
    assert_eq!(files(&installed_dir), before);

    // Unusable sources are refused before anything is fetched or copied.
    let missing = work.join("no-such-folder");
    for source in [
        "relative/folder",
        arg(&missing),
        "http://example.invalid/tools.git",
        "https://user:secret@example.invalid/tools.git",
        "https://example.invalid/tools.git#../outside",
    ] {
        let (status, raw) = install(source);
        assert_eq!(status, 400, "{source}: {raw}");
        assert!(!raw.contains("secret"), "{raw}");
    }

    // A host that cannot be reached is reported as such.
    let (status, raw) = install("https://127.0.0.1:9/tools.git");
    assert_eq!(status, 502, "{raw}");

    // A bundle with a link that leads out of it is refused.
    #[cfg(unix)]
    {
        let linked = bundle(&work, "linked", false);
        std::os::unix::fs::symlink(&home, linked.join("skills").join("outside")).expect("link");
        let (status, raw) = install(arg(&linked));
        assert_eq!(status, 400, "{raw}");
        assert!(!home.join("extensions").join("linked").exists());
    }

    // Removing, showing or toggling what is not installed, or a bad name.
    assert_eq!(
        call(&runtime, "DELETE", &format!("{EXTENSIONS}/nothing"), None).0,
        404
    );
    assert_eq!(
        call(&runtime, "DELETE", &format!("{EXTENSIONS}/bad.name"), None).0,
        400
    );
    assert_eq!(
        call(&runtime, "GET", &format!("{EXTENSIONS}/nothing"), None).0,
        404
    );
    assert_eq!(
        call(
            &runtime,
            "PUT",
            &format!("{EXTENSIONS}/nothing/enabled"),
            Some(r#"{"enabled":true}"#)
        )
        .0,
        404
    );

    // After all of it: exactly the one extension, as installed, and no
    // staging or removal leftovers.
    let (_, raw) = call(&runtime, "GET", EXTENSIONS, None);
    let listed = parsed(&raw);
    assert_eq!(
        listed["extensions"].as_array().map(Vec::len),
        Some(1),
        "{listed}"
    );
    assert_eq!(listed["extensions"][0]["version"], "1.0.0", "{listed}");
    assert_eq!(files(&installed_dir), before);
    assert!(leftovers(&home.join("extensions")).is_empty());
}

#[test]
fn a_profile_install_stays_with_its_profile_and_mcp_waits_to_be_asked() {
    let work = scratch("profile");
    let home = work.join("home");
    std::fs::create_dir_all(&home).expect("home");
    cli_ok(&home, &["config", "set", "extensions.enabled", "true"]);
    cli_ok(&home, &["profile", "create", "work"]);
    cli_ok(&home, &["profile", "create", "other"]);
    cli_ok(&home, &["profile", "use", "work"]);
    let runtime = start(&home);
    let source = bundle(&work, "tooled", true);

    let body = json!({ "source": source, "scope": "profile" }).to_string();
    let (status, raw) = call(&runtime, "POST", EXTENSIONS, Some(&body));
    assert_eq!(status, 201, "{raw}");
    let installed = parsed(&raw);
    assert_eq!(installed["scope"], "profile", "{installed}");
    assert_eq!(
        installed["mcp_enabled"], false,
        "an install does not switch MCP on unasked"
    );
    assert_eq!(config(&home)["mcp"]["enabled"], false);
    assert!(!home.join("extensions").join("tooled").exists());

    // Shown before it is trusted: what the server would run, and only the
    // names of its environment variables.
    let (status, raw) = call(&runtime, "GET", &format!("{EXTENSIONS}/tooled"), None);
    assert_eq!(status, 200, "{raw}");
    let shown = parsed(&raw);
    assert_eq!(shown["mcp"][0]["command"], "echo-server", "{shown}");
    assert_eq!(shown["mcp"][0]["env"], json!(["TOKEN"]), "{shown}");
    assert_eq!(shown["mcp_enabled"], false, "{shown}");
    assert!(!raw.contains("secret-value"), "{raw}");
    let shown_by_cli = cli_json(&home, &["extensions", "show", "tooled"]);
    assert!(!shown_by_cli.to_string().contains("secret-value"));

    // Another profile does not load it.
    cli_ok(&home, &["profile", "use", "other"]);
    assert_eq!(
        cli_json(&home, &["extensions", "list"])["extensions"],
        json!([])
    );
    cli_ok(&home, &["profile", "use", "work"]);
    assert_eq!(
        cli_json(&home, &["extensions", "list"])["extensions"][0]["scope"],
        "profile"
    );

    // The global root does not hold it; the profile's does.
    assert_eq!(
        call(
            &runtime,
            "DELETE",
            &format!("{EXTENSIONS}/tooled?scope=global"),
            None
        )
        .0,
        404
    );
    let (status, raw) = call(
        &runtime,
        "DELETE",
        &format!("{EXTENSIONS}/tooled?scope=profile"),
        None,
    );
    assert_eq!(status, 200, "{raw}");

    // MCP is switched on only when asked.
    let body = json!({ "source": source, "activate_mcp": true }).to_string();
    let (status, raw) = call(&runtime, "POST", EXTENSIONS, Some(&body));
    assert_eq!(status, 201, "{raw}");
    assert_eq!(parsed(&raw)["mcp_enabled"], true, "{raw}");
    assert_eq!(config(&home)["mcp"]["enabled"], true);
}
