=== message
Read a credential's secret with a native picker in the host (D8)

The desktop's `remote-credential add|rotate --secret-stdin`: the window
asks the host to add or rotate a grant with the non-secret fields it
has; the host opens the platform's own file picker (GTK on Linux,
NSOpenPanel on macOS, the common item dialog on Windows), reads the
chosen file itself and sends the secret to its runtime in the request
body, as the CLI pipes it on standard input. The window gets back what
happened, the file's name and the runtime's answer, never the secret.

- The API takes a `secret` in the add and rotate bodies only from the
  desktop host (FromDesktop); any other caller is told to name a file,
  and naming both is refused. A redacted InlineSecret carries it.
- The contract marks `secret` host_only_body on both operations; the
  host refuses it from the window (and the test driver).
- Cancel, an unreadable, oversized, non-text, empty or multi-line file
  fail with the reason and store nothing. The chosen file is never
  deleted or changed; the host keeps no copy.
- Backend tests (desktop vs an open loopback server), a host scenario
  with seeded secrets and leak scans of the home, the diagnostics bundle
  and the driver log, and a native scenario in the rendered window that
  also drives the real picker through OS input where the runner allows.

//! The desktop's `remote-credential add|rotate --secret-stdin` (D8): a
//! secret carried in the request, accepted only from the desktop host,
//! which read it from a file the person chose natively. The window cannot
//! send one; that half is the host's (desktop/src-tauri/src/secret.rs).

use std::io::{BufRead as _, BufReader, Read as _, Write as _};
use std::net::TcpStream;
use std::path::{Path, PathBuf};
use std::process::{Child, ChildStdin, Command, Stdio};
use std::sync::mpsc;
use std::time::{Duration, SystemTime, UNIX_EPOCH};

use serde_json::{Value, json};

const KEY: &str = "c5d6e7f8091a2b3c4d5e6f708192a3b4c5d6e7f8091a2b3c4d5e6f708192a3b4";
const CREDENTIALS: &str = "/api/lightagent/v1/remote-credentials";

struct Runtime {
    child: Child,
    _lifeline: Option<ChildStdin>,
    addr: String,
}

impl Drop for Runtime {
    fn drop(&mut self) {
        let _ = self.child.kill();
        let _ = self.child.wait();
    }
}

fn scratch(label: &str) -> PathBuf {
    let nanos = SystemTime::now()
        .duration_since(UNIX_EPOCH)
        .map(|elapsed| elapsed.as_nanos())
        .unwrap_or(0);
    let dir = std::env::temp_dir().join(format!(
        "lightagent-desktop-secret-input-{label}-{}-{nanos}",
        std::process::id()
    ));
    std::fs::create_dir_all(&dir).expect("scratch dir");
    dir
}

fn first_line(child: &mut Child) -> String {
    let stdout = child.stdout.take().expect("stdout");
    let (tx, rx) = mpsc::channel();
    std::thread::spawn(move || {
        let mut reader = BufReader::new(stdout);
        let mut line = String::new();
        let _ = reader.read_line(&mut line);
        let _ = tx.send(line);
        let mut rest = String::new();
        while reader.read_line(&mut rest).is_ok_and(|read| read > 0) {
            rest.clear();
        }
    });
    rx.recv_timeout(Duration::from_secs(60))
        .expect("first line")
}

fn command(home: &Path, key_file: &Path) -> Command {
    let mut command = Command::new(env!("CARGO_BIN_EXE_lightagent"));
    command
        .env("LIGHTAGENT_HOME", home)
        .env_remove("LIGHTAGENT_CREDENTIAL_KEY")
        .env("LIGHTAGENT_CREDENTIAL_KEY_FILE", key_file);
    command
}

fn start_desktop(home: &Path, key_file: &Path) -> Runtime {
    let mut child = command(home, key_file)
        .args(["serve", "--desktop", "--host", "127.0.0.1", "--port", "0"])
        .stdin(Stdio::piped())
        .stdout(Stdio::piped())
        .stderr(Stdio::null())
        .spawn()
        .expect("start lightagent serve --desktop");
    let mut lifeline = child.stdin.take().expect("stdin");
    writeln!(lifeline, "{KEY}").expect("key");
    lifeline.flush().expect("flush");
    let handshake: Value = serde_json::from_str(&first_line(&mut child)).expect("handshake JSON");
    let addr = handshake["addr"].as_str().expect("addr").to_owned();
    Runtime {
        child,
        _lifeline: Some(lifeline),
        addr,
    }
}

/// `lightagent serve` on loopback without a key: a local operator, but not
/// the desktop host.
fn start_open(home: &Path, key_file: &Path) -> Runtime {
    let mut child = command(home, key_file)
        .args(["serve", "--host", "127.0.0.1", "--port", "0"])
        .env_remove("LIGHTAGENT_API_KEY")
        .stdin(Stdio::null())
        .stdout(Stdio::piped())
        .stderr(Stdio::null())
        .spawn()
        .expect("start lightagent serve");
    let line = first_line(&mut child);
    let addr = line
        .split("http://")
        .nth(1)
        .and_then(|rest| rest.split('/').next())
        .unwrap_or_else(|| panic!("no address in {line:?}"))
        .trim()
        .to_owned();
    Runtime {
        child,
        _lifeline: None,
        addr,
    }
}

fn call(runtime: &Runtime, method: &str, path: &str, body: &Value) -> (u16, String) {
    let mut stream = TcpStream::connect(&runtime.addr).expect("connect");
    stream
        .set_read_timeout(Some(Duration::from_secs(60)))
        .expect("timeout");
    let body = body.to_string();
    let head = format!(
        "{method} {path} HTTP/1.1\r\nHost: {}\r\nAuthorization: Bearer {KEY}\r\nContent-Type: application/json\r\nContent-Length: {}\r\nConnection: close\r\n\r\n{body}",
        runtime.addr,
        body.len()
    );
    stream.write_all(head.as_bytes()).expect("send");
    let mut response = String::new();
    let _ = stream.read_to_string(&mut response);
    let status = response
        .split_whitespace()
        .nth(1)
        .and_then(|code| code.parse().ok())
        .unwrap_or(0);
    let raw = response
        .split_once("\r\n\r\n")
        .map(|(_, body)| body.to_owned())
        .unwrap_or_default();
    (status, raw)
}

fn cli(home: &Path, key_file: &Path, args: &[&str]) -> Value {
    let mut all = vec!["--json"];
    all.extend_from_slice(args);
    let output = command(home, key_file)
        .args(&all)
        .stdin(Stdio::null())
        .output()
        .expect("run lightagent");
    assert!(
        output.status.success(),
        "lightagent {args:?}: {}",
        String::from_utf8_lossy(&output.stderr)
    );
    serde_json::from_slice(&output.stdout).unwrap_or(Value::Null)
}

/// Fail if any file under `dir` holds `secret` in plaintext.
fn assert_nowhere(dir: &Path, secret: &str) {
    let Ok(entries) = std::fs::read_dir(dir) else {
        return;
    };
    for entry in entries.flatten() {
        let path = entry.path();
        if path.is_dir() {
            assert_nowhere(&path, secret);
        } else if let Ok(bytes) = std::fs::read(&path) {
            assert!(
                !String::from_utf8_lossy(&bytes).contains(secret),
                "{} holds a secret in plaintext",
                path.display()
            );
        }
    }
}

fn grants(home: &Path, key_file: &Path) -> Vec<Value> {
    cli(home, key_file, &["remote-credential", "list"])["credentials"]
        .as_array()
        .cloned()
        .unwrap_or_default()
}

fn a_secret_in_the_request_is_taken_only_from_the_desktop_host() {
    let home = scratch("home");
    let outside = scratch("outside");
    let key_file = outside.join("vault.key");
    cli(&home, &key_file, &["profile", "create", "work"]);
    let key_arg = key_file.to_str().expect("UTF-8 path").to_owned();
    cli(&home, &key_file, &["remote-credential", "keygen", &key_arg]);
    let fields = json!({
        "profile": "work",
        "service": "web-search",
        "destinations": ["https://search.example.com"],
    });
    let with = |extra: Value| {
        let mut body = fields.clone();
        for (name, value) in extra.as_object().expect("object") {
            body[name] = value.clone();
        }
        body
    };

    let desktop = start_desktop(&home, &key_file);
    // Added and rotated from the secret itself, as `--secret-stdin` does;
    // no answer carries it.
    let secret = "inline-secret-5c6d7e8f9a0b1c2d";
    let (code, raw) = call(&desktop, "POST", CREDENTIALS, &with(json!({"secret": format!("{secret}\n")})));
    assert_eq!(code, 200, "{raw}");
    assert!(!raw.contains(secret), "the answer carries the secret: {raw}");
    let added: Value = serde_json::from_str(&raw).expect("JSON");
    let id = added["added"].as_str().expect("grant id").to_owned();
    let rotated = "inline-rotated-1a2b3c4d5e6f";
    let rotate = format!("{CREDENTIALS}/{id}/rotate");
    let (code, raw) = call(&desktop, "POST", &rotate, &json!({"secret": rotated}));
    assert_eq!(code, 200, "{raw}");
    assert!(!raw.contains(rotated), "{raw}");
    let listed = grants(&home, &key_file);
    assert_eq!(listed.len(), 1, "{listed:?}");
    assert_eq!(listed[0]["version"], 2, "{listed:?}");

    // A file and a secret together, an invalid secret, and neither: refused,
    // without echoing the secret, and nothing stored.
    let file = outside.join("web.secret");
    std::fs::write(&file, "file-secret-0a1b2c3d\n").expect("secret file");
    let (code, raw) = call(&desktop, "POST", CREDENTIALS, &with(json!({"secret": "both-secret-9f8e7d", "secret_file": file})));
    assert_eq!(code, 400, "{raw}");
    assert!(raw.contains("not both"), "{raw}");
    assert!(!raw.contains("both-secret-9f8e7d"), "{raw}");
    let (code, raw) = call(&desktop, "POST", CREDENTIALS, &with(json!({"secret": "line-one-secret\nline-two-secret"})));
    assert_eq!(code, 400, "{raw}");
    assert!(!raw.contains("line-one-secret") && !raw.contains("line-two-secret"), "{raw}");
    let (code, raw) = call(&desktop, "POST", &rotate, &json!({"secret": ""}));
    assert_eq!(code, 400, "{raw}");
    let (code, raw) = call(&desktop, "POST", CREDENTIALS, &fields);
    assert_eq!(code, 400, "{raw}");

=== write desktop/src-tauri/src/secret.rs
//! Host-side secret input (D8): the desktop's `lightagent remote-credential
//! add|rotate --secret-stdin`.
//!
//! The window never holds the secret. It asks the host to add or rotate a
//! grant with the non-secret fields it has. The host opens the platform's
//! own file picker (GTK on Linux, NSOpenPanel on macOS, the common item
//! dialog on Windows), reads the chosen file itself, and sends the secret
//! to its runtime in the request body, as the CLI pipes it on standard
//! input. The window gets back what happened, the file's name and the
//! runtime's answer, which carries no secret. A request from the window
//! that names a secret itself is refused (`host_only_body` in the
//! contract).
//!
//! Nothing is written. The chosen file is left exactly as it was: deleting
//! it is the person's choice. The host keeps no copy once the request is
//! sent; the secret is stored only as the vault seals it.

use std::io::Read as _;
use std::path::{Path, PathBuf};
use std::sync::atomic::Ordering;

use serde::Serialize;
use serde_json::Value;

use crate::Host;

/// The most a secret file may hold, as the CLI reads from standard input.
const MAX_FILE: u64 = 16 * 1024;
/// The vault's own limit on a secret (lightagent_core::credentials).
const MAX_SECRET: usize = 8 * 1024;
const TITLE: &str = "Choose the file that holds the secret";

/// What happened. Never carries the secret.
#[derive(Debug, Serialize)]
pub struct Outcome {
    /// `stored`; `cancelled` (no file chosen, nothing sent); `invalid` (the
    /// file cannot be used, nothing sent); `refused` (the runtime said no).
    pub state: &'static str,
    /// The chosen file's name only, never its path or contents.
    pub file_name: Option<String>,
    pub status: Option<u16>,
    /// The runtime's answer: grant metadata or its error.
    pub body: Option<String>,
    pub reason: Option<String>,
}

impl Outcome {
    fn new(state: &'static str, file_name: Option<String>) -> Self {
        Self {
            state,
            file_name,
            status: None,
            body: None,
            reason: None,
        }
    }
}

/// Add or rotate (`target`, a contract operation that takes a host-only
/// secret) with `fields` from the window and a secret the host reads.
pub async fn submit(
    host: &Host,
    app: &tauri::AppHandle,
    target: String,
    fields: String,
) -> Result<Outcome, String> {
    let operation = host.contract.resolve("POST", &target)?.clone();
    if !operation.host_only_body.iter().any(|field| field == "secret") {
        return Err(format!("{} takes no secret from the host", operation.id));
    }
    if host.switching.load(Ordering::Acquire) && crate::switch::blocked_while_switching(&operation.id) {
        return Err("a profile switch is in progress; try again when it has finished".to_owned());
    }
    let Ok(Value::Object(mut fields)) = serde_json::from_str::<Value>(&fields) else {
        return Err("the request fields must be a JSON object".to_owned());
    };
    if fields.contains_key("secret") || fields.contains_key("secret_file") {
        return Err(
            "the window names no secret and no secret file: the host asks for the file".to_owned(),
        );
    }
    let Some(path) = pick(app).await? else {
        return Ok(Outcome::new("cancelled", None));
    };
    let file_name = path
        .file_name()
        .map(|name| name.to_string_lossy().into_owned());
    let secret = match read_secret(&path) {
        Ok(secret) => secret,
        Err(reason) => {
            return Ok(Outcome {
                reason: Some(reason),
                ..Outcome::new("invalid", file_name)
            });
        }
    };
    fields.insert("secret".to_owned(), Value::String(secret));
    let body = Value::Object(fields).to_string();
    let answer = crate::forward(host, &operation, "POST", &target, Some(body), None, None).await?;
    let state = if (200..300).contains(&answer.status) {
        "stored"
    } else {
        "refused"
    };
    Ok(Outcome {
        status: Some(answer.status),
        body: Some(answer.body),
        ..Outcome::new(state, file_name)
    })
}

/// The secret in `path`, checked as the vault checks it, or why it cannot
/// be used. A reason never quotes the file.
fn read_secret(path: &Path) -> Result<String, String> {
    let file = std::fs::File::open(path)
        .map_err(|error| format!("the chosen file could not be opened ({})", error.kind()))?;
    let metadata = file
        .metadata()
        .map_err(|error| format!("the chosen file could not be read ({})", error.kind()))?;
    if !metadata.is_file() {
        return Err("the chosen item is not a regular file".to_owned());
    }
    let too_large = || {
        format!(
            "the chosen file is over {} KiB; a secret file holds only the secret",
            MAX_FILE / 1024
        )
    };
    if metadata.len() > MAX_FILE {
        return Err(too_large());
    }
    let mut bytes = Vec::new();
    if let Err(error) = file.take(MAX_FILE + 1).read_to_end(&mut bytes) {
        wipe(bytes);
        return Err(format!("the chosen file could not be read ({})", error.kind()));
    }
    if bytes.len() as u64 > MAX_FILE {
        wipe(bytes);
        return Err(too_large());
    }
    let text = match String::from_utf8(bytes) {
        Ok(text) => text,
        Err(error) => {
            wipe(error.into_bytes());
            return Err("the chosen file is not text (UTF-8)".to_owned());
        }
    };
    let problem = {
        let value = text.trim();
        if value.is_empty() {
            Some("the chosen file is empty")
        } else if value.contains(['\r', '\n', '\0']) {
            Some("a secret is one line, and the chosen file has more than one")
        } else if value.len() > MAX_SECRET {
            Some("the secret is over 8 KiB")
        } else {
            None
        }
    };
    if let Some(problem) = problem {
        wipe(text.into_bytes());
        return Err(problem.to_owned());
    }
    Ok(text)
}

/// Overwrite a buffer that held secret bytes before it is released.
fn wipe(mut bytes: Vec<u8>) {
    bytes.fill(0);
    std::hint::black_box(&bytes);
}

        };

        const CANCELLED: u32 = 0x8007_04C7;
        // SAFETY: COM is initialised on this thread for the dialog's
        // lifetime, and the returned path's memory is freed once copied.
        unsafe {
            CoInitializeEx(None, COINIT_APARTMENTTHREADED)
                .ok()
                .map_err(|error| format!("the file picker could not start: {}", error.message()))?;
            let result = (|| {
                let dialog: IFileOpenDialog =
                    CoCreateInstance(&FileOpenDialog, None, CLSCTX_INPROC_SERVER)
                        .map_err(|error| format!("the file picker could not open: {}", error.message()))?;
                let _ = dialog.SetTitle(w!("Choose the file that holds the secret"));
                if let Err(error) = dialog.Show(None) {
                    if error.code().0 as u32 == CANCELLED {
                        return Ok(None);
                    }
                    return Err(format!("the file picker failed: {}", error.message()));
                }
                let item = dialog
                    .GetResult()
                    .map_err(|error| format!("the file picker returned nothing: {}", error.message()))?;
                let name = item
                    .GetDisplayName(SIGDN_FILESYSPATH)
                    .map_err(|error| format!("the chosen item has no file path: {}", error.message()))?;
                let path = name.to_string();
                CoTaskMemFree(Some(name.0 as *const _));
                path.map(|path| Some(PathBuf::from(path)))
                    .map_err(|_| "the chosen path is not valid Unicode".to_owned())
            })();
            CoUninitialize();
            result
        }
    }

  state: "stored" | "cancelled" | "invalid" | "refused";
  file_name: string | null;
  status: number | null;
  body: string | null;
  reason: string | null;
};

const CREDENTIALS = "/api/lightagent/v1/remote-credentials";

/**
 * Add a grant (`grant` null) or rotate one, with a secret the host reads
 * from a file the person picks. Resolves with the sentence to show for a
 * stored or cancelled outcome; throws with the reason for anything else.
 */
export async function sealFromChosenFile(
  grant: string | null,
  fields: Record<string, unknown>,
  stored: (answer: Record<string, unknown>) => string,
): Promise<string> {
  const target = grant === null ? CREDENTIALS : `${CREDENTIALS}/${grant}/rotate`;
  const outcome = await invoke<SecretOutcome>("remote_credential_secret", { target, fields: JSON.stringify(fields) });
  let answer: Record<string, unknown> = {};
  try {
    answer = outcome.body ? (JSON.parse(outcome.body) as Record<string, unknown>) : {};
  } catch {
    answer = {};
  }
  const file = outcome.file_name ?? "the chosen file";
  switch (outcome.state) {
    case "cancelled":
      return "No file was chosen, so nothing was stored.";
    case "invalid":
      throw new Error(`${file} cannot be used: ${outcome.reason ?? "unreadable"}. Nothing was stored; choose a file that holds only the secret.`);
    case "refused":
      throw new Error(`${typeof answer.error === "string" ? answer.error : `The runtime refused it (${outcome.status ?? "no status"})`}. Nothing was stored.`);
    default:
      return `${stored(answer)} The app read the secret from ${file} itself and sealed it; this window never held it. ` +
        "The file is left as it was: delete it yourself if you no longer need it.";
  }
}

def type_path_into_picker(path):
    """Answer the open native file picker with `path` through OS input.
    None when the keys were sent; otherwise why not."""
    time.sleep(2.0)
    try:
        if LINUX:
            subprocess.run(["xdotool", "key", "--clearmodifiers", "ctrl+l"], check=True, timeout=10)
            time.sleep(0.5)
            subprocess.run(["xdotool", "type", "--delay", "20", path], check=True, timeout=60)
            time.sleep(0.5)
            subprocess.run(["xdotool", "key", "Return"], check=True, timeout=10)
        elif MACOS:
            script = ('tell application "System Events"\n keystroke "g" using {command down, shift down}\n delay 1\n'
                      f' keystroke "{path}"\n delay 0.5\n key code 36\n delay 1\n key code 36\nend tell')
            out = subprocess.run(["osascript", "-e", script], capture_output=True, text=True, timeout=60)
            if out.returncode != 0:
                return f"System Events refused the key presses: {out.stderr.strip()[:240]}"
        else:
            escaped = "".join("{" + ch + "}" if ch in "+^%~(){}[]" else ch for ch in path)
            sends = f"[System.Windows.Forms.SendKeys]::SendWait('{escaped}'); Start-Sleep -Milliseconds 500; [System.Windows.Forms.SendKeys]::SendWait('{{ENTER}}')"
            out = subprocess.run(["powershell", "-NoProfile", "-Command", f"Add-Type -AssemblyName System.Windows.Forms; {sends}"],
                                 capture_output=True, text=True, timeout=60)
            if out.returncode != 0:
                return f"SendKeys failed: {out.stderr.strip()[:240]}"
        return None
    except Exception as error:
        return f"{type(error).__name__}: {error}"


def scenario_secret_input(view, host, home, notes):
    """The `--secret-stdin` rows in the rendered window: the Add form's
    host-read button stores a seeded secret the page never holds; a cancel
    and an unusable file say so; then the real native picker, answered
    through OS input where the runner allows it."""
    import secrets
    import tempfile

    check = Checks("secret_input")
    work = tempfile.mkdtemp(prefix="lightagent-native-secret-")
    seeds = ["LA-NATIVE-SEED-" + secrets.token_urlsafe(15), "LA-NATIVE-PICKER-" + secrets.token_urlsafe(15)]
    paths = []
    for index, seed in enumerate(seeds):
        paths.append(os.path.join(work, f"seed-{index}.secret"))
        with open(paths[-1], "w", encoding="utf-8") as handle:
            handle.write(seed + "\n")
    lines = os.path.join(work, "lines.secret")
    with open(lines, "w", encoding="utf-8") as handle:
        handle.write("LA-NATIVE-LINE-one\nLA-NATIVE-LINE-two\n")
    active = host.call("status")["profile"]
    view.go("Remote credentials", "#/admin/credentials")

    def submit(service, wanted):
        return view.js(T("""
          fill(await need("profile", () => document.querySelector("#rc-add-profile")), $PROFILE);
          fill(await need("service", () => document.querySelector("#rc-add-service")), $SERVICE);
          fill(await need("destinations", () => document.querySelector("#rc-add-destinations")), "https://docs.native.invalid");
          await sleep(200);
          press(await need("the host-read button", () => all("button").find((el) => nameOf(el) === "Add with a secret file chosen here…" && !el.disabled)));
          return await until(() => {
            const said = [...all("status"), ...all("alert")].map((el) => nameOf(el)).filter((s) => $WANTED.test(s));
            return said.length ? said.join(" | ") : null;
          }, 90000);""", PROFILE=active, SERVICE=service).replace('"$WANTED"', wanted).replace("$WANTED", wanted), 100)

    def page_dump():
        return view.js("""return document.documentElement.outerHTML + JSON.stringify(Object.entries(localStorage))
          + JSON.stringify(Object.entries(sessionStorage)) + document.body.innerText;""")

    host.call("pick_file", path=paths[0])
    stored = submit("mcp:native-docs", "/^Added /")
    view.shot("secret-stored")
    dump = page_dump()
    check.expect("the host-read button stores the secret and says where it came from", bool(stored) and "seed-0.secret" in stored, stored)
    check.expect("the page never holds the seeded secret (DOM, storage, text)", seeds[0] not in dump)
    host.call("pick_file", cancel=True)
    cancelled = submit("mcp:native-cancel", "/nothing was stored/")
    check.expect("a cancel says nothing was stored", bool(cancelled) and "No file was chosen" in cancelled, cancelled)
    host.call("pick_file", path=lines)
    invalid = submit("mcp:native-lines", "/cannot be used/")
    check.expect("an unusable file says why, near the form", bool(invalid) and "one line" in invalid and "LA-NATIVE-LINE" not in invalid, invalid)
    # The real picker: the page asks, the OS dialog opens, OS input answers it.
    host.call("pick_file", native=True)
    focus_window(host)
    view.js(T("""
      fill(document.querySelector("#rc-add-service"), "mcp:native-picker");
      await sleep(200);
      press(all("button").find((el) => nameOf(el) === "Add with a secret file chosen here…"));
      return true;""", ))
    time.sleep(1.0)
    view.shot("secret-native-picker")
    why = type_path_into_picker(paths[1])
    picked = view.js("""return await until(() => {
        const said = [...all("status"), ...all("alert")].map((el) => nameOf(el)).filter((s) => /seed-1\\.secret|No file was chosen|cannot be used/.test(s));
        return said.length ? said.join(" | ") : null;
      }, 45000);""", 60)
    view.shot("secret-native-done")
    if not picked:
        os_keys(["Escape"])
    leaked = seeds[1] in page_dump()
    notes["native_picker"] = {"os_input": why or "sent", "outcome": picked}
    if LINUX:
        check.expect("the real GTK picker, answered through OS input, stores the chosen file", why is None and bool(picked) and "seed-1.secret" in picked, notes["native_picker"])
    check.expect("the page never holds the secret read through the real picker", not leaked)
    check.expect("the chosen files are left as they were", all(harness.read(p) == s + "\n" for p, s in zip(paths, seeds)))
    check.seeds = seeds
    return check
