/// When a workspace change takes effect.
pub const EFFECT: &str = "Applies to the next run. A run in progress keeps its folder; an editor's ACP session keeps using the editor's folder.";

/// The workspace as the next run of the active profile uses it.
#[derive(Clone, Debug, PartialEq, Eq, Serialize)]
pub struct WorkspaceView {
    /// The configured folder, when one is set.
    pub configured: Option<String>,
    /// The folder the next run of the active profile uses: the configured
    /// one, or that profile's own `workspace/`.
    pub effective: Option<String>,
    /// `config` or `profile`.
    pub source: &'static str,
    /// The active profile.
    pub profile: Option<String>,
    /// Whether the effective folder exists now; a run creates a missing one.
    pub exists: bool,
    pub tools_enabled: bool,
    pub terminal_enabled: bool,
}

/// The workspace for the next run of the active profile.
pub fn view(config: &Config, paths: &LightagentPaths) -> WorkspaceView {
    let profiles = ProfileStore::new(paths.root());
    let profile = profiles.active().ok().flatten();
    let effective: Option<PathBuf> = match &config.tools.workspace {
        Some(folder) => Some(PathBuf::from(folder)),
        None => profile
            .as_ref()
            .map(|id| profiles.handle(id).workspace_dir()),
    };
    WorkspaceView {
        configured: config.tools.workspace.clone(),
        exists: effective.as_ref().is_some_and(|folder| folder.is_dir()),
        effective: effective.map(|folder| folder.display().to_string()),
        source: if config.tools.workspace.is_some() {
            "config"
        } else {
            "profile"
        },
        profile: profile.map(|id| id.as_str().to_owned()),
        tools_enabled: config.tools.enabled,
        terminal_enabled: config.tools.enabled && config.tools.allow_terminal,
    }
}

/// A folder typed by a person, trimmed: absolute, so the terminal and the
/// desktop runtime mean the same folder, and either an existing folder or a
/// new one whose parent folder exists (the next run creates it).
pub fn validate(folder: &str) -> Result<String, String> {
    let folder = folder.trim();
    if folder.chars().any(char::is_control) {
        return Err("the workspace folder contains a control character".to_owned());
    }
    let path = Path::new(folder);
    if !path.is_absolute() {
        return Err(
            "enter an absolute folder path: a relative one would mean a different folder for each process"
                .to_owned(),
        );
    }
    match std::fs::metadata(path) {
        Ok(found) if found.is_dir() => Ok(folder.to_owned()),
        Ok(_) => Err("that path is a file, not a folder".to_owned()),
        Err(_) => match path.parent().map(std::fs::metadata) {
            Some(Ok(parent)) if parent.is_dir() => Ok(folder.to_owned()),
            _ => Err("the folder does not exist, and neither does the folder it would be created in".to_owned()),
        },
    }
}

/// Set the workspace folder; `None` or blank clears it, so runs use the
/// active profile's own `workspace/`. An invalid folder changes nothing.
pub fn set(config: &mut Config, folder: Option<&str>) -> Result<(), String> {
    config.tools.workspace = match folder.map(str::trim).filter(|folder| !folder.is_empty()) {
        Some(folder) => Some(validate(folder)?),
        None => None,
    };
    Ok(())
}

/// The workspace for the next run of the active profile, the revision a
/// change must name, and when a change applies.
fn workspace_view(
    config: &lightagent_core::Config,
    revision: &ConfigRevision,
    root: &std::path::Path,
) -> serde_json::Value {
    json!({
        "workspace": lightagent_core::workspaces::view(config, &LightagentPaths::rooted_at(root)),
        "revision": revision.as_str(),
        "effect": lightagent_core::workspaces::EFFECT,
    })
}

/// The workspace folder `lightagent setup tools` sets.
async fn get_workspace(State(state): State<Arc<AppState>>, headers: HeaderMap) -> Response {
    if let Some(rejection) = deny(&state, &headers, Scope::Admin) {
        return rejection;
    }
    let (Some(store), Some(root)) = (&state.config_store, home(&state)) else {
        return internal("Lightagent settings are unavailable in this embedding");
    };
    match store.load_with_revision() {
        Ok((config, revision)) => Json(workspace_view(&config, &revision, &root)).into_response(),
        Err(error) => config_error(error),
    }
}

/// Set or clear the workspace folder as setup does, on the revision the
/// caller read: a stale one gets 409 and writes nothing; a folder that is
/// not absolute, is a file or has no parent folder gets 400.
async fn put_workspace(
    State(state): State<Arc<AppState>>,
    headers: HeaderMap,
    Json(body): Json<WorkspaceBody>,
) -> Response {
    if let Some(rejection) = deny(&state, &headers, Scope::Admin) {
        return rejection;
    }
    let (Some(store), Some(root)) = (&state.config_store, home(&state)) else {
        return internal("Lightagent settings are unavailable in this embedding");
    };
    let expected = body.expected_revision.as_deref().map(ConfigRevision::parse);
    match store.update(expected.as_ref(), |config| {
        lightagent_core::workspaces::set(config, body.path.as_deref())
    }) {
        Ok(((), revision)) => match store.load() {
            Ok(config) => Json(workspace_view(&config, &revision, &root)).into_response(),
            Err(error) => config_error(error),
        },
        Err(UpdateError::Change(message)) => bad_request(&message),
        Err(UpdateError::Store(error)) => config_error(error),
    }
}

struct Runtime {

fn scratch_home(label: &str) -> PathBuf {
    let nanos = SystemTime::now()
        .duration_since(UNIX_EPOCH)
        .map(|elapsed| elapsed.as_nanos())
        .unwrap_or(0);
    let home = std::env::temp_dir().join(format!(
        "lightagent-desktop-workspace-{label}-{}-{nanos}",
        std::process::id()
    ));
    std::fs::create_dir_all(&home).expect("home");
    home
}

fn start(home: &Path) -> Runtime {
    let mut child = Command::new(env!("CARGO_BIN_EXE_lightagent"))
        .args(["serve", "--desktop", "--host", "127.0.0.1", "--port", "0"])
        .env("LIGHTAGENT_HOME", home)
        .stdin(Stdio::piped())
        .stdout(Stdio::piped())
        .stderr(Stdio::null())
        .spawn()
        .expect("start lightagent serve --desktop");
    let mut lifeline = child.stdin.take().expect("stdin");
    writeln!(lifeline, "{KEY}").expect("key");
    lifeline.flush().expect("flush");
    let stdout = child.stdout.take().expect("stdout");
    let (tx, rx) = mpsc::channel();
    std::thread::spawn(move || {
        let mut line = String::new();
        let _ = BufReader::new(stdout).read_line(&mut line);
        let _ = tx.send(line);
    });
    let line = rx.recv_timeout(Duration::from_secs(60)).expect("handshake");
    let handshake: Value = serde_json::from_str(&line).expect("handshake JSON");
    let addr = handshake["addr"].as_str().expect("addr").to_owned();
    Runtime {
        child,
        _lifeline: lifeline,
        addr,
    }
}

/// One authenticated request: the status, the raw body and its JSON.
fn call(runtime: &Runtime, method: &str, path: &str, body: Option<&str>) -> (u16, String, Value) {
    let mut stream = TcpStream::connect(&runtime.addr).expect("connect");
    stream
        .set_read_timeout(Some(Duration::from_secs(60)))
        .expect("timeout");
    let body = body.unwrap_or("");
    let head = format!(
        "{method} {path} HTTP/1.1\r\nHost: {}\r\nAuthorization: Bearer {KEY}\r\nContent-Type: application/json\r\nContent-Length: {}\r\nConnection: close\r\n\r\n{body}",
        runtime.addr,
        body.len()
    );
    stream.write_all(head.as_bytes()).expect("send");
    let mut response = String::new();
    let _ = stream.read_to_string(&mut response);
    let status = response
        .split_whitespace()
        .nth(1)
        .and_then(|code| code.parse().ok())
        .unwrap_or(0);
    let raw = response
        .split_once("\r\n\r\n")
        .map(|(_, body)| body.to_owned())
        .unwrap_or_default();
    let json = serde_json::from_str(&raw).unwrap_or(Value::Null);
    (status, raw, json)
}

fn put(runtime: &Runtime, body: Value) -> (u16, String, Value) {
    call(runtime, "PUT", WORKSPACE, Some(&body.to_string()))
}

fn cli(home: &Path, args: &[&str]) -> Output {

        .args(args)
        .env("LIGHTAGENT_HOME", home)
        .stdin(Stdio::null())
        .output()
        .expect("run lightagent")
}

fn the_workspace_is_one_validated_shared_setting() {
    let home = scratch_home("set");
    cli_ok(&home, &["config", "set", "inference.model", "m1"]);
    cli_ok(&home, &["profile", "create", "work"]);
    cli_ok(&home, &["profile", "create", "other"]);
    cli_ok(&home, &["profile", "use", "work"]);
    let runtime = start(&home);

    // None set: the active profile's own folder, and it follows a switch.
    let (status, raw, view) = call(&runtime, "GET", WORKSPACE, None);
    assert_eq!(status, 200, "{raw}");
    assert_eq!(view["workspace"]["source"], "profile", "{view}");
    assert_eq!(view["workspace"]["effective"], profile_workspace(&home, "work"));
    assert!(
        view["effect"]
            .as_str()
            .unwrap_or_default()
            .contains("next run"),
        "{view}"
    );
    cli_ok(&home, &["profile", "use", "other"]);
    let (_, _, view) = call(&runtime, "GET", WORKSPACE, None);
    assert_eq!(view["workspace"]["effective"], profile_workspace(&home, "other"));
    assert_eq!(view["workspace"]["profile"], "other");

    // Refused folders write nothing.
    let file = home.join("a-file.txt");
    std::fs::write(&file, "x").expect("file");
    let before = std::fs::read(home.join("config.json")).expect("config");
    for folder in [
        "relative/folder".to_owned(),
        file.display().to_string(),
        home.join("missing").join("deeper").display().to_string(),
    ] {
        let (status, raw, _) = put(&runtime, json!({ "path": folder }));
        assert_eq!(status, 400, "{folder}: {raw}");
    }
    assert_eq!(
        std::fs::read(home.join("config.json")).expect("config"),
        before,
        "a refused folder wrote the config"
    );

    // A change on a revision read before a CLI write is refused.
    let folder = home.join("project").display().to_string();
    let read_at = view["revision"].clone();
    cli_ok(&home, &["config", "set", "inference.model", "m2"]);
    let (status, raw, _) = put(
        &runtime,
        json!({ "path": folder, "expected_revision": read_at }),
    );
    assert_eq!(status, 409, "{raw}");
    assert!(configured(&home).is_null());

    // On the current revision it is saved: a new folder inside an
    // existing one, created by the next run.
    let (_, _, view) = call(&runtime, "GET", WORKSPACE, None);
    let (status, raw, view) = put(
        &runtime,
        json!({ "path": folder, "expected_revision": view["revision"] }),
    );
    assert_eq!(status, 200, "{raw}");
    assert_eq!(view["workspace"]["source"], "config");
    assert_eq!(view["workspace"]["configured"], folder);
    assert_eq!(view["workspace"]["exists"], false);
    assert_eq!(configured(&home), folder);

    // Cleared: back to the active profile's own folder.
    let (status, raw, view) = put(&runtime, json!({ "path": null }));
    assert_eq!(status, 200, "{raw}");
    assert_eq!(view["workspace"]["source"], "profile");
    assert!(configured(&home).is_null());
}

              Use the profile's own folder
            </button>
          )}
        </div>
      </div>
    </Card>
  );
