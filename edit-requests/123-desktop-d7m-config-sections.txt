struct Runtime {
    child: Child,
    _lifeline: ChildStdin,
    addr: String,
}

impl Drop for Runtime {
    fn drop(&mut self) {
        let _ = self.child.kill();
        let _ = self.child.wait();
    }
}

fn scratch_home(label: &str) -> PathBuf {
    let nanos = SystemTime::now()
        .duration_since(UNIX_EPOCH)
        .map(|elapsed| elapsed.as_nanos())
        .unwrap_or(0);
    let home = std::env::temp_dir().join(format!(
        "lightagent-desktop-config-sections-{label}-{}-{nanos}",
        std::process::id()
    ));
    std::fs::create_dir_all(&home).expect("home");
    home
}

fn start(home: &Path) -> Runtime {
    let mut child = Command::new(env!("CARGO_BIN_EXE_lightagent"))
        .args(["serve", "--desktop", "--host", "127.0.0.1", "--port", "0"])
        .env("LIGHTAGENT_HOME", home)
        .stdin(Stdio::piped())
        .stdout(Stdio::piped())
        .stderr(Stdio::null())
        .spawn()
        .expect("start lightagent serve --desktop");
    let mut lifeline = child.stdin.take().expect("stdin");
    writeln!(lifeline, "{KEY}").expect("key");
    lifeline.flush().expect("flush");
    let stdout = child.stdout.take().expect("stdout");
    let (tx, rx) = mpsc::channel();
    std::thread::spawn(move || {
        let mut reader = BufReader::new(stdout);
        let mut line = String::new();
        let _ = reader.read_line(&mut line);
        let _ = tx.send(line);
        let mut rest = String::new();
        while reader.read_line(&mut rest).is_ok_and(|read| read > 0) {
            rest.clear();
        }
    });
    let line = rx.recv_timeout(Duration::from_secs(60)).expect("handshake");
    let handshake: Value = serde_json::from_str(&line).expect("handshake JSON");
    let addr = handshake["addr"].as_str().expect("addr").to_owned();
    Runtime {
        child,
        _lifeline: lifeline,
        addr,
    }
}

fn call(runtime: &Runtime, method: &str, path: &str, body: Option<&str>) -> (u16, String) {
    let mut stream = TcpStream::connect(&runtime.addr).expect("connect");
    stream
        .set_read_timeout(Some(Duration::from_secs(60)))
        .expect("timeout");
    let body = body.unwrap_or("");
    let head = format!(
        "{method} {path} HTTP/1.1\r\nHost: {}\r\nAuthorization: Bearer {KEY}\r\nContent-Type: application/json\r\nContent-Length: {}\r\nConnection: close\r\n\r\n{body}",
        runtime.addr,
        body.len()
    );
    stream.write_all(head.as_bytes()).expect("send");
    let mut response = String::new();
    let _ = stream.read_to_string(&mut response);
    let status = response
        .split_whitespace()
        .nth(1)
        .and_then(|code| code.parse().ok())
        .unwrap_or(0);
    let raw = response
        .split_once("\r\n\r\n")
        .map(|(_, body)| body.to_owned())
        .unwrap_or_default();
    (status, raw)
}

fn cli(home: &Path, args: &[&str]) -> Output {
    Command::new(env!("CARGO_BIN_EXE_lightagent"))
        .args(args)
        .env("LIGHTAGENT_HOME", home)
        .stdin(Stdio::null())
        .output()
        .expect("run lightagent")
}

fn cli_ok(home: &Path, args: &[&str]) -> String {
    let output = cli(home, args);
    assert!(
        output.status.success(),
        "lightagent {args:?}: {}",
        String::from_utf8_lossy(&output.stderr)
    );
    String::from_utf8_lossy(&output.stdout).trim().to_owned()
}

#[test]
fn one_key_of_every_settable_section_reads_and_writes_alike() {
    let home = scratch_home("sections");
    let runtime = start(&home);
    for (key, by_api, by_cli) in SECTIONS {
        let body = json!({ "value": by_api }).to_string();
        let (status, raw) = call(&runtime, "PUT", &format!("{CONFIG}/{key}"), Some(&body));
        assert_eq!(status, 200, "{key}: {raw}");
        assert_eq!(cli_ok(&home, &["config", "get", key]), by_api, "{key}");
        cli_ok(&home, &["config", "set", key, by_cli]);
        let (status, raw) = call(&runtime, "GET", &format!("{CONFIG}/{key}"), None);
        assert_eq!(status, 200, "{key}: {raw}");
        let read: Value = serde_json::from_str(&raw).expect("JSON");
        assert_eq!(read["value"], by_cli, "{key}: {raw}");
    }
}

// One key of every configuration section the CLI can set (inventory rows

const SECTIONS = [
  ["inference.model", "sections-model-a"],
  ["agent.max_turns", "17"],

export async function checkConfigSections(browser, base, outDir) {
  const failures = [];
  const page = await browser.newPage({ viewport: { width: 1280, height: 1100 } });
  const errors = [];
  page.on("pageerror", (error) => errors.push(error.message));
  const url = (key) => `${base}/api/lightagent/v1/config/${key}`;
  const valueOf = async (key) => (await (await fetch(url(key))).json()).value;
  const restore = (key, value) =>
    fetch(url(key), { method: "PUT", headers: { "Content-Type": "application/json" }, body: JSON.stringify({ value }) });
  try {
    await page.goto(`${base}/#/settings`);
    await page.getByRole("heading", { name: "Advanced configuration" }).waitFor({ timeout: 15000 });
    for (const [key, value] of SECTIONS) {
      const original = await valueOf(key);
      try {
        await page.getByLabel("Key", { exact: true }).fill(key);
        await page.getByRole("button", { name: "Read", exact: true }).click();
        await page.getByText(`${key} = `, { exact: false }).first().waitFor({ timeout: 5000 });
        await page.getByLabel("Value", { exact: true }).fill(value);
        await page.getByRole("button", { name: "Set", exact: true }).click();
        await page.getByText(`Set ${key} = ${value}.`, { exact: false }).waitFor({ timeout: 5000 });
        const now = await valueOf(key);
        if (now !== value) failures.push(`config sections: ${key} is ${JSON.stringify(now)} after the card set ${value}`);
      } catch (err) {
        failures.push(`config sections: ${key}: ${err instanceof Error ? err.message.split("\n")[0] : String(err)}`);
        await page.screenshot({ path: `${outDir}/config-sections-failed-${key.replace(/\W+/g, "-")}.png` }).catch(() => {});
      } finally {
        await restore(key, original ?? "").catch(() => {});
      }
    }
    if (errors.length) failures.push(`config sections: page errors: ${errors.join("; ")}`);
  } catch (err) {
    failures.push(`config sections: ${err instanceof Error ? err.message.split("\n")[0] : String(err)}`);
  } finally {
    await page.close();
  }
  return failures;
}
