/// What happens when a run calls `definition` under `policy`: `auto` (it
/// runs), `ask` (it waits for approval) or `block`. Shared by `tools list`
/// and the API's tool catalog, so both report the same permission.
pub(crate) fn tool_permission(
    policy: &lightagent_core::PolicyEngine,
    definition: &lightagent_tools::ToolDefinition,
) -> &'static str {
    let request = lightagent_core::ApprovalRequest::new(
        &definition.name,
        definition.risk,
        definition.scopes.clone(),
        "{}",
    );
    match policy.evaluate(&request, std::time::SystemTime::now()) {
        lightagent_core::ApprovalNeed::AutoApprove => "auto",
        lightagent_core::ApprovalNeed::Require(_) => "ask",
        lightagent_core::ApprovalNeed::Deny(_) => "block",
    }
}

struct Runtime {
    child: Child,
    _lifeline: ChildStdin,
    addr: String,
}

impl Drop for Runtime {
    fn drop(&mut self) {
        let _ = self.child.kill();

fn scratch_home(label: &str) -> PathBuf {
    let nanos = SystemTime::now()
        .duration_since(UNIX_EPOCH)
        .map(|elapsed| elapsed.as_nanos())
        .unwrap_or(0);
    let home = std::env::temp_dir().join(format!(
        "lightagent-desktop-tools-{label}-{}-{nanos}",
        std::process::id()
    ));
    std::fs::create_dir_all(&home).expect("home");
    home
}

fn start(home: &Path) -> Runtime {
    let mut child = Command::new(env!("CARGO_BIN_EXE_lightagent"))
        .args(["serve", "--desktop", "--host", "127.0.0.1", "--port", "0"])
        .env("LIGHTAGENT_HOME", home)
        .stdin(Stdio::piped())
        .stdout(Stdio::piped())
        .stderr(Stdio::null())
        .spawn()
        .expect("start lightagent serve --desktop");
    let mut lifeline = child.stdin.take().expect("stdin");
    writeln!(lifeline, "{KEY}").expect("key");
    lifeline.flush().expect("flush");
    let stdout = child.stdout.take().expect("stdout");
    let (tx, rx) = mpsc::channel();
    std::thread::spawn(move || {
        let mut line = String::new();
        let _ = BufReader::new(stdout).read_line(&mut line);
        let _ = tx.send(line);
    });
    let line = rx.recv_timeout(Duration::from_secs(60)).expect("handshake");
    let handshake: Value = serde_json::from_str(&line).expect("handshake JSON");
    let addr = handshake["addr"].as_str().expect("addr").to_owned();
    Runtime {
        child,
        _lifeline: lifeline,
        addr,
    }
}

/// One authenticated request: the status and the raw body.
fn call(runtime: &Runtime, method: &str, path: &str, body: Option<&str>) -> (u16, String) {
    let mut stream = TcpStream::connect(&runtime.addr).expect("connect");
    stream
        .set_read_timeout(Some(Duration::from_secs(60)))
        .expect("timeout");
    let body = body.unwrap_or("");
    let head = format!(
        "{method} {path} HTTP/1.1\r\nHost: {}\r\nAuthorization: Bearer {KEY}\r\nContent-Type: application/json\r\nContent-Length: {}\r\nConnection: close\r\n\r\n{body}",
        runtime.addr,
        body.len()
    );
    stream.write_all(head.as_bytes()).expect("send");
    let mut response = String::new();
    let _ = stream.read_to_string(&mut response);
    let status = response
        .split_whitespace()
        .nth(1)
        .and_then(|code| code.parse().ok())
        .unwrap_or(0);
    let raw = response
        .split_once("\r\n\r\n")
        .map(|(_, body)| body.to_owned())
        .unwrap_or_default();
    (status, raw)
}

/// (name, risk, permission, description), sorted.
fn rows(tools: &Value) -> Vec<(String, String, String, String)> {
    let mut rows: Vec<_> = tools
        .as_array()
        .expect("tools")
        .iter()
        .map(|tool| {
            let field = |key: &str| tool[key].as_str().unwrap_or_default().to_owned();
            (field("name"), field("risk"), field("permission"), field("description"))

    let source = home.join("bundle-src");

    skill_dir(&source.join("skills"), "beta", "Extension beta");

        assert_eq!(status, 200, "{raw}");
        parsed(&raw)["skills"].clone()
    };

    // The active extension's alpha overrides the global one, as in a run.
    let listed = skills(&runtime);
    let by_name = |name: &str| {
        listed
            .as_array()
            .expect("skills")
            .iter()
            .find(|skill| skill["name"] == name)
            .cloned()
            .unwrap_or(Value::Null)
    };

    );
    assert_eq!(by_name("beta")["source"]["scope"], "extension", "{listed}");

    // Disabled, it contributes nothing: alpha is the global one again.
    cli_ok(&home, &["extensions", "disable", "tooling"]);
    let listed = skills(&runtime);
    let names: Vec<&str> = listed
        .as_array()
        .expect("skills")
        .iter()
        .filter_map(|skill| skill["name"].as_str())
        .collect();
    assert_eq!(names, vec!["alpha"], "{listed}");
    assert_eq!(listed[0]["source"], json!({ "scope": "global" }));
    assert_eq!(listed[0]["description"], "Global alpha");

export async function checkToolCatalog(browser, base, outDir) {
  const failures = [];
  const dir = await mkdtemp(join(tmpdir(), "lightagent-render-skills-"));
  const bundle = join(dir, EXTENSION);
  await mkdir(join(bundle, "skills", "render-probe-skill"), { recursive: true });
  await writeFile(join(bundle, "extension.json"), JSON.stringify({ name: EXTENSION, version: "1.0.0" }));
  await writeFile(
    join(bundle, "skills", "render-probe-skill", "SKILL.md"),
    "---\nname: render-probe-skill\ndescription: Probe skill\nversion: 2.1.0\n---\nDo it.\n",
  );
  const page = await browser.newPage({ viewport: { width: 1280, height: 1300 } });
  const errors = [];
  page.on("pageerror", (error) => errors.push(error.message));
  const toolRequests = [];
  page.on("request", (request) => {
    const url = new URL(request.url());
    if (url.pathname.endsWith("/api/lightagent/v1/tools")) toolRequests.push(url.search);
  });
  try {
    const installed = await fetch(`${base}/api/lightagent/v1/extensions`, {
      method: "POST",
      headers: { "content-type": "application/json" },
      body: JSON.stringify({ source: bundle }),
    });
    if (installed.status !== 201) throw new Error(`the probe extension did not install (${installed.status})`);
    const catalog = await (await fetch(`${base}/api/lightagent/v1/tools?mcp=skip`)).json();
    for (const hash of ["#/", "#/settings", "#/tools"]) {
      await page.goto(`${base}/${hash}`);
      await page.waitForLoadState("networkidle").catch(() => {});
    }
    await page.getByText("datetime.now", { exact: true }).first().waitFor({ timeout: 15000 });
    const connecting = toolRequests.filter((search) => !search.includes("mcp=skip"));
    if (connecting.length) failures.push(`tools: ${connecting.length} catalog request(s) asked to connect MCP servers`);
    for (const tool of catalog.tools) {
      const label = LABEL[tool.permission];
      if (!label) {
        failures.push(`tools: ${tool.name} has permission ${tool.permission}`);
        continue;
      }
      const row = page.locator("div", { has: page.getByText(tool.name, { exact: true }) }).filter({ hasText: label });
      if ((await row.count()) === 0) failures.push(`tools: ${tool.name} does not show "${label}"`);
    }
    const skill = page.locator("div", { has: page.getByText("render-probe-skill", { exact: true }) });
    await skill.filter({ hasText: `Extension ${EXTENSION}` }).first().waitFor({ timeout: 10000 })
      .catch(() => failures.push("tools: the extension's skill is not listed with its source"));
    if (!(await skill.first().innerText().catch(() => "")).includes("2.1.0")) {
      failures.push("tools: the skill's version is not shown");
    }
    await page.screenshot({ path: `${outDir}/tools-permissions.png`, fullPage: true });
    if (errors.length) failures.push(`tools: page errors: ${errors.join("; ")}`);
  } catch (err) {
    failures.push(`tools: ${err instanceof Error ? err.message.split("\n")[0] : String(err)}`);
  } finally {
    await page.close();
    await fetch(`${base}/api/lightagent/v1/extensions/${EXTENSION}`, { method: "DELETE" }).catch(() => {});
    await rm(dir, { recursive: true, force: true });
  }
  return failures;
}
